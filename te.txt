async function getMenuNamesByEmpId(req, res) {
  const empId = req.body.emp_id;

  try {
    // Get a connection from the Oracle database pool
    const connection = await oracledb.getConnection(dbConfig);

    // Define the output bind variable
    const resultBind = {
      dir: oracledb.BIND_OUT,
      type: oracledb.CURSOR
    };

    // Prepare and execute the SQL query with input values
    const query = `
    SELECT DISTINCT m.id, m.MENU_NAME,m.WEB_URL,m.PARENT_ID
    FROM DMS.C_MENU_ACCESS_T m
    LEFT JOIN DMS.C_ROLE_MANAGEMENT_T r ON m.id = r.menu_id
    LEFT JOIN DMS.C_USER_MANAGEMENT_T u ON r.ROLE_ID = u.ROLE_ID
    WHERE u.EMP_ID = :empId AND u.USER_STATUS_CONTROL = 0
       OR m.id IN (
          SELECT menu_id
          FROM DMS.C_USER_MANAGEMENT_T
          WHERE EMP_ID = :empId AND DEL_FLAG = 0 AND u.USER_STATUS_CONTROL = 0 
       ) ORDER BY ID
    `;

    const result = await connection.execute(query, [empId]);

    // Retrieve the result from the output bind variable

    // Close the result set and release the connection
    const columns = result.metaData.map(column => column.name);
        const data = result.rows.map(row => {
          const rowObject = {};
          columns.forEach((column, index) => {
            rowObject[column] = row[index];
          });
          return rowObject;
        });
    await connection.close();

    // Send the response with the retrieved data
    res.json({ empId, menu: data });
  } catch (error) {
    console.error('Error executing SQL query:', error);
    res.status(500).json({ error: 'Internal server error' });
  }
}
