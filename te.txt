PROCEDURE INSERT_MEAL_SCAN_LOG(
    V_EMP_ID    IN VARCHAR2,
    V_SITE      IN VARCHAR2,
    V_FX_BADGE  IN VARCHAR2,
    V_RES       OUT VARCHAR2,
    V_RESULT    OUT VARCHAR2,
    P_CURSOR    OUT SYS_REFCURSOR
) IS
    V_COUNT         NUMBER := 0;
    V_ALREADY_COUNT NUMBER := 0;

    V_GROUPS         CMS.C_F1_MEAL_INFO.GROUPS%TYPE;
    V_SITE_VAL       CMS.C_F1_MEAL_INFO.PROPERTY_01%TYPE;
    V_NAME           CMS.C_F1_MEAL_INFO.NAME%TYPE;
    V_MEAL_PREF      CMS.C_F1_MEAL_INFO.MEAL_PREFERENCE%TYPE;
    V_MEAL_PREF_NAME CMS.C_BASE_CODE_T.NAME%TYPE;
    V_FIRST_DATE     CMS.C_F1_MEAL_INFO.FIRST_DATE_FXBL%TYPE;
    V_LAST_DATE      CMS.C_F1_MEAL_INFO.LAST_DATE_FXBL%TYPE;

    V_STATUS  VARCHAR2(50);
    V_MESSAGE VARCHAR2(200);

BEGIN

    IF IS_ACTIVE_CMS_USER(V_EMP_ID) = 0 THEN
        V_RES := '0';
        V_RESULT := 'Active CMS user not found for employee: '
                    || V_EMP_ID;
        RETURN;
    END IF;

    IF V_SITE IS NOT NULL AND TRIM(V_SITE) IS NOT NULL THEN
        IF HAS_SITE_ACCESS(V_EMP_ID, V_SITE) = 0 THEN
            V_RES := '0';
            V_RESULT := 'You are not authorized for the selected site.';
            RETURN;
        END IF;
    END IF;

    /*
     * IMPORTANT:
     * Only records added today can be scanned.
     */
    BEGIN
        SELECT
            GROUPS,
            PROPERTY_01,
            NAME,
            MEAL_PREFERENCE,
            FIRST_DATE_FXBL,
            LAST_DATE_FXBL
        INTO
            V_GROUPS,
            V_SITE_VAL,
            V_NAME,
            V_MEAL_PREF,
            V_FIRST_DATE,
            V_LAST_DATE
        FROM (
            SELECT
                M.*,
                ROW_NUMBER() OVER (
                    ORDER BY M.ADD_DATE DESC, M.ID DESC
                ) AS RN
            FROM CMS.C_F1_MEAL_INFO M
            WHERE UPPER(TRIM(M.FX_BADGE)) =
                  UPPER(TRIM(V_FX_BADGE))
              AND NVL(M.DEL_FLAG, 0) = 0
              AND M.ADD_DATE >= TRUNC(SYSDATE)
              AND M.ADD_DATE < TRUNC(SYSDATE) + 1
        )
        WHERE RN = 1;

        V_COUNT := 1;

    EXCEPTION
        WHEN NO_DATA_FOUND THEN
            V_COUNT := 0;
    END;

    /*
     * Badge not found or added before today.
     */
    IF V_COUNT = 0 THEN

        V_RES := '0';
        V_RESULT := 'FX Badge not found';

        OPEN P_CURSOR FOR
            SELECT
                V_FX_BADGE AS FX_BADGE,
                CAST(NULL AS VARCHAR2(200)) AS NAME,
                CAST(NULL AS VARCHAR2(100)) AS SITE,
                CAST(NULL AS VARCHAR2(100)) AS SITE_NAME,
                CAST(NULL AS VARCHAR2(200)) AS MEAL_PREFERENCE,
                TO_CHAR(SYSDATE, 'DD-MM-YYYY HH24:MI:SS')
                    AS SCAN_TIME,
                'INVALID' AS STATUS,
                'FX Badge not found' AS STATUS_MESSAGE
            FROM DUAL;

        RETURN;

    END IF;

    /*
     * Validate site ownership.
     */
    IF HAS_SITE_ACCESS(V_EMP_ID, V_SITE_VAL) = 0 THEN
        V_RES := '0';
        V_RESULT := 'This FX Badge does not belong to your assigned site.';
        RETURN;
    END IF;

    /*
     * Get meal preference name.
     */
    SELECT NVL(MAX(BC.NAME), TO_CHAR(V_MEAL_PREF))
    INTO V_MEAL_PREF_NAME
    FROM CMS.C_BASE_CODE_T BC
    WHERE BC.TYPE_ID = 1001
      AND TO_CHAR(BC.ID) = TO_CHAR(V_MEAL_PREF);

    /*
     * Validate meal eligibility.
     */
    IF UPPER(TRIM(V_MEAL_PREF_NAME)) = 'NO MEAL REQUIRED' THEN

        V_STATUS := 'INVALID';
        V_MESSAGE := 'NO MEAL REQUIRED BUT COME TO COLLECT';

    ELSIF V_FIRST_DATE IS NULL OR V_LAST_DATE IS NULL THEN

        V_STATUS := 'INVALID';
        V_MESSAGE := 'Dates not updated';

    ELSIF TRUNC(SYSDATE) NOT BETWEEN
          TRUNC(V_FIRST_DATE) AND TRUNC(V_LAST_DATE) THEN

        V_STATUS := 'INVALID';
        V_MESSAGE := 'LAST DATE EXPIRED';

    ELSE

        V_STATUS := 'SUCCESS';
        V_MESSAGE := 'Eligible for meal';

    END IF;

    /*
     * Check whether the meal was already collected today.
     */
    IF V_STATUS = 'SUCCESS' THEN

        SELECT COUNT(1)
        INTO V_ALREADY_COUNT
        FROM CMS.C_F1_SCAN_LOG
        WHERE UPPER(TRIM(FX_BADGE)) = UPPER(TRIM(V_FX_BADGE))
          AND TRUNC(MEAL_DATE) = TRUNC(SYSDATE)
          AND NVL(DEL_FLAG, 0) = 0
          AND COLLECTED_DATA = 'SUCCESS';

        IF V_ALREADY_COUNT > 0 THEN
            V_STATUS := 'ALREADY_SCANNED';
            V_MESSAGE := 'Meal already collected today';
        END IF;

    END IF;

    /*
     * Insert today's scan result.
     */
    INSERT INTO CMS.C_F1_SCAN_LOG (
        ID,
        GROUPS,
        PROPERTY_01,
        MEAL_PREFERENCE,
        FX_BADGE,
        NAME,
        FIRST_DATE_FXBL,
        LAST_DATE_FXBL,
        COLLECTED_DATA,
        STATUS_MESSAGE,
        MEAL_DATE,
        FIRST_SCANTIME,
        LAST_SCANTIME,
        DEL_FLAG
    )
    VALUES (
        CMS.SEQ_C_F1_SCAN_LOG.NEXTVAL,
        V_GROUPS,
        V_SITE_VAL,
        V_MEAL_PREF,
        V_FX_BADGE,
        V_NAME,
        V_FIRST_DATE,
        V_LAST_DATE,
        V_STATUS,
        V_MESSAGE,
        SYSDATE,
        SYSDATE,
        SYSDATE,
        0
    );

    COMMIT;

    OPEN P_CURSOR FOR
        SELECT
            V_FX_BADGE AS FX_BADGE,
            V_NAME AS NAME,
            V_SITE_VAL AS SITE,
            GET_SITE_NAME(V_SITE_VAL) AS SITE_NAME,
            V_MEAL_PREF_NAME AS MEAL_PREFERENCE,
            TO_CHAR(SYSDATE, 'DD-MM-YYYY HH24:MI:SS')
                AS SCAN_TIME,
            V_STATUS AS STATUS,
            V_MESSAGE AS STATUS_MESSAGE
        FROM DUAL;

    V_RES := '1';
    V_RESULT := 'Success';

EXCEPTION
    WHEN DUP_VAL_ON_INDEX THEN
        ROLLBACK;

        V_RES := '1';
        V_RESULT := 'Already scanned today';

        OPEN P_CURSOR FOR
            SELECT
                V_FX_BADGE AS FX_BADGE,
                V_NAME AS NAME,
                V_SITE_VAL AS SITE,
                GET_SITE_NAME(V_SITE_VAL) AS SITE_NAME,
                V_MEAL_PREF_NAME AS MEAL_PREFERENCE,
                TO_CHAR(SYSDATE, 'DD-MM-YYYY HH24:MI:SS')
                    AS SCAN_TIME,
                'ALREADY_SCANNED' AS STATUS,
                'Meal already collected today' AS STATUS_MESSAGE
            FROM DUAL;

    WHEN OTHERS THEN
        ROLLBACK;

        V_RES := '0';
        V_RESULT := 'Error: ' || SQLERRM;

END INSERT_MEAL_SCAN_LOG;
