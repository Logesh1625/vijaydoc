DownloadExcelfile() {

  // =====================================================
  // 1. CHECK MEAL PREFERENCE MASTER DATA
  // =====================================================

  const mealValues = (this.mealPreferenceList || [])
    .map((item: any) => item.label)
    .filter((value: any) =>
      value !== null &&
      value !== undefined &&
      value !== ""
    )
    .map((value: any) => String(value).trim());

  if (mealValues.length === 0) {

    this.msgSvr.warning(
      "Meal Preference data is not loaded. Please try again."
    );

    this.loadMealPreferences();

    return;
  }

  console.log(
    "Meal Preference values:",
    mealValues
  );


  // =====================================================
  // 2. CREATE WORKBOOK
  // =====================================================

  XlsxPopulate.fromBlankAsync()
    .then((workbook: any) => {

      const worksheet = workbook.sheet(0);

      worksheet.name("Template");


      // =================================================
      // 3. ADD HEADER
      // =================================================

      worksheet.range("A1:G1").value([[
        "Groups",
        "Sub Group",
        "Meal Preference",
        "FX Badge",
        "Name",
        "First Day at FXBL",
        "Last Day at FXBL"
      ]]);


      // =================================================
      // 4. HEADER STYLE
      // =================================================

      worksheet.range("A1:G1").style({
        bold: true,
        fontSize: 11,
        horizontalAlignment: "center",
        verticalAlignment: "center",
        wrapText: true
      });

      worksheet.row(1).height(25);


      // =================================================
      // 5. COLUMN WIDTH
      // =================================================

      worksheet.column("A").width(20);
      worksheet.column("B").width(20);
      worksheet.column("C").width(25);
      worksheet.column("D").width(20);
      worksheet.column("E").width(30);
      worksheet.column("F").width(22);
      worksheet.column("G").width(22);


      // =================================================
      // 6. DATE FORMAT
      // =================================================

      worksheet
        .range("F2:G500")
        .style(
          "numberFormat",
          "dd-mm-yyyy"
        );


      // =================================================
      // 7. MEAL PREFERENCE DROPDOWN
      // =================================================

      const masterSheet =
        workbook.addSheet("MealMaster");

      masterSheet
        .cell("A1")
        .value("Meal Preference");


      for (let i = 0; i < mealValues.length; i++) {

        masterSheet
          .cell(`A${i + 2}`)
          .value(mealValues[i]);

      }


      // Hide master sheet
      masterSheet.hidden(true);


      const lastMealRow =
        mealValues.length + 1;


      // =================================================
      // 8. MEAL DROPDOWN C2:C500
      // =================================================

      worksheet
        .range("C2:C500")
        .dataValidation({

          type: "list",

          allowBlank: true,

          showErrorMessage: true,

          errorTitle:
            "Invalid Meal Preference",

          error:
            "Please select Meal Preference from the dropdown.",

          formula1:
            `'MealMaster'!$A$2:$A$${lastMealRow}`

        });


      // =================================================
      // 9. FIRST DATE VALIDATION
      // =================================================

      worksheet
        .range("F2:F500")
        .dataValidation({

          type: "date",

          operator: "between",

          allowBlank: true,

          showErrorMessage: true,

          errorTitle:
            "Invalid Date",

          error:
            "Please enter a valid date in DD-MM-YYYY format.",

          formula1:
            "DATE(2000,1,1)",

          formula2:
            "DATE(2100,12,31)"

        });


      // =================================================
      // 10. LAST DATE VALIDATION
      // =================================================

      worksheet
        .range("G2:G500")
        .dataValidation({

          type: "date",

          operator: "between",

          allowBlank: true,

          showErrorMessage: true,

          errorTitle:
            "Invalid Date",

          error:
            "Please enter a valid date in DD-MM-YYYY format.",

          formula1:
            "DATE(2000,1,1)",

          formula2:
            "DATE(2100,12,31)"

        });


      // =================================================
      // 11. FREEZE HEADER
      // =================================================

      worksheet.freezePanes(0, 1);


      // =================================================
      // 12. OUTPUT
      // =================================================

      return workbook.outputAsync();

    })
    .then((buffer: any) => {

      // =================================================
      // 13. CREATE BLOB
      // =================================================

      const blob = new Blob(
        [buffer],
        {
          type:
            "application/vnd.openxmlformats-officedocument.spreadsheetml.sheet"
        }
      );


      // =================================================
      // 14. DOWNLOAD
      // =================================================

      FileSaver.saveAs(
        blob,
        "F1Meal-info Template.xlsx"
      );


      console.log(
        "F1 Meal Excel Template downloaded successfully."
      );

      this.msgSvr.success(
        "Excel template downloaded successfully."
      );

    })
    .catch((error: any) => {

      console.error(
        "===================================="
      );

      console.error(
        "Excel generation error:",
        error
      );

      console.error(
        "Error message:",
        error && error.message
      );

      console.error(
        "Error stack:",
        error && error.stack
      );

      console.error(
        "===================================="
      );

      this.msgSvr.error(
        "Unable to generate Excel template."
      );

    });

}
