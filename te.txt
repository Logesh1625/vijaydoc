DownloadExcelfile() {

  // =====================================================
  // 1. CHECK MEAL PREFERENCE MASTER DATA
  // =====================================================

  const mealValues = (this.mealPreferenceList || [])
    .map((item: any) => item.label)
    .filter((value: any) =>
      value !== null &&
      value !== undefined &&
      value !== ""
    );

  if (mealValues.length === 0) {

    this.msgSvr.warning(
      "Meal Preference data is not loaded. Please try again."
    );

    this.loadMealPreferences();

    return;
  }

  console.log(
    "Meal Preference values:",
    mealValues
  );


  // =====================================================
  // 2. CREATE WORKBOOK
  // =====================================================

  const workbook = new ExcelJS.Workbook();

  const worksheet = workbook.addWorksheet("Template");


  // =====================================================
  // 3. ADD HEADER
  // =====================================================

  worksheet.addRow([
    "Groups",
    "Sub Group",
    "Meal Preference",
    "FX Badge",
    "Name",
    "First Day at FXBL",
    "Last Day at FXBL"
  ]);


  // =====================================================
  // 4. HEADER STYLE
  // =====================================================

  const headerRow = worksheet.getRow(1);

  headerRow.height = 25;

  headerRow.eachCell((cell: any) => {

    cell.font = {
      bold: true,
      size: 11
    };

    cell.alignment = {
      horizontal: "center",
      vertical: "middle"
    };

    cell.protection = {
      locked: true
    };

  });


  // =====================================================
  // 5. COLUMN WIDTH
  // =====================================================

  worksheet.getColumn(1).width = 20; // Groups

  worksheet.getColumn(2).width = 20; // Sub Group

  worksheet.getColumn(3).width = 25; // Meal Preference

  worksheet.getColumn(4).width = 20; // FX Badge

  worksheet.getColumn(5).width = 30; // Name

  worksheet.getColumn(6).width = 22; // First Day at FXBL

  worksheet.getColumn(7).width = 22; // Last Day at FXBL


  // =====================================================
  // 6. MEAL PREFERENCE DROPDOWN
  //    C2:C500
  // =====================================================

  const mealDropdownFormula =
    '"' + mealValues.join(",") + '"';


  // =====================================================
  // 7. DATA ROWS
  // =====================================================

  for (let row = 2; row <= 500; row++) {

    // -----------------------------------------------
    // UNLOCK DATA CELLS
    // -----------------------------------------------

    for (let col = 1; col <= 7; col++) {

      worksheet.getCell(row, col).protection = {
        locked: false
      };

    }


    // -----------------------------------------------
    // MEAL PREFERENCE DROPDOWN
    // -----------------------------------------------

    const mealCell =
      worksheet.getCell(row, 3);

    mealCell.dataValidation = {

      type: "list",

      allowBlank: true,

      formulae: [
        mealDropdownFormula
      ],

      showErrorMessage: true,

      errorTitle:
        "Invalid Meal Preference",

      error:
        "Please select Meal Preference from the dropdown."

    };


    // -----------------------------------------------
    // FIRST DATE
    // -----------------------------------------------

    const firstDateCell =
      worksheet.getCell(row, 6);

    firstDateCell.numFmt =
      "dd-mm-yyyy";


    firstDateCell.dataValidation = {

      type: "date",

      operator: "between",

      allowBlank: true,

      formulae: [
        new Date(2000, 0, 1),
        new Date(2100, 11, 31)
      ],

      showErrorMessage: true,

      errorTitle:
        "Invalid Date",

      error:
        "Please enter a valid date."

    };


    // -----------------------------------------------
    // LAST DATE
    // -----------------------------------------------

    const lastDateCell =
      worksheet.getCell(row, 7);

    lastDateCell.numFmt =
      "dd-mm-yyyy";


    lastDateCell.dataValidation = {

      type: "date",

      operator: "between",

      allowBlank: true,

      formulae: [
        new Date(2000, 0, 1),
        new Date(2100, 11, 31)
      ],

      showErrorMessage: true,

      errorTitle:
        "Invalid Date",

      error:
        "Please enter a valid date."

    };

  }


  // =====================================================
  // 8. FREEZE HEADER
  // =====================================================

  worksheet.views = [
    {
      state: "frozen",
      ySplit: 1
    }
  ];


  // =====================================================
  // 9. FILTER
  // =====================================================

  worksheet.autoFilter = {
    from: "A1",
    to: "G500"
  };


  // =====================================================
  // 10. PROTECT SHEET
  // =====================================================
  // Header = LOCKED
  // Rows 2-500 = EDITABLE
  //
  // Password can be changed if required.
  // =====================================================

  worksheet.protect("F1Template2026", {

    selectLockedCells: false,

    selectUnlockedCells: true,

    formatCells: false,

    formatColumns: false,

    formatRows: false,

    insertColumns: false,

    insertRows: false,

    deleteColumns: false,

    deleteRows: false,

    sort: false,

    autoFilter: true

  }).then(() => {


    // =================================================
    // 11. GENERATE EXCEL
    // =================================================

    return workbook.xlsx.writeBuffer();

  }).then((buffer: any) => {


    // =================================================
    // 12. CREATE BLOB
    // =================================================

    const blob = new Blob(
      [buffer],
      {
        type:
          "application/vnd.openxmlformats-officedocument.spreadsheetml.sheet"
      }
    );


    // =================================================
    // 13. DOWNLOAD
    // =================================================

    FileSaver.saveAs(
      blob,
      "F1Meal-info Template.xlsx"
    );


    console.log(
      "F1 Meal Excel Template downloaded successfully."
    );

  }).catch((error: any) => {

    console.error(
      "Excel generation error:",
      error
    );

    this.msgSvr.error(
      "Unable to generate Excel template."
    );

  });

}
