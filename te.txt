  parseDate(value: any): Date | null {
    if (!value) return null;
    if (value instanceof Date) return value;

    if (typeof value === 'string') {
      const parts = value.split(/[-\/]/);
      if (parts.length === 3) {
        const day = parseInt(parts[0], 10);
        const month = parseInt(parts[1], 10) - 1;
        const year = parseInt(parts[2], 10);
        const parsed = new Date(year, month, day);
        if (!isNaN(parsed.getTime())) return parsed;
      }
    }
    return null;
  }

  add() {
    const params = {
      Groups: this.groups,
      SubGroup: this.subGroup,
      MealPreference: this.mealPreference,
      FxBadge: this.fxBadge,
      Name: this.name,
      FirstDateFxbl: this.firstDateFxbl ? this.publicSvr.dateFormat(this.firstDateFxbl, 'DD-MM-YYYY') : '',
      LastDateFxbl: this.lastDateFxbl ? this.publicSvr.dateFormat(this.lastDateFxbl, 'DD-MM-YYYY') : '',
      AddBy: this.publicSvr.getSettingUserInfo().eeid,
      Func: "CMS-InsertF1Menu"
    }

    this.publicSvr.iwxPostF1(params, {
      headers: {
        apiID: "ffff-1789631992006-10208193125-1291",
        userKey: "E9785F13CCEAFB251A6AB147AFD24336"
      }
    }, true).then((res: any) => {
      if (res.IsOK == "1") {
        this.msgSvr.success("Success");
        this.ref.close(true);
      } else {
        this.msgSvr.error(res.Msg);
      }
    })
  }

edit() {
