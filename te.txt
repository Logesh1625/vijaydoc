getGuardPassCode(
  params?: any,
  site?: any,
  isShowLabel?: any
): Observable<any> {

  if (!params || typeof params !== "string" || params.length < 10) {
    console.error("Invalid getGuardPassCode params:", params);
    return of([]);
  }

  const code = params.slice(10);

  const requestParams = {
    TypeName: code,
    Func: "CMS-GetBaseTypeInfo",
    Site: site || ""
  };

  const config = {
    headers: {
      apiID: "ffff-1710991146185-10208193125-0170",
      userKey: this.USER_TN
    }
  };

  const requestBody = this.dealPostParamsEncrypt(requestParams);

  // Always call LIVE microservice for this API.
  const liveUrl =
    "https://microservice.abgind.foxconn.com/india/gateway";

  console.log("getGuardPassCode API URL:", liveUrl);
  console.log("getGuardPassCode params:", requestParams);

  return this.http
    .post(liveUrl, requestBody, null, config)
    .pipe(
      map((resp: any) => {

        if (!resp || !resp.r) {
          console.error("Empty getGuardPassCode response:", resp);
          return [];
        }

        try {
          const result = JSON.parse(
            this.security.decrypt(resp.r, this.iwxKInfo)
          );

          console.log("Live Base Type API response:", result);

          if (
            String(result.IsOK) !== "1" ||
            !Array.isArray(result.List)
          ) {
            console.error("Live Base Type API failed:", result);
            return [];
          }

          const options: any[] = [];

          result.List.forEach((item: any) => {
            if (!item) {
              return;
            }

            const name = String(
              item.Name !== undefined
                ? item.Name
                : item.NAME !== undefined
                  ? item.NAME
                  : ""
            ).trim();

            const id =
              item.BaseCodeId !== undefined
                ? item.BaseCodeId
                : item.BASE_CODE_ID !== undefined
                  ? item.BASE_CODE_ID
                  : item.Id !== undefined
                    ? item.Id
                    : item.ID;

            if (!name || id === undefined || id === null) {
              console.warn("Invalid Meal Preference item:", item);
              return;
            }

            options.push({
              label: name,
              value: isShowLabel ? name : Number(id)
            });
          });

          console.log("LIVE MEAL PREFERENCE LIST:", options);

          return options;

        } catch (error) {
          console.error("Unable to decrypt/parse Live API response:", error);
          return [];
        }
      })
    );
}
