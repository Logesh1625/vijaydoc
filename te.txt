const verifyOTP = async () => {
  setotploading(true);
  setErrorMessage('');

  try {
    const acnt = sessionStorage.getItem('account');

    if (!acnt) {
      setErrorMessage('Session expired. Please login again.');
      return;
    }

    const decryptemp = decrypt(acnt, getdk(iwxKey));

    if (!decryptemp) {
      setErrorMessage('Unable to read employee session.');
      return;
    }

    const empnumber = decryptemp.replace(/"/g, '');

    const headers = {
      apiID: 'ffff-1743480426986-1020912129-2253',
      userKey: '10A33A67A312E649D907BBC927995B8A'
    };

    const params = {
      EmpId: empnumber,
      EmpName: '',
      Status: '',
      StartPage: 1,
      PageSize: 999999,
      Func: 'UserManagement-Get_user_details_new'
    };

    const result = await iwxPost(params, { headers }, true);

    const secretkeyget = result?.List?.[0]?.SecretKey;

    if (
      isscuessqr !== encryptedBooleannum &&
      !secretkeyget
    ) {
      setErrorMessage(
        'MFA secret is not configured for this account.'
      );
      return;
    }

    await verifyOTP22(secretkeyget);
  } catch (error) {
    console.error(
      'OTP initialization failed:',
      error.response?.status || error.message
    );

    setErrorMessage('Unable to initialize OTP verification.');
  } finally {
    setotploading(false);
  }
};


const verifyOTP22 = async (secretkeyget) => {
  setotploading(true);
  setErrorMessage('');

  try {
    const headers = {
      apiID: 'ffff-1694232123378-10208193125-0103',
      userKey: '433F1BB575A5F60E2547453C13FAFA32'
    };

    const selectedSecret =
      isscuessqr === encryptedBooleannum
        ? secretkey
        : secretkeyget;

    if (!selectedSecret || !otp.trim()) {
      setErrorMessage('Secret or OTP is missing.');
      return;
    }

    const params = dealPostParamsEncrypt({
      secret: selectedSecret,
      otp: otp.trim()
    });

    const response = await axios.post(
      'https://microservice.abgind.foxconn.com/india/gateway',
      params,
      { headers }
    );

    const data = response?.data;

    console.log('OTP HTTP status:', response.status);
    console.log(
      'OTP response fields:',
      Object.keys(data || {})
    );

    let result;

    if (typeof data?.r === 'string' && data.r.trim()) {
      const decryptedResponse = decrypt(
        data.r,
        getdk(iwxKey)
      );

      if (!decryptedResponse) {
        throw new Error('Unable to decrypt OTP API response.');
      }

      try {
        result = JSON.parse(decryptedResponse);
      } catch {
        throw new Error('OTP API response is not valid JSON.');
      }
    } else if (data && typeof data.message === 'string') {
      console.error('OTP API rejected the request.');

      setErrorMessage(data.message);
      return;
    } else {
      console.error('Unexpected OTP response structure.');

      setErrorMessage('Unexpected response from OTP service.');
      return;
    }

    if (Number(result?.IsOK) === 1) {
      await updatateqrstatua();

      sessionStorage.setItem(
        'isLoggedIn',
        encryptedBoolean
      );

      const loginResponse = sessionStorage.getItem(
        'loginresponse'
      );

      sessionStorage.setItem('data', loginResponse || '');

      await StoreOTPtologinlog(
        result.Msg || 'OTP verification successful'
      );

      navigate('/home/Dashboard', {
        state: { account }
      });
    } else {
      const message =
        result?.Msg || 'OTP verification failed.';

      setErrorMessage(message);

      await StoreOTPtologinlog(message);
    }
  } catch (error) {
    console.error(
      'OTP verification failed:',
      error.response?.status || error.message
    );

    setErrorMessage(
      'Unable to verify OTP. Check the OTP API configuration.'
    );
  } finally {
    setotploading(false);
  }
};
