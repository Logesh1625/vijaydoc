// ======================================================
// LOGIN USER DETAILS
// ======================================================

async function loginUserDetails(req, res) {

  let connection;

  try {

    const { EmpId } = req.body;

    console.log(
      'LOGIN USER DETAILS REQUEST:',
      req.body
    );


    // ----------------------------------------------
    // Validate EmpId
    // ----------------------------------------------

    if (!EmpId) {

      return res.status(400).json({
        IsOK: '0',
        Msg: 'EmpId is required',
        List: []
      });

    }


    // ----------------------------------------------
    // Oracle connection
    // ----------------------------------------------

    connection =
      await oracledb.getConnection(
        dbConfig
      );


    // ----------------------------------------------
    // Get user
    // ----------------------------------------------

    const query = `

      SELECT

        ID,

        EMP_ID,

        EMP_NAME,

        DEPARTMENT,

        USER_STATUS,

        QR_STATUS,

        SECRET_KEY,

        DEL_FLAG,

        CREATE_BY,

        CREATED_DATE,

        LAST_EDIT_BY,

        LAST_EDIT_DATE

      FROM DMS.C_NEW_USER_MANAGEMENT_T

      WHERE UPPER(TRIM(EMP_ID))
            =
            UPPER(TRIM(:EmpId))

    `;


    const result =
      await connection.execute(
        query,
        {
          EmpId: EmpId
        },
        {
          outFormat:
            oracledb.OUT_FORMAT_OBJECT
        }
      );


    console.log(
      'LOGIN USER DB RESULT:',
      result.rows
    );


    const rows =
      result.rows || [];


    // ----------------------------------------------
    // Convert DB response
    // ----------------------------------------------

    const List =
      rows.map((row) => {

        return {

          Id:
            row.ID,

          EmpId:
            row.EMP_ID,

          EmpName:
            row.EMP_NAME,

          Department:
            row.DEPARTMENT,

          UserStatus:
            row.USER_STATUS,

          QrStatus:
            row.QR_STATUS,

          SecretKey:
            row.SECRET_KEY,

          DelFlag:
            row.DEL_FLAG,

          CreateBy:
            row.CREATE_BY,

          CreatedDate:
            row.CREATED_DATE,

          LastEditBy:
            row.LAST_EDIT_BY,

          LastEditDate:
            row.LAST_EDIT_DATE

        };

      });


    // ----------------------------------------------
    // No user
    // ----------------------------------------------

    if (List.length === 0) {

      return res.status(200).json({

        IsOK: '0',

        Msg:
          'User not found',

        List: []

      });

    }


    // ----------------------------------------------
    // Success
    // ----------------------------------------------

    return res.status(200).json({

      IsOK: '1',

      Msg:
        'Success',

      List:
        List

    });


  } catch (error) {

    console.error(
      'loginUserDetails ERROR:',
      error
    );


    return res.status(500).json({

      IsOK: '0',

      Msg:
        'Internal server error',

      List: []

    });


  } finally {

    // ----------------------------------------------
    // Close connection
    // ----------------------------------------------

    if (connection) {

      try {

        await connection.close();

      } catch (closeError) {

        console.error(
          'DB CLOSE ERROR:',
          closeError
        );

      }

    }

  }

}
