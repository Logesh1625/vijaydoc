CREATE OR REPLACE PACKAGE BODY CMS.F1_MEAL_MANAGEMENT
AS

    /* ============================================================
       PRIVATE: Active CMS user check
       ============================================================ */
    FUNCTION IS_ACTIVE_CMS_USER(
        P_EMP_ID IN VARCHAR2
    ) RETURN NUMBER
    IS
        L_COUNT NUMBER;
    BEGIN
        SELECT COUNT(1)
        INTO L_COUNT
        FROM CMS.C_USER_MANAGE_T U
        WHERE UPPER(TRIM(U.EMP_ID)) = UPPER(TRIM(P_EMP_ID))
          AND UPPER(TRIM(U.SYSTEM_TYPE)) = 'CMS'
          AND NVL(U.DEL_FLAG, 0) = 0;

        RETURN CASE WHEN L_COUNT > 0 THEN 1 ELSE 0 END;

    EXCEPTION
        WHEN OTHERS THEN
            RETURN 0;
    END IS_ACTIVE_CMS_USER;


    /* ============================================================
       PRIVATE: Check site access
       ============================================================ */
    FUNCTION HAS_SITE_ACCESS(
        P_EMP_ID IN VARCHAR2,
        P_SITE   IN VARCHAR2
    ) RETURN NUMBER
    IS
        L_COUNT NUMBER;
    BEGIN
        SELECT COUNT(1)
        INTO L_COUNT
        FROM CMS.C_USER_MANAGE_T U
        WHERE UPPER(TRIM(U.EMP_ID)) = UPPER(TRIM(P_EMP_ID))
          AND UPPER(TRIM(U.SYSTEM_TYPE)) = 'CMS'
          AND NVL(U.DEL_FLAG, 0) = 0
          AND UPPER(TRIM(U.SITE)) = UPPER(TRIM(P_SITE));

        RETURN CASE WHEN L_COUNT > 0 THEN 1 ELSE 0 END;

    EXCEPTION
        WHEN OTHERS THEN
            RETURN 0;
    END HAS_SITE_ACCESS;


    /* ============================================================
       PRIVATE: Get user's site
       ============================================================ */
    FUNCTION GET_USER_SITE(
        P_EMP_ID IN VARCHAR2
    ) RETURN CMS.C_F1_MEAL_INFO.PROPERTY_01%TYPE
    IS
        L_SITE CMS.C_F1_MEAL_INFO.PROPERTY_01%TYPE;
    BEGIN
        SELECT MIN(U.SITE)
        INTO L_SITE
        FROM CMS.C_USER_MANAGE_T U
        WHERE UPPER(TRIM(U.EMP_ID)) = UPPER(TRIM(P_EMP_ID))
          AND UPPER(TRIM(U.SYSTEM_TYPE)) = 'CMS'
          AND NVL(U.DEL_FLAG, 0) = 0;

        RETURN L_SITE;

    EXCEPTION
        WHEN OTHERS THEN
            RETURN NULL;
    END GET_USER_SITE;


    /* ============================================================
       PUBLIC: Site code -> site name
       ============================================================ */
    FUNCTION GET_SITE_NAME(
        P_SITE IN VARCHAR2
    ) RETURN VARCHAR2
    IS
    BEGIN
        RETURN CASE TRIM(P_SITE)
                   WHEN '95326' THEN 'FXBL'
                   WHEN '69941' THEN 'FXCN'
                   ELSE P_SITE
               END;
    END GET_SITE_NAME;


    /* ============================================================
       INSERT F1 MENU
       ============================================================ */
    PROCEDURE INSERT_F1_MENU(
        V_GROUPS            IN VARCHAR2,
        V_MEAL_PREFERENCE   IN VARCHAR2,
        V_FX_BADGE          IN VARCHAR2,
        V_NAME              IN VARCHAR2,
        V_FIRST_DATE_FXBL   IN VARCHAR2,
        V_LAST_DATE_FXBL    IN VARCHAR2,
        V_ADD_BY            IN VARCHAR2,
        V_RES               OUT VARCHAR2,
        V_RESULT            OUT VARCHAR2
    )
    IS
        V_SITE_VAL CMS.C_F1_MEAL_INFO.PROPERTY_01%TYPE;
        V_FIRST_DATE DATE;
        V_LAST_DATE  DATE;
    BEGIN

        IF IS_ACTIVE_CMS_USER(V_ADD_BY) = 0 THEN
            V_RES := '0';
            V_RESULT := 'Active CMS user not found for employee: '
                        || V_ADD_BY;
            RETURN;
        END IF;

        V_SITE_VAL := GET_USER_SITE(V_ADD_BY);

        IF V_SITE_VAL IS NULL THEN
            V_RES := '0';
            V_RESULT := 'Active CMS user/site not found for employee: '
                        || V_ADD_BY;
            RETURN;
        END IF;

        BEGIN
            V_FIRST_DATE := TO_DATE(
                V_FIRST_DATE_FXBL,
                'DD-MM-YYYY'
            );

            V_LAST_DATE := TO_DATE(
                V_LAST_DATE_FXBL,
                'DD-MM-YYYY'
            );

        EXCEPTION
            WHEN OTHERS THEN
                V_RES := '0';
                V_RESULT := 'Invalid date. Expected format DD-MM-YYYY';
                RETURN;
        END;

        IF V_FIRST_DATE > V_LAST_DATE THEN
            V_RES := '0';
            V_RESULT := 'First date cannot be greater than last date';
            RETURN;
        END IF;

        INSERT INTO CMS.C_F1_MEAL_INFO (
            ID,
            GROUPS,
            PROPERTY_01,
            MEAL_PREFERENCE,
            FX_BADGE,
            NAME,
            FIRST_DATE_FXBL,
            LAST_DATE_FXBL,
            ADD_BY,
            ADD_DATE,
            DEL_FLAG
        )
        VALUES (
            CMS.SEQ_C_F1_MEAL_INFO.NEXTVAL,
            V_GROUPS,
            V_SITE_VAL,
            V_MEAL_PREFERENCE,
            V_FX_BADGE,
            V_NAME,
            V_FIRST_DATE,
            V_LAST_DATE,
            V_ADD_BY,
            SYSDATE,
            0
        );

        COMMIT;

        V_RES := '1';
        V_RESULT := 'Success';

    EXCEPTION
        WHEN OTHERS THEN
            ROLLBACK;
            V_RES := '0';
            V_RESULT := 'Error: ' || SQLERRM;

    END INSERT_F1_MENU;


    /* ============================================================
       BULK F1 MENU
       ============================================================ */
    PROCEDURE BULK_F1_MENU(
        V_GROUPS            IN VARCHAR2,
        V_MEAL_PREFERENCE   IN VARCHAR2,
        V_FX_BADGE          IN VARCHAR2,
        V_NAME              IN VARCHAR2,
        V_FIRST_DATE_FXBL   IN VARCHAR2,
        V_LAST_DATE_FXBL    IN VARCHAR2,
        V_ADD_BY            IN VARCHAR2,
        V_RES               OUT VARCHAR2,
        V_RESULT            OUT VARCHAR2
    )
    IS
    BEGIN

        INSERT_F1_MENU(
            V_GROUPS            => V_GROUPS,
            V_MEAL_PREFERENCE   => V_MEAL_PREFERENCE,
            V_FX_BADGE          => V_FX_BADGE,
            V_NAME              => V_NAME,
            V_FIRST_DATE_FXBL   => V_FIRST_DATE_FXBL,
            V_LAST_DATE_FXBL    => V_LAST_DATE_FXBL,
            V_ADD_BY            => V_ADD_BY,
            V_RES               => V_RES,
            V_RESULT            => V_RESULT
        );

    EXCEPTION
        WHEN OTHERS THEN
            ROLLBACK;
            V_RES := '0';
            V_RESULT := 'Error: ' || SQLERRM;

    END BULK_F1_MENU;


    /* ============================================================
       GET F1 MENU MANAGEMENT
       Supports TYPE_ID 81 and TYPE_ID 1001.
       ============================================================ */
    PROCEDURE GET_F1_MENU_MANAGE(
        V_EMP_ID        IN VARCHAR2,
        V_GROUPS        IN VARCHAR2,
        V_SITE          IN VARCHAR2,
        V_FX_BADGE      IN VARCHAR2,
        V_ADD_BY        IN VARCHAR2,
        L_START_PAGE    IN VARCHAR2,
        L_PAGE_SIZE     IN VARCHAR2,
        V_RES           OUT VARCHAR2,
        V_RESULT        OUT VARCHAR2,
        P_CURSOR        OUT SYS_REFCURSOR
    )
    IS
        L_START_PAGE_NUM NUMBER;
        L_PAGE_SIZE_NUM  NUMBER;
    BEGIN

        IF IS_ACTIVE_CMS_USER(V_EMP_ID) = 0 THEN
            V_RES := '0';
            V_RESULT := 'Active CMS user not found for employee: '
                        || V_EMP_ID;
            RETURN;
        END IF;

        BEGIN
            L_START_PAGE_NUM :=
                TO_NUMBER(
                    NVL(NULLIF(TRIM(L_START_PAGE), ''), '1')
                );

            L_PAGE_SIZE_NUM :=
                TO_NUMBER(
                    NVL(NULLIF(TRIM(L_PAGE_SIZE), ''), '10')
                );

            IF L_START_PAGE_NUM < 1 THEN
                L_START_PAGE_NUM := 1;
            END IF;

            IF L_PAGE_SIZE_NUM < 1 THEN
                L_PAGE_SIZE_NUM := 10;
            END IF;

        EXCEPTION
            WHEN VALUE_ERROR THEN
                V_RES := '0';
                V_RESULT := 'Invalid pagination values';
                RETURN;
        END;

        OPEN P_CURSOR FOR
        SELECT
            RN,
            ROWSCOUNT,
            ID,
            GROUPS,
            SITE,
            SITE_NAME,
            MEAL_PREFERENCE,
            FX_BADGE,
            NAME,
            FIRST_DATE_FXBL,
            LAST_DATE_FXBL,
            ADD_BY,
            ADD_DATE,
            EDIT_BY,
            EDIT_DATE
        FROM
        (
            SELECT
                ROW_NUMBER() OVER (
                    ORDER BY V.ADD_DATE DESC, V.ID DESC
                ) AS RN,

                COUNT(1) OVER () AS ROWSCOUNT,

                V.ID,
                V.GROUPS,
                V.PROPERTY_01 AS SITE,
                GET_SITE_NAME(V.PROPERTY_01) AS SITE_NAME,

                BC.NAME AS MEAL_PREFERENCE,

                V.FX_BADGE,
                V.NAME,

                TO_CHAR(
                    V.FIRST_DATE_FXBL,
                    'DD-MM-YYYY'
                ) AS FIRST_DATE_FXBL,

                TO_CHAR(
                    V.LAST_DATE_FXBL,
                    'DD-MM-YYYY'
                ) AS LAST_DATE_FXBL,

                V.ADD_BY,

                TO_CHAR(
                    V.ADD_DATE,
                    'DD-MM-YYYY HH24:MI:SS'
                ) AS ADD_DATE,

                V.EDIT_BY,

                TO_CHAR(
                    V.EDIT_DATE,
                    'DD-MM-YYYY HH24:MI:SS'
                ) AS EDIT_DATE

            FROM CMS.C_F1_MEAL_INFO V

            LEFT JOIN CMS.C_BASE_CODE_T BC
                ON TO_CHAR(V.MEAL_PREFERENCE) = TO_CHAR(BC.ID)
               AND BC.TYPE_ID IN (81, 1001)

            WHERE NVL(V.DEL_FLAG, 0) = 0

              AND (
                    V_GROUPS IS NULL
                    OR TRIM(V_GROUPS) IS NULL
                    OR UPPER(V.GROUPS)
                       LIKE '%' || UPPER(TRIM(V_GROUPS)) || '%'
                  )

              AND (
                    V_SITE IS NULL
                    OR TRIM(V_SITE) IS NULL
                    OR UPPER(TRIM(V.PROPERTY_01))
                       = UPPER(TRIM(V_SITE))
                  )

              AND (
                    V_FX_BADGE IS NULL
                    OR TRIM(V_FX_BADGE) IS NULL
                    OR UPPER(V.FX_BADGE)
                       LIKE '%' || UPPER(TRIM(V_FX_BADGE)) || '%'
                  )

              AND (
                    V_ADD_BY IS NULL
                    OR TRIM(V_ADD_BY) IS NULL
                    OR UPPER(TRIM(V.ADD_BY))
                       = UPPER(TRIM(V_ADD_BY))
                  )

              AND EXISTS (
                    SELECT 1
                    FROM CMS.C_USER_MANAGE_T U
                    WHERE UPPER(TRIM(U.EMP_ID))
                          = UPPER(TRIM(V_EMP_ID))
                      AND UPPER(TRIM(U.SYSTEM_TYPE)) = 'CMS'
                      AND NVL(U.DEL_FLAG, 0) = 0
                      AND UPPER(TRIM(U.SITE))
                          = UPPER(TRIM(V.PROPERTY_01))
                  )
        )
        WHERE RN > (L_START_PAGE_NUM - 1) * L_PAGE_SIZE_NUM
          AND RN <= L_START_PAGE_NUM * L_PAGE_SIZE_NUM;

        V_RES := '1';
        V_RESULT := 'Success';

    EXCEPTION
        WHEN OTHERS THEN
            V_RES := '0';
            V_RESULT := 'Error: ' || SQLERRM;

    END GET_F1_MENU_MANAGE;


    /* ============================================================
       UPDATE F1 MENU MANAGEMENT
       ============================================================ */
    PROCEDURE UPDATE_F1_MENU_MANAGE(
        V_ID                IN NUMBER,
        V_GROUPS            IN VARCHAR2,
        V_MEAL_PREFERENCE   IN VARCHAR2,
        V_FX_BADGE          IN VARCHAR2,
        V_NAME              IN VARCHAR2,
        V_FIRST_DATE_FXBL   IN VARCHAR2,
        V_LAST_DATE_FXBL    IN VARCHAR2,
        V_EDIT_BY           IN VARCHAR2,
        V_RES               OUT VARCHAR2,
        V_RESULT            OUT VARCHAR2
    )
    IS
        V_FIRST_DATE DATE;
        V_LAST_DATE  DATE;
    BEGIN

        BEGIN
            V_FIRST_DATE :=
                TO_DATE(V_FIRST_DATE_FXBL, 'DD-MM-YYYY');

            V_LAST_DATE :=
                TO_DATE(V_LAST_DATE_FXBL, 'DD-MM-YYYY');

        EXCEPTION
            WHEN OTHERS THEN
                V_RES := '0';
                V_RESULT := 'Invalid date. Expected format DD-MM-YYYY';
                RETURN;
        END;

        IF V_FIRST_DATE > V_LAST_DATE THEN
            V_RES := '0';
            V_RESULT := 'First date cannot be greater than last date';
            RETURN;
        END IF;

        UPDATE CMS.C_F1_MEAL_INFO V
        SET
            V.GROUPS = V_GROUPS,
            V.MEAL_PREFERENCE = V_MEAL_PREFERENCE,
            V.FX_BADGE = V_FX_BADGE,
            V.NAME = V_NAME,
            V.FIRST_DATE_FXBL = V_FIRST_DATE,
            V.LAST_DATE_FXBL = V_LAST_DATE,
            V.EDIT_BY = V_EDIT_BY,
            V.EDIT_DATE = SYSDATE

        WHERE V.ID = V_ID
          AND NVL(V.DEL_FLAG, 0) = 0

          AND EXISTS (
                SELECT 1
                FROM CMS.C_USER_MANAGE_T U
                WHERE UPPER(TRIM(U.EMP_ID))
                      = UPPER(TRIM(V_EDIT_BY))
                  AND UPPER(TRIM(U.SYSTEM_TYPE)) = 'CMS'
                  AND NVL(U.DEL_FLAG, 0) = 0
                  AND UPPER(TRIM(U.SITE))
                      = UPPER(TRIM(V.PROPERTY_01))
              );

        IF SQL%ROWCOUNT = 0 THEN
            ROLLBACK;
            V_RES := '0';
            V_RESULT :=
                'No authorized active record found with ID = '
                || V_ID;
            RETURN;
        END IF;

        COMMIT;

        V_RES := '1';
        V_RESULT := 'Success';

    EXCEPTION
        WHEN OTHERS THEN
            ROLLBACK;
            V_RES := '0';
            V_RESULT := 'Error: ' || SQLERRM;

    END UPDATE_F1_MENU_MANAGE;


    /* ============================================================
       DELETE F1 MENU MANAGEMENT
       ============================================================ */
    PROCEDURE DELETE_F1_MENU_MANAGE(
        L_IDS       IN VARCHAR2,
        V_EDIT_BY   IN VARCHAR2,
        V_RES       OUT VARCHAR2,
        V_RESULT    OUT VARCHAR2
    )
    IS
    BEGIN

        IF L_IDS IS NULL OR TRIM(L_IDS) IS NULL THEN
            V_RES := '0';
            V_RESULT := 'No ID selected for delete';
            RETURN;
        END IF;

        UPDATE CMS.C_F1_MEAL_INFO V
        SET
            V.DEL_FLAG = 1,
            V.EDIT_DATE = SYSDATE,
            V.EDIT_BY = V_EDIT_BY

        WHERE INSTR(
                  ',' || REPLACE(L_IDS, ' ', '') || ',',
                  ',' || TO_CHAR(V.ID) || ','
              ) > 0

          AND NVL(V.DEL_FLAG, 0) = 0

          AND EXISTS (
                SELECT 1
                FROM CMS.C_USER_MANAGE_T U
                WHERE UPPER(TRIM(U.EMP_ID))
                      = UPPER(TRIM(V_EDIT_BY))
                  AND UPPER(TRIM(U.SYSTEM_TYPE)) = 'CMS'
                  AND NVL(U.DEL_FLAG, 0) = 0
                  AND UPPER(TRIM(U.SITE))
                      = UPPER(TRIM(V.PROPERTY_01))
              );

        IF SQL%ROWCOUNT = 0 THEN
            ROLLBACK;
            V_RES := '0';
            V_RESULT :=
                'No authorized active record found for selected ID(s)';
            RETURN;
        END IF;

        COMMIT;

        V_RES := '1';
        V_RESULT := 'Success';

    EXCEPTION
        WHEN OTHERS THEN
            ROLLBACK;
            V_RES := '0';
            V_RESULT := 'Error: ' || SQLERRM;

    END DELETE_F1_MENU_MANAGE;


    /* ============================================================
       GET F1 MENU BY FX BADGE
       Supports TYPE_ID 81 and TYPE_ID 1001.
       ============================================================ */
    PROCEDURE GET_F1_MENU_BY_FXBADGE(
        V_EMP_ID    IN VARCHAR2,
        V_SITE      IN VARCHAR2,
        V_FX_BADGE  IN VARCHAR2,
        V_RES       OUT VARCHAR2,
        V_RESULT    OUT VARCHAR2,
        P_CURSOR    OUT SYS_REFCURSOR
    )
    IS
        V_COUNT NUMBER;
        V_ALREADY NUMBER;
        V_RECORD_SITE CMS.C_F1_MEAL_INFO.PROPERTY_01%TYPE;
    BEGIN

        IF IS_ACTIVE_CMS_USER(V_EMP_ID) = 0 THEN
            V_RES := '0';
            V_RESULT := 'Active CMS user not found for employee: '
                        || V_EMP_ID;

            OPEN P_CURSOR FOR
                SELECT CAST(NULL AS VARCHAR2(100)) AS FX_BADGE
                FROM DUAL
                WHERE 1 = 0;

            RETURN;
        END IF;

        IF V_SITE IS NOT NULL AND TRIM(V_SITE) IS NOT NULL THEN
            IF HAS_SITE_ACCESS(V_EMP_ID, V_SITE) = 0 THEN
                V_RES := '0';
                V_RESULT :=
                    'You are not authorized for the selected site.';

                OPEN P_CURSOR FOR
                    SELECT CAST(NULL AS VARCHAR2(100)) AS FX_BADGE
                    FROM DUAL
                    WHERE 1 = 0;

                RETURN;
            END IF;
        END IF;

        SELECT COUNT(1)
        INTO V_COUNT
        FROM CMS.C_F1_MEAL_INFO
        WHERE UPPER(TRIM(FX_BADGE)) = UPPER(TRIM(V_FX_BADGE))
          AND NVL(DEL_FLAG, 0) = 0;

        IF V_COUNT = 0 THEN
            V_RES := '0';
            V_RESULT := 'No record found for this FX Badge.';

            OPEN P_CURSOR FOR
                SELECT CAST(NULL AS VARCHAR2(100)) AS FX_BADGE
                FROM DUAL
                WHERE 1 = 0;

            RETURN;
        END IF;

        SELECT PROPERTY_01
        INTO V_RECORD_SITE
        FROM (
            SELECT
                M.*,
                ROW_NUMBER() OVER (
                    ORDER BY M.ADD_DATE DESC, M.ID DESC
                ) AS RN
            FROM CMS.C_F1_MEAL_INFO M
            WHERE UPPER(TRIM(M.FX_BADGE))
                  = UPPER(TRIM(V_FX_BADGE))
              AND NVL(M.DEL_FLAG, 0) = 0
        )
        WHERE RN = 1;

        IF HAS_SITE_ACCESS(V_EMP_ID, V_RECORD_SITE) = 0 THEN
            V_RES := '0';
            V_RESULT :=
                'This FX Badge does not belong to your assigned site.';

            OPEN P_CURSOR FOR
                SELECT CAST(NULL AS VARCHAR2(100)) AS FX_BADGE
                FROM DUAL
                WHERE 1 = 0;

            RETURN;
        END IF;

        SELECT COUNT(1)
        INTO V_ALREADY
        FROM CMS.C_F1_SCAN_LOG
        WHERE UPPER(TRIM(FX_BADGE)) = UPPER(TRIM(V_FX_BADGE))
          AND TRUNC(MEAL_DATE) = TRUNC(SYSDATE)
          AND NVL(DEL_FLAG, 0) = 0
          AND COLLECTED_DATA = 'SUCCESS';

        OPEN P_CURSOR FOR
        SELECT
            V.GROUPS,
            V.SUB_GROUP,
            V.PROPERTY_01 AS SITE,
            GET_SITE_NAME(V.PROPERTY_01) AS SITE_NAME,
            BC.NAME AS MEAL_PREFERENCE,
            V.FX_BADGE,
            V.NAME,

            TO_CHAR(
                V.FIRST_DATE_FXBL,
                'DD-MM-YYYY'
            ) AS FIRST_DATE_FXBL,

            TO_CHAR(
                V.LAST_DATE_FXBL,
                'DD-MM-YYYY'
            ) AS LAST_DATE_FXBL,

            CASE
                WHEN UPPER(TRIM(BC.NAME)) = 'NO MEAL REQUIRED'
                    THEN 'NOT_ENTITLED'

                WHEN V.FIRST_DATE_FXBL IS NULL
                  OR V.LAST_DATE_FXBL IS NULL
                    THEN 'NOT_UPDATED'

                WHEN TRUNC(SYSDATE) NOT BETWEEN
                     TRUNC(V.FIRST_DATE_FXBL)
                     AND TRUNC(V.LAST_DATE_FXBL)
                    THEN 'LAST DATE EXPIRED'

                WHEN V_ALREADY > 0
                    THEN 'ALREADY_SCANNED'

                ELSE 'ENTITLED'
            END AS ENTITLEMENT_STATUS

        FROM (
            SELECT
                M.*,
                ROW_NUMBER() OVER (
                    ORDER BY M.ADD_DATE DESC, M.ID DESC
                ) AS RN
            FROM CMS.C_F1_MEAL_INFO M
            WHERE UPPER(TRIM(M.FX_BADGE))
                  = UPPER(TRIM(V_FX_BADGE))
              AND NVL(M.DEL_FLAG, 0) = 0
        ) V

        LEFT JOIN CMS.C_BASE_CODE_T BC
            ON TO_CHAR(V.MEAL_PREFERENCE) = TO_CHAR(BC.ID)
           AND BC.TYPE_ID IN (81, 1001)

        WHERE V.RN = 1;

        V_RES := '1';
        V_RESULT := 'Success';

    EXCEPTION
        WHEN OTHERS THEN
            V_RES := '0';
            V_RESULT := 'Error: ' || SQLERRM;

    END GET_F1_MENU_BY_FXBADGE;


    /* ============================================================
       INSERT MEAL SCAN LOG
       Supports TYPE_ID 81 and TYPE_ID 1001.
       ============================================================ */
    PROCEDURE INSERT_MEAL_SCAN_LOG(
        V_EMP_ID    IN VARCHAR2,
        V_SITE      IN VARCHAR2,
        V_FX_BADGE  IN VARCHAR2,
        V_RES       OUT VARCHAR2,
        V_RESULT    OUT VARCHAR2,
        P_CURSOR    OUT SYS_REFCURSOR
    )
    IS
        V_COUNT NUMBER;
        V_ALREADY_COUNT NUMBER;

        V_GROUPS         CMS.C_F1_MEAL_INFO.GROUPS%TYPE;
        V_SITE_VAL       CMS.C_F1_MEAL_INFO.PROPERTY_01%TYPE;
        V_NAME           CMS.C_F1_MEAL_INFO.NAME%TYPE;
        V_MEAL_PREF      CMS.C_F1_MEAL_INFO.MEAL_PREFERENCE%TYPE;
        V_MEAL_PREF_NAME CMS.C_BASE_CODE_T.NAME%TYPE;
        V_FIRST_DATE     CMS.C_F1_MEAL_INFO.FIRST_DATE_FXBL%TYPE;
        V_LAST_DATE      CMS.C_F1_MEAL_INFO.LAST_DATE_FXBL%TYPE;

        V_STATUS  VARCHAR2(50);
        V_MESSAGE VARCHAR2(200);

    BEGIN

        IF IS_ACTIVE_CMS_USER(V_EMP_ID) = 0 THEN
            V_RES := '0';
            V_RESULT := 'Active CMS user not found for employee: '
                        || V_EMP_ID;
            RETURN;
        END IF;

        IF V_SITE IS NOT NULL AND TRIM(V_SITE) IS NOT NULL THEN
            IF HAS_SITE_ACCESS(V_EMP_ID, V_SITE) = 0 THEN
                V_RES := '0';
                V_RESULT :=
                    'You are not authorized for the selected site.';
                RETURN;
            END IF;
        END IF;

        BEGIN
            SELECT
                GROUPS,
                PROPERTY_01,
                NAME,
                MEAL_PREFERENCE,
                FIRST_DATE_FXBL,
                LAST_DATE_FXBL

            INTO
                V_GROUPS,
                V_SITE_VAL,
                V_NAME,
                V_MEAL_PREF,
                V_FIRST_DATE,
                V_LAST_DATE

            FROM (
                SELECT
                    M.*,
                    ROW_NUMBER() OVER (
                        ORDER BY M.ADD_DATE DESC, M.ID DESC
                    ) AS RN
                FROM CMS.C_F1_MEAL_INFO M
                WHERE UPPER(TRIM(M.FX_BADGE))
                      = UPPER(TRIM(V_FX_BADGE))
                  AND NVL(M.DEL_FLAG, 0) = 0
            )
            WHERE RN = 1;

            V_COUNT := 1;

        EXCEPTION
            WHEN NO_DATA_FOUND THEN
                V_COUNT := 0;
        END;

        IF V_COUNT = 0 THEN

            V_STATUS := 'INVALID';
            V_MESSAGE := 'FX Badge not found';

            V_GROUPS := NULL;
            V_SITE_VAL := NULL;
            V_NAME := NULL;
            V_MEAL_PREF := NULL;
            V_MEAL_PREF_NAME := NULL;
            V_FIRST_DATE := NULL;
            V_LAST_DATE := NULL;

        ELSE

            IF HAS_SITE_ACCESS(V_EMP_ID, V_SITE_VAL) = 0 THEN
                V_RES := '0';
                V_RESULT :=
                    'This FX Badge does not belong to your assigned site.';
                RETURN;
            END IF;

            SELECT NVL(
                       MAX(BC.NAME),
                       TO_CHAR(V_MEAL_PREF)
                   )
            INTO V_MEAL_PREF_NAME
            FROM CMS.C_BASE_CODE_T BC
            WHERE BC.TYPE_ID IN (81, 1001)
              AND TO_CHAR(BC.ID) = TO_CHAR(V_MEAL_PREF);

            IF UPPER(TRIM(V_MEAL_PREF_NAME))
               = 'NO MEAL REQUIRED' THEN

                V_STATUS := 'INVALID';
                V_MESSAGE :=
                    'NO MEAL REQUIRED BUT COME TO COLLECT';

            ELSIF V_FIRST_DATE IS NULL
               OR V_LAST_DATE IS NULL THEN

                V_STATUS := 'INVALID';
                V_MESSAGE := 'Dates not updated';

            ELSIF TRUNC(SYSDATE) NOT BETWEEN
                  TRUNC(V_FIRST_DATE)
                  AND TRUNC(V_LAST_DATE) THEN

                V_STATUS := 'INVALID';
                V_MESSAGE := 'LAST DATE EXPIRED';

            ELSE

                V_STATUS := 'SUCCESS';
                V_MESSAGE := 'Eligible for meal';

            END IF;

            IF V_STATUS = 'SUCCESS' THEN

                SELECT COUNT(1)
                INTO V_ALREADY_COUNT
                FROM CMS.C_F1_SCAN_LOG
                WHERE UPPER(TRIM(FX_BADGE))
                      = UPPER(TRIM(V_FX_BADGE))
                  AND TRUNC(MEAL_DATE) = TRUNC(SYSDATE)
                  AND NVL(DEL_FLAG, 0) = 0
                  AND COLLECTED_DATA = 'SUCCESS';

                IF V_ALREADY_COUNT > 0 THEN
                    V_STATUS := 'ALREADY_SCANNED';
                    V_MESSAGE := 'Meal already collected today';
                END IF;

            END IF;

        END IF;

        INSERT INTO CMS.C_F1_SCAN_LOG (
            ID,
            GROUPS,
            PROPERTY_01,
            MEAL_PREFERENCE,
            FX_BADGE,
            NAME,
            FIRST_DATE_FXBL,
            LAST_DATE_FXBL,
            COLLECTED_DATA,
            STATUS_MESSAGE,
            MEAL_DATE,
            FIRST_SCANTIME,
            LAST_SCANTIME,
            DEL_FLAG
        )
        VALUES (
            CMS.SEQ_C_F1_SCAN_LOG.NEXTVAL,
            V_GROUPS,
            V_SITE_VAL,
            V_MEAL_PREF,
            V_FX_BADGE,
            V_NAME,
            V_FIRST_DATE,
            V_LAST_DATE,
            V_STATUS,
            V_MESSAGE,
            SYSDATE,
            SYSDATE,
            SYSDATE,
            0
        );

        COMMIT;

        OPEN P_CURSOR FOR
        SELECT
            V_FX_BADGE AS FX_BADGE,
            V_NAME AS NAME,
            V_SITE_VAL AS SITE,
            GET_SITE_NAME(V_SITE_VAL) AS SITE_NAME,
            V_MEAL_PREF_NAME AS MEAL_PREFERENCE,
            TO_CHAR(
                SYSDATE,
                'DD-MM-YYYY HH24:MI:SS'
            ) AS SCAN_TIME,
            V_STATUS AS STATUS,
            V_MESSAGE AS STATUS_MESSAGE
        FROM DUAL;

        V_RES := '1';
        V_RESULT := 'Success';

    EXCEPTION
        WHEN DUP_VAL_ON_INDEX THEN

            ROLLBACK;

            V_RES := '1';
            V_RESULT := 'Already scanned today';

            OPEN P_CURSOR FOR
            SELECT
                V_FX_BADGE AS FX_BADGE,
                V_NAME AS NAME,
                V_SITE_VAL AS SITE,
                GET_SITE_NAME(V_SITE_VAL) AS SITE_NAME,
                V_MEAL_PREF_NAME AS MEAL_PREFERENCE,
                TO_CHAR(
                    SYSDATE,
                    'DD-MM-YYYY HH24:MI:SS'
                ) AS SCAN_TIME,
                'ALREADY_SCANNED' AS STATUS,
                'Meal already collected today' AS STATUS_MESSAGE
            FROM DUAL;

        WHEN OTHERS THEN
            ROLLBACK;
            V_RES := '0';
            V_RESULT := 'Error: ' || SQLERRM;

    END INSERT_MEAL_SCAN_LOG;


    /* ============================================================
       GET MEAL SCAN LOG
       ============================================================ */
    PROCEDURE GET_MEAL_SCAN_LOG(
        V_EMP_ID            IN VARCHAR2,
        V_SEARCH_TEXT       IN VARCHAR2,
        V_STATUS_FILTER     IN VARCHAR2,
        V_SITE              IN VARCHAR2,
        V_RES               OUT VARCHAR2,
        V_RESULT            OUT VARCHAR2,
        P_CURSOR            OUT SYS_REFCURSOR,
        P_SUMMARY_CURSOR    OUT SYS_REFCURSOR
    )
    IS
    BEGIN

        IF IS_ACTIVE_CMS_USER(V_EMP_ID) = 0 THEN
            V_RES := '0';
            V_RESULT := 'Active CMS user not found for employee: '
                        || V_EMP_ID;

            OPEN P_CURSOR FOR
                SELECT CAST(NULL AS VARCHAR2(100)) AS FX_BADGE
                FROM DUAL
                WHERE 1 = 0;

            OPEN P_SUMMARY_CURSOR FOR
                SELECT
                    0 AS TOTAL_SCANS,
                    0 AS SUCCESS_COUNT,
                    0 AS ALREADY_SCANNED_COUNT,
                    0 AS INVALID_COUNT
                FROM DUAL
                WHERE 1 = 0;

            RETURN;
        END IF;

        IF V_SITE IS NOT NULL
           AND TRIM(V_SITE) IS NOT NULL
           AND UPPER(TRIM(V_SITE)) <> 'ALL' THEN

            IF HAS_SITE_ACCESS(V_EMP_ID, V_SITE) = 0 THEN
                V_RES := '0';
                V_RESULT :=
                    'You are not authorized for the selected site.';

                OPEN P_CURSOR FOR
                    SELECT CAST(NULL AS VARCHAR2(100)) AS FX_BADGE
                    FROM DUAL
                    WHERE 1 = 0;

                OPEN P_SUMMARY_CURSOR FOR
                    SELECT
                        0 AS TOTAL_SCANS,
                        0 AS SUCCESS_COUNT,
                        0 AS ALREADY_SCANNED_COUNT,
                        0 AS INVALID_COUNT
                    FROM DUAL
                    WHERE 1 = 0;

                RETURN;
            END IF;
        END IF;

        OPEN P_CURSOR FOR
        SELECT
            S.FX_BADGE,
            S.NAME,
            S.GROUPS,
            S.PROPERTY_01 AS SITE,
            GET_SITE_NAME(S.PROPERTY_01) AS SITE_NAME,
            BC.NAME AS MEAL_PREFERENCE,

            TO_CHAR(
                S.FIRST_DATE_FXBL,
                'DD-MM-YYYY'
            ) AS FIRST_DATE_FXBL,

            TO_CHAR(
                S.LAST_DATE_FXBL,
                'DD-MM-YYYY'
            ) AS LAST_DATE_FXBL,

            TO_CHAR(
                NVL(S.LAST_SCANTIME, S.FIRST_SCANTIME),
                'DD-MM-YYYY HH24:MI:SS'
            ) AS SCAN_TIME,

            S.COLLECTED_DATA AS STATUS,
            S.STATUS_MESSAGE

        FROM CMS.C_F1_SCAN_LOG S

        LEFT JOIN CMS.C_BASE_CODE_T BC
            ON TO_CHAR(S.MEAL_PREFERENCE) = TO_CHAR(BC.ID)
           AND BC.TYPE_ID IN (81, 1001)

        WHERE NVL(S.DEL_FLAG, 0) = 0

          AND (
                V_SEARCH_TEXT IS NULL
                OR TRIM(V_SEARCH_TEXT) IS NULL
                OR UPPER(S.FX_BADGE)
                   LIKE '%' || UPPER(TRIM(V_SEARCH_TEXT)) || '%'
                OR UPPER(S.NAME)
                   LIKE '%' || UPPER(TRIM(V_SEARCH_TEXT)) || '%'
              )

          AND (
                V_STATUS_FILTER IS NULL
                OR TRIM(V_STATUS_FILTER) IS NULL
                OR UPPER(TRIM(V_STATUS_FILTER)) = 'ALL'
                OR S.COLLECTED_DATA = V_STATUS_FILTER
              )

          AND (
                V_SITE IS NULL
                OR TRIM(V_SITE) IS NULL
                OR UPPER(TRIM(V_SITE)) = 'ALL'
                OR UPPER(TRIM(S.PROPERTY_01)) = UPPER(TRIM(V_SITE))
              )

          AND EXISTS (
                SELECT 1
                FROM CMS.C_USER_MANAGE_T U
                WHERE UPPER(TRIM(U.EMP_ID))
                      = UPPER(TRIM(V_EMP_ID))
                  AND UPPER(TRIM(U.SYSTEM_TYPE)) = 'CMS'
                  AND NVL(U.DEL_FLAG, 0) = 0
                  AND UPPER(TRIM(U.SITE))
                      = UPPER(TRIM(S.PROPERTY_01))
              )

        ORDER BY S.MEAL_DATE DESC, S.ID DESC;


        OPEN P_SUMMARY_CURSOR FOR
        SELECT
            COUNT(1) AS TOTAL_SCANS,

            NVL(
                SUM(
                    CASE
                        WHEN S.COLLECTED_DATA = 'SUCCESS'
                        THEN 1 ELSE 0
                    END
                ),
                0
            ) AS SUCCESS_COUNT,

            NVL(
                SUM(
                    CASE
                        WHEN S.COLLECTED_DATA = 'ALREADY_SCANNED'
                        THEN 1 ELSE 0
                    END
                ),
                0
            ) AS ALREADY_SCANNED_COUNT,

            NVL(
                SUM(
                    CASE
                        WHEN S.COLLECTED_DATA = 'INVALID'
                        THEN 1 ELSE 0
                    END
                ),
                0
            ) AS INVALID_COUNT

        FROM CMS.C_F1_SCAN_LOG S

        WHERE NVL(S.DEL_FLAG, 0) = 0

          AND (
                V_SITE IS NULL
                OR TRIM(V_SITE) IS NULL
                OR UPPER(TRIM(V_SITE)) = 'ALL'
                OR UPPER(TRIM(S.PROPERTY_01)) = UPPER(TRIM(V_SITE))
              )

          AND EXISTS (
                SELECT 1
                FROM CMS.C_USER_MANAGE_T U
                WHERE UPPER(TRIM(U.EMP_ID))
                      = UPPER(TRIM(V_EMP_ID))
                  AND UPPER(TRIM(U.SYSTEM_TYPE)) = 'CMS'
                  AND NVL(U.DEL_FLAG, 0) = 0
                  AND UPPER(TRIM(U.SITE))
                      = UPPER(TRIM(S.PROPERTY_01))
              );

        V_RES := '1';
        V_RESULT := 'Success';

    EXCEPTION
        WHEN OTHERS THEN
            V_RES := '0';
            V_RESULT := 'Error: ' || SQLERRM;

    END GET_MEAL_SCAN_LOG;


    /* ============================================================
       GET MEAL DASHBOARD
       Supports TYPE_ID 81 and TYPE_ID 1001.
       ============================================================ */
    PROCEDURE GET_MEAL_DASHBOARD(
        V_EMP_ID            IN VARCHAR2,
        V_DATE              IN VARCHAR2,
        V_SITE              IN VARCHAR2,
        V_RES               OUT VARCHAR2,
        V_RESULT            OUT VARCHAR2,
        P_CURSOR            OUT SYS_REFCURSOR,
        P_SUMMARY_CURSOR    OUT SYS_REFCURSOR
    )
    IS
        V_DAY DATE;

        V_TOTAL NUMBER := 0;
        V_NO_MEAL NUMBER := 0;
        V_COLLECTED NUMBER := 0;
        V_NOT_REQ_CAME NUMBER := 0;
        V_NOT_COLLECTED NUMBER := 0;

    BEGIN

        IF IS_ACTIVE_CMS_USER(V_EMP_ID) = 0 THEN
            V_RES := '0';
            V_RESULT := 'Active CMS user not found for employee: '
                        || V_EMP_ID;
            RETURN;
        END IF;

        BEGIN
            V_DAY := TO_DATE(V_DATE, 'DD-MM-YYYY');

        EXCEPTION
            WHEN OTHERS THEN
                V_RES := '0';
                V_RESULT := 'Invalid date. Expected format DD-MM-YYYY';
                RETURN;
        END;


        WITH EMP AS
        (
            SELECT
                M.FX_BADGE,
                M.MEAL_PREFERENCE,
                BC.NAME AS MEAL_PREFERENCE_NAME,
                M.PROPERTY_01 AS SITE

            FROM CMS.C_F1_MEAL_INFO M

            LEFT JOIN CMS.C_BASE_CODE_T BC
                ON TO_CHAR(M.MEAL_PREFERENCE) = TO_CHAR(BC.ID)
               AND BC.TYPE_ID IN (81, 1001)

            WHERE NVL(M.DEL_FLAG, 0) = 0
              AND M.FIRST_DATE_FXBL IS NOT NULL
              AND M.LAST_DATE_FXBL IS NOT NULL

              AND V_DAY BETWEEN
                  TRUNC(M.FIRST_DATE_FXBL)
                  AND TRUNC(M.LAST_DATE_FXBL)

              AND EXISTS (
                    SELECT 1
                    FROM CMS.C_USER_MANAGE_T U
                    WHERE UPPER(TRIM(U.EMP_ID))
                          = UPPER(TRIM(V_EMP_ID))
                      AND UPPER(TRIM(U.SYSTEM_TYPE)) = 'CMS'
                      AND NVL(U.DEL_FLAG, 0) = 0
                      AND UPPER(TRIM(U.SITE))
                          = UPPER(TRIM(M.PROPERTY_01))

                      AND (
                            V_SITE IS NULL
                            OR TRIM(V_SITE) IS NULL
                            OR UPPER(TRIM(V_SITE)) = 'ALL'
                            OR UPPER(TRIM(U.SITE))
                               = UPPER(TRIM(V_SITE))
                          )
                  )
        ),

        COLLECTED AS
        (
            SELECT DISTINCT S.FX_BADGE

            FROM CMS.C_F1_SCAN_LOG S

            WHERE NVL(S.DEL_FLAG, 0) = 0
              AND TRUNC(S.MEAL_DATE) = V_DAY
              AND S.COLLECTED_DATA = 'SUCCESS'

              AND EXISTS (
                    SELECT 1
                    FROM CMS.C_USER_MANAGE_T U
                    WHERE UPPER(TRIM(U.EMP_ID))
                          = UPPER(TRIM(V_EMP_ID))
                      AND UPPER(TRIM(U.SYSTEM_TYPE)) = 'CMS'
                      AND NVL(U.DEL_FLAG, 0) = 0
                      AND UPPER(TRIM(U.SITE))
                          = UPPER(TRIM(S.PROPERTY_01))

                      AND (
                            V_SITE IS NULL
                            OR TRIM(V_SITE) IS NULL
                            OR UPPER(TRIM(V_SITE)) = 'ALL'
                            OR UPPER(TRIM(U.SITE))
                               = UPPER(TRIM(V_SITE))
                          )
                  )
        )

        SELECT
            (
                SELECT COUNT(DISTINCT FX_BADGE)
                FROM EMP
            ),

            (
                SELECT COUNT(DISTINCT FX_BADGE)
                FROM EMP
                WHERE UPPER(TRIM(MEAL_PREFERENCE_NAME))
                      = 'NO MEAL REQUIRED'
            ),

            (
                SELECT COUNT(*)
                FROM COLLECTED
            ),

            (
                SELECT COUNT(DISTINCT S.FX_BADGE)

                FROM CMS.C_F1_SCAN_LOG S

                LEFT JOIN CMS.C_BASE_CODE_T BC
                    ON TO_CHAR(S.MEAL_PREFERENCE) = TO_CHAR(BC.ID)
                   AND BC.TYPE_ID IN (81, 1001)

                WHERE NVL(S.DEL_FLAG, 0) = 0
                  AND TRUNC(S.MEAL_DATE) = V_DAY
                  AND UPPER(TRIM(BC.NAME)) = 'NO MEAL REQUIRED'

                  AND EXISTS (
                        SELECT 1
                        FROM CMS.C_USER_MANAGE_T U
                        WHERE UPPER(TRIM(U.EMP_ID))
                              = UPPER(TRIM(V_EMP_ID))
                          AND UPPER(TRIM(U.SYSTEM_TYPE)) = 'CMS'
                          AND NVL(U.DEL_FLAG, 0) = 0
                          AND UPPER(TRIM(U.SITE))
                              = UPPER(TRIM(S.PROPERTY_01))

                          AND (
                                V_SITE IS NULL
                                OR TRIM(V_SITE) IS NULL
                                OR UPPER(TRIM(V_SITE)) = 'ALL'
                                OR UPPER(TRIM(U.SITE))
                                   = UPPER(TRIM(V_SITE))
                              )
                      )
            ),

            (
                SELECT COUNT(DISTINCT E.FX_BADGE)
                FROM EMP E
                WHERE UPPER(
                          NVL(TRIM(E.MEAL_PREFERENCE_NAME), 'X')
                      ) <> 'NO MEAL REQUIRED'

                  AND NOT EXISTS (
                        SELECT 1
                        FROM COLLECTED C
                        WHERE C.FX_BADGE = E.FX_BADGE
                      )
            )

        INTO
            V_TOTAL,
            V_NO_MEAL,
            V_COLLECTED,
            V_NOT_REQ_CAME,
            V_NOT_COLLECTED

        FROM DUAL;


        OPEN P_CURSOR FOR
        SELECT
            V_TOTAL AS TOTAL_EMPLOYEES,
            V_COLLECTED AS MEALS_COLLECTED,
            V_NOT_COLLECTED AS MEALS_NOT_COLLECTED,
            V_NO_MEAL AS NO_MEAL_REQUIRED,
            V_NOT_REQ_CAME AS NOT_REQUIRED_CAME
        FROM DUAL;


        OPEN P_SUMMARY_CURSOR FOR

        WITH CATS AS
        (
            SELECT 1 AS SORT_NO, 'Indian Veg' AS CATEGORY
            FROM DUAL

            UNION ALL
            SELECT 2, 'Indian Non-Veg'
            FROM DUAL

            UNION ALL
            SELECT 3, 'Western Veg'
            FROM DUAL

            UNION ALL
            SELECT 4, 'Western Non-Veg'
            FROM DUAL

            UNION ALL
            SELECT 5, 'Chinese Non-Veg'
            FROM DUAL

            UNION ALL
            SELECT 6, 'Chinese Veg'
            FROM DUAL

            UNION ALL
            SELECT 7, 'Jain'
            FROM DUAL

            UNION ALL
            SELECT 8, 'Vegan'
            FROM DUAL
        ),

        CNT AS
        (
            SELECT
                UPPER(TRIM(BC.NAME)) AS PREF,
                COUNT(DISTINCT S.FX_BADGE) AS TOTAL_COUNT

            FROM CMS.C_F1_SCAN_LOG S

            LEFT JOIN CMS.C_BASE_CODE_T BC
                ON TO_CHAR(S.MEAL_PREFERENCE) = TO_CHAR(BC.ID)
               AND BC.TYPE_ID IN (81, 1001)

            WHERE NVL(S.DEL_FLAG, 0) = 0
              AND TRUNC(S.MEAL_DATE) = V_DAY
              AND S.COLLECTED_DATA = 'SUCCESS'

              AND EXISTS (
                    SELECT 1
                    FROM CMS.C_USER_MANAGE_T U
                    WHERE UPPER(TRIM(U.EMP_ID))
                          = UPPER(TRIM(V_EMP_ID))
                      AND UPPER(TRIM(U.SYSTEM_TYPE)) = 'CMS'
                      AND NVL(U.DEL_FLAG, 0) = 0
                      AND UPPER(TRIM(U.SITE))
                          = UPPER(TRIM(S.PROPERTY_01))

                      AND (
                            V_SITE IS NULL
                            OR TRIM(V_SITE) IS NULL
                            OR UPPER(TRIM(V_SITE)) = 'ALL'
                            OR UPPER(TRIM(U.SITE))
                               = UPPER(TRIM(V_SITE))
                          )
                  )

            GROUP BY UPPER(TRIM(BC.NAME))
        )

        SELECT
            C.CATEGORY,

            NVL(N.TOTAL_COUNT, 0) AS COLLECTED_COUNT,

            CASE
                WHEN V_COLLECTED = 0 THEN 0
                ELSE ROUND(
                    NVL(N.TOTAL_COUNT, 0) * 100 / V_COLLECTED
                )
            END AS PERCENTAGE

        FROM CATS C

        LEFT JOIN CNT N
            ON N.PREF = UPPER(TRIM(C.CATEGORY))

        ORDER BY C.SORT_NO;


        V_RES := '1';
        V_RESULT := 'Success';

    EXCEPTION
        WHEN OTHERS THEN
            V_RES := '0';
            V_RESULT := 'Error: ' || SQLERRM;

    END GET_MEAL_DASHBOARD;


END F1_MEAL_MANAGEMENT;
