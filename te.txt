// ============================================================
// GENERATE MFA SECRET
// ============================================================

const handleGenerateSecret = async () => {
  const headers = {
    apiID: 'YOUR_EXISTING_SECRET_API_ID',
    userKey: 'YOUR_EXISTING_USER_KEY',
  };

  try {
    setShowQRCode(false);
    setsecretkey('');
    setErrorMessage('');

    const response = await axios.post(
      'https://microservice.abgind.foxconn.com/india/gateway',
      {},
      { headers }
    );

    console.log('MFA API Status:', response.status);

    const result = response.data;

    console.log(
      'MFA API Response Fields:',
      Object.keys(result || {})
    );

    // API returns secret directly
    if (
      !result ||
      typeof result.secret !== 'string' ||
      !result.secret.trim()
    ) {
      throw new Error(
        'MFA Secret is missing from API response'
      );
    }

    // Store secret in component state
    setsecretkey(result.secret);

    // Display QR code
    setShowQRCode(true);

    console.log('MFA Secret generated successfully');

  } catch (error) {
    console.error(
      'MFA Secret Generation Failed:',
      error.response?.data || error.message
    );

    setErrorMessage(
      'Unable to generate MFA secret. Please contact IT support.'
    );
  }
};
