submit(userInfo?: string) {
  this.loading = true;

  try {
    if (!userInfo || typeof userInfo !== 'string') {
      throw new Error('OUAC UserInfo is empty or invalid');
    }

    console.log('OUAC UserInfo length:', userInfo.length);

    // Step 1: Get the OUAC decryption key
    const encodedKey =
      'NzQxRTMzNzlBODA0MjM3OEY4NEE4MDc4NzNCMUJCNDk=';

    const decryptKey =
      this.publicSvr.security.getDK(encodedKey);

    // Step 2: Validate the key length
    if (![16, 24, 32].includes(decryptKey.length)) {
      throw new Error(
        'Invalid AES key length. Verify the OUAC key.'
      );
    }

    // Step 3: Decrypt OUAC UserInfo
    let decryptedUserInfo: string;

    try {
      decryptedUserInfo = security.decrypt(
        userInfo,
        decryptKey
      );
    } catch (error) {
      console.error('OUAC decryption failed:', error);

      throw new Error(
        'OUAC decryption failed. Verify the encryption key, IV, mode and padding.'
      );
    }

    // Step 4: Validate the decrypted data
    if (!decryptedUserInfo) {
      throw new Error(
        'Decrypted UserInfo is empty. Check the OUAC encryption configuration.'
      );
    }

    // Step 5: Parse JSON
    let data: any;

    try {
      data = JSON.parse(decryptedUserInfo);
    } catch (error) {
      console.error('Invalid decrypted JSON:', error);

      throw new Error(
        'Decrypted UserInfo is not valid JSON.'
      );
    }

    // Step 6: Encrypt UserInfo for the login API
    const userInfo2 = security.encrypt(
      JSON.stringify(data),
      '6jxaHMIhvefCOokm5XbDdW4qzl30U7Kt'
    );

    const param = {
      userInfo: userInfo2
    };

    // Step 7: Encrypt the request body
    const body =
      this.publicSvr.security.encryptForWeb(
        JSON.stringify(param)
      );

    // Step 8: Call the login API
    this.http.post(
      '/oneAccountLogin?_allow_anonymous=true',
      body,
      null,
      {
        headers: {
          'Content-Type': 'application/json;charset=UTF-8'
        }
      }
    ).subscribe(
      (response: any) => {
        try {
          const decryptedResponse =
            this.publicSvr.security.decryptForWeb(response);

          const res = JSON.parse(decryptedResponse);

          if (res.status === 200 && res.data) {
            const user = {
              avatar: './assets/tmp/img/avatar.jpg',
              eeid: res.data.eeid,
              name: res.data.empName,
              time: new Date().getTime()
            };

            this.loginSuccess(
              user,
              res.data.JsessionId,
              res.data.ClientIp
            );
          } else {
            this.msg.error(res.msg || 'Login failed');
          }
        } catch (error) {
          console.error(
            'Login response processing failed:',
            error
          );

          this.msg.error(
            'Unable to process login response.'
          );
        } finally {
          this.loading = false;
        }
      },
      (error) => {
        this.loading = false;

        console.error('Login API error:', error);

        this.msg.error(
          'Login failed. Please try again.'
        );
      }
    );

  } catch (error) {
    this.loading = false;

    console.error('Login processing error:', error);

    this.msg.error(
      error.message || 'Unable to process login information.'
    );
  }
}
