const data = response?.data;

console.log('OTP API raw response:', data);

let res = null;

if (typeof data?.r === 'string' && data.r.length > 0) {
  const decryptedResponse = decrypt(data.r, getdk(iwxKey));

  if (!decryptedResponse) {
    throw new Error('Unable to decrypt OTP API response.');
  }

  res = JSON.parse(decryptedResponse);
} else if (data && typeof data.IsOK !== 'undefined') {
  res = data;
} else {
  console.error('Unexpected OTP response:', data);

  setErrorMessage(
    data?.message || 'OTP verification response is incomplete.'
  );

  return;
}

console.log('Parsed OTP response:', res);
