const response = await axios.post(
  'https://microservice.abgind.foxconn.com/india/gateway',
  params,
  { headers }
);

console.log('OTP API status:', response.status);
console.log(
  'OTP API response keys:',
  Object.keys(response.data || {})
);

const data = response.data;

let res;

if (typeof data?.r === 'string') {
  const decryptedResponse = decrypt(data.r, getdk(iwxKey));

  if (!decryptedResponse) {
    throw new Error('OTP API response decryption failed.');
  }

  res = JSON.parse(decryptedResponse);
} else {
  console.error('OTP API response has no encrypted r field.');
  console.error('OTP API response format:', Object.keys(data || {}));

  setErrorMessage(
    data?.message || 'Unexpected response from OTP service.'
  );

  return;
}

console.log('OTP verification result:', {
  IsOK: res?.IsOK,
  Msg: res?.Msg
});
