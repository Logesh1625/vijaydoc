const handleGenerateSecret = async () => {
  const headers = {
    apiID: 'ffff-1694231787059-10208193125-0099',
    userKey: 'YOUR_EXISTING_USER_KEY',
  };

  try {
    const response = await axios.post(
      'https://microservice.abgind.foxconn.com/india/gateway',
      {},
      { headers }
    );

    // Debug: inspect response structure without logging the secret
    console.log('Secret API status:', response.status);
    console.log('Secret API response keys:', Object.keys(response.data || {}));

    let res;

    // Case 1: API returns secret directly
    if (typeof response.data?.secret === 'string') {
      res = response.data;
    }
    // Case 2: API returns an object inside Data
    else if (typeof response.data?.Data?.secret === 'string') {
      res = response.data.Data;
    }
    // Case 3: API returns encrypted response in r
    else if (typeof response.data?.r === 'string') {
      const decryptedResponse = decrypt(
        response.data.r,
        getdk(iwxKey)
      );

      if (!decryptedResponse) {
        throw new Error('Unable to decrypt MFA response.');
      }

      res = JSON.parse(decryptedResponse);
    } else {
      console.error('Unexpected secret API response structure.');
      setErrorMessage('Unexpected response from MFA service.');
      return;
    }

    if (typeof res?.secret !== 'string' || !res.secret.trim()) {
      console.error('MFA secret is missing from API response.');
      setErrorMessage('MFA secret was not returned by the service.');
      return;
    }

    setsecretkey(res.secret);
    setShowQRCode(true);
    setErrorMessage('');

  } catch (error) {
    console.error(
      'Secret generation failed:',
      error.response?.status || error.message
    );

    setErrorMessage('Unable to generate MFA secret.');
  }
};
