export function decrypt(
  str: string,
  keyStr?: any,
  ivStr?: any
): string {
  const key = keyStr
    ? CryptoJS.enc.Utf8.parse(keyStr)
    : CryptoJS.enc.Utf8.parse(
        "abcdefghijklmnopqrstuvwxyz123456"
      );

  const iv = ivStr
    ? CryptoJS.enc.Hex.parse(ivStr)
    : CryptoJS.enc.Hex.parse("0000000000000000");

  if (!str || !keyStr) {
    throw new Error("Encrypted data or key is missing.");
  }

  console.log("AES Debug:", {
    inputLength: str.length,
    keyLength: keyStr.length,
    ivLength: iv.sigBytes,
    mode: "CBC",
    padding: "Pkcs7"
  });

  const decrypted = CryptoJS.AES.decrypt(str, key, {
    iv,
    mode: CryptoJS.mode.CBC,
    padding: CryptoJS.pad.Pkcs7
  });

  console.log("Decrypted byte length:", decrypted.sigBytes);

  if (decrypted.sigBytes <= 0) {
    throw new Error(
      "Decryption produced empty data. Verify the key and AES configuration."
    );
  }

  try {
    return decrypted.toString(CryptoJS.enc.Utf8);
  } catch (error) {
    console.error("AES UTF-8 conversion failed:", error);
    throw error;
  }
}
