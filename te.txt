  /* ============================================================
     LOAD LOGGED-IN USER SITE
     ============================================================ */

  private loadLoggedInUserSite(): void {

    const userInfo =
      this.publicSvr.getSettingUserInfo();


    console.log(
      'Dashboard logged-in user:',
      userInfo
    );


    if (
      !userInfo ||
      !userInfo.eeid
    ) {

      this.resetDashboard();

      console.error(
        'Logged-in user EEID is missing'
      );

      return;
    }


    /*
     * Get user/site information.
     *
     * Backend CMS-Get_user_manage should already
     * exclude DEL_FLAG = 1.
     */
    this.publicSvr
      .getSitebasedUser()
      .subscribe(

        (userList: any[]) => {

          console.log(
            'Dashboard CMS-Get_user_manage:',
            userList
          );


          if (
            !userList ||
            userList.length === 0
          ) {

            this.resetDashboard();

            console.error(
              'No active user/site data'
            );

            return;
          }


          /*
           * Extract only active sites
           * belonging to logged-in employee.
           */
          this.siteList =
            this.extractAllSites(
              userList,
              userInfo.eeid
            );


          console.log(
            'Dashboard active site list:',
            this.siteList
          );


          if (
            this.siteList.length === 0
          ) {

            this.resetDashboard();

            console.error(
              'No active site found'
            );

            return;
          }


          /*
           * Default site.
           */
          this.site =
            this.siteList[0];


          /*
           * Empty = All My Sites.
           */
          this.selectedSite = '';


          /*
           * Build dropdown.
           */
          this.buildLocationList();


          /*
           * Load dashboard.
           */
          this.getDashboardData();

        },

        error => {

          console.error(
            'CMS-Get_user_manage error:',
            error
          );

          this.resetDashboard();

        }

      );

  }


  /* ============================================================
     GET EMPLOYEE ID
     ============================================================ */

  private getItemEeid(
    item: any
  ): string {

    if (!item) {
      return '';
    }


    return String(

      item.Eeid ||

      item.EEID ||

      item.EeidNo ||

      item.EmpId ||

      item.EMP_ID ||

      item.EmployeeId ||

      item.EmployeeCode ||

      item.eeid ||

      ''

    )
      .trim()
      .toUpperCase();

  }


  /* ============================================================
     ACTIVE USER CHECK
     ============================================================ */

  private isActiveUser(
    item: any
  ): boolean {

    if (!item) {
      return false;
    }


    /*
     * Compatible with old TypeScript.
     *
     * 0 / NULL = active
     * 1 = deleted
     */

    var delFlag = 0;


    if (
      item.DelFlag !== undefined &&
      item.DelFlag !== null &&
      item.DelFlag !== ''
    ) {

      delFlag =
        item.DelFlag;

    }

    else if (
      item.DEL_FLAG !== undefined &&
      item.DEL_FLAG !== null &&
      item.DEL_FLAG !== ''
    ) {

      delFlag =
        item.DEL_FLAG;

    }

    else if (
      item.delFlag !== undefined &&
      item.delFlag !== null &&
      item.delFlag !== ''
    ) {

      delFlag =
        item.delFlag;

    }

    else if (
      item.del_flag !== undefined &&
      item.del_flag !== null &&
      item.del_flag !== ''
    ) {

      delFlag =
        item.del_flag;

    }


    return Number(delFlag) === 0;

  }


  /* ============================================================
     EXTRACT ALL ACTIVE SITES
     ============================================================ */

  private extractAllSites(
    userList: any[],
    eeid: string
  ): string[] {

    const loginEeid =
      String(eeid)
        .trim()
        .toUpperCase();


    const sites =
      userList

        .filter(item => {

          if (!item) {
            return false;
          }


          /*
           * Employee must match.
           */
          if (
            this.getItemEeid(item)
            !==
            loginEeid
          ) {

            return false;
          }


          /*
           * Deleted user must not be used.
           */
          if (
            !this.isActiveUser(item)
          ) {

            return false;
          }


          return true;

        })


        .map(item =>
          this.extractSite(item)
        )


        .filter(
          site => !!site
        );


    return Array.from(
      new Set(sites)
    );

  }


  /* ============================================================
     EXTRACT SITE
     ============================================================ */

  private extractSite(
    user: any
  ): string {

    if (!user) {
      return '';
    }


    const site =
      user.Site ||

      user.site ||

      user.SITE ||

      user.SiteName ||

      user.siteName ||

      user.WorkSite ||

      user.Worksite ||

      user.BasicCode ||

      user.Basiccode ||

      user.Location ||

      user.location ||

      '';


    return String(site)
      .trim();

  }


  /* ============================================================
     BUILD SITE DROPDOWN
     ============================================================ */

  private buildLocationList(): void {

    this.locationList = [

      {
        value: '',
        label: 'All My Sites'
      }

    ];


    this.siteList.forEach(
      site => {

        this.locationList.push({

          value: site,

          label: site

        });

      }
    );


    console.log(
      'Dashboard location list:',
      this.locationList
    );

  }


  /* ============================================================
     DATE CHANGE
     ============================================================ */

  onDateChange(
    date: Date
  ): void {

    this.selectedDate =
      date || new Date();


    this.getDashboardData();

  }


  /* ============================================================
     SITE CHANGE
     ============================================================ */

  onLocationChange(
    location: string
  ): void {

    /*
     * Only allow sites belonging
     * to logged-in user.
     */
    if (
      location &&
      this.siteList.indexOf(location) === -1
    ) {

      console.warn(
        'Invalid site selected:',
        location
      );

      this.selectedSite = '';

    }

    else {

      this.selectedSite =
        location || '';

    }


    console.log(
      'Dashboard selected site:',
      this.selectedSite
    );


    this.getDashboardData();

  }


  /* ============================================================
     FORMAT DATE
     ============================================================ */

  private formatDate(
    d: Date
  ): string {

    const dd =
      ('0' + d.getDate())
        .slice(-2);


    const mm =
      ('0' + (d.getMonth() + 1))
        .slice(-2);


    return (
      dd +
      '-' +
      mm +
      '-' +
      d.getFullYear()
    );

  }


  /* ============================================================
     PERCENTAGE
     ============================================================ */

  pct(
    n: number
  ): number {

    return this.totalEmployees

      ? Math.round(
          (
            n /
            this.totalEmployees
          ) *
          100
        )

      : 0;

  }


  /* ============================================================
     GET DASHBOARD DATA
     ============================================================ */

  getDashboardData(): void {

    const userInfo =
      this.publicSvr
        .getSettingUserInfo();


    if (
      !userInfo ||
      !userInfo.eeid
    ) {

      this.resetDashboard();

      return;
    }


    /*
     * Do not call dashboard if
     * logged-in user has no active site.
     */
    if (
      !this.siteList ||
      this.siteList.length === 0
    ) {

      this.resetDashboard();

      return;
    }


    /*
     * Validate selected site.
     */
    if (
      this.selectedSite &&
      this.siteList.indexOf(
        this.selectedSite
      ) === -1
    ) {

      this.selectedSite = '';

    }


    this.loading = true;


    const params: any = {

      Date:
        this.formatDate(
          this.selectedDate
        ),

      /*
       * Empty = All My Sites.
       */
      Site:
        this.selectedSite || '',

      /*
       * Send logged-in employee.
       * Backend should use this for authorization.
       */
      EmpId:
        userInfo.eeid,

      SystemName:
        'CMS',

      Func:
        'CMS-GetMealDashboard'

    };


    console.log(
      'F1 Dashboard params:',
      params
    );


    this.publicSvr
      .iwxPostF1(

        params,

        {

          headers: {

            apiID:
              'ffff-1789987849667-10208193125-1519',

            userKey:
              'E9785F13CCEAFB251A6AB147AFD24336'

          }

        },

        true

      )


      .then(
        (res: any) => {

          this.loading = false;


          console.log(
            'F1 Dashboard response:',
            res
          );


          if (
            !res ||
            String(res.IsOK) !== '1'
          ) {

            this.resetDashboard();

            return;
          }


          /* ======================================================
             CURSOR 1
             Dashboard cards
             ====================================================== */

          const c =
            (
              res.List &&
              res.List.length > 0
            )

              ? res.List[0]

              : {};


          this.totalEmployees =
            Number(
              c.TotalEmployees || 0
            );


          this.mealsCollected =
            Number(
              c.MealsCollected || 0
            );


          this.mealsNotCollected =
            Number(
              c.MealsNotCollected || 0
            );


          this.noMealRequired =
            Number(
              c.NoMealRequired || 0
            );


          this.notCameToCollect =
            Number(
              c.NotRequiredCame || 0
            );


          /* ======================================================
             CURSOR 2
             Category summary
             ====================================================== */

          const cats: any[] =

            res.List2 ||

            res.Summary ||

            res.CategoryList ||

            [];


          if (
            cats.length > 0
          ) {

            this.categoryList =
              cats.map(
                (item: any) => {

                  return {

                    name:
                      item.Category,

                    count:
                      Number(
                        item.CollectedCount || 0
                      ),

                    percent:
                      Number(
                        item.Percentage || 0
                      ),

                    color:
                      this.getColorForCategory(
                        item.Category
                      ),

                    icon:
                      this.getIconForCategory(
                        item.Category
                      )

                  };

                }
              );

          }

          else {

            /*
             * Reset category values
             * when API returns no rows.
             */
            this.resetCategories();

          }


          this.updateSummary();

        }

      )


      .catch(
        err => {

          this.loading = false;

          console.error(
            'F1 Dashboard API error:',
            err
          );

          this.resetDashboard();

        }
      );

  }


  /* ============================================================
     RESET DASHBOARD

    this.noMealRequired = 0;

    this.notCameToCollect = 0;


    this.resetCategories();

    this.updateSummary();

  }

  /* ============================================================
     COLOR
     ============================================================ */

  getColorForCategory(
    name: string
  ): string {

    return (
      this.colorMap[name]
      ||
      '#999'
    );

  }


  /* ============================================================
     ICON
     ============================================================ */

  getIconForCategory(
    name: string
  ): string {

    return (
      this.iconMap[name]
      ||
      '🍽️'
    );

  }


  /* ============================================================
     UPDATE SUMMARY
     ============================================================ */

  private updateSummary(): void {

    this.summaryList = [

      {
        label:
          'Meals Collected',

        count:
          this.mealsCollected,

        percent:
          this.pct(
            this.mealsCollected
          ),

        color:
          '#1890ff'
      },


      {
        label:
          'Meals Not Collected',

        count:
          this.mealsNotCollected,

        percent:
          this.pct(
            this.mealsNotCollected
          ),

        color:
          '#b4bfd4'
      },


      {
        label:
          'No Meal Required',

        count:
          this.noMealRequired,

        percent:
          this.pct(
            this.noMealRequired
          ),

        color:
          '#dde3ee'
      },


      {
        label:
          'Not Required - Came to Collect',

        count:
          this.notCameToCollect,

        percent:
          this.pct(
            this.notCameToCollect
          ),

        color:
          '#b39ddb'
      }

    ];


    /*
     * First 3 slices are used
     * for donut.
     *
     * Not Required - Came to Collect
     * is a subset.
     */

    const ringParts =
      this.summaryList.slice(
        0,
        3
      );


    const sum =
      ringParts.reduce(
        (s, x) =>
          s + x.count,
        0
      );


    if (!sum) {

      this.donutGradient =
        'conic-gradient(#e6ebf5 0% 100%)';

      return;

    }


    let acc = 0;


    const stops =
      ringParts.map(
        x => {

          const start =
            acc;


          acc +=
            (
              x.count /
              sum
            ) *
            100;


          return (
            x.color +
            ' ' +
            start +
            '%' +
            ' ' +
            acc +
            '%'
          );

        }
      );


    this.donutGradient =
      'conic-gradient(' +
      stops.join(', ') +
      ')';

  }
