import {
  Component,
  OnInit,
  ViewChild,
  ElementRef
} from '@angular/core';

import {
  HttpClient
} from '@angular/common/http';

import {
  DomSanitizer,
  SafeUrl
} from '@angular/platform-browser';

import {
  NzMessageService
} from 'ng-zorro-antd';

import {
  PublicNetService
} from 'src/app/routes/a_Public_net/Public_net.service';


@Component({
  selector: 'app-Meal-scan',

  templateUrl:
    './Meal-scan.component.html',

  styleUrls:
    ['./Meal-scan.component.less']
})


export class MealScanComponent implements OnInit {


  /* ============================================================
     SCAN INPUT
     ============================================================ */

  @ViewChild(
    'scanInput',
    { static: false }
  )
  scanInput: ElementRef;


  empNo: string = '';

  loading: boolean = false;

  proceeding: boolean = false;


  /* ============================================================
     EMPLOYEE
     ============================================================ */

  employeeDetail: any = null;

  entitlementStatus: string = '';


  /* ============================================================
     PHOTO
     ============================================================ */

  employeePhotoUrl:
    SafeUrl | null = null;

  photoLoading: boolean = false;


  private readonly PHOTO_API_BASE =
    'http://10.209.110.187:3000/api/image';


  /* ============================================================
     SITE
     ============================================================ */

  /*
   * Raw site IDs from API.
   *
   * 69941 = FXCN
   * 95326 = FXBL
   */

  siteList: string[] = [];


  /*
   * Selected RAW site ID.
   *
   * Example:
   *
   * FXCN selected -> 69941
   * FXBL selected -> 95326
   *
   * Empty = All
   */

  selectedSite: string = '';


  /*
   * Dropdown options.
   *
   * value = RAW site ID
   * label = FXBL / FXCN
   */

  siteOptions:
    Array<{
      value: string;
      label: string;
    }> = [];


  /* ============================================================
     SITE NAME MAP
     ============================================================ */

  private readonly siteNameMap: {
    [key: string]: string;
  } = {

    '69941': 'FXCN',

    '95326': 'FXBL'

  };


  /* ============================================================
     LOGGED-IN USER
     ============================================================ */

  loginEmployeeId: string = '';


  /* ============================================================
     CONSTRUCTOR
     ============================================================ */

  constructor(

    private publicSvr:
      PublicNetService,

    private msgSvr:
      NzMessageService,

    private http:
      HttpClient,

    private sanitizer:
      DomSanitizer

  ) {}


  /* ============================================================
     INIT
     ============================================================ */

  ngOnInit(): void {

    this.loadLoggedInUserSite();

  }


  /* ============================================================
     LOAD LOGGED-IN USER SITE
     ============================================================ */

  private loadLoggedInUserSite(): void {

    const userInfo =
      this.publicSvr
        .getSettingUserInfo();


    console.log(
      '=========================================='
    );

    console.log(
      'MEAL SCAN - LOGGED IN USER'
    );

    console.log(
      userInfo
    );

    console.log(
      '=========================================='
    );


    /*
     * Employee ID mandatory.
     */

    if (
      !userInfo ||
      !userInfo.eeid
    ) {

      this.msgSvr.error(
        'Logged-in employee information not found.'
      );

      return;

    }


    /*
     * Store logged-in EEID.
     */

    this.loginEmployeeId =
      String(
        userInfo.eeid
      )
      .trim()
      .toUpperCase();


    /*
     * Get user/site mapping.
     */

    this.publicSvr
      .getSitebasedUser()
      .subscribe(

        (userList: any[]) => {

          console.log(
            'Meal Scan - User/site list:',
            userList
          );


          if (
            !userList ||
            userList.length === 0
          ) {

            this.msgSvr.error(
              'No active site found for logged-in user.'
            );

            return;

          }


          /*
           * Get active sites for
           * logged-in employee.
           */

          this.siteList =
            this.extractAllSites(
              userList,
              this.loginEmployeeId
            );


          console.log(
            'Meal Scan - Raw Site IDs:',
            this.siteList
          );


          /*
           * No site assigned.
           */

          if (
            this.siteList.length === 0
          ) {

            this.msgSvr.error(
              'No active site assigned to the logged-in user.'
            );

            return;

          }


          /*
           * Build dropdown.
           *
           * IMPORTANT:
           *
           * Value = Site ID
           * Label = FXBL / FXCN
           *
           * Therefore API receives ID,
           * UI shows only site name.
           */

          this.siteOptions =
            this.siteList.map(
              siteId => {

                return {

                  value:
                    siteId,

                  label:
                    this.getSiteName(siteId)

                };

              }
            );


          /*
           * Default:
           *
           * All
           */

          this.selectedSite = '';


          console.log(
            'Meal Scan - Site Options:',
            this.siteOptions
          );


          /*
           * Focus scanner.
           */

          this.focusInput();

        },


        error => {

          console.error(
            'Meal Scan - getSitebasedUser error:',
            error
          );


          this.msgSvr.error(
            'Failed to load user site information.'
          );

        }

      );

  }


  /* ============================================================
     GET EMPLOYEE ID
     ============================================================ */

  private getItemEeid(
    item: any
  ): string {

    if (!item) {

      return '';

    }


    const value =

      item.Eeid ||

      item.EEID ||

      item.EeidNo ||

      item.EmpId ||

      item.EMP_ID ||

      item.EmployeeId ||

      item.EmployeeCode ||

      item.eeid ||

      '';


    return String(value)
      .trim()
      .toUpperCase();

  }


  /* ============================================================
     ACTIVE USER CHECK
     ============================================================ */

  private isActiveUser(
    item: any
  ): boolean {

    if (!item) {

      return false;

    }


    let delFlag: any = 0;


    if (
      item.DelFlag !== undefined &&
      item.DelFlag !== null &&
      item.DelFlag !== ''
    ) {

      delFlag =
        item.DelFlag;

    }

    else if (
      item.DEL_FLAG !== undefined &&
      item.DEL_FLAG !== null &&
      item.DEL_FLAG !== ''
    ) {

      delFlag =
        item.DEL_FLAG;

    }

    else if (
      item.delFlag !== undefined &&
      item.delFlag !== null &&
      item.delFlag !== ''
    ) {

      delFlag =
        item.delFlag;

    }

    else if (
      item.del_flag !== undefined &&
      item.del_flag !== null &&
      item.del_flag !== ''
    ) {

      delFlag =
        item.del_flag;

    }


    /*
     * 0 / NULL = ACTIVE
     * 1 = DELETED
     */

    return Number(delFlag) === 0;

  }


  /* ============================================================
     EXTRACT ALL ACTIVE SITES
     ============================================================ */

  private extractAllSites(
    userList: any[],
    eeid: string
  ): string[] {

    const loginEeid =
      String(eeid)
        .trim()
        .toUpperCase();


    const sites: string[] = [];


    userList.forEach(
      item => {

        if (!item) {

          return;

        }


        /*
         * Employee match.
         */

        if (
          this.getItemEeid(item)
          !==
          loginEeid
        ) {

          return;

        }


        /*
         * Ignore deleted user.
         */

        if (
          !this.isActiveUser(item)
        ) {

          return;

        }


        /*
         * Get site.
         */

        const site =
          this.extractSite(item);


        if (site) {

          sites.push(site);

        }

      }
    );


    /*
     * Remove duplicates.
     */

    const uniqueSites: string[] = [];


    sites.forEach(
      site => {

        const normalized =
          String(site)
            .trim();


        const exists =
          uniqueSites.some(
            existing =>
              existing
                .toUpperCase() ===
              normalized
                .toUpperCase()
          );


        if (!exists) {

          uniqueSites.push(
            normalized
          );

        }

      }
    );


    return uniqueSites;

  }


  /* ============================================================
     EXTRACT SITE
     ============================================================ */

  private extractSite(
    user: any
  ): string {

    if (!user) {

      return '';

    }


    /*
     * Prefer Site / SiteId.
     */

    const site =

      user.Site ||

      user.site ||

      user.SITE ||

      user.SiteId ||

      user.siteId ||

      '';


    return String(site)
      .trim();

  }


  /* ============================================================
     GET SITE NAME
     ============================================================ */

  getSiteName(
    siteId: any
  ): string {

    const id =
      String(
        siteId || ''
      )
      .trim()
      .toUpperCase();


    if (!id) {

      return '';

    }


    /*
     * Already a site name.
     */

    if (
      id === 'FXBL' ||
      id === 'FXCN'
    ) {

      return id;

    }


    /*
     * Convert ID -> Site Name.
     */

    return this.siteNameMap[id] || id;

  }


  /* ============================================================
     GET SITE DISPLAY NAME
     ============================================================ */

  getSiteDisplayName(
    siteId: any
  ): string {

    if (
      !siteId
    ) {

      return 'Not Updated';

    }


    return this.getSiteName(
      siteId
    );

  }


  /* ============================================================
     SITE CHANGE
     ============================================================ */

  onLocationChange(
    location: string
  ): void {

    console.log(
      '=========================================='
    );

    console.log(
      'MEAL SCAN - SITE CHANGE'
    );

    console.log(
      'Selected Raw Site:',
      location
    );

    console.log(
      'Selected Display Site:',
      this.getSiteName(location)
    );

    console.log(
      '=========================================='
    );


    /*
     * All
     */

    if (
      !location
    ) {

      this.selectedSite = '';

    }


    else {

      /*
       * Validate selected site.
       */

      const validSite =
        this.siteList.some(
          site =>
            site
              .toUpperCase() ===
            String(location)
              .trim()
              .toUpperCase()
        );


      if (
        validSite
      ) {

        /*
         * Store RAW site ID.
         */

        this.selectedSite =
          String(location)
            .trim();

      }

      else {

        this.selectedSite = '';

        this.msgSvr.warning(
          'Invalid site selected.'
        );

        return;

      }

    }


    /*
     * Clear previous employee.
     */

    this.employeeDetail = null;

    this.entitlementStatus = '';

    this.employeePhotoUrl = null;


    /*
     * Clear badge.

     */

    this.empNo = '';


    /*
     * Focus scanner.
     */

    this.focusInput();

  }


  /* ============================================================
     FOCUS INPUT
     ============================================================ */

  focusInput(): void {

    setTimeout(
      () => {

        if (
          this.scanInput
        ) {

          this.scanInput
            .nativeElement
            .focus();

        }

      },
      100
    );

  }


  /* ============================================================
     SCAN ENTER
     ============================================================ */

  onScanEnter(): void {

    this.search();

  }


  /* ============================================================
     SEARCH
     ============================================================ */

  search(
    fromButton?: boolean
  ): void {

    /*
     * Badge validation.
     */

    if (
      !this.empNo ||
      this.empNo.trim() === ''
    ) {

      this.msgSvr.error(
        'Please scan or enter an FX Badge.'
      );

      this.focusInput();

      return;

    }


    /*
     * Site validation.
     */

    if (
      this.siteList.length === 0
    ) {

      this.msgSvr.error(
        'No active site available.'
      );

      return;

    }


    /*
     * Validate selected site.
     */

    if (
      this.selectedSite
    ) {

      const validSite =
        this.siteList.some(
          site =>
            site
              .toUpperCase() ===
            this.selectedSite
              .toUpperCase()
        );


      if (
        !validSite
      ) {

        this.msgSvr.error(
          'Selected site is not assigned to you.'
        );

        return;

      }

    }


    /*
     * Start loading.
     */

    this.loading = true;


    this.employeeDetail = null;

    this.entitlementStatus = '';

    this.employeePhotoUrl = null;


    /* ==========================================================
       API PARAMETER
       ========================================================== */

    const params: any = {

      /*
       * Employee badge.
       */

      FxBadge:
        this.empNo.trim(),


      /*
       * RAW SITE ID.
       *
       * FXCN = 69941
       * FXBL = 95326
       *
       * Empty = All
       */

      Site:
        this.selectedSite || '',


      /*
       * Logged-in user.
       */

      EmpId:
        this.loginEmployeeId,


      SystemName:
        'CMS',


      Func:
        'CMS-GetF1MenuByFxbadge'

    };


    console.log(
      '=========================================='
    );

    console.log(
      'MEAL SCAN SEARCH'
    );

    console.log(
      'Badge:',
      params.FxBadge
    );

    console.log(
      'RAW Site:',
      params.Site || 'All'
    );

    console.log(
      'DISPLAY Site:',
      this.getSiteName(params.Site) || 'All'
    );

    console.log(
      'Logged-in Employee:',
      params.EmpId
    );

    console.log(
      'Request:',
      params
    );

    console.log(
      '=========================================='
    );


    this.publicSvr
      .iwxPostF1(

        params,

        {

          headers: {

            apiID:
              'ffff-1789807467770-10208193125-1466',

            userKey:
              'E9785F13CCEAFB251A6AB147AFD24336'

          }

        },

        true

      )


      .then(
        (res: any) => {

          this.loading = false;


          console.log(
            'Meal Scan Search Response:',
            res
          );


          /* ==================================================
             SUCCESS
             ================================================== */

          if (

            res &&

            String(res.IsOK) === '1' &&

            res.List &&

            res.List.length > 0

          ) {

            const data =
              res.List[0];


            /*
             * Employee details.
             */

            this.employeeDetail = {

              FxBadge:
                data.FxBadge ||
                data.FX_BADGE ||
                this.empNo.trim(),


              Name:
                data.Name ||
                data.NAME ||
                '',


              Site:
                data.Site ||
                data.SITE ||
                '',


              MealPreference:
                data.MealPreference ||
                data.MEAL_PREFERENCE ||
                '',


              FirstDateFxbl:
                data.FirstDateFxbl ||
                data.FIRST_DATE_FXBL ||
                '',


              LastDateFxbl:
                data.LastDateFxbl ||
                data.LAST_DATE_FXBL ||
                ''

            };


            /*
             * Entitlement status.
             */

            this.entitlementStatus =
              data.EntitlementStatus ||
              data.ENTITLEMENT_STATUS ||
              '';


            /* ==================================================
               SITE MISMATCH
               ================================================== */

            if (

              this.selectedSite &&

              this.employeeDetail.Site &&

              !this.isSameSite(
                this.employeeDetail.Site,
                this.selectedSite
              )

            ) {

              this.entitlementStatus =
                'SITE_MISMATCH';


              this.msgSvr.error(
                'This FX Badge does not belong to the selected site.'
              );


              /*
               * Still show employee photo.
               */

              this.fetchEmployeePhoto(
                this.employeeDetail.FxBadge
              );


              this.clearInputOnly();


              return;

            }


            /*
             * Fetch photo.
             */

            this.fetchEmployeePhoto(
              this.employeeDetail.FxBadge
            );


            /*
             * Insert scan.
             */

            this.proceed();

          }


          /* ==================================================
             BADGE NOT FOUND
             ================================================== */

          else {

            this.employeeDetail = {

              FxBadge:
                this.empNo.trim(),

              Name:
                '',

              Site:
                '',

              MealPreference:
                '',

              FirstDateFxbl:
                '',

              LastDateFxbl:
                ''

            };


            this.entitlementStatus =
              'NOT_UPDATED';


            /*
             * Try employee photo.
             */

            this.fetchEmployeePhoto(
              this.empNo.trim()
            );


            this.msgSvr.error(
              'FX Badge not found for the selected site.'
            );


            this.clearInputOnly();

          }

        }

      )


      .catch(
        err => {

          this.loading = false;


          console.error(
            'Meal Scan Search Error:',
            err
          );


          this.msgSvr.error(
            'Failed to fetch employee details.'
          );


          this.focusInput();

        }
      );

  }


  /* ============================================================
     COMPARE SITE
     ============================================================ */

  private isSameSite(
    site1: any,
    site2: any
  ): boolean {

    const first =
      String(
        site1 || ''
      )
      .trim()
      .toUpperCase();


    const second =
      String(
        site2 || ''
      )
      .trim()
      .toUpperCase();


    if (
      !first ||
      !second
    ) {

      return false;

    }


    /*
     * Direct ID comparison.
     */

    if (
      first === second
    ) {

      return true;

    }


    /*
     * Convert both to display names.
     */

    const firstName =
      this.getSiteName(first)
        .toUpperCase();


    const secondName =
      this.getSiteName(second)
        .toUpperCase();


    return (
      firstName ===
      secondName
    );

  }


  /* ============================================================
     FETCH EMPLOYEE PHOTO
     ============================================================ */

  fetchEmployeePhoto(
    fxBadge: string
  ): void {

    if (
      !fxBadge
    ) {

      return;

    }


    this.photoLoading = true;


    /*
     * Clear old photo.
     */

    this.employeePhotoUrl = null;


    this.http
      .get(

        `${this.PHOTO_API_BASE}/${fxBadge}`,

        {
          responseType:
            'blob'
        }

      )


      .subscribe({

        next:
          (blob) => {

            /*
             * Create object URL.
             */

            const objectUrl =
              URL.createObjectURL(
                blob
              );


            /*
             * Safe URL.
             */

            this.employeePhotoUrl =
              this.sanitizer
                .bypassSecurityTrustUrl(
                  objectUrl
                );


            this.photoLoading =
              false;

          },


        error:
          (error) => {

            console.error(
              'Employee photo error:',
              error
            );


            this.employeePhotoUrl =
              null;


            this.photoLoading =
              false;

          }

      });

  }


  /* ============================================================
     INSERT MEAL SCAN
     ============================================================ */

  proceed(): void {

    if (
      !this.employeeDetail
    ) {

      return;

    }


    this.proceeding = true;


    /* ============================================================
       INSERT PARAMETERS
       ============================================================ */

    const params: any = {

      /*
       * Employee badge.
       */

      FxBadge:
        this.employeeDetail.FxBadge,


      /*
       * RAW SITE ID.
       *
       * FXCN = 69941
       * FXBL = 95326
       */

      Site:
        this.selectedSite || '',


      /*
       * Logged-in employee.
       */

      EmpId:
        this.loginEmployeeId,


      SystemName:
        'CMS',


      Func:
        'CMS-InsertMealScanLog'

    };


    console.log(
      '=========================================='
    );

    console.log(
      'MEAL SCAN INSERT'
    );

    console.log(
      'Badge:',
      params.FxBadge
    );

    console.log(
      'RAW Site:',
      params.Site || 'All'
    );

    console.log(
      'DISPLAY Site:',
      this.getSiteName(params.Site) || 'All'
    );

    console.log(
      'Logged-in Employee:',
      params.EmpId
    );

    console.log(
      'Request:',
      params
    );

    console.log(
      '=========================================='
    );


    this.publicSvr
      .iwxPostF1(

        params,

        {

          headers: {

            apiID:
              'ffff-1789818962176-10208193125-1483',

            userKey:
              'E9785F13CCEAFB251A6AB147AFD24336'

          }

        },

        true

      )


      .then(
        (res: any) => {

          this.proceeding = false;


          console.log(
            'Meal Scan Insert Response:',
            res
          );


          if (

            res &&

            String(res.IsOK) === '1' &&

            res.List &&

            res.List.length > 0

          ) {

            const logResult =
              res.List[0];


            const status =
              logResult.Status ||
              logResult.STATUS ||
              '';


            const message =
              logResult.StatusMessage ||
              logResult.STATUS_MESSAGE ||
              '';


            /* ==================================================
               SUCCESS
               ================================================== */

            if (
              status === 'SUCCESS'
            ) {

              this.entitlementStatus =
                'ENTITLED';


              this.msgSvr.success(
                'Meal collection recorded successfully!'
              );

            }


            /* ==================================================
               ALREADY SCANNED
               ================================================== */

            else if (
              status ===
              'ALREADY_SCANNED'
            ) {

              this.entitlementStatus =
                'ALREADY_SCANNED';


              this.msgSvr.error(

                message ||
                'Meal already collected today'

              );

            }


            /* ==================================================
               SITE MISMATCH
               ================================================== */

            else if (
              status ===
              'SITE_MISMATCH'
            ) {

              this.entitlementStatus =
                'SITE_MISMATCH';


              this.msgSvr.error(

                message ||
                'Employee does not belong to selected site.'

              );

            }


            /* ==================================================
               INVALID
               ================================================== */

            else {

              this.msgSvr.error(

                message ||
                'Invalid scan.'

              );

            }


            /*
             * Clear scanner input only.
             */

            this.clearInputOnly();

          }


          else {

            this.msgSvr.error(

              res.Msg ||
              'Failed to log scan.'

            );

          }

        }

      )


      .catch(
        err => {

          this.proceeding = false;


          console.error(
            'Meal Scan Insert Error:',
            err
          );


          this.msgSvr.error(
            'Failed to log scan. Please try again.'
          );

        }
      );

  }


  /* ============================================================
     CLEAR INPUT ONLY
     ============================================================ */

  clearInputOnly(): void {

    this.empNo = '';


    /*
     * Keep employee result visible.
     */

    this.focusInput();

  }


  /* ============================================================
     CANCEL
     ============================================================ */

  cancelScan(): void {

    this.employeeDetail =
      null;


    this.entitlementStatus =
      '';


    this.empNo =
      '';


    this.employeePhotoUrl =
      null;


    this.proceeding =
      false;


    this.loading =
      false;


    this.photoLoading =
      false;


    this.focusInput();

  }

}

const ftp = require("basic-ftp");
const dotenv = require("dotenv");
const path = require("path");

dotenv.config();

const app = express();
const PORT = 3000;

app.use(express.static(path.join(__dirname, "public")));

// Get image from FTP
app.get("/api/image/:id", async (req, res) => {
    const id = req.params.id.trim();

    // Only allow numbers
    /*if (!/^\d+$/.test(id)) {
        return res.status(400).json({
            error: "Image ID must contain numbers only."
        });
    }*/

    const possibleFiles = [
        `${id}.png`,
        `${id}.jpeg`,
        `${id}.jpg`
    ];

    const client = new ftp.Client();

    try {
        await client.access({
            host: process.env.FTP_HOST,
            port: Number(process.env.FTP_PORT || 21),
            user: process.env.FTP_USER,
            password: process.env.FTP_PASSWORD,
            secure: process.env.FTP_SECURE === "true"
        });

        await client.cd("/photos");

        let foundFile = null;

        // Check which file exists
        for (const filename of possibleFiles) {
            const files = await client.list();

            const exists = files.some(
                file => file.name.toLowerCase() === filename.toLowerCase()
            );

            if (exists) {
                foundFile = filename;
                break;
            }
        }

        if (!foundFile) {
            return res.status(404).json({
                error: `No image found for ${id}`
            });
        }

        // Download the image into memory
        const chunks = [];

        const writable = new (require("stream").Writable)({
            write(chunk, encoding, callback) {
                chunks.push(chunk);
                callback();
            }
        });

        await client.downloadTo(writable, foundFile);

        const imageBuffer = Buffer.concat(chunks);

        // Determine content type
        const extension = path.extname(foundFile).toLowerCase();

        let contentType = "application/octet-stream";

        if (extension === ".png") {
            contentType = "image/png";
        } else if (extension === ".jpg" || extension === ".jpeg") {
            contentType = "image/jpeg";
        }

        res.setHeader("Content-Type", contentType);
        res.setHeader("Cache-Control", "no-cache");

        res.send(imageBuffer);

    } catch (error) {
        console.error("FTP error:", error);

        res.status(500).json({
            error: "Unable to retrieve image from FTP server."
        });

    } finally {
        client.close();
    }
});

app.listen(PORT, () => {
    console.log(`Image viewer running at http://localhost:${PORT}`);
});
