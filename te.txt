import { Component, OnInit } from '@angular/core';
import * as XLSX from "xlsx";
import * as FileSaver from "file-saver";
import XlsxPopulate from "xlsx-populate/browser/xlsx-populate";

import { SharedModule } from "@shared";
import { PublicNetService } from "src/app/routes/a_Public_net/Public_net.service";
import {
  NzMessageService,
  UploadFile,
  NzDrawerRef
} from "ng-zorro-antd";

@Component({
  selector: 'app-F1-Details-import',
  templateUrl: './F1-Details-import.html',
})
export class F1ManageImportComponent implements OnInit {

  fileList: UploadFile[] = [];

  addLoading: boolean = false;
  isSpinning: boolean = false;

  sussessUpLoad = {
    total: 0,
    suc: 0,
    fail: 0,
  };

  allData: any[] = [];
  subData: any[] = [];
  excelData: any[] = [];

  fileForm = {
    fileListStr: "",
  };

  siteOptions = [];

  mealPreferenceList: any[] = [];

  // Header names must match the downloaded template exactly
  templateData = {
    "Groups": '',
    "Meal Preference": '',
    "FX Badge": '',
    "Name": '',
    "First Day at Factory": '',
    "Last Day at Factory": ''
  };

  constructor(
    private apiSvr: PublicNetService,
    private shareSvr: SharedModule,
    private msgSvr: NzMessageService,
    private ref: NzDrawerRef
  ) { }


  // ==========================================
  // INIT
  // ==========================================

  ngOnInit() {
    this.loadMealPreferences();
  }


  // ==========================================
  // LOAD MEAL PREFERENCE
  // ==========================================

  loadMealPreferences() {
    this.apiSvr
      .getGuardPassCode("GuardPass-Meal Preference")
      .subscribe(
        (res: any) => {
          this.mealPreferenceList = res || [];
          console.log(
            "IMPORT - MEAL PREFERENCE LIST:",
            JSON.stringify(this.mealPreferenceList)
          );
        },
        (error: any) => {
          console.error("Failed to load Meal Preference:", error);
          this.msgSvr.error("Unable to load Meal Preference master data.");
        }
      );
  }


  // ==========================================
  // FILE BEFORE UPLOAD
  // ==========================================

  beforeUpload = (file: UploadFile): boolean => {

    this.fileForm.fileListStr = file.name;
    this.fileList = [file];

    const fileType = file.name
      .substring(file.name.lastIndexOf("."), file.name.length)
      .toLocaleUpperCase();

    if (fileType !== ".XLS" && fileType !== ".XLSX") {

      this.shareSvr.modalSvr.confirm({
        nzCancelText: null,
        nzContent: "Attachment format must be excel!",
        nzOnOk: () => {
          this.fileForm.fileListStr = "";
          this.fileList = [];
        },
      });

    } else {
      this.addLoading = false;
    }

    return false;
  };


  // ==========================================
  // DOWNLOAD EXCEL TEMPLATE
  // ==========================================

  DownloadExcelfile() {

    // 1. CHECK MEAL PREFERENCE MASTER DATA
    const mealValues = (this.mealPreferenceList || [])
      .map((item: any) => item.label)
      .filter((value: any) =>
        value !== null && value !== undefined && value !== ""
      )
      .map((value: any) => String(value).trim());

    if (mealValues.length === 0) {
      this.msgSvr.warning("Meal Preference data is not loaded. Please try again.");
      this.loadMealPreferences();
      return;
    }

    console.log("Meal Preference values:", mealValues);

    // 2. CREATE WORKBOOK
    XlsxPopulate.fromBlankAsync()
      .then((workbook: any) => {

        const worksheet = workbook.sheet(0);
        worksheet.name("Template");

        // 3. HEADER
        worksheet.range("A1:F1").value([[
          "Groups",
          "Meal Preference",
          "FX Badge",
          "Name",
          "First Day at Factory",
          "Last Day at Factory"
        ]]);

        // 4. HEADER STYLE
        worksheet.range("A1:F1").style({
          bold: true,
          fontSize: 11,
          horizontalAlignment: "center",
          verticalAlignment: "center",
          wrapText: true
        });

        worksheet.row(1).height(25);

        // 5. COLUMN WIDTH
        worksheet.column("A").width(20);
        worksheet.column("B").width(25);
        worksheet.column("C").width(25);
        worksheet.column("D").width(20);
        worksheet.column("E").width(22);
        worksheet.column("F").width(22);

        // 6. DATE COLUMNS AS TEXT
        // Text format keeps exactly what the user types (DD-MM-YYYY).
        // Note: pasted data can replace this format with real Excel dates;
        // formatDateValue() below handles both text and real dates.
        worksheet
          .range("E2:F500")
          .style("numberFormat", "@");

        // 6b. SAMPLE ROW (shows the expected date format)
        worksheet.cell("E2").value("DD-MM-YYYY");
        worksheet.cell("F2").value("DD-MM-YYYY");

        // 7. MEAL PREFERENCE DROPDOWN SOURCE (hidden sheet)
        const masterSheet = workbook.addSheet("MealMaster");

        masterSheet.cell("A1").value("Meal Preference");

        for (let i = 0; i < mealValues.length; i++) {
          masterSheet.cell(`A${i + 2}`).value(mealValues[i]);
        }

        masterSheet.hidden(true);

        const lastMealRow = mealValues.length + 1;

        // 8. MEAL DROPDOWN B2:B500
        worksheet
          .range("B2:B500")
          .dataValidation({
            type: "list",
            allowBlank: true,
            showErrorMessage: true,
            errorTitle: "Invalid Meal Preference",
            error: "Please select Meal Preference from the dropdown.",
            formula1: `'MealMaster'!$A$2:$A$${lastMealRow}`
          });

        // 9. No validation on date columns: the user types DD-MM-YYYY
        //    and formatDateValue() handles parsing on import.

        // 10. FREEZE HEADER
        worksheet.freezePanes(0, 1);

        // 11. OUTPUT
        return workbook.outputAsync();
      })
      .then((buffer: any) => {

        const blob = new Blob(
          [buffer],
          {
            type: "application/vnd.openxmlformats-officedocument.spreadsheetml.sheet"
          }
        );

        FileSaver.saveAs(blob, "F1Meal-info Template.xlsx");

        console.log("F1 Meal Excel Template downloaded successfully.");
        this.msgSvr.success("Excel template downloaded successfully.");
      })
      .catch((error: any) => {

        console.error("====================================");
        console.error("Excel generation error:", error);
        console.error("Error message:", error && error.message);
        console.error("Error stack:", error && error.stack);
        console.error("====================================");

        this.msgSvr.error("Unable to generate Excel template.");
      });
  }


  // ==========================================
  // ADD FILE
  // ==========================================

  addFile() {

    this.addLoading = true;
    this.isSpinning = true;

    this.sussessUpLoad = {
      total: 0,
      suc: 0,
      fail: 0
    };

    this.allData = [];
    this.subData = [];

    this.fileList.forEach((file) => {

      this.getFileData(file, (data) => {

        // NOTE: do NOT use cellDates:true here.
        // SheetJS builds JS Dates with a timezone error (in India a date
        // becomes 23:59:50 of the PREVIOUS day, so every date moved back
        // by one day). Date cells are read as Excel serial numbers and
        // converted safely in formatDateValue().
        const wb: XLSX.WorkBook = XLSX.read(
          data,
          {
            type: "binary"
          }
        );

        const wsname: string = wb.SheetNames[0];
        const ws: XLSX.WorkSheet = wb.Sheets[wsname];
        const json = XLSX.utils.sheet_to_json(ws);

        this.batchAdd(json);
      });
    });
  }


  // ==========================================
  // READ FILE
  // ==========================================

  getFileData(file, callback: (fle: string) => void): void {

    const reader = new FileReader();

    reader.readAsBinaryString(file);

    reader.addEventListener("load", () => {
      callback(reader.result!.toString());
    });
  }


  // ==========================================
  // BATCH ADD
  // ==========================================

  batchAdd(data: any[]) {

    // Ignore the sample row from the template (dates still say "DD-MM-YYYY")
    const isSampleRow = (row: any): boolean => {
      const first = String(row["First Day at Factory"] || "").trim().toUpperCase();
      const last = String(row["Last Day at Factory"] || "").trim().toUpperCase();
      return first === "DD-MM-YYYY" && last === "DD-MM-YYYY";
    };

    this.excelData = (data || []).filter((row: any) => !isSampleRow(row));

    this.sussessUpLoad.total = this.excelData.length;

    if (this.excelData.length > 0) {

      this.excelData.forEach((item) => {
        this.validateAndProcessRow(item);
      });

      this.isSpinning = false;

    } else {

      this.fileForm.fileListStr = "";
      this.fileList = [];
      this.isSpinning = false;

      this.msgSvr.error("Excel data does not exist");
    }
  }


  // ==========================================
  // VALIDATE ROW
  // ==========================================

  validateAndProcessRow(item: any) {

    const requiredKeys = [
      "Groups",
      "Meal Preference",
      "FX Badge",
      "Name",
      "First Day at Factory",
      "Last Day at Factory"
    ];

    const hasAllProps = requiredKeys.every(
      (key) => item.hasOwnProperty(key)
    );

    const rowRecord = {
      ...this.templateData,
      ...item
    };

    if (!hasAllProps) {

      this.sussessUpLoad.fail++;

      rowRecord["Wrong Reason"] = "All of template content is required.";

      this.allData.push(rowRecord);

      return;
    }

    // MEAL PREFERENCE VALIDATION
    const mealName = item["Meal Preference"]
      ? item["Meal Preference"].toString().trim()
      : "";

    const selectedMeal = this.getMealPreferenceByName(mealName);

    if (!selectedMeal) {

      this.sussessUpLoad.fail++;

      rowRecord["Wrong Reason"] =
        "Invalid Meal Preference. Please select a value from the Excel dropdown.";

      this.allData.push(rowRecord);

      return;
    }

    // DATE VALIDATION (DD-MM-YYYY, real calendar date, first <= last)
    const firstDate = this.formatDateValue(item["First Day at Factory"]);
    const lastDate = this.formatDateValue(item["Last Day at Factory"]);

    if (!this.isValidDmy(firstDate) || !this.isValidDmy(lastDate)) {

      this.sussessUpLoad.fail++;

      rowRecord["Wrong Reason"] =
        "Invalid date. Use DD-MM-YYYY for First Day / Last Day at Factory.";

      this.allData.push(rowRecord);

      return;
    }

    if (this.dmyToTime(firstDate) > this.dmyToTime(lastDate)) {

      this.sussessUpLoad.fail++;

      rowRecord["Wrong Reason"] =
        "First Day at Factory cannot be after Last Day at Factory.";

      this.allData.push(rowRecord);

      return;
    }

    // SUCCESS
    this.sussessUpLoad.suc++;

    this.allData.push(rowRecord);
    this.subData.push(rowRecord);
  }


  // ==========================================
  // FIND MEAL PREFERENCE BY NAME
  // ==========================================

  getMealPreferenceByName(mealName: string): any {

    if (!mealName) {
      return null;
    }

    return this.mealPreferenceList.find(
      (item: any) =>
        String(item.label).trim().toUpperCase() ===
        String(mealName).trim().toUpperCase()
    );
  }


  // ==========================================
  // EXPORT FAILED EXCEL
  // ==========================================

  exportFailedExcel() {

    if (this.allData.length > 0) {
      this.apiSvr.exportHandle(this.allData, 68);
    } else {
      this.msgSvr.error("No Data.");
    }
  }


  // ==========================================
  // SUBMIT
  // ==========================================

  submit() {

    if (this.subData.length === 0) {
      this.msgSvr.error("No valid data to submit. Please check your file.");
      return;
    }

    this.isSpinning = true;

    let completedCount = 0;
    let successCount = 0;
    let failCount = 0;

    const validSubData = [...this.subData];

    this.subData = [];
    this.allData = [];

    validSubData.forEach((item) => {

      this.submitData(item, (isSuccess) => {

        completedCount++;

        if (isSuccess) {
          successCount++;
        } else {
          failCount++;
        }

        if (completedCount === validSubData.length) {

          this.isSpinning = false;

          this.sussessUpLoad.suc = successCount;
          this.sussessUpLoad.fail = failCount;

          if (failCount === 0) {

            this.msgSvr.success("Success.");
            this.ref.close("refresh");

          } else {

            this.msgSvr.error(
              `${failCount} records failed to import. Check error report.`
            );
          }
        }
      });
    });
  }


  // ==========================================
  // DATE HELPERS
  // ==========================================

  /**
   * Converts any Excel cell value to "DD-MM-YYYY".
   * - Excel date cell  -> serial number -> UTC getters (no timezone shift)
   * - Text cell        -> trimmed; "D/M/YYYY" and "D.M.YYYY" are normalised (day first)
   * - JS Date          -> +12h so a 23:59:50 value still lands on the right day
   */
  formatDateValue(value: any): string {

    if (value === null || value === undefined || value === "") {
      return "";
    }

    const pad = (n: number) => String(n).padStart(2, '0');

    // EXCEL SERIAL NUMBER (real date cells)
    if (typeof value === 'number') {
      const d = new Date(Math.round((Math.floor(value) - 25569) * 86400 * 1000));
      return `${pad(d.getUTCDate())}-${pad(d.getUTCMonth() + 1)}-${d.getUTCFullYear()}`;
    }

    // JS DATE (only if cellDates is ever turned on again)
    if (value instanceof Date && !isNaN(value.getTime())) {
      const d = new Date(value.getTime() + 12 * 60 * 60 * 1000);
      return `${pad(d.getDate())}-${pad(d.getMonth() + 1)}-${d.getFullYear()}`;
    }

    // TEXT (typed as DD-MM-YYYY, also accepts D/M/YYYY and D.M.YYYY)
    if (typeof value === 'string') {
      const text = value.trim();
      const m = /^(\d{1,2})[-\/.](\d{1,2})[-\/.](\d{4})$/.exec(text);
      if (m) {
        return `${pad(+m[1])}-${pad(+m[2])}-${m[3]}`;
      }
      return text;
    }

    return "";
  }

  /** True only for a real calendar date written as DD-MM-YYYY. */
  private isValidDmy(value: string): boolean {

    const m = /^(\d{2})-(\d{2})-(\d{4})$/.exec(value || "");

    if (!m) {
      return false;
    }

    const day = +m[1];
    const month = +m[2];
    const year = +m[3];

    const dt = new Date(Date.UTC(year, month - 1, day));

    return (
      dt.getUTCFullYear() === year &&
      dt.getUTCMonth() === month - 1 &&
      dt.getUTCDate() === day
    );
  }

  /** "DD-MM-YYYY" -> UTC timestamp (call only after isValidDmy). */
  private dmyToTime(value: string): number {

    const m = /^(\d{2})-(\d{2})-(\d{4})$/.exec(value);

    return Date.UTC(+m[3], +m[2] - 1, +m[1]);
  }


  // ==========================================
  // SUBMIT SINGLE DATA
  // ==========================================

  submitData(item: any, callback: (success: boolean) => void) {

    // GET MEAL NAME
    const mealName = item["Meal Preference"]
      ? item["Meal Preference"].toString().trim()
      : "";

    // FIND MASTER ID
    const selectedMeal = this.getMealPreferenceByName(mealName);

    if (!selectedMeal) {

      item["Wrong Reason"] = "Invalid Meal Preference.";

      this.allData.push({
        ...this.templateData,
        ...item
      });

      callback(false);

      return;
    }

    // NAME -> ID
    const mealPreferenceId = selectedMeal.value;

    console.log("IMPORT MEAL:", mealName, "-> ID:", mealPreferenceId);

    // API PARAMS (API field names unchanged)
    const params = {

      Groups: item["Groups"] ? item["Groups"].toString() : "",

      MealPreference: mealPreferenceId,

      FxBadge: item["FX Badge"] ? item["FX Badge"].toString() : "",

      Name: item["Name"] ? item["Name"].toString() : "",

      FirstDateFxbl: this.formatDateValue(item["First Day at Factory"]),

      LastDateFxbl: this.formatDateValue(item["Last Day at Factory"]),

      AddBy:
        (
          this.apiSvr.getSettingUserInfo() &&
          this.apiSvr.getSettingUserInfo().eeid
        ) || "",

      Func: "CMS-BulkF1Menu"
    };

    console.log("BULK IMPORT PARAMS:", JSON.stringify(params));

    // API CALL
    this.apiSvr.iwxPostF1(
      params,
      {
        headers: {
          apiID: "ffff-1789627520107-10208193125-1282",
          userKey: "E9785F13CCEAFB251A6AB147AFD24336"
        }
      },
      true
    )
      .then((res: any) => {

        if (res && res.IsOK === '1') {

          item["Import Result"] = "Success.";

          this.allData.push({
            ...this.templateData,
            ...item
          });

          callback(true);

        } else {

          item["Wrong Reason"] = (res && res.Msg) ? res.Msg : "API Error";

          this.allData.push({
            ...this.templateData,
            ...item
          });

          callback(false);
        }
      })
      .catch((err) => {

        console.error("Import API error:", err);

        item["Wrong Reason"] = "Network or Server Error";

        this.allData.push({
          ...this.templateData,
          ...item
        });

        callback(false);
      });
  }
