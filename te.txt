async function getPicture() {

    const input = document.getElementById("imageId");
    const image = document.getElementById("resultImage");
    const status = document.getElementById("status");

    const id = input.value.trim();

    // Validate
    if (!id) {
        status.className = "error";
        status.textContent = "Please enter an image number.";
        image.style.display = "none";
        return;
    }
/*
    if (!/^\d+$/.test(id)) {
        status.className = "error";
        status.textContent = "Please enter numbers only.";
        image.style.display = "none";
        return;
    }*/

    status.className = "loading";
    status.textContent = "Searching FTP server...";

    image.style.display = "none";

    try {

        const response = await fetch(
            `/api/image/${encodeURIComponent(id)}`
        );

        if (!response.ok) {

            let message = "Image not found.";

            try {
                const data = await response.json();

                if (data.error) {
                    message = data.error;
                }
            } catch (e) {
                // Ignore JSON parsing error
            }

            throw new Error(message);
        }

        // Convert response to image blob
        const blob = await response.blob();

        const imageUrl = URL.createObjectURL(blob);

        image.onload = function () {
            URL.revokeObjectURL(imageUrl);
        };

        image.src = imageUrl;
        image.style.display = "inline-block";

        status.className = "success";
        status.textContent = `Image ${id} found.`;

    } catch (error) {

        image.style.display = "none";

        status.className = "error";
        status.textContent = error.message;

    }
}


// Allow pressing Enter
document.getElementById("imageId").addEventListener(
    "keydown",
    function(event) {

        if (event.key === "Enter") {
            getPicture();
        }

    }
);

</script>

const express = require("express");
const ftp = require("basic-ftp");
const dotenv = require("dotenv");
const path = require("path");
const cors = require("cors"); 
dotenv.config();

const app = express();
const PORT = 3001;

app.use(cors());   
app.use(express.static(path.join(__dirname, "public")));

// Get image from FTP
app.get("/api/image/:id", async (req, res) => {
    const id = req.params.id.trim();

    const possibleFiles = [
        `${id}.png`,
        `${id}.jpeg`,
        `${id}.jpg`
    ];

    const client = new ftp.Client();

    try {
        await client.access({
            host: process.env.FTP_HOST,
            port: Number(process.env.FTP_PORT || 21),
            user: process.env.FTP_USER,
            password: process.env.FTP_PASSWORD,
            secure: process.env.FTP_SECURE === "true"
        });

        await client.cd("/photos");

        let foundFile = null;
        const files = await client.list();

        for (const filename of possibleFiles) {
            const exists = files.some(
                file => file.name.toLowerCase() === filename.toLowerCase()
            );

            if (exists) {
                foundFile = filename;
                break;
            }
        }

        if (!foundFile) {
            return res.status(404).json({
                error: `No image found for ${id}`
            });
        }

        const chunks = [];

        const writable = new (require("stream").Writable)({
            write(chunk, encoding, callback) {
                chunks.push(chunk);
                callback();
            }
        });

        await client.downloadTo(writable, foundFile);

        const imageBuffer = Buffer.concat(chunks);

        const extension = path.extname(foundFile).toLowerCase();

        let contentType = "application/octet-stream";

        if (extension === ".png") {
            contentType = "image/png";
        } else if (extension === ".jpg" || extension === ".jpeg") {
            contentType = "image/jpeg";
        }

        res.setHeader("Content-Type", contentType);
        res.setHeader("Cache-Control", "no-cache");

        res.send(imageBuffer);

    } catch (error) {
        console.error("FTP error:", error);

        res.status(500).json({
            error: "Unable to retrieve image from FTP server."
        });

    } finally {
        client.close();
    }
});

app.listen(PORT, () => {
    console.log(`Image viewer running at http://localhost:${PORT}`);
});
