import * as CryptoJS from 'crypto-js';

/**
 * AES Encryption
 */
export function encrypt(
  str: string,
  keyStr?: string,
  ivStr?: string
): string {
  const key = keyStr
    ? CryptoJS.enc.Utf8.parse(keyStr)
    : CryptoJS.enc.Utf8.parse(
        'abcdefghijklmnopqrstuvwxyz123456'
      );

  const iv = ivStr
    ? CryptoJS.enc.Hex.parse(ivStr)
    : CryptoJS.enc.Hex.parse('0000000000000000');

  return CryptoJS.AES.encrypt(str, key, {
    iv,
    mode: CryptoJS.mode.CBC,
    padding: CryptoJS.pad.Pkcs7
  }).toString();
}

/**
 * AES Decryption
 */
export function decrypt(
  str: string,
  keyStr?: string,
  ivStr?: string
): string {
  if (!str || typeof str !== 'string') {
    throw new Error('Invalid encrypted input.');
  }

  const key = keyStr
    ? CryptoJS.enc.Utf8.parse(keyStr)
    : CryptoJS.enc.Utf8.parse(
        'abcdefghijklmnopqrstuvwxyz123456'
      );

  const iv = ivStr
    ? CryptoJS.enc.Hex.parse(ivStr)
    : CryptoJS.enc.Hex.parse('0000000000000000');

  const decrypted = CryptoJS.AES.decrypt(str, key, {
    iv,
    mode: CryptoJS.mode.CBC,
    padding: CryptoJS.pad.Pkcs7
  });

  if (decrypted.sigBytes <= 0) {
    throw new Error(
      'Decryption returned empty data. Verify the key and encryption format.'
    );
  }

  try {
    return decrypted.toString(CryptoJS.enc.Utf8);
  } catch (error) {
    console.error(
      'AES decryption failed. Verify the key, IV, mode and padding.'
    );

    throw error;
  }
}

/**
 * Application Security
 */
export const security = {
  cfg: {
    iv: CryptoJS.enc.Hex.parse('0000000000000000'),
    mode: CryptoJS.mode.CBC,
    padding: CryptoJS.pad.Pkcs7
  },

  encryptForWeb(str: string): string {
    const key = CryptoJS.enc.Utf8.parse(
      '794fc586a5a5a589c4ac47402f7419c2'
    );

    return CryptoJS.AES.encrypt(str, key, this.cfg).toString();
  },

  decryptForWeb(str: string): string {
    const key = CryptoJS.enc.Utf8.parse(
      '794fc586a5a5a589c4ac47402f7419c2'
    );

    const result = CryptoJS.AES.decrypt(
      str,
      key,
      this.cfg
    );

    return result.toString(CryptoJS.enc.Utf8);
  },

  encrypt(str: string, keyStr: string): string {
    const key = CryptoJS.enc.Utf8.parse(keyStr);

    return CryptoJS.AES.encrypt(
      str,
      key,
      this.cfg
    ).toString();
  },

  decrypt(str: string, keyStr: string): string {
    const key = CryptoJS.enc.Utf8.parse(keyStr);

    const result = CryptoJS.AES.decrypt(
      str,
      key,
      this.cfg
    );

    return result.toString(CryptoJS.enc.Utf8);
  },

  md5(str: string): string {
    return CryptoJS.MD5(str).toString().toUpperCase();
  },

  genVldCode(str: string, key: string): string {
    return this.encrypt(this.md5(str + key), key);
  }
};
