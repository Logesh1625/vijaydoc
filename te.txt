import { Component, OnInit } from '@angular/core';
import * as XLSX from "xlsx";
import { SharedModule } from "@shared";
import * as FileSaver from "file-saver";
import { PublicNetService } from "src/app/routes/a_Public_net/Public_net.service";
import { NzMessageService, UploadFile, NzDrawerRef } from "ng-zorro-antd";

@Component({
  selector: 'app-F1-Details-import',
  templateUrl: './F1-Details-import.html',
})
export class F1ManageImportComponent implements OnInit {
  fileList: UploadFile[] = [];
  addLoading: boolean = false;
  isSpinning: boolean = false;

  sussessUpLoad = {
    total: 0,
    suc: 0,
    fail: 0,
  };

  allData: any[] = [];
  subData: any[] = [];
  excelData: any[] = [];

  fileForm = {
    fileListStr: "",
  };

  siteOptions = [];

  templateData = {
    "Groups": '',
    "Sub Group": '',
    "Meal Preference": '',
    "FX Badge": '',
    "Name": '',
    "First Day at FXBL": '',
    "Last Day at FXBL": ''
  };

  constructor(
    private apiSvr: PublicNetService,
    private shareSvr: SharedModule,
    private msgSvr: NzMessageService,
    private ref: NzDrawerRef
  ) { }

  ngOnInit() { }

  beforeUpload = (file: UploadFile): boolean => {
    this.fileForm.fileListStr = file.name;
    this.fileList = [file];

    let fileType = file.name.substring(file.name.lastIndexOf("."), file.name.length).toLocaleUpperCase();

    if (fileType !== ".XLS" && fileType !== ".XLSX") {
      this.shareSvr.modalSvr.confirm({
        nzCancelText: null,
        nzContent: "Attachment format must be excel!",
        nzOnOk: () => {
          this.fileForm.fileListStr = "";
          this.fileList = [];
        },
      });
    } else {
      this.addLoading = false;
    }
    return false;
  };

  DownloadExcelfile() {
    const worksheet: XLSX.WorkSheet = XLSX.utils.json_to_sheet([this.templateData]);
    const workbook: XLSX.WorkBook = {
      Sheets: { template: worksheet },
      SheetNames: ["Template"],
    };
    const excelBuffer: any = XLSX.write(workbook, { bookType: "xlsx", type: "array" });
    const data: Blob = new Blob([excelBuffer], {
      type: "application/vnd.openxmlformats-officedocument.spreadsheetml.sheet;charset=UTF-8",
    });
    FileSaver.saveAs(data, "F1Meal-info Template.xlsx");
  }

  addFile() {
    this.addLoading = true;
    this.isSpinning = true;
    this.sussessUpLoad = { total: 0, suc: 0, fail: 0 };
    this.allData = [];
    this.subData = [];

    this.fileList.forEach((file) => {
      this.getFileData(file, (data) => {
        // ✅ cellDates: true — Excel date cells ah proper JS Date objects ah parse pண்ணும்
        const wb: XLSX.WorkBook = XLSX.read(data, { type: "binary", cellDates: true });
        const wsname: string = wb.SheetNames[0];
        const ws: XLSX.WorkSheet = wb.Sheets[wsname];
        const json = XLSX.utils.sheet_to_json(ws);
        this.batchAdd(json);
      });
    });
  }

  getFileData(file, callback: (fle: string) => void): void {
    const reader = new FileReader();
    reader.readAsBinaryString(file);
    reader.addEventListener("load", () => callback(reader.result!.toString()));
  }

  batchAdd(data: any[]) {
    this.excelData = data;
    this.sussessUpLoad.total = this.excelData.length;

    if (this.excelData && this.excelData.length > 0) {
      this.excelData.forEach((item) => {
        this.validateAndProcessRow(item);
      });
      this.isSpinning = false;
    } else {
      this.fileForm.fileListStr = "";
      this.fileList = [];
      this.isSpinning = false;
      this.msgSvr.error("Excel data does not exist");
    }
  }

  validateAndProcessRow(item: any) {
    const requiredKeys = ["Groups", "Sub Group", "Meal Preference", "FX Badge", "Name", "First Day at FXBL", "Last Day at FXBL"];

    const hasAllProps = requiredKeys.every((key) => item.hasOwnProperty(key));

    let rowRecord = { ...this.templateData, ...item };

    if (!hasAllProps) {
      this.sussessUpLoad.fail++;
      rowRecord["Wrong Reason"] = "All of template content is required.";
      this.allData.push(rowRecord);
      return;
    }

    if (item['Phone No']) {
      const phoneReg = /^[6789]\d{9}$/;
      if (!phoneReg.test(item['Phone No'])) {
        this.sussessUpLoad.fail++;
        rowRecord["Wrong Reason"] = "Phone Number format is incorrect.";
        this.allData.push(rowRecord);
        return;
      }
    }

    this.sussessUpLoad.suc++;
    this.allData.push(rowRecord);
    this.subData.push(rowRecord);
  }

  exportFailedExcel() {
    if (this.allData.length > 0) {
      this.apiSvr.exportHandle(this.allData, 68);
    } else {
      this.msgSvr.error("No Data.");
    }
  }

  submit() {
    if (this.subData.length === 0) {
      this.msgSvr.error("No valid data to submit. Please check your file.");
      return;
    }

    this.isSpinning = true;
    let completedCount = 0;
    let successCount = 0;
    let failCount = 0;

    const validSubData = [...this.subData];
    this.subData = [];
    this.allData = [];

    validSubData.forEach((item) => {
      this.submitData(item, (isSuccess) => {
        completedCount++;
        if (isSuccess) {
          successCount++;
        } else {
          failCount++;
        }

        if (completedCount === validSubData.length) {
          this.isSpinning = false;
          this.sussessUpLoad.suc = successCount;
          this.sussessUpLoad.fail = failCount;

          if (failCount === 0) {
            this.msgSvr.success("Success.");
            this.ref.close("refresh");
          } else {
            this.msgSvr.error(`${failCount} records failed to import. Check error report.`);
          }
        }
      });
    });
  }

  // ✅ Helper — Date object, string, illa Excel serial number ah safe ah 'DD-MM-YYYY' string ah maathும்
  formatDateValue(value: any): string {
    if (!value) return "";

    // Case 1: already a proper JS Date object (cellDates: true set pண்ணின apparam idhே varும்)
    if (value instanceof Date && !isNaN(value.getTime())) {
      const day = String(value.getDate()).padStart(2, '0');
      const month = String(value.getMonth() + 1).padStart(2, '0');
      const year = value.getFullYear();
      return `${day}-${month}-${year}`;
    }

    // Case 2: already a string (manually typed in Excel, e.g. "03-09-2026")
    if (typeof value === 'string') {
      return value.trim();
    }

    // Case 3: fallback — Excel serial number
    if (typeof value === 'number') {
      const utcDays = Math.floor(value - 25569);
      const utcValue = utcDays * 86400;
      const dateInfo = new Date(utcValue * 1000);
      const day = String(dateInfo.getDate()).padStart(2, '0');
      const month = String(dateInfo.getMonth() + 1).padStart(2, '0');
      const year = dateInfo.getFullYear();
      return `${day}-${month}-${year}`;
    }

    return "";
  }

  submitData(item, callback: (success: boolean) => void) {
    const params = {
      Groups: item["Groups"] ? item["Groups"].toString() : "",
      SubGroup: item["Sub Group"] ? item["Sub Group"].toString() : "",
      MealPreference: item["Meal Preference"] ? item["Meal Preference"].toString() : "",
      FxBadge: item["FX Badge"] ? item["FX Badge"].toString() : "",
      Name: item["Name"] ? item["Name"].toString() : "",
      FirstDateFxbl: this.formatDateValue(item["First Day at FXBL"]),   // ✅ fixed
      LastDateFxbl: this.formatDateValue(item["Last Day at FXBL"]),     // ✅ fixed
      AddBy: (this.apiSvr.getSettingUserInfo() && this.apiSvr.getSettingUserInfo().eeid) || "",   // ✅ fixed
      Func: "CMS-BulkF1Menu",
    };

    this.apiSvr.iwxPostF1(params, {
      headers: {
        apiID: "ffff-1789627520107-10208193125-1282",
        userKey: "E9785F13CCEAFB251A6AB147AFD24336",
      }
    }, true).then((res: any) => {
      if (res && res.IsOK === '1') {
        item["Import Result"] = "Success.";
        this.allData.push({ ...this.templateData, ...item });
        callback(true);
      } else {
        item["Wrong Reason"] = (res && res.Msg) ? res.Msg : "API Error";
        this.allData.push({ ...this.templateData, ...item });
        callback(false);
      }
    }).catch(err => {
      item["Wrong Reason"] = "Network or Server Error";
      this.allData.push({ ...this.templateData, ...item });
      callback(false);
    });
  }
}
