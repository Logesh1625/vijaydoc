// ============================================================
// GET DECRYPTION KEY
// ============================================================

export function getdk(encodedKey) {
  if (
    typeof encodedKey !== 'string' ||
    !encodedKey
  ) {
    throw new Error('Invalid encoded encryption key');
  }

  const decoded = window.atob(encodedKey);
  const middle = Math.floor(decoded.length / 2);

  let left = middle - 1;

  let right =
    decoded.length % 2 === 0
      ? middle
      : middle + 1;

  let result = '';

  while (
    left >= 0 &&
    right < decoded.length
  ) {
    result += decoded[right];
    result += decoded[left];

    left--;
    right++;
  }

  return (
    result +
    decoded.slice(right) +
    decoded.slice(0, left + 1)
  );
}

// ============================================================
// AES ENCRYPTION
// ============================================================

export const encrypt = (value, key) => {
  if (
    value === null ||
    value === undefined
  ) {
    throw new Error('Cannot encrypt an empty value');
  }

  if (
    typeof key !== 'string' ||
    !key
  ) {
    throw new Error('Invalid encryption key');
  }

  return CryptoJS.AES.encrypt(
    String(value),
    CryptoJS.enc.Utf8.parse(key),
    {
      iv: CryptoJS.enc.Hex.parse(
        '0000000000000000'
      ),
      mode: CryptoJS.mode.CBC,
      padding: CryptoJS.pad.Pkcs7,
    }
  ).toString();
};

// ============================================================
// AES DECRYPTION
// ============================================================

export const decrypt = (value, key) => {
  try {
    if (
      typeof value !== 'string' ||
      !value.trim()
    ) {
      throw new Error(
        'Encrypted response is empty'
      );
    }

    if (
      typeof key !== 'string' ||
      !key
    ) {
      throw new Error('Invalid decryption key');
    }

    const result = CryptoJS.AES.decrypt(
      value,
      CryptoJS.enc.Utf8.parse(key),
      {
        iv: CryptoJS.enc.Hex.parse(
          '0000000000000000'
        ),
        mode: CryptoJS.mode.CBC,
        padding: CryptoJS.pad.Pkcs7,
      }
    );

    const plaintext = result.toString(
      CryptoJS.enc.Utf8
    );

    if (!plaintext) {
      throw new Error(
        'Unable to decrypt response'
      );
    }

    return plaintext;
  } catch (error) {
    console.error(
      'Decryption failed:',
      error.message
    );

    return null;
  }
};

// ============================================================
// ENCRYPT API PARAMETERS
// ============================================================

export const dealPostParamsEncrypt = (
  data = {}
) => {
  const key = getdk(iwxKey);

  return {
    o: encrypt(
      JSON.stringify({
        Os: '',
        DeviceId: 'Utility',
        AppVer: '0',
      }),
      key
    ),

    s: encrypt(
      JSON.stringify(data),
      key
    ),
  };
};

// ============================================================
// NORMAL API PARAMETERS
// ============================================================

export const dealPostParams = (
  data = {}
) => {
  const params = Object.entries(data)
    .map(([key, value]) => {
      return (
        `${key}=${String(value ?? '')
          .replace(/=/g, '%3D')
          .replace(/&/g, '%26')}`
      );
    })
    .join('&');

  return {
    inputString: params,
  };
};

// ============================================================
// ENCRYPTED API POST
// ============================================================

export const iwxPost = async (
  params = {},
  config = {},
  useEncryption = false
) => {
  const body = useEncryption
    ? dealPostParamsEncrypt(params)
    : dealPostParams(params);

  const response = await axios.post(
    'https://microservice.abgind.foxconn.com/india/gateway',
    body,
    {
      headers: {
        ...(config.headers || {}),
      },
    }
  );

  if (!useEncryption) {
    return response.data;
  }

  const encryptedResponse =
    response?.data?.r;

  if (
    typeof encryptedResponse !== 'string' ||
    !encryptedResponse.trim()
  ) {
    throw new Error(
      'Encrypted API response field "r" is missing'
    );
  }

  const plaintext = decrypt(
    encryptedResponse,
    getdk(iwxKey)
  );

  if (!plaintext) {
    throw new Error(
      'Unable to decrypt API response'
    );
  }

  try {
    return JSON.parse(plaintext);
  } catch {
    throw new Error(
      'Decrypted API response is not valid JSON'
    );
  }
};

export const getuserAccount = () => {
  try {
    const storedData =
      sessionStorage.getItem('data');

    if (!storedData) {
      return null;
    }

    const plaintext = decrypt(
      storedData,
      getdk(iwxKey)
    );

    if (!plaintext) {
      return null;
    }

    const result = JSON.parse(plaintext);

    return (
      result?.List?.[0]?.AccountId ?? null
    );
  } catch (error) {
    console.error(
      'Unable to get user account:',
      error.message
    );

    return null;
  }
};

          <Box className={classes.formm}>
            <img
              src="/images/logo1.png"
              alt="Company Logo"
              style={{
                maxWidth: '180px',
                height: 'auto',
              }}
            />

            {isMfaScreen ? (
              <Box>
                <Typography
                  sx={{
                    fontWeight: 600,
                    fontSize: '1rem',
                  }}
                >
                  MFA Auth
                </Typography>

                {isNewMfa && (
                  <Box sx={{ mt: 2 }}>
                    {secretLoading ? (
                      <CircularProgress
                        size={28}
                      />
                    ) : (
                      showQRCode &&
                      secretkey && (
                        <>
                          <QRCodeCanvas
                            value={generateOTPURI()}
                            size={128}
                          />

                          <Typography
                            sx={{
                              fontSize: '10px',
                              overflowWrap: 'anywhere',
                              mt: 1,
                            }}
                          >
                            Scan the QR code with
                            your authenticator app.
                          </Typography>
                        </>
                      )
                    )}
                  </Box>
                )}

                {errorMessage && (
                  <Typography
                    color="error"
                    sx={{ mt: 2 }}
                  >
                    {errorMessage}
                  </Typography>
                )}

                <TextField
                  margin="normal"
                  fullWidth
                  label="Enter OTP"
                  value={otp}
                  onChange={(event) => {
                    setOTP(
                      event.target.value.replace(
                        /\D/g,
                        ''
                      )
                    );
                  }}
                  inputProps={{
                    inputMode: 'numeric',
                    maxLength: 8,
                  }}
                />

                <Button
                  fullWidth
                  variant="contained"
                  sx={{ mt: 2, mb: 2 }}
                  disabled={
                    !otp.trim() ||
                    otploading
                  }
                  onClick={verifyOTP}
                >
                  {otploading ? (
                    <CircularProgress
                      size={24}
                    />
                  ) : (
                    'Verify OTP'
                  )}
                </Button>

                <Button
                  fullWidth
                  variant="contained"
                  onClick={handlebacklogin}
                >
                  Back to Login
                </Button>
              </Box>
            ) : (
              <form onSubmit={handleLogin}>
                <Typography
                  sx={{
                    fontWeight: 700,
                    fontSize: '1rem',
                    mb: 2,
                  }}
                >
                  Login
                </Typography>

                <TextField
                  margin="normal"
                  fullWidth
                  label="Username"
                  value={account}
                  autoComplete="username"
                  onChange={(event) => {
                    setAccount(
                      event.target.value
                    );
                  }}
                />

                <TextField
                  margin="normal"
                  fullWidth
                  label="Password"
                  type={
                    showPassword
                      ? 'text'
                      : 'password'
                  }
                  value={password}
                  autoComplete="current-password"
                  onChange={(event) => {
                    setPassword(
                      event.target.value
                    );
                  }}
                  InputProps={{
                    endAdornment: (
                      <InputAdornment
                        position="end"
                      >
                        <IconButton
                          onClick={() => {
                            setShowPassword(
                              (previous) =>
                                !previous
                            );
                          }}
                          edge="end"
                          aria-label="Toggle password visibility"
                        >
                          {showPassword ? (
                            <Visibility />
                          ) : (
                            <VisibilityOff />
                          )}
                        </IconButton>
                      </InputAdornment>
                    ),
                  }}
                />

                {errorMessage && (
                  <Typography
                    color="error"
                    sx={{ mt: 2 }}
                  >
                    {errorMessage}
                  </Typography>
                )}

                <Button
                  type="submit"
                  fullWidth
                  variant="contained"
                  sx={{ mt: 3, mb: 2 }}
                  disabled={isLoading}
                >
                  {isLoading ? (
                    <CircularProgress
                      size={24}
                    />
                  ) : (
                    'Login'
                  )}
                </Button>
              </form>
            )}
          </Box>
        </Box>
      </Grid>

      <Grid
        item
        xs={12}
        sx={{
          display: 'flex',
          alignItems: 'center',
          justifyContent: 'center',
          textAlign: 'center',
          padding: 2,
          fontSize: '12px',
          color: '#000',
        }}
      >
        <div>
          <div>
            FXCN IT: Tel:533-26038 |
            Email:idpbg-it-tn-dev@mail.foxconn.com
          </div>

          <div>
            Copyright 2020-2025 Foxconn
            A Business Group IT.
            All Rights Reserved
          </div>
        </div>
      </Grid>
    </Grid>
  );
};

export default Login;
