export const encrypt = (str, key) => {
  const k = CryptoJS.enc.Utf8.parse(key);
  return CryptoJS.AES.encrypt(str, k, {
    iv: CryptoJS.enc.Hex.parse('0000000000000000'),
    mode: CryptoJS.mode.CBC,
    padding: CryptoJS.pad.Pkcs7,
  }).toString();
};

export const decrypt = (str, key) => {
  try {
    const k = CryptoJS.enc.Utf8.parse(key);
    const decrypted = CryptoJS.AES.decrypt(str, k, {
      iv: CryptoJS.enc.Hex.parse('0000000000000000'),
      mode: CryptoJS.mode.CBC,
      padding: CryptoJS.pad.Pkcs7,
    }).toString(CryptoJS.enc.Utf8);
    return decrypted;
  } catch (error) {
    console.error('Decryption error:', error);
    return null;
  }
};

export const md5 = (str) => {
  return CryptoJS.MD5(str)
    .toString()
    .toUpperCase();
};

export const dealPostParams = (data = {}) => {
  let params = "";
  let res = null;

  for (const key in data) {
    // Replacing "=" and "&" with encoded values
    data[key] = data[key].replace("=", "%3D").replace("&", "%26");
    // Appending key-value pair to params string
    params += `${key}=${data[key]}&`;
  }

  // Removing the last "&"
  params = params.substring(0, params.length - 1);

  // Returning the resulting params as an object
  res = {
    inputString: params,
  };

  return res;
};

// ======================================================
// Gateway switch: change only the GATEWAY_URL line.
//   GATEWAYS.micro : the gateway that worked before
//   GATEWAYS.local : http://10.208.193.125:8086 (needs every apiID
//                    registered there with your userKey)
// ======================================================
// ======================================================
// Gateway selection (automatic, no .env needed)
//   npm start      -> development -> http://10.208.193.125:8086  (test)
//   npm run build  -> production  -> microservice gateway        (live)
// ======================================================
const GATEWAYS = {
  micro: "https://microservice.abgind.foxconn.com/india/gateway",
  local: "http://10.208.193.125:8086/gateway/manager",
};

const GATEWAY_URL =
  process.env.NODE_ENV === 'production' ? GATEWAYS.micro : GATEWAYS.local;



export const iwxPost = async (params = {}, config = {}, isEncrypt) => {
  const param = isEncrypt ? dealPostParamsEncrypt(params) : dealPostParams(params);

  try {
    const result = await axios.post(GATEWAY_URL, param, {
      headers: {
        ...config.headers, // apiID and userKey sent by the caller
      },
    });

    // Encrypted responses come back as { r: "..." }
    const res = isEncrypt
      ? JSON.parse(decrypt(result.data.r, getdk(iwxKey)))
      : result.data;

    return res;
  } catch (error) {
    console.error("Request failed:", error);
  }
};
