DownloadExcelfile() {

  const headers = [
    "Groups",
    "Sub Group",
    "Meal Preference",
    "FX Badge",
    "Name",
    "First Day at FXBL",
    "Last Day at FXBL"
  ];

  // DB-la irundhu Meal Preference names
  const mealNames = this.mealPreferenceList
    .map((item: any) => item.label)
    .filter((x: any) => x)
    .map((x: any) => String(x).trim());

  console.log("Meal Preference Dropdown:", mealNames);

  XlsxPopulate.fromBlankAsync()
    .then((workbook: any) => {

      const sheet = workbook.sheet(0);
      sheet.name("Template");

      // =========================
      // HEADER
      // =========================
      sheet.range("A1:G1").value([headers]);

      // Header style
      sheet.range("A1:G1").style({
        bold: true,
        horizontalAlignment: "center",
        verticalAlignment: "center",
        wrapText: true
      });

      // =========================
      // COLUMN WIDTH
      // =========================
      sheet.column("A").width(20);
      sheet.column("B").width(20);
      sheet.column("C").width(25);
      sheet.column("D").width(20);
      sheet.column("E").width(30);
      sheet.column("F").width(22);
      sheet.column("G").width(22);

      // =========================
      // DATE FORMAT
      // =========================
      sheet.range("F2:G500").style(
        "numberFormat",
        "dd-mm-yyyy"
      );

      // =========================
      // MEAL PREFERENCE DROPDOWN
      // =========================
      if (mealNames.length > 0) {

        // Hidden master sheet
        const masterSheet = workbook.addSheet("MealMaster");

        masterSheet.cell("A1").value("Meal Preference");

        mealNames.forEach((name: string, index: number) => {
          masterSheet
            .cell(`A${index + 2}`)
            .value(name);
        });

        // Hide master sheet
        masterSheet.hidden(true);

        const lastRow = mealNames.length + 1;

        // Dropdown source
        const validationFormula =
          `'MealMaster'!$A$2:$A$${lastRow}`;

        sheet
          .range("C2:C500")
          .dataValidation({
            type: "list",
            allowBlank: true,
            showErrorMessage: true,
            error: "Please select a valid Meal Preference.",
            errorTitle: "Invalid Meal Preference",
            formula1: validationFormula
          });
      }

      // =========================
      // UNLOCK DATA ROWS
      // =========================
      sheet.range("A2:G500").style(
        "locked",
        false
      );

      // =========================
      // LOCK HEADER
      // =========================
      sheet.range("A1:G1").style(
        "locked",
        true
      );

      // =========================
      // FREEZE HEADER
      // =========================
      sheet.freezePanes(0, 1);

      // =========================
      // OUTPUT
      // =========================
      return workbook.outputAsync();
    })
    .then((data: any) => {

      const blob = new Blob(
        [data],
        {
          type:
            "application/vnd.openxmlformats-officedocument.spreadsheetml.sheet"
        }
      );

      FileSaver.saveAs(
        blob,
        "F1Meal-info Template.xlsx"
      );

    })
    .catch((error: any) => {

      console.error(
        "Excel Template Error:",
        error
      );

      this.msgSvr.error(
        "Unable to generate Excel template."
      );

    });
}
