PROCEDURE GET_F1_MENU_BY_FXBADGE(
    V_FX_BADGE IN VARCHAR2,
    V_SITE IN VARCHAR2,
    V_EMP_ID IN VARCHAR2,
    V_RES OUT VARCHAR2,
    V_RESULT OUT VARCHAR2,
    P_CURSOR OUT SYS_REFCURSOR
)
IS

    V_COUNT NUMBER := 0;
    V_ALREADY NUMBER := 0;
    V_USER_SITE_COUNT NUMBER := 0;

BEGIN

    /* ============================================================
       1. CHECK ACTIVE CMS USER
       ============================================================ */

    SELECT COUNT(1)
    INTO V_USER_SITE_COUNT
    FROM CMS.C_USER_MANAGE_T U
    WHERE UPPER(TRIM(U.EMP_ID))
          =
          UPPER(TRIM(V_EMP_ID))

      AND U.SYSTEM_TYPE = 'CMS'

      AND NVL(U.DEL_FLAG, 0) = 0

      AND U.SITE IS NOT NULL;


    IF V_USER_SITE_COUNT = 0 THEN

        V_RES := '0';

        V_RESULT :=
            'Active CMS user/site not found.';

        OPEN P_CURSOR FOR
        SELECT
            NULL AS FX_BADGE
        FROM DUAL
        WHERE 1 = 0;

        RETURN;

    END IF;


    /* ============================================================
       2. VALIDATE SELECTED SITE BELONGS TO USER
       ============================================================ */

    IF V_SITE IS NOT NULL
       AND TRIM(V_SITE) <> ''
       AND UPPER(TRIM(V_SITE)) <> 'ALL'
    THEN

        SELECT COUNT(1)
        INTO V_USER_SITE_COUNT
        FROM CMS.C_USER_MANAGE_T U
        WHERE UPPER(TRIM(U.EMP_ID))
              =
              UPPER(TRIM(V_EMP_ID))

          AND U.SYSTEM_TYPE = 'CMS'

          AND NVL(U.DEL_FLAG, 0) = 0

          AND U.SITE IS NOT NULL

          AND UPPER(TRIM(U.SITE))
              =
              UPPER(TRIM(V_SITE));


        IF V_USER_SITE_COUNT = 0 THEN

            V_RES := '0';

            V_RESULT :=
                'Selected site is not assigned to this user.';

            OPEN P_CURSOR FOR
            SELECT
                NULL AS FX_BADGE
            FROM DUAL
            WHERE 1 = 0;

            RETURN;

        END IF;

    END IF;


    /* ============================================================
       3. CHECK BADGE
       ============================================================ */

    SELECT COUNT(1)
    INTO V_COUNT
    FROM CMS.C_F1_MEAL_INFO M
    WHERE UPPER(TRIM(M.FX_BADGE))
          =
          UPPER(TRIM(V_FX_BADGE))

      AND NVL(M.DEL_FLAG, 0) = 0

      /* Site */
      AND
      (
          V_SITE IS NULL
          OR TRIM(V_SITE) = ''
          OR UPPER(TRIM(V_SITE)) = 'ALL'
          OR UPPER(TRIM(M.PROPERTY_01))
             =
             UPPER(TRIM(V_SITE))
      )

      /* User allowed site */
      AND EXISTS
      (
          SELECT 1
          FROM CMS.C_USER_MANAGE_T U
          WHERE UPPER(TRIM(U.EMP_ID))
                =
                UPPER(TRIM(V_EMP_ID))

            AND U.SYSTEM_TYPE = 'CMS'

            AND NVL(U.DEL_FLAG, 0) = 0

            AND U.SITE IS NOT NULL

            AND UPPER(TRIM(U.SITE))
                =
                UPPER(TRIM(M.PROPERTY_01))
      );


    /* ============================================================
       4. BADGE NOT FOUND IN SELECTED SITE
       ============================================================ */

    IF V_COUNT = 0 THEN

        V_RES := '0';

        V_RESULT :=
            'FX Badge not found in selected site.';

        OPEN P_CURSOR FOR
        SELECT
            NULL AS FX_BADGE
        FROM DUAL
        WHERE 1 = 0;

        RETURN;

    END IF;


    /* ============================================================
       5. ALREADY SCANNED TODAY
       ============================================================ */

    SELECT COUNT(1)
    INTO V_ALREADY
    FROM CMS.C_F1_SCAN_LOG S
    WHERE UPPER(TRIM(S.FX_BADGE))
          =
          UPPER(TRIM(V_FX_BADGE))

      AND TRUNC(S.MEAL_DATE)
          =
          TRUNC(SYSDATE)

      AND NVL(S.DEL_FLAG, 0) = 0

      AND S.COLLECTED_DATA = 'SUCCESS'

      AND
      (
          V_SITE IS NULL
          OR TRIM(V_SITE) = ''
          OR UPPER(TRIM(V_SITE)) = 'ALL'
          OR UPPER(TRIM(S.PROPERTY_01))
             =
             UPPER(TRIM(V_SITE))
      );


    /* ============================================================
       6. GET LATEST ACTIVE RECORD
       ============================================================ */

    OPEN P_CURSOR FOR

    SELECT

        V.GROUPS,

        V.SUB_GROUP,

        V.PROPERTY_01 AS SITE,

        BC.NAME AS MEAL_PREFERENCE,

        V.FX_BADGE,

        V.NAME,

        TO_CHAR(
            V.FIRST_DATE_FXBL,
            'DD-MM-YYYY'
        ) AS FIRST_DATE_FXBL,

        TO_CHAR(
            V.LAST_DATE_FXBL,
            'DD-MM-YYYY'
        ) AS LAST_DATE_FXBL,

        CASE

            WHEN UPPER(TRIM(BC.NAME))
                 =
                 'NO MEAL REQUIRED'

            THEN 'NOT_ENTITLED'

            WHEN V.FIRST_DATE_FXBL IS NULL
              OR V.LAST_DATE_FXBL IS NULL

            THEN 'NOT_UPDATED'

            WHEN TRUNC(SYSDATE)
                 NOT BETWEEN
                 TRUNC(V.FIRST_DATE_FXBL)
                 AND
                 TRUNC(V.LAST_DATE_FXBL)

            THEN 'LAST DATE EXPIRED'

            WHEN V_ALREADY > 0

            THEN 'ALREADY_SCANNED'

            ELSE 'ENTITLED'

        END AS ENTITLEMENT_STATUS

    FROM
    (
        SELECT *
        FROM CMS.C_F1_MEAL_INFO M
        WHERE UPPER(TRIM(M.FX_BADGE))
              =
              UPPER(TRIM(V_FX_BADGE))

          AND NVL(M.DEL_FLAG, 0) = 0

          AND
          (
              V_SITE IS NULL
              OR TRIM(V_SITE) = ''
              OR UPPER(TRIM(V_SITE)) = 'ALL'
              OR UPPER(TRIM(M.PROPERTY_01))
                 =
                 UPPER(TRIM(V_SITE))
          )

        ORDER BY
            M.ADD_DATE DESC,
            M.ID DESC

    ) V

    LEFT JOIN CMS.C_BASE_CODE_T BC
        ON TO_CHAR(V.MEAL_PREFERENCE)
           =
           TO_CHAR(BC.ID)

       AND BC.TYPE_ID = 81

    WHERE ROWNUM = 1;


    V_RES := '1';

    V_RESULT := 'Success';


EXCEPTION

    WHEN OTHERS THEN

        V_RES := '0';

        V_RESULT :=
            'Error: ' || SQLERRM;

END GET_F1_MENU_BY_FXBADGE;
