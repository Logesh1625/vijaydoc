DownloadExcelfile() {

  // DB master values
  const mealValues = (this.mealPreferenceList || [])
    .map((item: any) => item.label)
    .filter((value: any) => value);

  if (mealValues.length === 0) {
    this.msgSvr.warning(
      "Meal Preference master data is not loaded yet. Please try again."
    );
    return;
  }

  const workbook = new ExcelJS.Workbook();

  // Main Template sheet
  const worksheet = workbook.addWorksheet("Template");

  // Headers
  worksheet.addRow([
    "Groups",
    "Sub Group",
    "Meal Preference",
    "FX Badge",
    "Name",
    "First Day at FXBL",
    "Last Day at FXBL"
  ]);

  // Header style
  const headerRow = worksheet.getRow(1);

  headerRow.eachCell((cell: any) => {
    cell.font = {
      bold: true
    };

    cell.alignment = {
      horizontal: "center",
      vertical: "middle"
    };
  });

  // Column widths
  worksheet.getColumn(1).width = 20;
  worksheet.getColumn(2).width = 20;
  worksheet.getColumn(3).width = 25;
  worksheet.getColumn(4).width = 20;
  worksheet.getColumn(5).width = 30;
  worksheet.getColumn(6).width = 22;
  worksheet.getColumn(7).width = 22;

  /*
   * Meal Preference dropdown
   *
   * C2:C500
   */
  const dropdownFormula =
    '"' + mealValues.join(",") + '"';

  for (let row = 2; row <= 500; row++) {

    worksheet.getCell(row, 3).dataValidation = {
      type: "list",
      allowBlank: true,
      formulae: [dropdownFormula],
      showErrorMessage: true,
      errorTitle: "Invalid Meal Preference",
      error: "Please select Meal Preference from the dropdown."
    };

  }

  // Date format
  for (let row = 2; row <= 500; row++) {
    worksheet.getCell(row, 6).numFmt = "dd-mm-yyyy";
    worksheet.getCell(row, 7).numFmt = "dd-mm-yyyy";
  }

  // Freeze header
  worksheet.views = [
    {
      state: "frozen",
      ySplit: 1
    }
  ];

  // Generate Excel
  workbook.xlsx.writeBuffer().then((buffer: any) => {

    const blob = new Blob(
      [buffer],
      {
        type:
          "application/vnd.openxmlformats-officedocument.spreadsheetml.sheet"
      }
    );

    FileSaver.saveAs(
      blob,
      "F1Meal-info Template.xlsx"
    );

  }).catch((error: any) => {

    console.error(
      "Excel generation error:",
      error
    );

    this.msgSvr.error(
      "Unable to generate Excel template."
    );

  });
}
