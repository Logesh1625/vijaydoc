submit(userInfo?: string): void {
  this.loading = true;

  try {
    if (
      !userInfo ||
      typeof userInfo !== "string"
    ) {
      throw new Error(
        "OUAC UserInfo is empty or invalid."
      );
    }

    console.log(
      "OUAC UserInfo received. Length:",
      userInfo.length
    );

    // Check the received value format.
    console.log(
      "UserInfo starts with:",
      userInfo.substring(0, 20)
    );

    const decryptKey =
      this.publicSvr.security.getDK(
        "NzQxRTMzNzlBODA0MjM3OEY4NEE4MDc4NzNCMUJCNDk="
      );

    if (!decryptKey) {
      throw new Error(
        "OUAC decryption key is empty."
      );
    }

    console.log(
      "OUAC decryption key received."
    );

    // Decrypt only using the verified OUAC configuration.
    const decryptedUserInfo =
      security.decrypt(
        userInfo,
        decryptKey
      );

    if (!decryptedUserInfo) {
      throw new Error(
        "OUAC decryption returned empty data."
      );
    }

    console.log(
      "OUAC decryption successful."
    );

    const data = JSON.parse(
      decryptedUserInfo
    );

    console.log(
      "OUAC UserInfo JSON parsed successfully."
    );

    const userInfo2 =
      security.encrypt(
        JSON.stringify(data),
        "6jxaHMIhvefCOokm5XbDdW4qzl30U7Kt"
      );

    const param = {
      userInfo: userInfo2
    };

    const body =
      this.publicSvr.security.encryptForWeb(
        JSON.stringify(param)
      );

    this.http.post(
      "/oneAccountLogin?_allow_anonymous=true",
      body,
      null,
      {
        headers: {
          "Content-Type":
            "application/json;charset=UTF-8"
        }
      }
    ).subscribe(
      (response: any) => {
        try {
          const decryptedResponse =
            this.publicSvr.security.decryptForWeb(
              response
            );

          const res = JSON.parse(
            decryptedResponse
          );

          if (res.status === 200) {
            const user = {
              avatar: "./assets/tmp/img/avatar.jpg",
              eeid: res.data.eeid,
              name: res.data.empName,
              time: new Date().getTime()
            };

            this.loginSuccess(
              user,
              res.data.JsessionId,
              res.data.ClientIp
            );
          } else {
            this.msg.error(
              res.msg || "Login failed."
            );
          }
        } catch (error) {
          console.error(
            "Login response processing failed:",
            error
          );

          this.msg.error(
            "Unable to process login response."
          );
        } finally {
          this.loading = false;
        }
      },
      (error) => {
        this.loading = false;

        console.error(
          "Login API error:",
          error
        );

        this.msg.error(
          "Login failed. Please try again."
        );
      }
    );

  } catch (error) {
    this.loading = false;

    console.error(
      "Login processing error:",
      error
    );

    this.msg.error(
      error && error.message
        ? error.message
        : "Unable to process login information."
    );
  }
}











@HostListener("window:message", ["$event"])
onMessage(event: MessageEvent): void {
  const info: any = event.data;

  console.log("OUAC callback received:", {
    msgType: info && info.msgType,
    hasUserInfo: !!(info && info.UserInfo),
    userInfoLength:
      info && typeof info.UserInfo === "string"
        ? info.UserInfo.length
        : 0
  });

  if (!info || info.msgType !== "login") {
    return;
  }

  if (
    !info.UserInfo ||
    typeof info.UserInfo !== "string"
  ) {
    this.loading = false;

    this.msg.error(
      "Login information was not received."
    );

    return;
  }

  this.submit(info.UserInfo);
}









decrypt(str, key) {
  const k = CryptoJS.enc.Utf8.parse(key);

  return CryptoJS.enc.Utf8.stringify(
    CryptoJS.AES.decrypt(str, k, this.cfg)
  );
}
