export class UserLoginComponent implements OnInit, OnDestroy {

  loading = false;

  iframeSrc: string = '';

  iframeSafeSrc: any;


  constructor(
    fb: FormBuilder,
    modalSrv: NzModalService,

    private router: Router,

    @Optional()
    @Inject(ReuseTabService)
    private reuseTabService: ReuseTabService,

    @Inject(DA_SERVICE_TOKEN)
    private tokenService: ITokenService,

    private settingService: SettingsService,

    private cookieService: CookieService,

    public location: PlatformLocation,

    private publicSvr: PublicNetService,

  ) {

    modalSrv.closeAll();
  }


  // =====================================================
  // Fields
  // =====================================================

  get userName() {
    return this.form.controls.userName;
  }

  get password() {
    return this.form.controls.pwdInfo;
  }

  get remember() {
    return this.form.controls.remember;
  }


  form: FormGroup;

  error = "2";

  type = 0;


  // =====================================================
  // Captcha
  // =====================================================

  count = 0;

  interval$: any;


  switch(ret: any) {

    this.type = ret.index;

  }


  // =====================================================
  // LOGIN
  // =====================================================

  submit(userInfo?: string) {

    try {

      // One Account response decrypt
      const data = JSON.parse(
        security.decrypt(
          userInfo,
          this.publicSvr.security.getDK(
            'NzQxRTMzNzlBODA0MjM3OEY4NEE4MDc4NzNCMUJCNDk='
          )
        )
      );


      const userInfo2 =
        security.encrypt(
          JSON.stringify(data),
          '6jxaHMIhvefCOokm5XbDdW4qzl30U7Kt'
        );


      const param = {

        url: "/oneAccountLogin?_allow_anonymous=true",

        data: {

          userInfo: userInfo2

        }

      };


      const body =
        security.encryptForWeb(
          JSON.stringify(param.data)
        );


      this.http
        .post(
          param.url,
          body,
          null,
          {
            headers: {
              "Content-Type":
                "application/json;charset=UTF-8"
            }
          }
        )
        .subscribe(

          (data) => {

            const res =
              JSON.parse(
                security.decryptForWeb(data)
              );


            this.loading = false;


            if (res.status === 200) {

              const user = {

                avatar:
                  "./assets/tmp/img/avatar.jpg",

                eeid:
                  res.data.eeid,

                name:
                  res.data.empName,

                time:
                  new Date().getTime()

              };


              this.loginSuccess(
                user,
                res.data.JsessionId,
                res.data.ClientIp
              );

            }

            else {

              this.msg.error(res.msg);


              let fromInfo =
                this.location.href
                  .indexOf("?from=iportal");


              if (fromInfo > -1) {

                this.router.navigateByUrl(
                  "/passport/login"
                );

                localStorage.removeItem(
                  "fromInfo"
                );

                localStorage.removeItem(
                  "toPageInfo"
                );
              }


              const iframeSrc =
                `${ouacServiceUrl}` +
                `&CallBack=/assets/login-callback.html?UserInfo` +
                `&CssLink=/assets/sso-login.css` +
                `&State=RxFdv2p7KsDq38CX` +
                `&SSO=0`;


              const iframe =
                document.querySelector(
                  '.login-iframe'
                );

              if (iframe) {

                iframe.setAttribute(
                  'src',
                  iframeSrc
                );
              }
            }

          },

          (error) => {

            this.loading = false;

            console.error(
              "Login API error:",
              error
            );

            this.msg.error(
              "Login failed. Please try again."
            );

          }

        );

    }

  async loginSuccess(

  ) {

    // User information
    this.settingService.setUser({

    });


    // Cookies
    this.cookieService.set(
      "USER_NO",
      this.publicSvr.security.encryptForWeb(
        user.eeid
      )
    );


    this.cookieService.set(
      "USER_NAME",
      this.publicSvr.security.encryptForWeb(
        user.name
      )
    );


    this.cookieService.set(
      "IS_FOREIGN",
      ''
    );


    this.cookieService.set(
      "SESSION_ID",
      ''
    );


    this.cookieService.set(
      'FIH_TOKEN',
      ''
    );


    sessionStorage.setItem(
      'ClientIp',
      ClientIP
    );


    console.log(
      "Login Details",
      user,
      token,
      ClientIP
    );


    // Login data
    try {

      await this.publicSvr.insertLoginData(
        user.eeid,
        ClientIP
      );

    }

    catch (error) {

      console.error(
        "insertLoginData error:",
        error
      );
    }


    // Clear route reuse
    this.reuseTabService.clear();


    // Token
    this.tokenService.set({
      token: token
    });


    // Startup
    this.startupSrv
      .load()
      .then(() => {

        // Load menu
        this.getSystemMenu();

      });

  }

  // =====================================================
  // INIT
  // =====================================================

  ngOnInit() {

    let url =
      window.location.href;


    // HTTPS redirect
    if (
      window.location.protocol !== 'https:' &&
      (
        url.indexOf(
          'lms.idpbgtn.efoxconn.com'
        ) != -1 ||

        url.indexOf(
          'lms.idpbgind.efoxconn.com'
        ) != -1
      )
    ) {

      window.location.href =
        'https://' +
        window.location.hostname +
        window.location.pathname +
        window.location.search +
        window.location.hash;

    }


    // TN -> IND
    if (
      window.location.hostname ==
      "lms.idpbgtn.efoxconn.com"
    ) {

      window.location.href =
        url.replace(
          "lms.idpbgtn.efoxconn.com",
          "lms.idpbgind.efoxconn.com"
        );
    }


    // Portal login
    let fromInfo =
      this.location.href
        .indexOf("?from=iportal");


    if (fromInfo > -1) {

      let origin =
        window.location.origin;

      if (!origin) {

        origin =
          `${window.location.protocol}//${window.location.host}`;

      }


      const iframeSrc =
        `${ouacServiceUrl}` +
        `&CallBack=/assets/login-callback.html?UserInfo` +
        `&CssLink=/assets/sso-login.css` +
        `&State=RxFdv2p7KsDq38CX` +
        `&SSO=1`;


      this.iframeSafeSrc =
        this.domSanitizer
          .bypassSecurityTrustResourceUrl(
            iframeSrc
          );


      // toPage
      let toPageIndex =
        this.location.href
          .indexOf("&toPage=");


      if (toPageIndex > -1) {

        this.toPage =
          this.location.href
            .slice(
              toPageIndex + 8
            )
            .replace(
              /%2F/g,
              '/'
            );
      }


      // menuId
      const menuIndex =
        this.location.href
          .indexOf("&menuId=");


      if (menuIndex > -1) {

        this.menuId =
          this.location.href
            .slice(
              menuIndex + 8
            );
      }

    }


    // Normal login
    else {

      let origin =
        window.location.origin;

      if (!origin) {

        origin =
          `${window.location.protocol}//${window.location.host}`;

      }


      const iframeSrc =
        `${ouacServiceUrl}` +
        `&CallBack=/assets/login-callback.html?UserInfo` +
        `&CssLink=/assets/sso-login.css` +
        `&State=RxFdv2p7KsDq38CX` +
        `&SSO=0`;


      this.iframeSafeSrc =
        this.domSanitizer
          .bypassSecurityTrustResourceUrl(
            iframeSrc
          );
    }
  }

  // =====================================================
  // GET SYSTEM MENU
  // =====================================================

  getSystemMenu() {

    const group = [

      {

        text: "主導航",

        group: true,

        children: []

      }

    ];


    const userInfo =
      this.publicSvr.getSettingUserINfo();


    if (
      !userInfo ||
      !userInfo.eeid
    ) {

      console.error(
        "User information not found"
      );

      this.msg.error(
        "User information not found"
      );

      return;
    }


    const params = {

      Func:
        "Permission management-GetUserPermitModuleBySystem",

      SystemId:
        30373,

      EmpNo:
        userInfo.eeid

    };


    const param =
      this.publicSvr.dealPostParamsEncrypt(
        params
      );


    const config = {

      headers: {

        apiID:
          "ffff-1687574205823-1020819373-1797",

        userKey:
          this.publicSvr.USER_APPLIATION_WEB

      }

    };


    // Debug
    console.log(
      "========== MENU API =========="
    );

    console.log(
      "Menu URL:",
      iwxServiceUrl
    );

    console.log(
      "Employee:",
      userInfo.eeid
    );

    console.log(
      "apiID:",
      config.headers.apiID
    );

    console.log(
      "=============================="
    );


    this.http
      .post(
        iwxServiceUrl,
        param,
        null,
        config
      )
      .subscribe(

        (data: any) => {

          console.log(
            "Menu API response:",
            data
          );


          try {

            if (
              !data ||
              !data.r
            ) {

              console.error(
                "Invalid menu API response:",
                data
              );

              this.msg.error(
                "Invalid menu response from server"
              );

              return;
            }


            const decrypted =
              this.publicSvr.security.decrypt(
                data.r,
                this.publicSvr.iwxKInfo
              );


            console.log(
              "Menu decrypted response:",
              decrypted
            );


            const res =
              typeof decrypted === "string"
                ? JSON.parse(decrypted)
                : decrypted;


            console.log(
              "Menu parsed response:",
              res
            );


            // =================================================
            // MENU ACCESS CHECK
            // =================================================

            const hasAccess =
              res &&
              res.IsOK == "1" &&
              res.List &&
              res.List.length > 0;


            if (!hasAccess) {

              this.msg.error(
                "您暫無可查看的菜單，請聯繫管理員開通！ (No menu access, please contact the administrator)",
                {
                  nzDuration: 3000
                }
              );


              this.publicSvr.logout();

              return;
            }


            // =================================================
            // MENU TREE
            // =================================================

            group[0].children =
              this.arrayToTree(
                res.List
              );


            this.menuService.add(
              group
            );


            // =================================================
            // NAVIGATION
            // =================================================

            if (this.toPage) {

              const userMenu =
                res.List
                  .filter(
                    (item) =>
                      item.webUrl
                  )
                  .map(
                    (item) => ({

                      id:
                        item.id,

                      webUrl:
                        item.webUrl

                    })
                  );


              this.goToPage(
                userMenu,
                this.toPage
              );

            }


            else if (this.menuId) {

              this.getToMenuData(
                group[0].children,
                this.menuId
              );

            }


            else {

              this.router.navigateByUrl(
                '/ism/dashboard'
              );

            }

          }


          catch (error) {

            console.error(
              "Menu decrypt/parse error:",
              error
            );

            this.msg.error(
              "Unable to process menu response"
            );
          }

        },


        // ===================================================
        // API ERROR
        // ===================================================

        (err) => {

          console.error(
            "Menu API error:",
            err
          );


          if (
            err &&
            err.status === 401
          ) {

            console.error(
              "401 Unauthorized"
            );

            console.error(
              "URL:",
              iwxServiceUrl
            );


            this.msg.error(
              "Menu API Unauthorized (401). Please contact administrator."
            );

          }

          else {

            this.msg.error(
              "Unable to load menu permissions"
            );

          }

        }

      );

  }


  // =====================================================
  // GO TO PAGE
  // =====================================================

  goToPage(
    userMenu,
    toPageInfo
  ) {

    const toMenuData =
      userMenu.filter(
        (item) => {

          return item.webUrl ==
            toPageInfo;

        }
      );


    if (
      toMenuData.length == 0
    ) {

      this.publicSvr.logout();

      this.ngOnInit();

    }

    else {

      this.router.navigateByUrl(
        toPageInfo
      );

    }

  }


  // =====================================================
  // GET MENU DATA
  // =====================================================

  getToMenuData(
    data,
    menuId
  ) {

    const toMenuData =
      data.filter(
        item => {

          return item.id ==
            menuId;

        }
      );


    if (
      toMenuData.length == 0
    ) {

      data.forEach(
        (item) => {

          if (
            item.children.length != 0
          ) {

            const list =
              item.children;

            this.getToMenuData(
              list,
              menuId
            );

          }

        }
      );

    }

    else {

      const menuData =
        toMenuData[0];

      this.getPage(
        menuData
      );

    }

  }


  // =====================================================
  // GET PAGE
  // =====================================================

  getPage(data) {

    if (
      data.children.length > 0
    ) {

      const list =
        data.children[0];

      this.getPage(
        list
      );

    }

    else {

      const toPage =
        data.link;

      this.router.navigateByUrl(
        toPage
      );


      localStorage.removeItem(
        "fromInfo"
      );

      localStorage.removeItem(
        "toPageInfo"
      );

      localStorage.removeItem(
        "userMenu"
      );

      localStorage.removeItem(
        "userMenuId"
      );

    }

  }


  // =====================================================
  // ARRAY TO TREE
  // =====================================================

  arrayToTree(
    list,
    parentId = 0
  ) {

    if (
      !list ||
      !Array.isArray(list)
    ) {

      return [];

    }


    return list
      .filter(
        item =>
          item.parentId === parentId
      )
      .map(
        item => {

          let icon = null;


          if (
            item.icon
          ) {

            const iconParts =
              item.icon.split("|");


            icon = {

              type: "icon",

              value:
                iconParts[0],

              theme:
                iconParts[1] ||
                "outline"

            } as MenuIcon;

          }


          return {

            text:
              item.name,

            icon:
              icon,

            link:
              item.menuType === "15"

                ? item.webUrl +
                  "/" +
                  item.permissionId

                : item.webUrl,

            parentId:
              item.parentId,

            children:
              this.arrayToTree(
                list,
                item.id
              ),

            menuType:
              item.menuType,

            id:
              item.id,

            permissionId:
              item.permissionId,

            show:
              true,

            urlKey:
              item.urlkey

          };

        }
      );

  }

  // =====================================================
  // PASSWORD CHANGE
  // =====================================================

  passwordChange(name) {

    const newPassword =
      this.resetInfo.newPassword;

    const confirmPassword =
      this.resetInfo.confirmPassword;


    if (
      newPassword ||
      confirmPassword
    ) {

      const reg =
        /^(?=.*[A-Za-z])(?=.*\d)(?=.*[`~!@#$%^&*()_+<>?:"{},.\/\\;'[\]])[A-Za-z\d`~!@#$%^&*()_+<>?:"{},.\/\\;'[\]]{8,}$/;


      if (
        name ==
        "newPassword"
      ) {

        const newPasswordResult =
          reg.test(
            newPassword
          );


        if (!newPasswordResult) {

          this.msg.error(
            'The password must have at least 8 digits and must contain digits, letters, and special characters.',
            {
              nzDuration: 10000
            }
          );


          this.resetInfo.newPassword =
            "";

          return false;

        }

      }


      if (
        name ==
        "confirmPassword"
      ) {

        const confirmResult =
          reg.test(
            confirmPassword
          );


        if (!confirmResult) {

          this.msg.error(
            'The password must have at least 8 digits and must contain digits, letters, and special characters.',
            {
              nzDuration: 10000
            }
          );


          this.resetInfo.confirmPassword =
            "";

          return false;

        }

      }


      if (
        newPassword &&
        confirmPassword
      ) {

        if (
          newPassword !=
          confirmPassword
        ) {

          this.msg.error(
            'The entered password information is inconsistent.',
            {
              nzDuration: 10000
            }
          );

        }

      }

    }

  }

  // =====================================================
  // RESET PASSWORD
  // =====================================================

  resetPassword() {

    if (
      this.resetInfo.newPassword &&
      this.resetInfo.confirmPassword &&
      this.resetInfo.newPassword !=
      this.resetInfo.confirmPassword
    ) {

      return false;

    }


    if (
      !this.resetInfo.account ||
      !this.resetInfo.code ||
      !this.resetInfo.newPassword ||
      !this.resetInfo.confirmPassword
    ) {

      this.msg.error(
        "All information is required."
      );

      return false;

    }


    const params = {

      EmpNo:
        this.resetInfo.account,

      Password:
        this.resetInfo.newPassword,

      CaptchaCode:
        this.resetInfo.code,

      Func:
        "H5-ChangeEmpPasswordV2-IWX"

    };


    this.publicSvr
      .permitRequest(

        params,

        {

          headers: {

            apiID:
              "ffff-1655376433685-16925411121-0217",

            userKey:
              this.publicSvr.USER_GQ

          }

        },

        true

      )
      .then(
        (res: any) => {

          if (
            res.IsOK == "1"
          ) {

            this.msg.success(
              "Password changed successfully"
            );

            this.closeForgetPasswordModal();

          }

          else {

            this.msg.error(
              res.Msg
            );

          }

        }
      );

  }
