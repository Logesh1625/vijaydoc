  /* ================================================================
     LOAD LOGGED-IN USER SITE
     ================================================================ */

  private loadLoggedInUserSite(): void {

    const userInfo =
      this.publicSvr.getSettingUserInfo();


    console.log(
      'F1 Dashboard - Logged in user:',
      userInfo
    );


    if (
      !userInfo ||
      !userInfo.eeid
    ) {

      console.error(
        'F1 Dashboard - EEID not found'
      );

      this.resetDashboard();

      return;

    }


    this.publicSvr
      .getSitebasedUser()
      .subscribe(

        (userList: any[]) => {

          console.log(
            'F1 Dashboard - User site list:',
            userList
          );


          if (
            !userList ||
            userList.length === 0
          ) {

            console.error(
              'F1 Dashboard - No user/site data'
            );

            this.resetDashboard();

            return;

          }


          /*
           * Get only active sites
           * belonging to logged-in user.
           */
          this.siteList =
            this.extractAllSites(
              userList,
              userInfo.eeid
            );


          console.log(
            'F1 Dashboard - Active raw sites:',
            this.siteList
          );


          if (
            this.siteList.length === 0
          ) {

            console.error(
              'F1 Dashboard - No active site found'
            );

            this.resetDashboard();

            return;

          }


          /*
           * Default = All My Sites
           */
          this.selectedSite = '';

          this.site = '';


          /*
           * Build dropdown.
           */
          this.buildLocationList();


          /*
           * Load dashboard.
           */
          this.getDashboardData();

        },


        error => {

          console.error(
            'F1 Dashboard - getSitebasedUser error:',
            error
          );

          this.resetDashboard();

        }

      );

  }


  /* ================================================================
     GET EMPLOYEE ID
     ================================================================ */

  private getItemEeid(
    item: any
  ): string {

    if (!item) {

      return '';

    }


    const value =

      item.Eeid ||

      item.EEID ||

      item.EeidNo ||

      item.EmpId ||

      item.EMP_ID ||

      item.EmployeeId ||

      item.EmployeeCode ||

      item.eeid ||

      '';


    return String(value)
      .trim()
      .toUpperCase();

  }


  /* ================================================================
     CHECK ACTIVE USER
     ================================================================ */

  private isActiveUser(
    item: any
  ): boolean {

    if (!item) {

      return false;

    }


    let delFlag: any = 0;


    if (
      item.DelFlag !== undefined &&
      item.DelFlag !== null &&
      item.DelFlag !== ''
    ) {

      delFlag =
        item.DelFlag;

    }

    else if (
      item.DEL_FLAG !== undefined &&
      item.DEL_FLAG !== null &&
      item.DEL_FLAG !== ''
    ) {

      delFlag =
        item.DEL_FLAG;

    }

    else if (
      item.delFlag !== undefined &&
      item.delFlag !== null &&
      item.delFlag !== ''
    ) {

      delFlag =
        item.delFlag;

    }

    else if (
      item.del_flag !== undefined &&
      item.del_flag !== null &&
      item.del_flag !== ''
    ) {

      delFlag =
        item.del_flag;

    }


    return Number(delFlag) === 0;

  }


  /* ================================================================
     EXTRACT ALL ACTIVE SITES
     ================================================================ */

  private extractAllSites(
    userList: any[],
    eeid: string
  ): string[] {

    const loginEeid =
      String(eeid)
        .trim()
        .toUpperCase();


    const sites: string[] = [];


    userList.forEach(
      item => {

        if (!item) {

          return;

        }


        const itemEeid =
          this.getItemEeid(item);


        if (
          itemEeid !== loginEeid
        ) {

          return;

        }


        if (
          !this.isActiveUser(item)
        ) {

          return;

        }


        const site =
          this.extractSite(item);


        if (site) {

          sites.push(site);

        }

      }
    );


    /*
     * Remove duplicates.
     */
    const uniqueSites: string[] = [];


    sites.forEach(
      site => {

        const exists =
          uniqueSites.some(
            x =>
              x.toUpperCase() ===
              site.toUpperCase()
          );


        if (!exists) {

          uniqueSites.push(site);

        }

      }
    );


    return uniqueSites;

  }


  /* ================================================================
     EXTRACT SITE
     ================================================================ */

  private extractSite(
    user: any
  ): string {

    if (!user) {

      return '';

    }


    const site =

      user.Site ||

      user.site ||

      user.SITE ||

      user.SiteName ||

      user.siteName ||

      user.WorkSite ||

      user.Worksite ||

      user.BasicCode ||

      user.Basiccode ||

      user.Location ||

      user.location ||

      '';


    return String(site)
      .trim();

  }


  /* ================================================================
     SITE ID -> DISPLAY NAME
     ================================================================ */

  getSiteName(
    siteId: any
  ): string {

    const id =
      String(siteId || '')
        .trim()
        .toUpperCase();


    if (!id) {

      return '';

    }


    /*
     * Already display value.
     */
    if (
      id === 'FXBL' ||
      id === 'FXCN'
    ) {

      return id;

    }


    return (
      this.siteNameMap[id] ||
      id
    );

  }


  /* ================================================================
     BUILD SITE DROPDOWN
     ================================================================ */

  private buildLocationList(): void {

    /*
     * All My Sites
     */
    this.locationList = [

      {
        value: '',
        label: 'All My Sites'
      }

    ];


    /*
     * Add user's active sites.
     *
     * Example:
     *
     * value = 95326
     * label = FXBL
     *
     * value = 69941
     * label = FXCN
     */
    this.siteList.forEach(
      siteId => {

        this.locationList.push({

          value: siteId,

          label:
            this.getSiteName(siteId)

        });

      }
    );


    console.log(
      'F1 Dashboard - Site Dropdown:',
      this.locationList
    );

  }


  /* ================================================================
     DATE CHANGE
     ================================================================ */

  onDateChange(
    date: Date
  ): void {

    if (date) {

      this.selectedDate = date;

    }

    else {

      this.selectedDate =
        new Date();

    }


    console.log(
      'F1 Dashboard - Date:',
      this.formatDate(
        this.selectedDate
      )
    );


    this.getDashboardData();

  }


  /* ================================================================
     SITE CHANGE
     ================================================================ */

  onLocationChange(
    location: string
  ): void {

    console.log(
      'F1 Dashboard - Selected raw site:',
      location
    );


    /*
     * Empty = All My Sites.
     */
    if (!location) {

      this.selectedSite = '';

    }

    else {

      /*
       * Validate raw site ID.
       */
      const validSite =
        this.siteList.some(
          site =>

            String(site)
              .trim()
              .toUpperCase() ===

            String(location)
              .trim()
              .toUpperCase()

        );


      if (validSite) {

        /*
         * Keep RAW site ID.
         *
         * 95326
         * 69941
         */
        this.selectedSite =
          String(location)
            .trim();

      }

      else {

        console.warn(
          'Invalid site selected:',
          location
        );

        this.selectedSite = '';

      }

    }


    console.log(
      'F1 Dashboard - Display site:',
      this.selectedSite
        ? this.getSiteName(
            this.selectedSite
          )
        : 'All My Sites'
    );


    console.log(
      'F1 Dashboard - API site:',
      this.selectedSite || 'ALL'
    );


    /*
     * Reload dashboard.
     */
    this.getDashboardData();

  }


  /* ================================================================
     FORMAT DATE
     ================================================================ */

  private formatDate(
    d: Date
  ): string {

    if (!d) {

      d = new Date();

    }


    const dd =
      ('0' + d.getDate())
        .slice(-2);


    const mm =
      ('0' + (d.getMonth() + 1))
        .slice(-2);


    return (

      dd +
      '-' +
      mm +
      '-' +
      d.getFullYear()

    );

  }


  /* ================================================================
     PERCENTAGE
     ================================================================ */

  pct(
    n: number
  ): number {

    if (
      !this.totalEmployees ||
      this.totalEmployees <= 0
    ) {

      return 0;

    }


    return Math.round(

      (
        Number(n || 0) /
        this.totalEmployees
      ) *

      100

    );

  }


  /* ================================================================
     GET DASHBOARD DATA
     ================================================================ */

  getDashboardData(): void {

    const userInfo =
      this.publicSvr
        .getSettingUserInfo();


    if (
      !userInfo ||
      !userInfo.eeid
    ) {

      this.resetDashboard();

      return;

    }


    if (
      !this.siteList ||
      this.siteList.length === 0
    ) {

      this.resetDashboard();

      return;

    }


    /*
     * Validate selected site.
     */
    if (
      this.selectedSite
    ) {

      const validSite =
        this.siteList.some(
          site =>

            site.toUpperCase() ===
            this.selectedSite
              .toUpperCase()

        );


      if (!validSite) {

        this.selectedSite = '';

      }

    }


    this.loading = true;


    /* ============================================================
       API PARAMETER
       ============================================================ */

    const params: any = {

      Date:
        this.formatDate(
          this.selectedDate
        ),

      /*
       * IMPORTANT:
       *
       * API gets raw ID.
       *
       * FXBL -> 95326
       * FXCN -> 69941
       */
      Site:
        this.selectedSite || '',

      EmpId:
        userInfo.eeid,

      SystemName:
        'CMS',

      Func:
        'CMS-GetMealDashboard'

    };


    console.log(
      '================================================'
    );

    console.log(
      'F1 DASHBOARD REQUEST'
    );

    console.log(
      'Employee:',
      userInfo.eeid
    );

    console.log(
      'Display Site:',
      this.selectedSite
        ? this.getSiteName(
            this.selectedSite
          )
        : 'All My Sites'
    );

    console.log(
      'API Site:',
      params.Site || 'ALL'
    );

    console.log(
      'Date:',
      params.Date
    );

    console.log(
      'Params:',
      params
    );

    console.log(
      '================================================'
    );


    this.publicSvr
      .iwxPostF1(

        params,

        {

          headers: {

            apiID:
              'ffff-1789987849667-10208193125-1519',

            userKey:
              'E9785F13CCEAFB251A6AB147AFD24336'

          }

        },

        true

      )


      .then(
        (res: any) => {

          this.loading = false;


          console.log(
            'F1 DASHBOARD RESPONSE',
            res
          );


          /*
           * API failed.
           */
          if (
            !res ||
            String(res.IsOK) !== '1'
          ) {

            this.resetDashboard();

            return;

          }


          /* ======================================================
             CURSOR 1
             ====================================================== */

          const c =

            (
              res.List &&
              res.List.length > 0
            )

              ? res.List[0]

              : {};


          this.totalEmployees =
            Number(

              c.TotalEmployees ||
              c.TOTAL_EMPLOYEES ||
              0

            );


          this.mealsCollected =
            Number(

              c.MealsCollected ||
              c.MEALS_COLLECTED ||
              0

            );


          this.mealsNotCollected =
            Number(

              c.MealsNotCollected ||
              c.MEALS_NOT_COLLECTED ||
              0

            );


          this.noMealRequired =
            Number(

              c.NoMealRequired ||
              c.NO_MEAL_REQUIRED ||
              0

            );


          this.notCameToCollect =
            Number(

              c.NotRequiredCame ||
              c.NOT_REQUIRED_CAME ||
              0

            );


          /* ======================================================
             CURSOR 2
             ====================================================== */

          const cats: any[] =

            res.List2 ||

            res.Summary ||

            res.CategoryList ||

            [];


          if (
            cats &&
            cats.length > 0
          ) {

            this.categoryList =
              cats.map(
                (item: any) => {

                  const categoryName =

                    item.Category ||

                    item.CATEGORY ||

                    '';


                  const count =

                    Number(

                      item.CollectedCount ||

                      item.COLLECTED_COUNT ||

                      0

                    );


                  const percentage =

                    Number(

                      item.Percentage ||

                      item.PERCENTAGE ||

                      0

                    );


                  return {

                    name:
                      categoryName,

                    count:
                      count,

                    percent:
                      percentage,

                    color:
                      this.getColorForCategory(
                        categoryName
                      ),

                    icon:
                      this.getIconForCategory(
                        categoryName
                      )

                  };

                }
              );

          }

          else {

            this.resetCategories();

          }


          this.updateSummary();

        }

      )


      .catch(
        err => {

          this.loading = false;


          console.error(
            'F1 Dashboard API error:',
            err
          );


          this.resetDashboard();

        }
      );

  }

  /* ================================================================
     COLOR
     ================================================================ */

  getColorForCategory(
    name: string
  ): string {

    if (!name) {

      return '#999';

    }


    if (
      this.colorMap[name]
    ) {

      return this.colorMap[name];

    }


    const key =
      Object.keys(
        this.colorMap
      ).find(
        x =>

          x.toUpperCase() ===

          String(name)
            .trim()
            .toUpperCase()

      );


    return key
      ? this.colorMap[key]
      : '#999';

  }


  /* ================================================================
     ICON
     ================================================================ */

  getIconForCategory(
    name: string
  ): string {

    if (!name) {

      return '🍽️';

    }


    if (
      this.iconMap[name]
    ) {

      return this.iconMap[name];

    }


    const key =
      Object.keys(
        this.iconMap
      ).find(
        x =>

          x.toUpperCase() ===

          String(name)
            .trim()
            .toUpperCase()

      );


    return key
      ? this.iconMap[key]
      : '🍽️';

  }


  /* ================================================================
     UPDATE SUMMARY
     ================================================================ */

  private updateSummary(): void {

    this.summaryList = [

      {
        label:
          'Meals Collected',

        count:
          this.mealsCollected,

        percent:
          this.pct(
            this.mealsCollected
          ),

        color:
          '#1890ff'

      },

      {
        label:
          'Meals Not Collected',

        count:
          this.mealsNotCollected,

        percent:
          this.pct(
            this.mealsNotCollected
          ),

        color:
          '#b4bfd4'

      },

      {
        label:
          'No Meal Required',

        count:
          this.noMealRequired,

        percent:
          this.pct(
            this.noMealRequired
          ),

        color:
          '#dde3ee'

      },

      {
        label:
          'Not Required - Came to Collect',

        count:
          this.notCameToCollect,

        percent:
          this.pct(
            this.notCameToCollect
          ),

        color:
          '#b39ddb'

      }

    ];


    const ringParts =
      this.summaryList.slice(
        0,
        3
      );


    const sum =
      ringParts.reduce(

        (
          total,
          item
        ) =>

          total +
          Number(
            item.count || 0
          ),

        0

      );


    if (
      sum <= 0
    ) {

      this.donutGradient =
        'conic-gradient(#e6ebf5 0% 100%)';

      return;

    }


    let acc = 0;


    const stops =
      ringParts.map(
        item => {

          const start =
            acc;


          acc +=

            (
              Number(
                item.count || 0
              ) /

              sum

            ) *

            100;


          return (

            item.color +

            ' ' +

            start +

            '%' +

            ' ' +

            acc +

            '%'

          );

        }
      );


    this.donutGradient =

      'conic-gradient(' +

      stops.join(', ') +

      ')';

  }
