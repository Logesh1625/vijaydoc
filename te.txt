/* ================================================================
   GET MEAL DASHBOARD
   ----------------------------------------------------------------
   Dashboard is based on:

   1. Logged-in employee
   2. Active CMS user
   3. User's active site(s)
   4. Selected date
   5. Active F1 meal records

   DATE LOGIC:
       FIRST_DATE_FXBL <= selected date <= LAST_DATE_FXBL

   SITE LOGIC:
       If V_SITE is supplied -> selected site only
       If V_SITE is NULL/empty/All -> all active sites of user

   USER LOGIC:
       C_USER_MANAGE_T.DEL_FLAG = 0
       SYSTEM_TYPE = CMS

   F1 LOGIC:
       C_F1_MEAL_INFO.DEL_FLAG = 0
   ================================================================ */

PROCEDURE GET_MEAL_DASHBOARD(
    V_EMP_ID IN VARCHAR2,
    V_DATE IN VARCHAR2,
    V_SITE IN VARCHAR2,
    V_RES OUT VARCHAR2,
    V_RESULT OUT VARCHAR2,
    P_CURSOR OUT SYS_REFCURSOR,
    P_SUMMARY_CURSOR OUT SYS_REFCURSOR
) IS

    V_DAY DATE;

BEGIN

    /* ============================================================
       DATE
       ============================================================ */

    IF V_DATE IS NULL
       OR TRIM(V_DATE) = ''
    THEN

        V_DAY := TRUNC(SYSDATE);

    ELSE

        V_DAY :=
            TO_DATE(
                V_DATE,
                'DD-MM-YYYY'
            );

    END IF;


    /* ============================================================
       DASHBOARD CARDS
       ============================================================ */

    OPEN P_CURSOR FOR

    WITH USER_SITES AS
    (
        /* ========================================================
           Get ACTIVE sites of logged-in user
           ======================================================== */

        SELECT DISTINCT
            UPPER(TRIM(U.SITE)) AS SITE

        FROM CMS.C_USER_MANAGE_T U

        WHERE UPPER(TRIM(U.EMP_ID))
              =
              UPPER(TRIM(V_EMP_ID))

          AND U.SYSTEM_TYPE = 'CMS'

          AND NVL(U.DEL_FLAG, 0) = 0

          AND U.SITE IS NOT NULL

          AND
          (
              V_SITE IS NULL

              OR TRIM(V_SITE) = ''

              OR UPPER(TRIM(V_SITE)) = 'ALL'

              OR UPPER(TRIM(U.SITE))
                 =
                 UPPER(TRIM(V_SITE))
          )
    ),


    EMP AS
    (
        /* ========================================================
           Latest active F1 record for each badge

           IMPORTANT:
           Date is checked against FIRST_DATE and LAST_DATE.
           ======================================================== */

        SELECT
            M.FX_BADGE,
            M.MEAL_PREFERENCE,
            BC.NAME AS MEAL_PREFERENCE_NAME,
            M.PROPERTY_01 AS SITE

        FROM
        (
            SELECT
                M.*,

                ROW_NUMBER() OVER
                (
                    PARTITION BY M.FX_BADGE

                    ORDER BY
                        M.ADD_DATE DESC,
                        M.ID DESC
                ) AS RN

            FROM CMS.C_F1_MEAL_INFO M

            INNER JOIN USER_SITES US
                ON UPPER(TRIM(M.PROPERTY_01))
                   =
                   US.SITE

            WHERE NVL(M.DEL_FLAG, 0) = 0

              /* ==================================================
                 Selected date must be within employee validity
                 ================================================== */

              AND M.FIRST_DATE_FXBL IS NOT NULL

              AND M.LAST_DATE_FXBL IS NOT NULL

              AND TRUNC(V_DAY)
                  BETWEEN
                  TRUNC(M.FIRST_DATE_FXBL)
                  AND
                  TRUNC(M.LAST_DATE_FXBL)
        ) M

        LEFT JOIN CMS.C_BASE_CODE_T BC

            ON TO_CHAR(M.MEAL_PREFERENCE)
               =
               TO_CHAR(BC.ID)

           AND BC.TYPE_ID = 81

        WHERE M.RN = 1
    ),


    COLLECTED AS
    (
        /* ========================================================
           Successful scans for selected date

           IMPORTANT:
           Scan must also belong to logged-in user's site(s).
           ======================================================== */

        SELECT DISTINCT
            S.FX_BADGE

        FROM CMS.C_F1_SCAN_LOG S

        INNER JOIN USER_SITES US

            ON UPPER(TRIM(S.PROPERTY_01))
               =
               US.SITE

        WHERE NVL(S.DEL_FLAG, 0) = 0

          AND TRUNC(S.MEAL_DATE)
              =
              TRUNC(V_DAY)

          AND S.COLLECTED_DATA = 'SUCCESS'
    ),


    NOT_REQUIRED_CAME AS
    (
        /* ========================================================
           No Meal Required employees who came to collect
           ======================================================== */

        SELECT DISTINCT
            S.FX_BADGE

        FROM CMS.C_F1_SCAN_LOG S

        INNER JOIN USER_SITES US

            ON UPPER(TRIM(S.PROPERTY_01))
               =
               US.SITE

        LEFT JOIN CMS.C_BASE_CODE_T BC

            ON TO_CHAR(S.MEAL_PREFERENCE)
               =
               TO_CHAR(BC.ID)

           AND BC.TYPE_ID = 81

        WHERE NVL(S.DEL_FLAG, 0) = 0

          AND TRUNC(S.MEAL_DATE)
              =
              TRUNC(V_DAY)

          AND UPPER(TRIM(BC.NAME))
              =
              'NO MEAL REQUIRED'
    )


    SELECT

        /* ========================================================
           TOTAL EMPLOYEES
           ======================================================== */

        (
            SELECT COUNT(1)
            FROM EMP
        ) AS TOTAL_EMPLOYEES,


        /* ========================================================
           MEALS COLLECTED
           ======================================================== */

        (
            SELECT COUNT(1)
            FROM COLLECTED
        ) AS MEALS_COLLECTED,


        /* ========================================================
           NO MEAL REQUIRED
           ======================================================== */

        (
            SELECT COUNT(1)
            FROM EMP

            WHERE UPPER(
                      TRIM(
                          NVL(
                              MEAL_PREFERENCE_NAME,
                              ''
                          )
                      )
                  )
                  =
                  'NO MEAL REQUIRED'
        ) AS NO_MEAL_REQUIRED,


        /* ========================================================
           NOT REQUIRED BUT CAME
           ======================================================== */

        (
            SELECT COUNT(1)
            FROM NOT_REQUIRED_CAME
        ) AS NOT_REQUIRED_CAME,


        /* ========================================================
           MEALS NOT COLLECTED

           Eligible employees excluding:
           - No Meal Required
           - Successfully collected
           ======================================================== */

        (
            SELECT COUNT(1)

            FROM EMP E

            WHERE UPPER(
                      TRIM(
                          NVL(
                              E.MEAL_PREFERENCE_NAME,
                              ''
                          )
                      )
                  )
                  <>
                  'NO MEAL REQUIRED'

              AND NOT EXISTS
              (
                  SELECT 1

                  FROM COLLECTED C

                  WHERE C.FX_BADGE =
                        E.FX_BADGE
              )
        ) AS MEALS_NOT_COLLECTED

    FROM DUAL;


    /* ============================================================
       CATEGORY-WISE SUMMARY
       ============================================================ */

    OPEN P_SUMMARY_CURSOR FOR

    WITH USER_SITES AS
    (
        SELECT DISTINCT
            UPPER(TRIM(U.SITE)) AS SITE

        FROM CMS.C_USER_MANAGE_T U

        WHERE UPPER(TRIM(U.EMP_ID))
              =
              UPPER(TRIM(V_EMP_ID))

          AND U.SYSTEM_TYPE = 'CMS'

          AND NVL(U.DEL_FLAG, 0) = 0

          AND U.SITE IS NOT NULL

          AND
          (
              V_SITE IS NULL

              OR TRIM(V_SITE) = ''

              OR UPPER(TRIM(V_SITE)) = 'ALL'

              OR UPPER(TRIM(U.SITE))
                 =
                 UPPER(TRIM(V_SITE))
          )
    ),


    CATS AS
    (
        SELECT 1 AS SORT_NO,
               'Indian Veg' AS CATEGORY
        FROM DUAL

        UNION ALL

        SELECT 2,
               'Indian Non-Veg'
        FROM DUAL

        UNION ALL

        SELECT 3,
               'Western Veg'
        FROM DUAL

        UNION ALL

        SELECT 4,
               'Western Non-Veg'
        FROM DUAL

        UNION ALL

        SELECT 5,
               'Chinese Non-Veg'
        FROM DUAL

        UNION ALL

        SELECT 6,
               'Chinese Veg'
        FROM DUAL

        UNION ALL

        SELECT 7,
               'Jain'
        FROM DUAL

        UNION ALL

        SELECT 8,
               'Vegan'
        FROM DUAL
    ),


    CNT AS
    (
        SELECT

            UPPER(
                TRIM(
                    BC.NAME
                )
            ) AS PREF,

            COUNT(
                DISTINCT
                S.FX_BADGE
            ) AS TOTAL_COUNT

        FROM CMS.C_F1_SCAN_LOG S

        INNER JOIN USER_SITES US

            ON UPPER(TRIM(S.PROPERTY_01))
               =
               US.SITE

        LEFT JOIN CMS.C_BASE_CODE_T BC

            ON TO_CHAR(
                S.MEAL_PREFERENCE
            )
            =
            TO_CHAR(
                BC.ID
            )

           AND BC.TYPE_ID = 81

        WHERE NVL(
                  S.DEL_FLAG,
                  0
              ) = 0

          AND TRUNC(
                  S.MEAL_DATE
              )
              =
              TRUNC(
                  V_DAY
              )

          AND S.COLLECTED_DATA =
              'SUCCESS'

        GROUP BY
            UPPER(
                TRIM(
                    BC.NAME
                )
            )
    )


    SELECT

        C.CATEGORY,

        NVL(
            N.TOTAL_COUNT,
            0
        ) AS COLLECTED_COUNT,

        CASE

            WHEN
            (
                SELECT COUNT(1)

                FROM CMS.C_F1_SCAN_LOG S

                INNER JOIN USER_SITES US

                    ON UPPER(TRIM(S.PROPERTY_01))
                       =
                       US.SITE

                WHERE NVL(
                          S.DEL_FLAG,
                          0
                      ) = 0

                  AND TRUNC(
                          S.MEAL_DATE
                      )
                      =
                      TRUNC(
                          V_DAY
                      )

                  AND S.COLLECTED_DATA =
                      'SUCCESS'
            ) = 0

            THEN 0

            ELSE ROUND(
                NVL(
                    N.TOTAL_COUNT,
                    0
                )
                * 100
                /
                (
                    SELECT COUNT(1)

                    FROM CMS.C_F1_SCAN_LOG S

                    INNER JOIN USER_SITES US

                        ON UPPER(TRIM(S.PROPERTY_01))
                           =
                           US.SITE

                    WHERE NVL(
                              S.DEL_FLAG,
                              0
                          ) = 0

                      AND TRUNC(
                              S.MEAL_DATE
                          )
                          =
                          TRUNC(
                              V_DAY
                          )

                      AND S.COLLECTED_DATA =
                          'SUCCESS'
                )
            )

        END AS PERCENTAGE

    FROM CATS C

    LEFT JOIN CNT N

        ON N.PREF =
           UPPER(
               C.CATEGORY
           )

    ORDER BY
        C.SORT_NO;


    V_RES :=
        '1';

    V_RESULT :=
        'Success';


EXCEPTION

    WHEN OTHERS THEN

        V_RES :=
            '0';

        V_RESULT :=
            'Error: '
            || SQLERRM;

END GET_MEAL_DASHBOARD;
