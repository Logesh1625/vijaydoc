import { SettingsService, _HttpClient } from "@delon/theme";
import {
  Component, OnInit, OnDestroy, Inject, Optional, HostListener
} from "@angular/core";
import { Router } from "@angular/router";
import { FormGroup, FormBuilder } from "@angular/forms";
import { NzMessageService, NzModalService } from "ng-zorro-antd";
import { ITokenService, DA_SERVICE_TOKEN } from "@delon/auth";
import { ReuseTabService } from "@delon/abc";
import { StartupService, I18NService } from "@core";
import { CookieService } from "ngx-cookie-service";
import { DomSanitizer } from "@angular/platform-browser";
import { PlatformLocation } from "@angular/common";
import { MenuService, MenuIcon } from "@delon/theme";
import { PublicNetService } from "../../a_Public_net/Public_net.service";
import { security } from "@shared";
import { ouacServiceUrl, iwxServiceUrl } from "../../a_Public_net/Url";

@Component({
  selector: "passport-login",
  templateUrl: "./login.component.html",
  styleUrls: ["./login.component.less"],
})
export class UserLoginComponent implements OnInit, OnDestroy {
  loading = false;
  iframeSrc = "";
  iframeSafeSrc: any;
  form: FormGroup;
  error = "2";
  type = 0;
  count = 0;
  interval$: any;
  isVisible = false;
  isForgetPassword = false;
  toPage: any;
  menuId: any;
  isCode = false;

  resetInfo = {
    account: "",
    code: "",
    newPassword: "",
    confirmPassword: "",
    email: ""
  };

  constructor(
    fb: FormBuilder,
    modalSrv: NzModalService,
    private router: Router,
    @Optional() @Inject(ReuseTabService) private reuseTabService: ReuseTabService,
    @Inject(DA_SERVICE_TOKEN) private tokenService: ITokenService,
    private startupSrv: StartupService,
    private settingService: SettingsService,
    private cookieService: CookieService,
    public http: _HttpClient,
    public msg: NzMessageService,
    public i18n: I18NService,
    public domSanitizer: DomSanitizer,
    public location: PlatformLocation,
    private menuService: MenuService,
    private publicSvr: PublicNetService,
  ) {
    modalSrv.closeAll();

    this.form = fb.group({
      userName: [""],
      pwdInfo: [""],
      remember: [true]
    });
  }

  get userName() {
    return this.form.controls.userName;
  }

  get password() {
    return this.form.controls.pwdInfo;
  }

  get remember() {
    return this.form.controls.remember;
  }

  switch(ret: any) {
    this.type = ret.index;
  }

  // =====================================================
  // SSO LOGIN
  // =====================================================

  submit(userInfo?: string) {
    this.loading = true;

    try {
      if (!userInfo || typeof userInfo !== "string") {
        throw new Error("OUAC UserInfo is empty or invalid.");
      }

      console.log(
        "OUAC UserInfo received. Length:",
        userInfo.length
      );

      // IMPORTANT:
      // This is the existing LMS key configuration.
      // Confirm the correct ISMS_INDIA key with OUAC admin
      // before changing this configuration.

      const decryptKey = this.publicSvr.security.getDK(
        "NzQxRTMzNzlBODA0MjM3OEY4NEE4MDc4NzNCMUJCNDk="
      );

      let decrypted: string;

      try {
        decrypted = security.decrypt(
          userInfo,
          decryptKey
        );
      } catch (decryptError) {
        console.error(
          "OUAC UserInfo decryption failed:",
          decryptError
        );

        throw new Error(
          "UserInfo decryption failed. Confirm the ISMS_INDIA key, AES mode, padding and IV with the OUAC administrator."
        );
      }

      let data: any;

      try {
        data = JSON.parse(decrypted);
      } catch (parseError) {
        console.error(
          "OUAC UserInfo JSON parsing failed:",
          parseError
        );

        throw new Error(
          "Decrypted UserInfo is not valid JSON."
        );
      }

      // Existing backend login encryption.
      // Do not change this key unless the backend administrator
      // confirms that a different key is required.

      const userInfo2 = security.encrypt(
        JSON.stringify(data),
        "6jxaHMIhvefCOokm5XbDdW4qzl30U7Kt"
      );

      const param = {
        url: "/oneAccountLogin?_allow_anonymous=true",
        data: {
          userInfo: userInfo2
        }
      };

      const body = security.encryptForWeb(
        JSON.stringify(param.data)
      );

      this.http.post(
        param.url,
        body,
        null,
        {
          headers: {
            "Content-Type": "application/json;charset=UTF-8"
          }
        }
      ).subscribe(
        (response: any) => {
          try {
            const res = JSON.parse(
              security.decryptForWeb(response)
            );

            this.loading = false;

            if (res && res.status === 200 && res.data) {
              const user = {
                avatar: "./assets/tmp/img/avatar.jpg",
                eeid: res.data.eeid,
                name: res.data.empName,
                time: new Date().getTime()
              };

              this.loginSuccess(
                user,
                res.data.JsessionId,
                res.data.ClientIp
              );
            } else {
              this.msg.error(
                (res && res.msg) || "Login failed."
              );

              this.resetLoginIframe();
            }
          } catch (error) {
            this.loading = false;

            console.error(
              "Login response processing failed:",
              error
            );

            this.msg.error(
              "Unable to process login response."
            );
          }
        },
        (error) => {
          this.loading = false;

          console.error(
            "Login API error:",
            error
          );

          this.msg.error(
            "Login API failed. Please try again."
          );
        }
      );

    } catch (error) {
      this.loading = false;

      console.error(
        "Login processing error:",
        error
      );

      this.msg.error(
        error instanceof Error
          ? error.message
          : "Unable to process login information."
      );
    }
  }

  private resetLoginIframe() {
    const iframe = document.querySelector(".login-iframe");

    if (iframe) {
      iframe.setAttribute(
        "src",
        this.buildIframeUrl(false)
      );
    }
  }

  private buildIframeUrl(sso: boolean): string {
    return `${ouacServiceUrl}&CallBack=/assets/login-callback.html?UserInfo` +
      `&CssLink=/assets/sso-login.css` +
      `&State=RxFdv2p7KsDq38CX` +
      `&SSO=${sso ? 1 : 0}`;
  }

  // =====================================================
  // IFRAME LOGIN CALLBACK
  // =====================================================

  @HostListener("window:message", ["$event"])
  onMessage(e: MessageEvent) {
    const info: any = e.data;

    if (!info || info.msgType !== "login") {
      return;
    }

    if (
      !info.UserInfo ||
      typeof info.UserInfo !== "string"
    ) {
      this.loading = false;

      console.error(
        "OUAC callback did not contain a valid UserInfo value."
      );

      this.msg.error(
        "Login information was not received."
      );

      return;
    }

    this.submit(info.UserInfo);
  }

  // =====================================================
  // LOGIN SUCCESS
  // =====================================================

  async loginSuccess(
    user: any,
    token: any,
    ClientIP: any
  ) {
    this.settingService.setUser({
      userInfo: this.publicSvr.security.encryptForWeb(
        JSON.stringify(user)
      )
    });

    this.cookieService.set(
      "USER_NO",
      this.publicSvr.security.encryptForWeb(user.eeid)
    );

    this.cookieService.set(
      "USER_NAME",
      this.publicSvr.security.encryptForWeb(user.name)
    );

    this.cookieService.set("IS_FOREIGN", "");
    this.cookieService.set("SESSION_ID", "");
    this.cookieService.set("FIH_TOKEN", "");

    sessionStorage.setItem(
      "ClientIp",
      ClientIP || ""
    );

    try {
      await this.publicSvr.insertLoginData(
        user.eeid,
        ClientIP
      );
    } catch (error) {
      console.error(
        "insertLoginData error:",
        error
      );
    }

    if (this.reuseTabService) {
      this.reuseTabService.clear();
    }

    this.tokenService.set({
      token: token
    });

    this.startupSrv.load()
      .then(() => {
        this.getSystemMenu();
      })
      .catch((error) => {
        console.error(
          "Startup loading failed:",
          error
        );

        this.msg.error(
          "Unable to load application startup data."
        );
      });
  }

  // =====================================================
  // INITIALIZATION
  // =====================================================

  ngOnInit() {
    const url = window.location.href;

    // HTTPS redirect for existing LMS hosts.
    if (
      window.location.protocol !== "https:" &&
      (
        url.indexOf("lms.idpbgtn.efoxconn.com") !== -1 ||
        url.indexOf("lms.idpbgind.efoxconn.com") !== -1
      )
    ) {
      window.location.href =
        "https://" +
        window.location.hostname +
        window.location.pathname +
        window.location.search +
        window.location.hash;

      return;
    }

    // Existing TN to IND redirect.
    if (
      window.location.hostname ===
      "lms.idpbgtn.efoxconn.com"
    ) {
      window.location.href = url.replace(
        "lms.idpbgtn.efoxconn.com",
        "lms.idpbgind.efoxconn.com"
      );

      return;
    }

    const currentUrl = this.location.href;

    const fromInfo =
      currentUrl.indexOf("?from=iportal") > -1;

    const iframeUrl = this.buildIframeUrl(fromInfo);

    this.iframeSafeSrc =
      this.domSanitizer.bypassSecurityTrustResourceUrl(
        iframeUrl
      );

    // Read target page.
    const toPageIndex =
      currentUrl.indexOf("&toPage=");

    if (toPageIndex > -1) {
      this.toPage = currentUrl
        .slice(toPageIndex + 8)
        .replace(/%2F/g, "/");
    }

    // Read menu ID.
    const menuIndex =
      currentUrl.indexOf("&menuId=");

    if (menuIndex > -1) {
      this.menuId = currentUrl.slice(menuIndex + 8);
    }
  }

  ngOnDestroy() {
    if (this.interval$) {
      clearInterval(this.interval$);
    }
  }

  // =====================================================
  // GET SYSTEM MENU
  // =====================================================

  getSystemMenu() {
    const group: any[] = [
      {
        text: "主導航",
        group: true,
        children: []
      }
    ];

    const userInfo =
      this.publicSvr.getSettingUserINfo();

    if (!userInfo || !userInfo.eeid) {
      console.error(
        "User information not found."
      );

      this.msg.error(
        "User information not found."
      );

      return;
    }

    const params = {
      Func: "Permission management-GetUserPermitModuleBySystem",
      SystemId: 30373,
      EmpNo: userInfo.eeid
    };

    const param =
      this.publicSvr.dealPostParamsEncrypt(params);

    const config = {
      headers: {
        apiID: "ffff-1687574205823-1020819373-1797",
        userKey: this.publicSvr.USER_APPLIATION_WEB
      }
    };

    this.http.post(
      iwxServiceUrl,
      param,
      null,
      config
    ).subscribe(
      (data: any) => {
        try {
          if (!data || !data.r) {
            console.error(
              "Invalid menu API response:",
              data
            );

            this.msg.error(
              "Invalid menu response from server."
            );

            return;
          }

          const decrypted =
            this.publicSvr.security.decrypt(
              data.r,
              this.publicSvr.iwxKInfo
            );

          const res = JSON.parse(decrypted);

          const hasAccess =
            res &&
            res.IsOK == "1" &&
            Array.isArray(res.List) &&
            res.List.length > 0;

          if (!hasAccess) {
            this.msg.error(
              "No menu access. Please contact the administrator.",
              {
                nzDuration: 3000
              }
            );

            this.publicSvr.logout();

            return;
          }

          group[0].children =
            this.arrayToTree(res.List);

          this.menuService.add(group);

          if (this.toPage) {
            const userMenu = res.List
              .filter((item: any) => item.webUrl)
              .map((item: any) => ({
                id: item.id,
                webUrl: item.webUrl
              }));

            this.goToPage(
              userMenu,
              this.toPage
            );
          } else if (this.menuId) {
            this.getToMenuData(
              group[0].children,
              this.menuId
            );
          } else {
            this.router.navigateByUrl(
              "/ism/dashboard"
            );
          }

        } catch (error) {
          console.error(
            "Menu decrypt/parse error:",
            error
          );

          this.msg.error(
            "Unable to process menu response."
          );
        }
      },
      (err) => {
        console.error(
          "Menu API error:",
          err
        );

        if (err && err.status === 401) {
          this.msg.error(
            "Menu API Unauthorized (401). Please contact administrator."
          );
        } else {
          this.msg.error(
            "Unable to load menu permissions."
          );
        }
      }
    );
  }

  // =====================================================
  // NAVIGATION
  // =====================================================

  goToPage(
    userMenu: any[],
    toPageInfo: string
  ) {
    const match = userMenu.filter(
      (item) => item.webUrl === toPageInfo
    );

    if (match.length === 0) {
      this.publicSvr.logout();
      return;
    }

    this.router.navigateByUrl(toPageInfo);
  }

  getToMenuData(
    data: any[],
    menuId: any
  ) {
    const matches = data.filter(
      (item) => item.id == menuId
    );

    if (matches.length === 0) {
      data.forEach((item) => {
        if (
          item.children &&
          item.children.length > 0
        ) {
          this.getToMenuData(
            item.children,
            menuId
          );
        }
      });
    } else {
      this.getPage(matches[0]);
    }
  }

  getPage(data: any) {
    if (
      data.children &&
      data.children.length > 0
    ) {
      this.getPage(data.children[0]);
    } else {
      this.router.navigateByUrl(data.link);

      localStorage.removeItem("fromInfo");
      localStorage.removeItem("toPageInfo");
      localStorage.removeItem("userMenu");
      localStorage.removeItem("userMenuId");
    }
  }

  // =====================================================
  // ARRAY TO TREE
  // =====================================================

  arrayToTree(
    list: any[],
    parentId: any = 0
  ): any[] {
    if (!Array.isArray(list)) {
      return [];
    }

    return list
      .filter(
        (item) => item.parentId === parentId
      )
      .map((item) => {
        let icon: MenuIcon = null;

        if (item.icon) {
          const parts = item.icon.split("|");

          icon = {
            type: "icon",
            value: parts[0],
            theme: parts[1] || "outline"
          } as MenuIcon;
        }

        return {
          text: item.name,
          icon: icon,

          link: item.menuType === "15"
            ? item.webUrl + "/" + item.permissionId
            : item.webUrl,

          parentId: item.parentId,
          children: this.arrayToTree(list, item.id),
          menuType: item.menuType,
          id: item.id,
          permissionId: item.permissionId,
          show: true,
          urlKey: item.urlkey
        };
      });
  }

  // =====================================================
  // FORGOT PASSWORD
  // =====================================================

  openForgetPasswordModal() {
    this.isForgetPassword = true;
  }

  closeForgetPasswordModal() {
    this.isForgetPassword = false;
    this.isCode = false;

    this.resetInfo = {
      account: "",
      code: "",
      newPassword: "",
      confirmPassword: "",
      email: ""
    };
  }

  // =====================================================
  // PASSWORD VALIDATION
  // =====================================================

  passwordChange(name: string) {
    const newPassword =
      this.resetInfo.newPassword;

    const confirmPassword =
      this.resetInfo.confirmPassword;

    if (newPassword || confirmPassword) {
      const reg =
        /^(?=.*[A-Za-z])(?=.*\d)(?=.*[`~!@#$%^&*()_+<>?:"{},.\/\\;'[\]])[A-Za-z\d`~!@#$%^&*()_+<>?:"{},.\/\\;'[\]]{8,}$/;

      if (
        name === "newPassword" &&
        !reg.test(newPassword)
      ) {
        this.msg.error(
          "The password must have at least 8 characters and include letters, numbers, and special characters.",
          {
            nzDuration: 10000
          }
        );

        this.resetInfo.newPassword = "";
        return false;
      }

      if (
        name === "confirmPassword" &&
        !reg.test(confirmPassword)
      ) {
        this.msg.error(
          "The password must have at least 8 characters and include letters, numbers, and special characters.",
          {
            nzDuration: 10000
          }
        );

        this.resetInfo.confirmPassword = "";
        return false;
      }

      if (
        newPassword &&
        confirmPassword &&
        newPassword !== confirmPassword
      ) {
        this.msg.error(
          "The entered passwords do not match.",
          {
            nzDuration: 10000
          }
        );
      }
    }

    return true;
  }

  // =====================================================
  // SEND VERIFICATION CODE
  // =====================================================

  sendVerifivationCode() {
    if (!this.resetInfo.account) {
      this.msg.error("Please enter your account");
      return false;
    }

    const params = {
      EmpNo: this.resetInfo.account,
      CaptchaCode: "",
      Func: "H5-SendVerificationCode-IWX"
    };

    this.publicSvr.permitRequest(
      params,
      {
        headers: {
          apiID: "ffff-1655357911834-16925411121-0153",
          userKey: this.publicSvr.USER_GQ
        }
      },
      true
    ).then((res: any) => {
      if (res && res.IsOK == "1") {
        this.resetInfo.email = res.Msg;
      } else {
        this.msg.error(
          (res && res.Msg) ||
          "Unable to send verification code."
        );
      }
    }).catch((error) => {
      console.error(
        "Send verification code error:",
        error
      );

      this.msg.error(
        "Unable to send verification code."
      );
    });
  }

  // =====================================================
  // VERIFY CODE
  // =====================================================

  nextOption() {
    if (!this.resetInfo.code) {
      this.msg.error("Please enter verification code");
      return false;
    }

    const params = {
      EmpNo: this.resetInfo.account,
      CaptchaCode: this.resetInfo.code,
      Func: "H5-CheckVerificationCode-IWX"
    };

    this.publicSvr.permitRequest(
      params,
      {
        headers: {
          apiID: "ffff-1655375927998-16925411121-0200",
          userKey: this.publicSvr.USER_GQ
        }
      },
      true
    ).then((res: any) => {
      if (res && res.IsOK == "1") {
        this.isCode = true;
      } else {
        this.msg.error(
          (res && res.Msg) ||
          "Verification failed."
        );
      }
    }).catch((error) => {
      console.error(
        "Verify code error:",
        error
      );

      this.msg.error(
        "Unable to verify code."
      );
    });
  }

  // =====================================================
  // RESET PASSWORD
  // =====================================================

  resetPassword() {
    if (
      this.resetInfo.newPassword &&
      this.resetInfo.confirmPassword &&
      this.resetInfo.newPassword !==
        this.resetInfo.confirmPassword
    ) {
      this.msg.error(
        "The entered passwords do not match."
      );

      return false;
    }

    if (
      !this.resetInfo.account ||
      !this.resetInfo.code ||
      !this.resetInfo.newPassword ||
      !this.resetInfo.confirmPassword
    ) {
      this.msg.error(
        "All information is required."
      );

      return false;
    }

    const params = {
      EmpNo: this.resetInfo.account,
      Password: this.resetInfo.newPassword,
      CaptchaCode: this.resetInfo.code,
      Func: "H5-ChangeEmpPasswordV2-IWX"
    };

    this.publicSvr.permitRequest(
      params,
      {
        headers: {
          apiID: "ffff-1655376433685-16925411121-0217",
          userKey: this.publicSvr.USER_GQ
        }
      },
      true
    ).then((res: any) => {
      if (res && res.IsOK == "1") {
        this.msg.success(
          "Password changed successfully"
        );

        this.closeForgetPasswordModal();
      } else {
        this.msg.error(
          (res && res.Msg) ||
          "Password reset failed."
        );
      }
    }).catch((error) => {
      console.error(
        "Reset password error:",
        error
      );

      this.msg.error(
        "Unable to reset password."
      );
    });
  }

  // =====================================================
  // CLOSE MODAL
  // =====================================================

  handleCancel() {
    this.isVisible = false;
  }
}
