import {
  Component,
  OnInit,
  ViewChild,
  ElementRef
} from '@angular/core';

import { HttpClient } from '@angular/common/http';

import {
  DomSanitizer,
  SafeUrl
} from '@angular/platform-browser';

import { NzMessageService } from 'ng-zorro-antd';

import {
  PublicNetService
} from 'src/app/routes/a_Public_net/Public_net.service';


@Component({
  selector: 'app-Meal-scan',

  templateUrl: './Meal-scan.component.html',

  styleUrls: ['./Meal-scan.component.less']
})


export class MealScanComponent implements OnInit {


  /* ============================================================
     SCAN INPUT
     ============================================================ */

  @ViewChild(
    'scanInput',
    { static: false }
  )
  scanInput: ElementRef;


  empNo: string = '';

  loading: boolean = false;

  proceeding: boolean = false;


  employeeDetail: any = null;

  entitlementStatus: string = '';


  /* ============================================================
     PHOTO
     ============================================================ */

  employeePhotoUrl: SafeUrl | null = null;

  photoLoading: boolean = false;


  private readonly PHOTO_API_BASE =
    'http://10.209.110.187:3000/api/image';


  /* ============================================================
     SITE
     ============================================================ */

  siteList: string[] = [];

  selectedSite: string = '';

  locationList: any[] = [];


  /* ============================================================
     LOGGED-IN USER
     ============================================================ */

  loginEmployeeId: string = '';


  /* ============================================================
     CONSTRUCTOR
     ============================================================ */

  constructor(

    private publicSvr: PublicNetService,

    private msgSvr: NzMessageService,

    private http: HttpClient,

    private sanitizer: DomSanitizer

  ) {}


  /* ============================================================
     INIT
     ============================================================ */

  ngOnInit(): void {

    /*
     * Get logged-in employee
     * and load active sites.
     */
    this.loadLoggedInUserSite();

  }


  /* ============================================================
     LOAD LOGGED-IN USER SITE
     ============================================================ */

  private loadLoggedInUserSite(): void {

    const userInfo =
      this.publicSvr.getSettingUserInfo();


    console.log(
      'Meal Scan - Logged-in user:',
      userInfo
    );


    /*
     * Employee ID mandatory.
     */
    if (
      !userInfo ||
      !userInfo.eeid
    ) {

      this.msgSvr.error(
        'Logged-in employee information not found.'
      );

      return;

    }


    this.loginEmployeeId =
      String(
        userInfo.eeid
      )
      .trim()
      .toUpperCase();


    /*
     * Get CMS user/site list.
     */
    this.publicSvr
      .getSitebasedUser()
      .subscribe(

        (userList: any[]) => {

          console.log(
            'Meal Scan - User/site list:',
            userList
          );


          if (
            !userList ||
            userList.length === 0
          ) {

            this.msgSvr.error(
              'No active site found for logged-in user.'
            );

            return;

          }


          /*
           * Extract only active sites
           * belonging to logged-in employee.
           */
          this.siteList =
            this.extractAllSites(
              userList,
              this.loginEmployeeId
            );


          console.log(
            'Meal Scan - Active sites:',
            this.siteList
          );


          if (
            this.siteList.length === 0
          ) {

            this.msgSvr.error(
              'No active site assigned to logged-in user.'
            );

            return;

          }


          /*
           * Default:
           *
           * Empty = All My Sites
           */
          this.selectedSite = '';


          /*
           * Build dropdown.
           */
          this.buildLocationList();


          /*
           * Focus scan input.
           */
          this.focusInput();

        },


        error => {

          console.error(
            'Meal Scan - getSitebasedUser error:',
            error
          );

          this.msgSvr.error(
            'Failed to load user site information.'
          );

        }

      );

  }


  /* ============================================================
     GET EEID FROM USER OBJECT
     ============================================================ */

  private getItemEeid(
    item: any
  ): string {

    if (!item) {

      return '';

    }


    const value =

      item.Eeid ||

      item.EEID ||

      item.EeidNo ||

      item.EmpId ||

      item.EMP_ID ||

      item.EmployeeId ||

      item.EmployeeCode ||

      item.eeid ||

      '';


    return String(value)
      .trim()
      .toUpperCase();

  }


  /* ============================================================
     ACTIVE USER CHECK
     ============================================================ */

  private isActiveUser(
    item: any
  ): boolean {

    if (!item) {

      return false;

    }


    let delFlag: any = 0;


    if (
      item.DelFlag !== undefined &&
      item.DelFlag !== null &&
      item.DelFlag !== ''
    ) {

      delFlag =
        item.DelFlag;

    }

    else if (
      item.DEL_FLAG !== undefined &&
      item.DEL_FLAG !== null &&
      item.DEL_FLAG !== ''
    ) {

      delFlag =
        item.DEL_FLAG;

    }

    else if (
      item.delFlag !== undefined &&
      item.delFlag !== null &&
      item.delFlag !== ''
    ) {

      delFlag =
        item.delFlag;

    }

    else if (
      item.del_flag !== undefined &&
      item.del_flag !== null &&
      item.del_flag !== ''
    ) {

      delFlag =
        item.del_flag;

    }


    /*
     * 0 / NULL = active
     * 1 = deleted
     */
    return Number(delFlag) === 0;

  }


  /* ============================================================
     EXTRACT ALL ACTIVE SITES
     ============================================================ */

  private extractAllSites(
    userList: any[],
    eeid: string
  ): string[] {

    const loginEeid =
      String(eeid)
        .trim()
        .toUpperCase();


    const sites: string[] = [];


    userList.forEach(
      item => {

        if (!item) {

          return;

        }


        /*
         * Employee match.
         */
        if (
          this.getItemEeid(item)
          !==
          loginEeid
        ) {

          return;

        }


        /*
         * Deleted user must not be used.
         */
        if (
          !this.isActiveUser(item)
        ) {

          return;

        }


        /*
         * Extract site.
         */
        const site =
          this.extractSite(item);


        if (site) {

          sites.push(site);

        }

      }
    );


    /*
     * Remove duplicate sites.
     */
    const uniqueSites: string[] = [];


    sites.forEach(
      site => {

        const exists =
          uniqueSites.some(
            x =>
              x.toUpperCase() ===
              site.toUpperCase()
          );


        if (!exists) {

          uniqueSites.push(site);

        }

      }
    );


    return uniqueSites;

  }


  /* ============================================================
     EXTRACT SITE
     ============================================================ */

  private extractSite(
    user: any
  ): string {

    if (!user) {

      return '';

    }


    const site =

      user.Site ||

      user.site ||

      user.SITE ||

      user.SiteName ||

      user.siteName ||

      user.WorkSite ||

      user.Worksite ||

      user.BasicCode ||

      user.Basiccode ||

      user.Location ||

      user.location ||

      '';


    return String(site)
      .trim();

  }


  /* ============================================================
     BUILD SITE DROPDOWN
     ============================================================ */

  private buildLocationList(): void {

    this.locationList = [

      {
        value: '',
        label: 'All My Sites'
      }

    ];


    this.siteList.forEach(
      site => {

        this.locationList.push({

          value: site,

          label: site

        });

      }
    );


    console.log(
      'Meal Scan - Site dropdown:',
      this.locationList
    );

  }


  /* ============================================================
     SITE CHANGE
     ============================================================ */

  onLocationChange(
    location: string
  ): void {

    console.log(
      'Meal Scan - Selected site:',
      location
    );


    /*
     * Empty = All My Sites.
     */
    if (
      !location
    ) {

      this.selectedSite = '';

    }

    else {

      /*
       * Validate selected site.
       */
      const validSite =
        this.siteList.some(
          site =>
            site.toUpperCase() ===
            String(location)
              .trim()
              .toUpperCase()
        );


      if (validSite) {

        this.selectedSite =
          String(location).trim();

      }

      else {

        this.selectedSite = '';

        this.msgSvr.warning(
          'Invalid site selected.'
        );

      }

    }


    /*
     * Clear current employee.
     */
    this.employeeDetail = null;

    this.entitlementStatus = '';

    this.employeePhotoUrl = null;


    /*
     * Clear badge input.
     */
    this.empNo = '';


    /*
     * Focus scanner.
     */
    this.focusInput();

  }


  /* ============================================================
     FOCUS INPUT
     * ============================================================ */

  focusInput(): void {

    setTimeout(
      () => {

        if (
          this.scanInput
        ) {

          this.scanInput.nativeElement.focus();

        }

      },
      100
    );

  }


  /* ============================================================
     SCAN ENTER
     * ============================================================ */

  onScanEnter(): void {

    this.search();

  }


  /* ============================================================
     SEARCH / GET EMPLOYEE
     * ============================================================ */

  search(
    fromButton?: boolean
  ): void {


    /* ============================================================
       CHECK BADGE
       ============================================================ */

    if (
      !this.empNo ||
      this.empNo.trim() === ''
    ) {

      this.msgSvr.error(
        'Please scan or enter an FX Badge.'
      );

      this.focusInput();

      return;

    }


    /* ============================================================
       CHECK SITE
       ============================================================ */

    if (
      this.siteList.length === 0
    ) {

      this.msgSvr.error(
        'No active site available.'
      );

      return;

    }


    /*
     * If selectedSite is specified,
     * it must belong to logged-in user.
     */
    if (
      this.selectedSite
    ) {

      const validSite =
        this.siteList.some(
          site =>
            site.toUpperCase() ===
            this.selectedSite.toUpperCase()
        );


      if (!validSite) {

        this.msgSvr.error(
          'Selected site is not assigned to you.'
        );

        return;

      }

    }


    /* ============================================================
       START LOADING
       ============================================================ */

    this.loading = true;

    this.employeeDetail = null;

    this.entitlementStatus = '';

    this.employeePhotoUrl = null;


    /* ============================================================
       GET F1 MENU
       ============================================================ */

    const params: any = {

      FxBadge:
        this.empNo.trim(),

      /*
       * Empty = All My Sites
       */
      Site:
        this.selectedSite || '',

      /*
       * Logged-in employee.
       */
      EmpId:
        this.loginEmployeeId,

      SystemName:
        'CMS',

      Func:
        'CMS-GetF1MenuByFxbadge'

    };


    console.log(
      '================================================'
    );

    console.log(
      'MEAL SCAN SEARCH'
    );

    console.log(
      'Badge:',
      params.FxBadge
    );

    console.log(
      'Site:',
      params.Site || 'ALL MY SITES'
    );

    console.log(
      'Logged-in Employee:',
      params.EmpId
    );

    console.log(
      'Request:',
      params
    );

    console.log(
      '================================================'
    );


    this.publicSvr
      .iwxPostF1(

        params,

        {

          headers: {

            apiID:
              'ffff-1789807467770-10208193125-1466',

            userKey:
              'E9785F13CCEAFB251A6AB147AFD24336'

          }

        },

        true

      )


      .then(
        (res: any) => {

          this.loading = false;


          console.log(
            'Meal Scan Search Response:',
            res
          );


          /*
           * ======================================================
           * SUCCESS
           * ======================================================
           */

          if (
            res &&
            String(res.IsOK) === '1' &&
            res.List &&
            res.List.length > 0
          ) {

            const data =
              res.List[0];


            this.employeeDetail = {

              FxBadge:
                data.FxBadge ||
                data.FX_BADGE,

              Name:
                data.Name ||
                data.NAME,

              Site:
                data.Site ||
                data.SITE,

              MealPreference:
                data.MealPreference ||
                data.MEAL_PREFERENCE,

              FirstDateFxbl:
                data.FirstDateFxbl ||
                data.FIRST_DATE_FXBL,

              LastDateFxbl:
                data.LastDateFxbl ||
                data.LAST_DATE_FXBL

            };


            this.entitlementStatus =
              data.EntitlementStatus ||
              data.ENTITLEMENT_STATUS ||
              '';


            /*
             * Safety check:
             *
             * If selected site is specified,
             * returned employee site must match.
             */
            if (
              this.selectedSite &&
              this.employeeDetail.Site &&
              this.employeeDetail.Site
                .toUpperCase() !==
              this.selectedSite
                .toUpperCase()
            ) {

              this.loading = false;

              this.msgSvr.error(
                'This FX Badge does not belong to the selected site.'
              );

              this.employeeDetail = null;

              this.entitlementStatus = '';

              this.fetchEmployeePhoto(
                this.empNo.trim()
              );

              this.clearInputOnly();

              return;

            }


            /*
             * Fetch employee photo.
             */
            this.fetchEmployeePhoto(
              this.employeeDetail.FxBadge
            );


            /*
             * Insert scan log.
             */
            this.proceed();

          }

          else {

            /*
             * Badge not found.
             */
            this.employeeDetail = {

              FxBadge:
                this.empNo.trim(),

              Name: '',

              Site: '',

              MealPreference: '',

              FirstDateFxbl: '',

              LastDateFxbl: ''

            };


            this.entitlementStatus =
              'NOT_UPDATED';


            this.fetchEmployeePhoto(
              this.empNo.trim()
            );


            this.msgSvr.error(
              'FX Badge not found for the selected site.'
            );


            this.clearInputOnly();

          }

        }

      )


      .catch(
        err => {

          this.loading = false;


          console.error(
            'Meal Scan Search Error:',
            err
          );


          this.msgSvr.error(
            'Failed to fetch employee details.'
          );


          this.focusInput();

        }
      );

  }


  /* ============================================================
     FETCH EMPLOYEE PHOTO
     ============================================================ */

  fetchEmployeePhoto(
    fxBadge: string
  ): void {

    if (!fxBadge) {

      return;

    }


    this.photoLoading = true;


    this.http
      .get(

        `${this.PHOTO_API_BASE}/${fxBadge}`,

        {
          responseType: 'blob'
        }

      )


      .subscribe({

        next:
          (blob) => {

            const objectUrl =
              URL.createObjectURL(
                blob
              );


            this.employeePhotoUrl =
              this.sanitizer
                .bypassSecurityTrustUrl(
                  objectUrl
                );


            this.photoLoading = false;

          },


        error:
          () => {

            this.employeePhotoUrl = null;

            this.photoLoading = false;

          }

      });

  }


  /* ============================================================
     PROCEED / INSERT SCAN LOG
     ============================================================ */

  proceed(): void {

    if (
      !this.employeeDetail
    ) {

      return;

    }


    this.proceeding = true;


    /* ============================================================
       INSERT SCAN LOG
       ============================================================ */

    const params: any = {

      FxBadge:
        this.employeeDetail.FxBadge,

      /*
       * Selected site.
       *
       * Empty = All My Sites.
       */
      Site:
        this.selectedSite || '',

      /*
       * Logged-in user.
       */
      EmpId:
        this.loginEmployeeId,

      SystemName:
        'CMS',

      Func:
        'CMS-InsertMealScanLog'

    };


    console.log(
      '================================================'
    );

    console.log(
      'MEAL SCAN INSERT'
    );

    console.log(
      'Badge:',
      params.FxBadge
    );

    console.log(
      'Site:',
      params.Site || 'ALL MY SITES'
    );

    console.log(
      'Logged-in Employee:',
      params.EmpId
    );

    console.log(
      'Request:',
      params
    );

    console.log(
      '================================================'
    );


    this.publicSvr
      .iwxPostF1(

        params,

        {

          headers: {

            apiID:
              'ffff-1789818962176-10208193125-1483',

            userKey:
              'E9785F13CCEAFB251A6AB147AFD24336'

          }

        },

        true

      )


      .then(
        (res: any) => {

          this.proceeding = false;


          console.log(
            'Meal Scan Insert Response:',
            res
          );


          if (
            res &&
            String(res.IsOK) === '1' &&
            res.List &&
            res.List.length > 0
          ) {

            const logResult =
              res.List[0];


            const status =
              logResult.Status ||
              logResult.STATUS ||
              '';


            const message =
              logResult.StatusMessage ||
              logResult.STATUS_MESSAGE ||
              '';


            /* ==================================================
               SUCCESS
               ================================================== */

            if (
              status === 'SUCCESS'
            ) {

              this.msgSvr.success(
                'Meal collection recorded successfully!'
              );

            }


            /* ==================================================
               ALREADY SCANNED
               ================================================== */

            else if (
              status ===
              'ALREADY_SCANNED'
            ) {

              this.entitlementStatus =
                'ALREADY_SCANNED';


              this.msgSvr.error(

                message ||
                'Meal already collected today'

              );

            }


            /* ==================================================
               SITE MISMATCH
               ================================================== */

            else if (
              status ===
              'SITE_MISMATCH'
            ) {

              this.entitlementStatus =
                'SITE_MISMATCH';


              this.msgSvr.error(

                message ||
                'Employee does not belong to selected site.'

              );

            }


            /* ==================================================
               INVALID
               ================================================== */

            else {

              this.msgSvr.error(

                message ||
                'Invalid scan.'

              );

            }


            /*
             * Clear scanner.
             */
            this.clearInputOnly();

          }

          else {

            this.msgSvr.error(

              res.Msg ||
              'Failed to log scan.'

            );

          }

        }

      )


      .catch(
        err => {

          this.proceeding = false;


          console.error(
            'Meal Scan Insert Error:',
            err
          );


          this.msgSvr.error(
            'Failed to log scan. Please try again.'
          );

        }
      );

  }


  /* ============================================================
     CLEAR INPUT ONLY
     ============================================================ */

  clearInputOnly(): void {

    this.empNo = '';


    /*
     * Keep employee detail visible
     * after scan result.
     *
     * Only clear scanner input.
     */


    this.focusInput();

  }


  /* ============================================================
     CANCEL
     ============================================================ */

  cancelScan(): void {

    this.employeeDetail = null;

    this.entitlementStatus = '';

    this.empNo = '';

    this.employeePhotoUrl = null;


    this.proceeding = false;

    this.loading = false;


    this.focusInput();

  }

}

      this.publicSvr.getSettingUserInfo();

    console.log(
      'Logged-in user information:',
      userInfo
    );

    if (!userInfo || !userInfo.eeid) {
      this.msgSvr.error(
        'Logged-in user EEID is missing'
      );

      return;
    }

    this.publicSvr.getSitebasedUser().subscribe(
      (userList: any[]) => {
        console.log(
          'User manage response:',
          userList
        );

        if (
          !userList ||
          userList.length === 0
        ) {
          this.msgSvr.error(
            'No active user data returned'
          );

          return;
        }

        this.siteList =
          this.extractAllSites(
            userList,
            userInfo.eeid
          );

        console.log(
          'Active site IDs:',
          this.siteList
        );

        if (
          !this.siteList ||
          this.siteList.length === 0
        ) {
          this.msgSvr.error(
            'No active site found for logged-in user'
          );

          this.source = [];
          this.total = 0;

          return;
        }

this.siteOptions = this.siteList.map(siteId => {
  return {
    value: siteId,
    label: this.getSiteName(siteId)
  };
});

        /*
         * Default selected site.
         */
        this.site =
          this.siteList[0];

        this.selectedSite =
          this.site;

        this.siteDisplay =
          this.getSiteName(this.site);

        console.log(
          'Site options:',
          this.siteOptions
        );

        console.log(
          'API site:',
          this.selectedSite
        );

        console.log(
          'Display site:',
          this.siteDisplay
        );

        this.search(true);
      },
      error => {
        console.error(
          'User site API error:',
          error
        );

        this.msgSvr.error(
          'Unable to load logged-in user sites'
        );
      }
    );
  }

  /**
   * Get employee ID.
   */
  private getItemEeid(item: any): string {
    if (!item) {
      return '';
    }

    return String(
      item.Eeid ||
      item.EEID ||
      item.EeidNo ||
      item.EmpId ||
      item.EMP_ID ||
      item.EmployeeId ||
      item.EmployeeCode ||
      item.eeid ||
      ''
    )
      .trim()
      .toUpperCase();
  }

  /**
   * Active user check.
   */
  private isActiveUser(item: any): boolean {
    if (!item) {
      return false;
    }

    let delFlag: any = 0;

    if (
      item.DelFlag !== undefined &&
      item.DelFlag !== null &&
      item.DelFlag !== ''
    ) {
      delFlag = item.DelFlag;
    } else if (
      item.DEL_FLAG !== undefined &&
      item.DEL_FLAG !== null &&
      item.DEL_FLAG !== ''
    ) {
      delFlag = item.DEL_FLAG;
    } else if (
      item.delFlag !== undefined &&
      item.delFlag !== null &&
      item.delFlag !== ''
    ) {
      delFlag = item.delFlag;
    } else if (
      item.del_flag !== undefined &&
      item.del_flag !== null &&
      item.del_flag !== ''
    ) {
      delFlag = item.del_flag;
    }

    return Number(delFlag) === 0;
  }

  /**
   * Extract active sites for logged-in user.
   */
  private extractAllSites(
    userList: any[],
    eeid: string
  ): string[] {
    const loginEeid =
      String(eeid)
        .trim()
        .toUpperCase();

    const sites =
      userList
        .filter(item => {
          if (!item) {
            return false;
          }

          const itemEeid =
            this.getItemEeid(item);

          if (itemEeid !== loginEeid) {
            return false;
          }

          return this.isActiveUser(item);
        })
        .map(item => {
          return this.extractSite(item);
        })
        .filter(site => !!site)
        .map(site => {
          return String(site).trim();
        });

    return Array.from(
      new Set(sites)
    );
  }

  /**
   * Extract raw site ID.
   */
  private extractSite(item: any): string {
    if (!item) {
      return '';
    }

    return String(
      item.Site ||
      item.site ||
      item.SITE ||
      item.SiteId ||
      item.siteId ||
      ''
    ).trim();
  }

  /**
   * Convert site ID to display text.
   *
   * 69941 -> 69941-FXCN
   * 95326 -> 95326-FXBL
   */
getSiteName(siteId: any): string {
  const id = String(siteId || '').trim();

  if (!id) {
    return '';
  }

  return this.siteNameMap[id] || id;
}

  /**
   * Site dropdown change.
   */
onSiteChange(siteId: string): void {
  this.selectedSite = siteId || '';

  this.siteDisplay = this.selectedSite
    ? this.getSiteName(this.selectedSite)
    : 'All My Sites';

  console.log(
    'UI site:',
    this.siteDisplay
  );

  console.log(
    'API site:',
    this.selectedSite
  );

  this.search(true);
}

  /**
   * Search.
   */
  search(first?: boolean): void {
    this.checkedTableData = [];

    if (first) {
      this.pi = 1;
    }

    this.loading = true;

    this.searchHandle(false);
  }

  /**
   * F1 API search.
   */
  searchHandle(isExport?: boolean): void {
    const userInfo =
      this.publicSvr.getSettingUserInfo();

    if (!userInfo || !userInfo.eeid) {
      this.loading = false;
      this.exportLoading = false;

      this.msgSvr.error(
        'Logged-in user EEID is missing'
      );

      return;
    }

    if (
      !this.siteList ||
      this.siteList.length === 0
    ) {
      this.loading = false;
      this.exportLoading = false;

      this.source = [];
      this.total = 0;

      this.msgSvr.error(
        'No active site found'
      );

      return;
    }

    const params: any = {
      Groups: this.groups || '',
      FxBadge: this.fxBadge || '',
      EmpId: userInfo.eeid,
      SystemName: 'CMS',

      /*
       * API receives 69941.
       * UI shows 69941-FXCN.
       */
      Site: this.selectedSite || '',

      StartPage: isExport
        ? 1
        : this.pi,

      PageSize: isExport
        ? 999999999999999
        : this.ps,

      Func: 'CMS-GetF1MenuManage'
    };

    const formValues =
      this.sf && this.sf.value
        ? this.sf.value
        : {};

    const finalParams: any =
      deepMerge(
        params,
        formValues
      );

    /*
     * Force site and employee values.
     */
    finalParams.Site =
      this.selectedSite || '';

    finalParams.EmpId =
      userInfo.eeid;

    console.log(
      'F1 Menu Manage params:',
      finalParams
    );

    this.publicSvr.iwxPostF1(
      finalParams,
      {
        headers: {
          apiID:
            'ffff-1789633596388-10208193125-1306',

          userKey:
            'E9785F13CCEAFB251A6AB147AFD24336'
        }
      },
      true
    )
    .then((res: any) => {
      this.loading = false;
      this.exportLoading = false;

      console.log(
        'F1 Menu Manage response:',
        res
      );

      if (
        !res ||
        String(res.IsOK) !== '1'
      ) {
        this.source = [];
        this.total = 0;

        this.msgSvr.error(
          res && res.Msg
            ? res.Msg
            : 'Unable to load F1 menu data'
        );

        return;
      }

      const list =
        Array.isArray(res.List)
          ? res.List
          : [];

      if (isExport) {
        this.exportExcel(list);
        return;
      }

      this.source =
        list;

      this.total =
        list.length > 0
          ? Number(
              list[0].Rowscount ||
              list.length
            )
          : 0;
    })
    .catch(error => {
      this.loading = false;
      this.exportLoading = false;

      console.error(
        'F1 Menu Manage API error:',
        error
      );

      this.msgSvr.error(
        'F1 Menu Manage API failed'
      );
    });
  }

  /**
   * Export Excel.
   */
  private exportExcel(
    list: any[]
  ): void {
    if (
      !list ||
      list.length === 0
    ) {
      this.msgSvr.error(
        'No Data!'
      );

      return;
    }

    const allData =
      list.map(item => {
        return {
          Item: item.Rn,
          Group: item.Groups,
          Site: this.getSiteName(
            item.Site ||
            this.selectedSite
          ),
          FxBadge: item.FxBadge,
          Name: item.Name,
          MealPreference: item.MealPreference,
          FirstDateFxbl: item.FirstDateFxbl,
          LastDateFxbl: item.LastDateFxbl,
          AddBy: item.AddBy,
          AddDate: item.AddDate,
          LastEditBy: item.EditBy,
          LastEditDate: item.EditDate
        };
      });

    const fileName =
      'F1_Meal_Management_' +
      new Date().getTime() +
      '.xlsx';

    this.publicSvr.exportHandle(
      allData,
      13,
      fileName
    );
  }

   */
  openDrawer(name: string): void {
    if (
      name === 'Edit' &&
      this.checkedTableData.length !== 1
    ) {
      this.shareSvr.showMsg(
        'error',
        this.i18n.fanyi(
          'table.content.edit-data-select'
        )
      );

      return;
    }

    this.shareSvr.openDrawer(
      name,
      AddDetailsDrawComponent,
      {
        option: name,
        editData:
          this.checkedTableData[0],

        /*
         * Raw site ID for API.
         */
        site:
          this.selectedSite ||
          this.site,

        /*
         * Display value for UI.
         */
        siteDisplay:
          this.siteDisplay,

        siteList:
          this.siteList,

        siteOptions:
          this.siteOptions
      },
      result => {
        if (result) {
          this.search(true);
        }
      }
    );
  }

  /**
   * Delete.
   */
  delete(): void {
    if (
      this.checkedTableData.length === 0
    ) {
      this.shareSvr.showMsg(
        'error',
        this.i18n.fanyi(
          'table.content.edit-data-select'
        )
      );

      return;
    }

    this.delIdx = 0;

    this.shareSvr.modalSvr.confirm({
      nzTitle: 'Prompt',

      nzContent:
        'Are you sure you want to delete the selected ' +
        this.checkedTableData.length +
        ' data?',

      nzOnOk: () => {
        this.delHandle();
      }
    });
  }

  /**
   * Delete API.
   */
  delHandle(): void {
    const ids: any[] = [];

    this.checkedTableData.forEach(item => {
      if (
        item &&
        item.Id !== undefined &&
        item.Id !== null
      ) {
        ids.push(item.Id);
      }
    });

    if (ids.length === 0) {
      this.msgSvr.error(
        'Selected records do not contain Id'
      );

      return;
    }

    const userInfo =
      this.publicSvr.getSettingUserInfo();

    const params = {
      Ids: ids.join(',') + ',',
      EditBy: userInfo.eeid,
      Func: 'CMS-DeleteF1MenuManage'
    };

    console.log(
      'Delete params:',
      params
    );

    this.publicSvr.iwxPostF1(
      params,
      {
        headers: {
          apiID:
            'ffff-1789634776545-10208193125-1341',

          userKey:
            'E9785F13CCEAFB251A6AB147AFD24336'
        }
      },
      true
    )
    .then((res: any) => {
      if (
        res &&
        String(res.IsOK) === '1'
      ) {
        this.msgSvr.success(
          'Success'
        );

        this.search(true);
      } else {
        this.msgSvr.error(
          res && res.Msg
            ? res.Msg
            : 'Delete failed'
        );
      }
    })
    .catch(error => {
      console.error(
        'Delete API error:',
        error
      );

      this.msgSvr.error(
        'Delete failed'
      );
    });
  }

  /**
   * Export.
   */
  exportData(): void {
    if (
      !this.siteList ||
      this.siteList.length === 0
    ) {
      this.msgSvr.error(
        'No active site found'
      );

      return;
    }

    this.exportLoading = true;

    this.searchHandle(true);
  }

  /**
   * Table changes.
   */
  change(ret: STChange): void {
    if (
      ret.type === 'checkbox'
    ) {
      this.checkedTableData =
        ret.checkbox || [];
    }

    if (
      ret.type === 'pi'
    ) {
      this.pi =
        ret.pi;

      this.search(false);
    }

    if (
      ret.type === 'ps'
    ) {
      this.ps =
        ret.ps;

      this.search(true);
    }
  }
