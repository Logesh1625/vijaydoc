DownloadExcelfile() {

  console.log("DOWNLOAD TEMPLATE START");

  const mealNames = this.mealPreferenceList
    .map((item: any) => item.label)
    .filter((x: any) => x)
    .map((x: any) => String(x).trim());

  console.log("MEAL LIST:", mealNames);

  XlsxPopulate.fromBlankAsync()
    .then((workbook: any) => {

      console.log("WORKBOOK CREATED");

      const sheet = workbook.sheet(0);
      sheet.name("Template");

      // ==========================
      // HEADER
      // ==========================
      const headers = [
        "Groups",
        "Sub Group",
        "Meal Preference",
        "FX Badge",
        "Name",
        "First Day at FXBL",
        "Last Day at FXBL"
      ];

      sheet.range("A1:G1").value([headers]);

      // ==========================
      // COLUMN WIDTH
      // ==========================
      sheet.column("A").width(20);
      sheet.column("B").width(20);
      sheet.column("C").width(25);
      sheet.column("D").width(20);
      sheet.column("E").width(30);
      sheet.column("F").width(22);
      sheet.column("G").width(22);

      // ==========================
      // HEADER STYLE
      // ==========================
      sheet.range("A1:G1").style({
        bold: true,
        horizontalAlignment: "center",
        verticalAlignment: "center"
      });

      // ==========================
      // DATE FORMAT
      // ==========================
      sheet.range("F2:G500").style(
        "numberFormat",
        "dd-mm-yyyy"
      );

      // ==========================
      // MEAL MASTER SHEET
      // ==========================
      if (mealNames.length > 0) {

        const masterSheet = workbook.addSheet("MealMaster");

        masterSheet.cell("A1").value("Meal Preference");

        for (let i = 0; i < mealNames.length; i++) {
          masterSheet
            .cell(`A${i + 2}`)
            .value(mealNames[i]);
        }

        console.log(
          "MEAL MASTER CREATED:",
          mealNames.length
        );

        // Hide master sheet
        masterSheet.hidden(true);

        const lastRow = mealNames.length + 1;

        // ==========================
        // DROPDOWN
        // ==========================
        sheet
          .range("C2:C500")
          .dataValidation({
            type: "list",
            allowBlank: true,
            showErrorMessage: true,
            error: "Please select a valid Meal Preference.",
            errorTitle: "Invalid Meal Preference",
            formula1: `'MealMaster'!$A$2:$A$${lastRow}`
          });

        console.log("DROPDOWN CREATED");
      }

      // ==========================
      // OUTPUT
      // ==========================
      return workbook.outputAsync();
    })
    .then((data: any) => {

      console.log("EXCEL BUFFER CREATED");

      const blob = new Blob(
        [data],
        {
          type:
            "application/vnd.openxmlformats-officedocument.spreadsheetml.sheet"
        }
      );

      FileSaver.saveAs(
        blob,
        "F1Meal-info Template.xlsx"
      );

      console.log("EXCEL DOWNLOAD SUCCESS");

      this.msgSvr.success(
        "Excel template downloaded successfully."
      );
    })
    .catch((error: any) => {

      console.error(
        "================================"
      );

      console.error(
        "EXCEL TEMPLATE ERROR:",
        error
      );

      console.error(
        "ERROR MESSAGE:",
        error && error.message
      );

      console.error(
        "ERROR STACK:",
        error && error.stack
      );

      console.error(
        "================================"
      );

      this.msgSvr.error(
        "Unable to generate Excel template. Check browser console."
      );
    });
}
