  @ViewChild('sf', { static: false })

  @ViewChild('st', { static: false })
  private st: STComponent;

  constructor(
    private publicSvr: PublicNetService,
    public settings: SettingsService,
    private msgSvr: NzMessageService,
    private shareSvr: SharedModule,
    private modelSvr: NzModalService,
    private router: ActivatedRoute,
    private i18n: I18NService
  ) {}

  title = 'F1 Meal Management';

  sfSchema: SFSchema;

  source: any[] = [];

  loading = false;

  exportLoading = false;

  pi = 1;

  ps = 10;

  total = 0;

  /**
   * Default active site.
   */
  site = '';

  /**
   * Active sites belonging to logged-in user.
   */
  siteList: string[] = [];

  /**
   * Selected site.
   * Empty string = All active sites.
   */
  selectedSite = '';

  groups = '';

  fxBadge = '';

  groupsList: any[] = [];

  checkedTableData: any[] = [];

  delIdx = 0;

  pages: STPage = {
    show: true,
    front: false,
    showSize: true
  };

  columns: STColumn[] = [

    {
      title: '',
      type: 'checkbox',
      fixed: 'left',
      width: '50px',

    },

    {

    {

    {

    {

    },

    {

    },

    {

    {

  ngOnInit(): void {

    const form = [
      {
        key: 'Groups',
        name: 'Group',
        type: 'string'
      },
      {
        key: 'FxBadge',
        name: 'FX Badge',
        type: 'string'
      },
      {
        key: 'AddBy',
        name: 'Add By',
        type: 'string'
      }
    ];

    this.sfSchema =
      this.publicSvr.jointSchema(form);

    console.log(
      'Current user from SettingsService:',
      this.publicSvr.getSettingUserInfo()
    );

    this.loadLoggedInUserSite();
  }


  /**
   * Load only active sites of logged-in user.
   */
  private loadLoggedInUserSite(): void {

    const userInfo =
      this.publicSvr.getSettingUserInfo();

    console.log(
      'Logged-in user information:',
      userInfo
    );

    if (!userInfo || !userInfo.eeid) {

      this.msgSvr.error(
        'Logged-in user EEID is missing'
      );

      return;
    }

    console.log(
      'Calling CMS-Get_user_manage for EEID:',
      userInfo.eeid
    );

    this.publicSvr.getSitebasedUser().subscribe(
      (userList: any[]) => {

        console.log(
          'CMS-Get_user_manage user list:',
          userList
        );

        if (
          !userList ||
          userList.length === 0
        ) {

          this.msgSvr.error(
            'No active user data returned from CMS-Get_user_manage'
          );

          return;
        }

        /*
         * Extract only active user sites.
         */
        this.siteList =
          this.extractAllSites(
            userList,
            userInfo.eeid
          );

        console.log(
          'Resolved active logged-in user sites:',
          this.siteList
        );

        if (
          this.siteList.length === 0
        ) {

          this.msgSvr.error(
            'No active site found for logged-in user'
          );

          this.source = [];
          this.total = 0;

          return;
        }

        /*
         * Default active site.
         */
        this.site =
          this.siteList[0];

        /*
         * Empty = All active sites.
         */
        this.selectedSite = '';

        /*
         * Load F1 data only after
         * active sites are available.
         */
        this.search(true);

      },
      error => {

        console.error(
          'CMS-Get_user_manage error:',
          error
        );

        this.msgSvr.error(
          'Unable to load logged-in user information'
        );

      }
    );
  }


  /**
   * Get employee ID from API response.
   */
  private getItemEeid(item: any): string {

    return String(
      item.Eeid ||
      item.EEID ||
      item.EeidNo ||
      item.EmpId ||
      item.EMP_ID ||
      item.EmployeeId ||
      item.EmployeeCode ||
      item.eeid ||
      ''
    )
      .trim()
      .toUpperCase();
  }


  /**
   * Check whether user is active.
   *
   * DEL_FLAG:
   * 0 / NULL = active
   * 1 = deleted
   */
  private isActiveUser(item: any): boolean {

    if (!item) {
      return false;
    }

    const delFlag =
      item.DelFlag ??
      item.DEL_FLAG ??
      item.delFlag ??
      item.del_flag ??
      0;

    return Number(delFlag) === 0;
  }


  /**
   * Return every unique ACTIVE site
   * belonging to logged-in employee.
   */
  private extractAllSites(
    userList: any[],
    eeid: string
  ): string[] {

    const loginEeid =
      String(eeid)
        .trim()
        .toUpperCase();

    const sites =
      userList

        .filter(item => {

          if (!item) {
            return false;
          }

          const itemEeid =
            this.getItemEeid(item);

          /*
           * Employee must match.
           */
          if (itemEeid !== loginEeid) {
            return false;
          }

          /*
           * Deleted users must be excluded.
           */
          if (!this.isActiveUser(item)) {
            return false;
          }

          return true;
        })

        .map(item =>
          this.extractSite(item)
        )

        .filter(site => !!site);

    return Array.from(
      new Set(sites)
    );
  }


  /**
   * Extract site value from API response.
   */
  private extractSite(user: any): string {

    if (!user) {
      return '';
    }

    const site =
      user.Site ||
      user.site ||
      user.SITE ||
      user.SiteName ||
      user.siteName ||
      user.WorkSite ||
      user.Worksite ||
      user.BasicCode ||
      user.Basiccode ||
      user.Location ||
      user.location ||
      '';

    return String(site).trim();
  }


  /**
   * Search / reload grid.
   */
  search(first?: boolean): void {

    this.checkedTableData = [];

    if (first) {
      this.pi = 1;
    }

    this.loading = true;

    this.searchHandle(false);
  }


  /**
   * Search API.
   */
  searchHandle(
    isExport?: boolean
  ): void {

    const userInfo =
      this.publicSvr.getSettingUserInfo();

    if (
      !userInfo ||
      !userInfo.eeid
    ) {

      this.loading = false;
      this.exportLoading = false;

      this.msgSvr.error(
        'Logged-in user EEID is missing'
      );

      return;
    }

    /*
     * No active sites = no data.
     */
    if (
      !this.siteList ||
      this.siteList.length === 0
    ) {

      this.loading = false;
      this.exportLoading = false;

      this.source = [];
      this.total = 0;

      this.msgSvr.error(
        'No active site found for logged-in user'
      );

      return;
    }


    const params: any = {

      Groups:
        this.groups || '',

      FxBadge:
        this.fxBadge || '',

      /*
       * Logged-in employee.
       */
      EmpId:
        userInfo.eeid,

      SystemName:
        'CMS',

      /*
       * Empty = All active sites.
       */
      Site:
        this.selectedSite || '',

      StartPage:
        isExport
          ? 1
          : this.pi,

      PageSize:
        isExport
          ? 999999999999999
          : this.ps,

      Func:
        'CMS-GetF1MenuManage'
    };


    const formValues =
      this.sf && this.sf.value
        ? this.sf.value
        : {};


    const finalParams: any =
      deepMerge(
        params,
        formValues
      );


    /*
     * Site must always come
     * from site dropdown.
     */
    finalParams.Site =
      this.selectedSite || '';


    /*
     * EmpId must always be
     * logged-in employee.
     */
    finalParams.EmpId =
      userInfo.eeid;


    console.log(
      'F1 Menu Manage final params:',
      finalParams
    );


    this.publicSvr.iwxPostF1(
      finalParams,
      {
        headers: {
          apiID:
            'ffff-1789633596388-10208193125-1306',

          userKey:
            'E9785F13CCEAFB251A6AB147AFD24336'
        }
      },
      true
    )

    .then((res: any) => {

      this.loading = false;
      this.exportLoading = false;


      console.log(
        'F1 Menu Manage response:',
        res
      );


      if (
        !res ||
        String(res.IsOK) !== '1'
      ) {

        if (!isExport) {

          this.source = [];
          this.total = 0;
        }

        this.msgSvr.error(
          res && res.Msg
            ? res.Msg
            : 'Unable to load F1 menu data'
        );

        return;
      }


      const list =
        Array.isArray(res.List)
          ? res.List
          : [];


      if (isExport) {

        this.exportExcel(list);

        return;
      }


      this.source =
        list;


      this.total =
        list.length > 0
          ? Number(
              list[0].Rowscount || 0
            )
          : 0;

    })

    .catch(error => {

      this.loading = false;
      this.exportLoading = false;

      console.error(
        'F1 Menu Manage API error:',
        error
      );

      this.msgSvr.error(
        'F1 Menu Manage API failed'
      );

    });
  }


  /**
   * Export Excel.
   */
  private exportExcel(
    list: any[]
  ): void {

    if (
      !list ||
      list.length === 0
    ) {

      this.msgSvr.error(
        'No Data!'
      );

      return;
    }


    const allData =
      list.map(item => {

        return {

          Item:
            item.Rn,

          Group:
            item.Groups,

          Site:
            item.Site,

          FxBadge:
            item.FxBadge,

          Name:
            item.Name,

          MealPreference:
            item.MealPreference,

          FirstDateFxbl:
            item.FirstDateFxbl,

          LastDateFxbl:
            item.LastDateFxbl,

          AddBy:
            item.AddBy,

          AddDate:
            item.AddDate,

          LastEditBy:
            item.EditBy,

          LastEditDate:
            item.EditDate
        };
      });


    const fileName =
      'F1_Meal_Management_' +
      new Date().getTime() +
      '.xlsx';


    this.publicSvr.exportHandle(
      allData,
      13,
      fileName
    );
  }


  /**
   * Add.
   */
  add(): void {

    this.openDrawer(
      'Add'
    );
  }


  /**
   * Edit.
   */
  update(): void {

    if (
      this.checkedTableData.length !== 1
    ) {

      this.shareSvr.showMsg(
        'error',
        this.i18n.fanyi(
          'table.content.edit-data-select'
        )
      );

      return;
    }

    this.openDrawer(
      'Edit'
    );
  }


  /**
   * Bulk import.
   */
  importbulk(): void {

    this.shareSvr.openDrawer(
      'Import',
      F1ManageImportComponent,
      {},
      result => {

        if (result) {

          this.search(true);
        }
      }
    );
  }


  /**
   * Open Add / Edit drawer.
   */
  openDrawer(
    name: string
  ): void {

    if (
      name === 'Edit' &&
      this.checkedTableData.length !== 1
    ) {

      this.shareSvr.showMsg(
        'error',
        this.i18n.fanyi(
          'table.content.edit-data-select'
        )
      );

      return;
    }


    console.log(
      'Selected row:',
      this.checkedTableData[0]
    );


    this.shareSvr.openDrawer(
      name,
      AddDetailsDrawComponent,
      {

        option:
          name,

        editData:
          this.checkedTableData[0],

        /*
         * Send active selected site.
         */
        site:
          this.selectedSite ||
          this.site,

        /*
         * Send only active sites.
         */
        siteList:
          this.siteList

      },

      result => {

        if (result) {

          this.search(true);
        }
      }
    );
  }


  /**
   * Delete selected F1 records.
   */
  delete(): void {

    if (
      this.checkedTableData.length === 0
    ) {

      this.shareSvr.showMsg(
        'error',
        this.i18n.fanyi(
          'table.content.edit-data-select'
        )
      );

      return;
    }


    this.delIdx = 0;


    this.shareSvr.modalSvr.confirm({

      nzTitle:
        'Prompt',

      nzContent:
        'Are you sure you want to delete the selected ' +
        this.checkedTableData.length +
        ' data?',

      nzOnOk: () => {

        this.delHandle();
      }
    });
  }


  /**
   * Delete F1 records.
   */
  delHandle(): void {

    const ids: any[] = [];


    this.checkedTableData.forEach(
      item => {

        if (
          item &&
          item.Id !== undefined &&
          item.Id !== null
        ) {

          ids.push(
            item.Id
          );
        }
      }
    );


    if (
      ids.length === 0
    ) {

      this.msgSvr.error(
        'Selected records do not contain Id'
      );

      return;
    }


    const userInfo =
      this.publicSvr.getSettingUserInfo();


    const params = {

      Ids:
        ids.join(',') + ',',

      EditBy:
        userInfo.eeid,

      Func:
        'CMS-DeleteF1MenuManage'
    };


    console.log(
      'Delete params:',
      params
    );


    this.publicSvr.iwxPostF1(
      params,
      {
        headers: {

          apiID:
            'ffff-1789634776545-10208193125-1341',

          userKey:
            'E9785F13CCEAFB251A6AB147AFD24336'
        }
      },
      true
    )

    .then((res: any) => {

      if (
        res &&
        String(res.IsOK) === '1'
      ) {

        this.msgSvr.success(
          'Success'
        );

        this.search(true);

      } else {

        this.msgSvr.error(
          res && res.Msg
            ? res.Msg
            : 'Delete failed'
        );
      }

    })

    .catch(error => {

      console.error(
        'Delete API error:',
        error
      );

      this.msgSvr.error(
        'Delete failed'
      );
    });
  }


  /**
   * Export.
   */
  exportData(): void {

    if (
      !this.siteList ||
      this.siteList.length === 0
    ) {

      this.msgSvr.error(
        'No active site found for logged-in user'
      );

      return;
    }


    this.exportLoading = true;

    this.searchHandle(
      true
    );
  }


  /**
   * Table change.
   */
  change(
    ret: STChange
  ): void {

    if (
      ret.type === 'checkbox'
    ) {

      this.checkedTableData =
        ret.checkbox || [];
    }


    if (
      ret.type === 'pi'
    ) {

      this.pi =
        ret.pi;

      this.search(
        false
      );
    }


    if (
      ret.type === 'ps'
    ) {

      this.ps =
        ret.ps;

      this.search(
        true
      );
    }
  }

}
