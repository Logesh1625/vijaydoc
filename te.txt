  STComponent,
  STChange,
  STPage

  Component,
  OnInit,
  ViewChild
} from '@angular/core';

  SFComponent,
  SFSchema
} from '@delon/form';

import { deepMerge } from '@delon/util';

  NzMessageService,
  NzModalService
} from 'ng-zorro-antd';

import { ActivatedRoute } from '@angular/router';

import { I18NService } from '@core/i18n/i18n.service';

import {
  PublicNetService
} from '@../../src/app/routes/a_Public_net/Public_net.service';

  AddDetailsDrawComponent
} from './F1-Deatails-add/F1-Details-add';

  F1ManageImportComponent
} from '../food-time-price/F1-Details-import/F1-Details-import';


@Component({
  selector: 'app-food-time-price',
  templateUrl: './food-time-price.component.html',
  styleUrls: ['./food-time-price.component.scss']
})
export class FoodTimePriceComponent implements OnInit {

  @ViewChild('sf', { static: false })
  sf: SFComponent;

  @ViewChild('st', { static: false })
  private st: STComponent;

  constructor(
    private publicSvr: PublicNetService,
    public settings: SettingsService,
    private msgSvr: NzMessageService,
    private shareSvr: SharedModule,
    private modelSvr: NzModalService,
    private router: ActivatedRoute,
    private i18n: I18NService
  ) {}

  title = 'F1 Meal Management';

  sfSchema: SFSchema;

  source: any[] = [];

  loading = false;

  exportLoading = false;

  pi = 1;

  ps = 10;

  total = 0;

  /** Default site (first site of the logged-in user), used by the Add drawer */
  site = '';

  /** All sites that belong to the logged-in user */
  siteList: string[] = [];

  /** Site chosen in the dropdown. '' = All My Sites */
  selectedSite = '';

  groups = '';

  fxBadge = '';

  groupsList: any[] = [];

  checkedTableData: any[] = [];

  delIdx = 0;

  pages: STPage = {
    show: true,
    front: false,
    showSize: true
  };

  columns: STColumn[] = [
    {
      title: '',
      type: 'checkbox',
      fixed: 'left',
      width: '50px',
      className: 'text-center'
    },

    {
      title: 'Item',
      index: 'Rn',
      className: 'text-center',
      width: '80px'
    },

    {
      title: 'Group',
      index: 'Groups',
      className: 'text-center',
      width: '100px'
    },

    {
      title: 'Site',
      index: 'Site',
      className: 'text-center',
      width: '100px'
    },

    {
      title: 'FX Badge',
      index: 'FxBadge',
      className: 'text-center',
      width: '120px'
    },

    {
      title: 'Name',
      index: 'Name',
      className: 'text-center',
      width: '150px'
    },

    {
      title: 'Meal Preference',
      index: 'MealPreference',
      className: 'text-center',
      width: '150px'
    },

    {
      title: 'First Date at FXBL',
      index: 'FirstDateFxbl',
      className: 'text-center',
      width: '160px'
    },

    {
      title: 'Last Date at FXBL',
      index: 'LastDateFxbl',
      className: 'text-center',
      width: '160px'
    },

    {
      title: 'Add By',
      index: 'AddBy',
      className: 'text-center',
      width: '150px'
    },

    {
      title: 'Add Date',
      index: 'AddDate',
      className: 'text-center',
      width: '160px'
    },

    {
      title: 'Last Edit By',
      index: 'EditBy',
      className: 'text-center',
      width: '150px'
    },

    {
      title: 'Last Edit Date',
      index: 'EditDate',
      className: 'text-center',
      width: '160px'
    }
  ];

  ngOnInit(): void {
    const form = [
      {
        key: 'Groups',
        name: 'Group',
        type: 'string'
      },
      {
        key: 'FxBadge',
        name: 'FX Badge',
        type: 'string'
      },
      {
        key: 'AddBy',
        name: 'Add By',
        type: 'string'
      }
    ];

    this.sfSchema = this.publicSvr.jointSchema(form);

    console.log(
      'Current user from SettingsService:',
      this.publicSvr.getSettingUserInfo()
    );

    this.loadLoggedInUserSite();
  }

  /**
   * Do not call search() from ngAfterViewInit().
   * Search will be called after the user's sites are loaded.
   */

  private loadLoggedInUserSite(): void {
    const userInfo = this.publicSvr.getSettingUserInfo();

    console.log(
      'Logged-in user information:',
      userInfo
    );

    if (!userInfo || !userInfo.eeid) {
      this.msgSvr.error(
        'Logged-in user EEID is missing'
      );

      return;
    }

    console.log(
      'Calling CMS-Get_user_manage for EEID:',
      userInfo.eeid
    );

    this.publicSvr.getSitebasedUser().subscribe(
      (userList: any[]) => {
        console.log(
          'CMS-Get_user_manage user list:',
          userList
        );

        if (!userList || userList.length === 0) {
          this.msgSvr.error(
            'No user data returned from CMS-Get_user_manage'
          );

          return;
        }

        // A user can have more than one site
        this.siteList = this.extractAllSites(
          userList,
          userInfo.eeid
        );

        console.log(
          'Resolved logged-in user sites:',
          this.siteList
        );

        if (this.siteList.length === 0) {
          this.msgSvr.error(
            'Logged-in user site is missing'
          );

          return;
        }

        this.site = this.siteList[0];

        // Default: show data of all my sites
        this.selectedSite = '';

        // Call F1 API only after sites are available
        this.search(true);
      },
      error => {
        console.error(
          'CMS-Get_user_manage error:',
          error
        );

        this.msgSvr.error(
          'Unable to load logged-in user information'
        );
      }
    );
  }

  private getItemEeid(item: any): string {
    return String(
      item.Eeid ||
      item.EEID ||
      item.EeidNo ||
      item.EmpId ||
      item.EmployeeId ||
      item.EmployeeCode ||
      item.eeid ||
      ''
    )
    .trim()
    .toUpperCase();
  }

  /**
   * Returns every unique site of the logged-in employee.
   */
  private extractAllSites(
    userList: any[],
    eeid: string
  ): string[] {
    const loginEeid = String(eeid)
      .trim()
      .toUpperCase();

    const sites = userList
      .filter(item => item && this.getItemEeid(item) === loginEeid)
      .map(item => this.extractSite(item))
      .filter(site => !!site);

    return Array.from(new Set(sites));
  }

  private extractSite(user: any): string {
    if (!user) {
      return '';
    }

    const site =
      user.Site ||
      user.site ||
      user.SITE ||
      user.SiteName ||
      user.siteName ||
      user.WorkSite ||
      user.Worksite ||
      user.BasicCode ||
      user.Basiccode ||
      user.Location ||
      user.location ||
      '';

    return String(site).trim();
  }

  search(first?: boolean): void {
    this.checkedTableData = [];

    if (first) {
      this.pi = 1;
    }

    this.loading = true;

    this.searchHandle(false);
  }

  searchHandle(isExport?: boolean): void {
    const userInfo = this.publicSvr.getSettingUserInfo();

    if (!userInfo || !userInfo.eeid) {
      this.loading = false;
      this.exportLoading = false;
      this.msgSvr.error('Logged-in user EEID is missing');
      return;
    }

    if (!this.siteList || this.siteList.length === 0) {
      this.loading = false;
      this.exportLoading = false;
      this.msgSvr.error('Logged-in user site is missing');
      return;
    }

    const params: any = {
      Groups: this.groups || '',
      FxBadge: this.fxBadge || '',
      EmpId: userInfo.eeid,
      SystemName: 'CMS',
      Site: this.selectedSite || '',
      StartPage: isExport ? 1 : this.pi,
      PageSize: isExport ? 999999999999999 : this.ps,
      Func: 'CMS-GetF1MenuManage'
    };

    const formValues =
      this.sf && this.sf.value
        ? this.sf.value
        : {};

    const finalParams: any = deepMerge(
      params,
      formValues
    );

    // Site always comes from the dropdown ('' = all my sites)
    finalParams.Site = this.selectedSite || '';

    console.log(
      'F1 Menu Manage final params:',
      finalParams
    );

    this.publicSvr.iwxPostF1(
      finalParams,
      {
        headers: {
          apiID: 'ffff-1789633596388-10208193125-1306',
          userKey: 'E9785F13CCEAFB251A6AB147AFD24336'
        }
      },
      true
    )
    .then((res: any) => {
      this.loading = false;
      this.exportLoading = false;

      console.log(
        'F1 Menu Manage response:',
        res
      );

      if (!res || String(res.IsOK) !== '1') {
        if (!isExport) {
          this.source = [];
          this.total = 0;
        }

        this.msgSvr.error(
          res && res.Msg
            ? res.Msg
            : 'Unable to load F1 menu data'
        );

        return;
      }

      const list = Array.isArray(res.List)
        ? res.List
        : [];

      if (isExport) {
        this.exportExcel(list);
        return;
      }

      this.source = list;

      this.total = list.length > 0
        ? Number(list[0].Rowscount || 0)
        : 0;
    })
    .catch(error => {
      this.loading = false;
      this.exportLoading = false;

      console.error(
        'F1 Menu Manage API error:',
        error
      );

      this.msgSvr.error(
        'F1 Menu Manage API failed'
      );
    });
  }

  private exportExcel(list: any[]): void {
    if (!list || list.length === 0) {
      this.msgSvr.error('No Data!');
      return;
    }

    const allData = list.map(item => {
      return {
        Item: item.Rn,
        Group: item.Groups,
        Site: item.Site,
        FxBadge: item.FxBadge,
        Name: item.Name,
        MealPreference: item.MealPreference,
        FirstDateFxbl: item.FirstDateFxbl,
        LastDateFxbl: item.LastDateFxbl,
        AddBy: item.AddBy,
        AddDate: item.AddDate,
        LastEditBy: item.EditBy,
        LastEditDate: item.EditDate
      };
    });

    const fileName =
      'F1_Meal_Management_' +
      new Date().getTime() +
      '.xlsx';

    // second argument = number of columns (was 12 for 11 fields, now 13 for 12 fields)
    this.publicSvr.exportHandle(
      allData,
      13,
      fileName
    );
  }

  add(): void {
    this.openDrawer('Add');
  }

  update(): void {
    if (this.checkedTableData.length !== 1) {
      this.shareSvr.showMsg(
        'error',
        this.i18n.fanyi(
          'table.content.edit-data-select'
        )
      );

      return;
    }

    this.openDrawer('Edit');
  }

  importbulk(): void {
    this.shareSvr.openDrawer(
      'Import',
      F1ManageImportComponent,
      {},
      result => {
        if (result) {
          this.search(true);
        }
      }
    );
  }

  openDrawer(name: string): void {
    if (
      name === 'Edit' &&
      this.checkedTableData.length !== 1
    ) {
      this.shareSvr.showMsg(
        'error',
        this.i18n.fanyi(
          'table.content.edit-data-select'
        )
      );

      return;
    }

    console.log(
      'Selected row:',
      this.checkedTableData[0]
    );

    this.shareSvr.openDrawer(
      name,
      AddDetailsDrawComponent,
      {
        option: name,
        editData: this.checkedTableData[0],
        site: this.selectedSite || this.site,
        siteList: this.siteList
      },
      result => {
        if (result) {
          this.search(true);
        }
      }
    );
  }

  delete(): void {
    if (this.checkedTableData.length === 0) {
      this.shareSvr.showMsg(
        'error',
        this.i18n.fanyi(
          'table.content.edit-data-select'
        )
      );

      return;
    }

    this.delIdx = 0;

    this.shareSvr.modalSvr.confirm({
      nzTitle: 'Prompt',

      nzContent:
        'Are you sure you want to delete the selected ' +
        this.checkedTableData.length +
        ' data?',

      nzOnOk: () => {
        this.delHandle();
      }
    });
  }

  delHandle(): void {
    const ids: any[] = [];

    this.checkedTableData.forEach(item => {
      if (
        item &&
        item.Id !== undefined &&
        item.Id !== null
      ) {
        ids.push(item.Id);
      }
    });

    if (ids.length === 0) {
      this.msgSvr.error(
        'Selected records do not contain Id'
      );

      return;
    }

    const userInfo = this.publicSvr.getSettingUserInfo();

    const params = {
      Ids: ids.join(',') + ',',
      EditBy: userInfo.eeid,
      Func: 'CMS-DeleteF1MenuManage'
    };

    console.log(
      'Delete params:',
      params
    );

    this.publicSvr.iwxPostF1(
      params,
      {
        headers: {
          apiID: 'ffff-1789634776545-10208193125-1341',
          userKey: 'E9785F13CCEAFB251A6AB147AFD24336'
        }
      },
      true
    )
    .then((res: any) => {
      if (res && String(res.IsOK) === '1') {
        this.msgSvr.success('Success');
        this.search(true);
      } else {
        this.msgSvr.error(
          res && res.Msg
            ? res.Msg
            : 'Delete failed'
        );
      }
    })
    .catch(error => {
      console.error(
        'Delete API error:',
        error
      );

      this.msgSvr.error(
        'Delete failed'
      );
    });
  }

  exportData(): void {
    if (!this.siteList || this.siteList.length === 0) {
      this.msgSvr.error(
        'Logged-in user site is missing'
      );

      return;
    }

    this.exportLoading = true;

    this.searchHandle(true);
  }

  change(ret: STChange): void {
    if (ret.type === 'checkbox') {
      this.checkedTableData = ret.checkbox || [];
    }

    if (ret.type === 'pi') {
      this.pi = ret.pi;
      this.search(false);
    }

    if (ret.type === 'ps') {
      this.ps = ret.ps;
      this.search(true);
    }
  }
}
