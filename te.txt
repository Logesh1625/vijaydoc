// ============================================================
// GET DECRYPTION KEY
// ============================================================

export function getdk(encodedKey) {
  if (typeof encodedKey !== 'string' || !encodedKey) {
    throw new Error('Invalid encoded encryption key');
  }

  const decoded = window.atob(encodedKey);
  const middle = Math.floor(decoded.length / 2);

  let left = middle - 1;
  let right = decoded.length % 2 === 0
    ? middle
    : middle + 1;

  let result = '';

  while (left >= 0 && right < decoded.length) {
    result += decoded[right];
    result += decoded[left];

    left--;
    right++;
  }

  return (
    result +
    decoded.slice(right) +
    decoded.slice(0, left + 1)
  );
}

// ============================================================
// AES ENCRYPTION
// ============================================================

export const encrypt = (str, key) => {
  if (str === null || str === undefined) {
    throw new Error('Cannot encrypt an empty value');
  }

  if (typeof key !== 'string' || !key) {
    throw new Error('Invalid encryption key');
  }

  const k = CryptoJS.enc.Utf8.parse(key);

  return CryptoJS.AES.encrypt(String(str), k, {
    iv: CryptoJS.enc.Hex.parse('0000000000000000'),
    mode: CryptoJS.mode.CBC,
    padding: CryptoJS.pad.Pkcs7,
  }).toString();
};

// ============================================================
// AES DECRYPTION
// ============================================================

export const decrypt = (str, key) => {
  try {
    if (
      typeof str !== 'string' ||
      !str.trim()
    ) {
      throw new Error(
        'Encrypted response is empty or undefined'
      );
    }

    if (typeof key !== 'string' || !key) {
      throw new Error('Invalid decryption key');
    }

    const k = CryptoJS.enc.Utf8.parse(key);

    const decrypted = CryptoJS.AES.decrypt(str, k, {
      iv: CryptoJS.enc.Hex.parse('0000000000000000'),
      mode: CryptoJS.mode.CBC,
      padding: CryptoJS.pad.Pkcs7,
    });

    const plaintext = decrypted.toString(
      CryptoJS.enc.Utf8
    );

    if (!plaintext) {
      throw new Error(
        'Decryption returned empty text. Check the key and ciphertext.'
      );
    }

    return plaintext;
  } catch (error) {
    console.error(
      'Decryption error:',
      error.message
    );

    return null;
  }
};

// ============================================================
// ENCRYPT API PARAMETERS
// ============================================================

export const dealPostParamsEncrypt = (data = {}) => {
  const key = getdk(iwxKey);

  const res = {};

  res.o = encrypt(
    JSON.stringify({
      Os: '',
      DeviceId: 'Utility',
      AppVer: '0',
    }),
    key
  );

  res.s = encrypt(
    JSON.stringify(data),
    key
  );

  return res;
};

// ============================================================
// NORMAL API PARAMETERS
// ============================================================

export const dealPostParams = (data = {}) => {
  let params = '';

  for (const key in data) {
    if (!Object.prototype.hasOwnProperty.call(data, key)) {
      continue;
    }

    const value = String(data[key] ?? '');

    params +=
      `${key}=${value
        .replace(/=/g, '%3D')
        .replace(/&/g, '%26')}&`;
  }

  params = params.substring(0, params.length - 1);

  return {
    inputString: params,
  };
};

// ============================================================
// MD5
// ============================================================

export const md5 = (str) => {
  return CryptoJS.MD5(str)
    .toString()
    .toUpperCase();
};

// ============================================================
// ENCRYPTED API POST
// ============================================================

export const iwxPost = async (
  params = {},
  config = {},
  useEncryption = false
) => {
  const param = useEncryption
    ? dealPostParamsEncrypt(params)
    : dealPostParams(params);

  try {
    const result = await axios.post(
      'https://microservice.abgind.foxconn.com/india/gateway',
      param,
      {
        headers: {
          ...(config.headers || {}),
        },
      }
    );

    if (!useEncryption) {
      return result.data;
    }

    const encryptedResponse = result?.data?.r;

    if (
      typeof encryptedResponse !== 'string' ||
      !encryptedResponse.trim()
    ) {
      console.error(
        'Invalid encrypted API response:',
        {
          status: result.status,
          responseKeys: Object.keys(result.data || {}),
        }
      );

      throw new Error(
        'Encrypted API response field "r" is missing'
      );
    }

    const decryptedResponse = decrypt(
      encryptedResponse,
      getdk(iwxKey)
    );

    if (!decryptedResponse) {
      throw new Error(
        'Unable to decrypt API response'
      );
    }

    try {
      return JSON.parse(decryptedResponse);
    } catch {
      throw new Error(
        'Decrypted API response is not valid JSON'
      );
    }
  } catch (error) {
    console.error(
      'Request failed:',
      error.response?.status,
      error.response?.data || error.message
    );

    throw error;
  }
};

export const getuserAccount = () => {
  try {
    const data = sessionStorage.getItem('data');

    if (!data) {
      return null;
    }

    const decryptedResponse = decrypt(
      data,
      getdk(iwxKey)
    );

    if (!decryptedResponse) {
      return null;
    }

    const decryptedResponseObj = JSON.parse(
      decryptedResponse
    );

    return decryptedResponseObj?.List?.[0]?.AccountId ?? null;
  } catch (error) {
    console.error(
      'Unable to get user account:',
      error.message
    );

    return null;
  }
};

    sessionStorage.removeItem('account');
    sessionStorage.removeItem('isLoggedIn');

    setsecretkey('');
    setOTP('');
    setShowQRCode(false);
    setErrorMessage('');
    setLoginStatus('');

    setAccount('');
    setPassword('');

    navigate('/login');
  };

  // ----------------------------------------------------------
  // CLEAR ERRORS WHEN INPUT CHANGES
  // ----------------------------------------------------------

  useEffect(() => {
    setErrorMessage('');
  }, [account, password, otp]);

  // ----------------------------------------------------------
  // GENERATE SECRET WHEN REQUIRED
  // ----------------------------------------------------------

  useEffect(() => {
    const isLoggedIn =
      sessionStorage.getItem('issuccessfullylogin');

    if (
      isLoggedIn === null ||
      isLoggedIn === encryptedBoolean
    ) {
      handleGenerateSecret();
    }
  }, [issuccessfullylogin, loginStatus]);

  // ==========================================================
  // UI
  // ==========================================================

  return (
    <Grid
      container
      component="main"
      onKeyDown={handleKeyPress}
      sx={{
        minHeight: '100vh',
        backgroundImage: `url(${loginimage})`,
        backgroundRepeat: 'no-repeat',
        backgroundSize: 'cover',
        backgroundPosition: 'center',
        opacity: 0.95,
      }}
    >
      <Grid
        item
        xs={12}
        sx={{
          display: 'flex',
          alignItems: 'center',
          justifyContent: 'center',
          minHeight: '85vh',
          padding: 2,
        }}
      >
        <Box className={classes.root}>
          <Box className={classes.formm}>
            <img
              src="/images/logo1.png"
              alt="Company Logo"
              style={{
                maxWidth: '180px',
                height: 'auto',
              }}
            />

            {/* ============================================= */}
            {/* MFA SCREEN */}
            {/* ============================================= */}

            {issuccessfullylogin === encryptedBoolean ? (
              <div>
                <Typography
                  sx={{
                    fontFamily:
                      '"Open Sans","Helvetica","Arial",sans-serif',
                    fontWeight: 600,
                    fontSize: '1rem',
                  }}
                >
                  MFA Auth
                </Typography>

                {isscuessqr === encryptedBooleannum && (
                  <>
                    <br />

                    {showQRCode && secretkey && (
                      <div>
                        <QRCodeCanvas
                          value={generateOTPURI()}
                          size={128}
                        />

                        <Typography
                          sx={{
                            fontFamily:
                              '"Open Sans","Helvetica","Arial",sans-serif',
                            fontWeight: 600,
                            fontSize: '10px',
                            overflowWrap: 'anywhere',
                          }}
                        >
                          Secret Key:
                          <br />
                          {secretkey}
                        </Typography>
                      </div>
                    )}
                  </>
                )}

                {errorMessage && (
                  <Typography
                    color="error"
                    sx={{ mt: 2 }}
                  >
                    {errorMessage}
                  </Typography>
                )}

                <TextField
                  margin="normal"
                  fullWidth
                  type="text"
                  label="Enter OTP"
                  value={otp}
                  error={Boolean(errorMessage)}
                  onChange={(e) => {
                    setOTP(
                      e.target.value.replace(/\D/g, '')
                    );
                  }}
                  inputProps={{
                    inputMode: 'numeric',
                    maxLength: 8,
                  }}
                />

                <Button
                  fullWidth
                  disabled={
                    !otp.trim() ||
                    otploading
                  }
                  onClick={verifyOTP}
                  sx={{ mt: 3, mb: 2 }}
                  variant="contained"
                >
                  {otploading ? (
                    <CircularProgress size={24} />
                  ) : (
                    'Verify OTP'
                  )}
                </Button>

                <Button
                  onClick={handlebacklogin}
                  variant="contained"
                  fullWidth
                >
                  Back to Login
                </Button>
              </div>
            ) : (
              /* =========================================== */
              /* LOGIN SCREEN */
              /* =========================================== */

              <form onSubmit={handleLogin}>
                <Typography
                  sx={{
                    fontFamily:
                      '"Open Sans","Helvetica","Arial",sans-serif',
                    fontWeight: 700,
                    fontSize: '1rem',
                    mb: 2,
                  }}
                >
                  Login
                </Typography>

                <TextField
                  margin="normal"
                  fullWidth
                  label="Username"
                  value={account}
                  autoComplete="username"
                  onChange={(e) =>
                    setAccount(e.target.value)
                  }
                />

                <TextField
                  margin="normal"
                  fullWidth
                  label="Password"
                  type={
                    showPassword
                      ? 'text'
                      : 'password'
                  }
                  value={password}
                  autoComplete="current-password"
                  onChange={(e) =>
                    setPassword(e.target.value)
                  }
                  InputProps={{
                    endAdornment: (
                      <InputAdornment position="end">
                        <IconButton
                          onClick={handleTogglePassword}
                          edge="end"
                          aria-label="Toggle password visibility"
                        >
                          {showPassword ? (
                            <Visibility />
                          ) : (
                            <VisibilityOff />
                          )}
                        </IconButton>
                      </InputAdornment>
                    ),
                  }}
                />

                {errorMessage && (
                  <Typography
                    color="error"
                    sx={{ mt: 2 }}
                  >
                    {errorMessage}
                  </Typography>
                )}

                <Button
                  type="submit"
                  fullWidth
                  variant="contained"
                  sx={{ mt: 3, mb: 2 }}
                  disabled={isLoading}
                >
                  {isLoading ? (
                    <CircularProgress size={24} />
                  ) : (
                    'Login'
                  )}
                </Button>
              </form>
            )}
          </Box>
        </Box>
      </Grid>

      {/* =============================================== */}
      {/* FOOTER */}
      {/* =============================================== */}

      <Grid
        item
        xs={12}
        sx={{
          display: 'flex',
          alignItems: 'center',
          justifyContent: 'center',
          textAlign: 'center',
          padding: 2,
          fontSize: '12px',
          color: '#000',
        }}
      >
        <div>
          <div>
            FXCN IT: Tel:533-26038 | Email:idpbg-it-tn-dev@mail.foxconn.com
          </div>

          <div>
            Copyright 2020-2025 Foxconn A Business Group IT.
            All Rights Reserved
          </div>
        </div>
      </Grid>
    </Grid>
  );
};
