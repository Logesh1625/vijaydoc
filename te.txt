getGuardPassCode(
  params?: any,
  site?: any,
  isShowLabel?: any
): Observable<any> {

  // Get Base Type Name
  if (!params || typeof params !== "string" || params.length < 10) {
    console.error("Invalid getGuardPassCode params:", params);
    return of([]);
  }

  const code = params.slice(10);

  const _param = {
    TypeName: code,
    Func: "CMS-GetBaseTypeInfo",
    Site: site
  };

  const config = {
    headers: {
      apiID: "ffff-1710991146185-10208193125-0170",
      userKey: this.USER_TN
    }
  };

  const param = this.dealPostParamsEncrypt(_param);

  return this.http
    .post(iwxServiceUrl, param, null, config)
    .pipe(
      map((resp: any) => {
        try {
          // Validate response
          if (!resp || !resp.r) {
            console.error(
              "getGuardPassCode: Empty API response",
              resp
            );
            return [];
          }

          // Decrypt API response
          const res = JSON.parse(
            this.security.decrypt(resp.r, this.iwxKInfo)
          );

          console.log(
            "Base Type API response:",
            res
          );

          // Validate API status
          if (
            !res ||
            String(res.IsOK) !== "1" ||
            !Array.isArray(res.List)
          ) {
            console.error(
              "getGuardPassCode API failed:",
              res
            );
            return [];
          }

          // Convert API data to dropdown options
          const _data: any[] = [];

          res.List.forEach((element: any) => {
            if (!element) {
              return;
            }

            const name =
              element.Name !== undefined &&
              element.Name !== null
                ? String(element.Name).trim()
                : "";

            const baseCodeId =
              element.BaseCodeId !== undefined &&
              element.BaseCodeId !== null
                ? element.BaseCodeId
                : element.ID !== undefined &&
                  element.ID !== null
                  ? element.ID
                  : element.Id;

            if (!name || baseCodeId === undefined || baseCodeId === null) {
              console.warn(
                "Invalid base code item:",
                element
              );
              return;
            }

            // When isShowLabel is true, use the name as the value.
            // Otherwise, use the numeric database ID.
            const item = isShowLabel
              ? {
                  label: name,
                  value: name
                }
              : {
                  label: name,
                  value: Number(baseCodeId)
                };

            _data.push(item);
          });

          console.log(
            "MEAL PREFERENCE LIST:",
            _data
          );

          return _data;

        } catch (error) {
          console.error(
            "Error loading base type dropdown:",
            error
          );

          return [];
        }
      })
    );
}
