import { Component, OnInit, ViewChild, ElementRef } from '@angular/core';
import { HttpClient } from '@angular/common/http';
import { DomSanitizer, SafeUrl } from '@angular/platform-browser';
import { NzMessageService } from 'ng-zorro-antd';
import { PublicNetService } from "src/app/routes/a_Public_net/Public_net.service";

@Component({
  selector: 'app-Meal-scan',
  templateUrl: './Meal-scan.component.html',
  styleUrls: ['./Meal-scan.component.less']
})
export class MealScanComponent implements OnInit {

  @ViewChild('scanInput', { static: false }) scanInput: ElementRef;

  empNo: string = '';
  loading: boolean = false;
  proceeding: boolean = false;
  employeeDetail: any = null;
  entitlementStatus: string = '';

  employeePhotoUrl: SafeUrl | null = null;
  photoLoading: boolean = false;

  private readonly PHOTO_API_BASE = 'http://10.209.110.187:3000/api/image';

  constructor(
    private publicSvr: PublicNetService,
    private msgSvr: NzMessageService,
    private http: HttpClient,
    private sanitizer: DomSanitizer,
  ) { }

  ngOnInit(): void {
    this.focusInput();
  }

  focusInput() {
    setTimeout(() => {
      if (this.scanInput) {
        this.scanInput.nativeElement.focus();
      }
    }, 100);
  }

  onScanEnter() {
    this.search();
  }

  search(fromButton?: boolean) {
    if (!this.empNo || this.empNo.trim() === '') {
      this.msgSvr.error('Please scan or enter an FX Badge.');
      return;
    }

    this.loading = true;
    this.employeeDetail = null;
    this.entitlementStatus = '';
    this.employeePhotoUrl = null;

    const params = {
      FxBadge: this.empNo.trim(),
      Func: "CMS-GetF1MenuByFxbadge"
    };

    this.publicSvr.iwxPostF1(params, {
      headers: {
        apiID: "ffff-1789807467770-10208193125-1466",
        userKey: "E9785F13CCEAFB251A6AB147AFD24336"
      }
    }, true).then((res: any) => {
      this.loading = false;

      if (res.IsOK == "1" && res.List && res.List.length > 0) {
        const data = res.List[0];
        this.employeeDetail = {
          FxBadge: data.FxBadge,
          Name: data.Name,
          MealPreference: data.MealPreference,
          FirstDateFxbl: data.FirstDateFxbl,
          LastDateFxbl: data.LastDateFxbl,
        };
        this.entitlementStatus = data.EntitlementStatus;
        this.fetchEmployeePhoto(data.FxBadge);
        this.proceed();
      } else {
        this.employeeDetail = {
          FxBadge: this.empNo.trim(),
          Name: '',
          MealPreference: '',
          FirstDateFxbl: '',
          LastDateFxbl: '',
        };
        this.entitlementStatus = 'NOT_UPDATED';
        this.fetchEmployeePhoto(this.empNo.trim());
        this.proceed();
      }
    }).catch((err) => {
      this.loading = false;
      this.msgSvr.error('Failed to fetch employee details.');
    });
  }

  fetchEmployeePhoto(fxBadge: string) {
    this.photoLoading = true;
    this.http.get(`${this.PHOTO_API_BASE}/${fxBadge}`, { responseType: 'blob' })
      .subscribe({
        next: (blob) => {
          const objectUrl = URL.createObjectURL(blob);
          this.employeePhotoUrl = this.sanitizer.bypassSecurityTrustUrl(objectUrl);
          this.photoLoading = false;
        },
        error: () => {
          this.employeePhotoUrl = null;
          this.photoLoading = false;
        }
      });
  }

  proceed() {
  if (!this.employeeDetail) return;

  this.proceeding = true;

  const params = {
    FxBadge: this.employeeDetail.FxBadge,
    Func: "CMS-InsertMealScanLog"
  };

  this.publicSvr.iwxPostF1(params, {
    headers: {
      apiID: "ffff-1789818962176-10208193125-1483",
      userKey: "E9785F13CCEAFB251A6AB147AFD24336"
    }
  }, true).then((res: any) => {
    this.proceeding = false;

    if (res.IsOK == "1" && res.List && res.List.length > 0) {
      const logResult = res.List[0];

      if (logResult.Status === 'SUCCESS') {
        this.msgSvr.success('Meal collection recorded successfully!');
      } else if (logResult.Status === 'ALREADY_SCANNED') {
        this.entitlementStatus = 'ALREADY_SCANNED';
        this.msgSvr.error(logResult.StatusMessage || 'Meal already collected today');
      } else {
        this.msgSvr.error(logResult.StatusMessage || 'Invalid scan.');
      }
      this.clearInputOnly();
    } else {
      this.msgSvr.error(res.Msg || 'Failed to log scan.');
    }
  }).catch((err) => {
    this.proceeding = false;
    this.msgSvr.error('Failed to log scan. Please try again.');
  });
}

  clearInputOnly() {
    this.empNo = '';
    this.focusInput();
  }

  cancelScan() {
    this.employeeDetail = null;
    this.entitlementStatus = '';
    this.empNo = '';
    this.employeePhotoUrl = null;
    this.focusInput();
  }
}
