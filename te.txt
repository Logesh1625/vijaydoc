submit(userInfo?: string) {
  this.loading = true;

  try {
    if (!userInfo || typeof userInfo !== 'string') {
      throw new Error('OUAC UserInfo is empty or invalid');
    }

    console.log('OUAC UserInfo received. Length:', userInfo.length);

    const decryptKey = this.publicSvr.security.getDK(
      'NzQxRTMzNzlBODA0MjM3OEY4NEE4MDc4NzNCMUJCNDk='
    );

    let decryptedUserInfo: string;

    try {
      decryptedUserInfo = security.decrypt(userInfo, decryptKey);
    } catch (decryptError) {
      console.error('OUAC UserInfo decryption failed:', decryptError);
      throw new Error(
        'UserInfo decryption failed. Verify the OUAC encryption key and AES configuration.'
      );
    }

    if (!decryptedUserInfo) {
      throw new Error('Decrypted UserInfo is empty');
    }

    let data: any;

    try {
      data = JSON.parse(decryptedUserInfo);
    } catch (parseError) {
      console.error('UserInfo JSON parsing failed:', parseError);
      throw new Error('Decrypted UserInfo is not valid JSON');
    }

    const userInfo2 = security.encrypt(
      JSON.stringify(data),
      '6jxaHMIhvefCOokm5XbDdW4qzl30U7Kt'
    );

    const param = {
      userInfo: userInfo2
    };

    const body = this.publicSvr.security.encryptForWeb(
      JSON.stringify(param)
    );

    this.http.post(
      '/oneAccountLogin?_allow_anonymous=true',
      body,
      null,
      {
        headers: {
          'Content-Type': 'application/json;charset=UTF-8'
        }
      }
    ).subscribe(
      (response: any) => {
        try {
          const decryptedResponse =
            this.publicSvr.security.decryptForWeb(response);

          const res = JSON.parse(decryptedResponse);

          if (res.status === 200) {
            const user = {
              avatar: './assets/tmp/img/avatar.jpg',
              eeid: res.data.eeid,
              name: res.data.empName,
              time: new Date().getTime()
            };

            this.loginSuccess(
              user,
              res.data.JsessionId,
              res.data.ClientIp
            );
          } else {
            this.msg.error(res.msg || 'Login failed');
          }
        } catch (error) {
          console.error('Login response processing failed:', error);
          this.msg.error('Unable to process login response');
        } finally {
          this.loading = false;
        }
      },
      (error) => {
        this.loading = false;
        console.error('Login API error:', error);
        this.msg.error('Login failed. Please try again.');
      }
    );

  } catch (error) {
    this.loading = false;
    console.error('Login processing error:', error);
    this.msg.error(error.message || 'Unable to process login information');
  }
}
