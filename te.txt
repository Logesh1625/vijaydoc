PROCEDURE GET_F1_MENU_BY_FXBADGE(
    V_EMP_ID    IN VARCHAR2,
    V_SITE      IN VARCHAR2,
    V_FX_BADGE  IN VARCHAR2,
    V_RES       OUT VARCHAR2,
    V_RESULT    OUT VARCHAR2,
    P_CURSOR    OUT SYS_REFCURSOR
) IS
    V_COUNT       NUMBER;
    V_ALREADY     NUMBER;
    V_RECORD_SITE CMS.C_F1_MEAL_INFO.PROPERTY_01%TYPE;
BEGIN

    IF IS_ACTIVE_CMS_USER(V_EMP_ID) = 0 THEN
        V_RES := '0';
        V_RESULT := 'Active CMS user not found for employee: ' || V_EMP_ID;

        OPEN P_CURSOR FOR
            SELECT CAST(NULL AS VARCHAR2(100)) AS FX_BADGE
            FROM DUAL WHERE 1 = 0;
        RETURN;
    END IF;

    IF V_SITE IS NOT NULL AND TRIM(V_SITE) IS NOT NULL THEN
        IF HAS_SITE_ACCESS(V_EMP_ID, V_SITE) = 0 THEN
            V_RES := '0';
            V_RESULT := 'You are not authorized for the selected site.';

            OPEN P_CURSOR FOR
                SELECT CAST(NULL AS VARCHAR2(100)) AS FX_BADGE
                FROM DUAL WHERE 1 = 0;
            RETURN;
        END IF;
    END IF;

    /*
     * Only badges added today are eligible for lookup.
     */
    SELECT COUNT(1)
    INTO V_COUNT
    FROM CMS.C_F1_MEAL_INFO M
    WHERE UPPER(TRIM(M.FX_BADGE)) = UPPER(TRIM(V_FX_BADGE))
      AND NVL(M.DEL_FLAG, 0) = 0
      AND M.ADD_DATE >= TRUNC(SYSDATE)
      AND M.ADD_DATE < TRUNC(SYSDATE) + 1;

    IF V_COUNT = 0 THEN
        V_RES := '0';
        V_RESULT := 'FX Badge not found';

        OPEN P_CURSOR FOR
            SELECT CAST(NULL AS VARCHAR2(100)) AS FX_BADGE
            FROM DUAL WHERE 1 = 0;
        RETURN;
    END IF;

    /*
     * Get the latest record added today.
     */
    SELECT PROPERTY_01
    INTO V_RECORD_SITE
    FROM (
        SELECT
            M.PROPERTY_01,
            ROW_NUMBER() OVER (
                ORDER BY M.ADD_DATE DESC, M.ID DESC
            ) AS RN
        FROM CMS.C_F1_MEAL_INFO M
        WHERE UPPER(TRIM(M.FX_BADGE)) = UPPER(TRIM(V_FX_BADGE))
          AND NVL(M.DEL_FLAG, 0) = 0
          AND M.ADD_DATE >= TRUNC(SYSDATE)
          AND M.ADD_DATE < TRUNC(SYSDATE) + 1
    )
    WHERE RN = 1;

    IF HAS_SITE_ACCESS(V_EMP_ID, V_RECORD_SITE) = 0 THEN
        V_RES := '0';
        V_RESULT := 'This FX Badge does not belong to your assigned site.';

        OPEN P_CURSOR FOR
            SELECT CAST(NULL AS VARCHAR2(100)) AS FX_BADGE
            FROM DUAL WHERE 1 = 0;
        RETURN;
    END IF;

    SELECT COUNT(1)
    INTO V_ALREADY
    FROM CMS.C_F1_SCAN_LOG
    WHERE UPPER(TRIM(FX_BADGE)) = UPPER(TRIM(V_FX_BADGE))
      AND TRUNC(MEAL_DATE) = TRUNC(SYSDATE)
      AND NVL(DEL_FLAG, 0) = 0
      AND COLLECTED_DATA = 'SUCCESS';

    OPEN P_CURSOR FOR
        SELECT
            V.GROUPS,
            V.PROPERTY_01 AS SITE,
            GET_SITE_NAME(V.PROPERTY_01) AS SITE_NAME,
            BC.NAME AS MEAL_PREFERENCE,
            V.FX_BADGE,
            V.NAME,
            TO_CHAR(V.FIRST_DATE_FXBL, 'DD-MM-YYYY')
                AS FIRST_DATE_FXBL,
            TO_CHAR(V.LAST_DATE_FXBL, 'DD-MM-YYYY')
                AS LAST_DATE_FXBL,
            CASE
                WHEN UPPER(TRIM(BC.NAME)) = 'NO MEAL REQUIRED'
                    THEN 'NOT_ENTITLED'

                WHEN V.FIRST_DATE_FXBL IS NULL
                  OR V.LAST_DATE_FXBL IS NULL
                    THEN 'NOT_UPDATED'

                WHEN TRUNC(SYSDATE) NOT BETWEEN
                     TRUNC(V.FIRST_DATE_FXBL)
                     AND TRUNC(V.LAST_DATE_FXBL)
                    THEN 'LAST DATE EXPIRED'

                WHEN V_ALREADY > 0
                    THEN 'ALREADY_SCANNED'

                ELSE 'ENTITLED'
            END AS ENTITLEMENT_STATUS
        FROM (
            SELECT
                M.*,
                ROW_NUMBER() OVER (
                    ORDER BY M.ADD_DATE DESC, M.ID DESC
                ) AS RN
            FROM CMS.C_F1_MEAL_INFO M
            WHERE UPPER(TRIM(M.FX_BADGE)) =
                  UPPER(TRIM(V_FX_BADGE))
              AND NVL(M.DEL_FLAG, 0) = 0
              AND M.ADD_DATE >= TRUNC(SYSDATE)
              AND M.ADD_DATE < TRUNC(SYSDATE) + 1
        ) V
        LEFT JOIN CMS.C_BASE_CODE_T BC
            ON TO_CHAR(V.MEAL_PREFERENCE) = TO_CHAR(BC.ID)
           AND BC.TYPE_ID = 1001
        WHERE V.RN = 1;

    V_RES := '1';
    V_RESULT := 'Success';

EXCEPTION
    WHEN NO_DATA_FOUND THEN
        V_RES := '0';
        V_RESULT := 'FX Badge not found';

        OPEN P_CURSOR FOR
            SELECT CAST(NULL AS VARCHAR2(100)) AS FX_BADGE
            FROM DUAL WHERE 1 = 0;

    WHEN OTHERS THEN
        V_RES := '0';
        V_RESULT := 'Error: ' || SQLERRM;

END GET_F1_MENU_BY_FXBADGE;
