  // =====================================================
  // SSO LOGIN
  // =====================================================

  submit(userInfo?: string): void {

    if (this.loginRequestInProgress) {
      console.warn("Login request is already in progress.");
      return;
    }

    if (!userInfo || typeof userInfo !== "string") {

      this.loading = false;

      this.msg.error(
        "OUAC login information is empty."
      );

      return;
    }

    this.loading = true;
    this.loginRequestInProgress = true;

    console.log("SSO Step 1: UserInfo received.");
    console.log("UserInfo length:", userInfo.length);

    try {

      // -------------------------------------------------
      // STEP 1: GET OUAC DECRYPTION KEY
      // -------------------------------------------------

      const encodedKey =
        "NzQxRTMzNzlBODA0MjM3OEY4NEE4MDc4NzNCMUJCNDk=";

      const decryptKey =
        this.publicSvr.security.getDK(encodedKey);

      if (
        !decryptKey ||
        ![16, 24, 32].includes(decryptKey.length)
      ) {

        throw new Error(
          "Invalid AES key length. Verify the OUAC key configuration."
        );
      }

      console.log(
        "SSO Step 2: Decryption key generated."
      );

      // -------------------------------------------------
      // STEP 2: DECRYPT OUAC USERINFO
      // -------------------------------------------------

      let decryptedUserInfo: string;

      try {

        decryptedUserInfo = security.decrypt(
          userInfo,
          decryptKey
        );

      } catch (decryptError) {

        console.error(
          "OUAC decryption failed:",
          decryptError
        );

        throw new Error(
          "OUAC decryption failed. Verify the encryption configuration."
        );
      }

      if (
        !decryptedUserInfo ||
        typeof decryptedUserInfo !== "string"
      ) {

        throw new Error(
          "Decrypted OUAC UserInfo is empty."
        );
      }

      console.log(
        "SSO Step 3: UserInfo decrypted successfully."
      );

      // -------------------------------------------------
      // STEP 3: PARSE USERINFO JSON
      // -------------------------------------------------

      let userData: any;

      try {

        userData = JSON.parse(decryptedUserInfo);

      } catch (parseError) {

        console.error(
          "UserInfo JSON parsing failed:",
          parseError
        );

        throw new Error(
          "Decrypted UserInfo is not valid JSON."
        );
      }

      if (
        !userData ||
        typeof userData !== "object" ||
        Array.isArray(userData)
      ) {

        throw new Error(
          "Invalid OUAC user information."
        );
      }

      console.log(
        "SSO Step 4: UserInfo JSON parsed successfully."
      );

      // -------------------------------------------------
      // STEP 4: ENCRYPT USERINFO FOR LOGIN API
      // -------------------------------------------------

      const userInfo2 = security.encrypt(
        JSON.stringify(userData),
        "6jxaHMIhvefCOokm5XbDdW4qzl30U7Kt"
      );

      if (!userInfo2) {

        throw new Error(
          "Unable to encrypt login UserInfo."
        );
      }

      const param = {
        userInfo: userInfo2
      };

      // -------------------------------------------------
      // STEP 5: ENCRYPT REQUEST BODY
      // -------------------------------------------------

      const body =
        this.publicSvr.security.encryptForWeb(
          JSON.stringify(param)
        );

      if (!body) {

        throw new Error(
          "Unable to encrypt the login request."
        );
      }

      console.log(
        "SSO Step 5: Login request prepared."
      );

      // -------------------------------------------------
      // STEP 6: CALL LOGIN API
      // -------------------------------------------------

      this.http.post(
        "/oneAccountLogin?_allow_anonymous=true",
        body,
        null,
        {
          headers: {
            "Content-Type": "application/json;charset=UTF-8"
          }
        }
      ).subscribe(

        (response: any) => {

          console.log(
            "SSO Step 6: Login API response received."
          );

          try {

            // -------------------------------------------
            // STEP 7: DECRYPT API RESPONSE
            // -------------------------------------------

            const decryptedResponse =
              this.publicSvr.security.decryptForWeb(
                response
              );

            if (!decryptedResponse) {

              throw new Error(
                "Decrypted login response is empty."
              );
            }

            // -------------------------------------------
            // STEP 8: PARSE API RESPONSE
            // -------------------------------------------

            let res: any;

            try {

              res = JSON.parse(decryptedResponse);

            } catch (parseError) {

              console.error(
                "Login response JSON parsing failed:",
                parseError
              );

              throw new Error(
                "Login API returned an invalid response."
              );
            }

            console.log(
              "Login response status:",
              res && res.status
            );

            // -------------------------------------------
            // STEP 9: VALIDATE LOGIN RESPONSE
            // -------------------------------------------

            if (
              res &&
              Number(res.status) === 200 &&
              res.data
            ) {

              const loginData = res.data;

              if (
                !loginData.eeid ||
                !loginData.JsessionId
              ) {

                throw new Error(
                  "Login response is missing required session information."
                );
              }

              const user = {

                avatar: "./assets/tmp/img/avatar.jpg",

                eeid: loginData.eeid,

                name: loginData.empName || loginData.eeid,

                time: new Date().getTime()

              };

              console.log(
                "SSO Step 7: Login successful. Initializing session."
              );

              // -----------------------------------------
              // STEP 10: LOGIN SUCCESS
              // -----------------------------------------

              this.loginSuccess(
                user,
                loginData.JsessionId,
                loginData.ClientIp
              );

            } else {

              console.error(
                "Login API rejected the request."
              );

              this.msg.error(
                (res && res.msg) || "Login failed."
              );

              this.finishLoginRequest();
            }

          } catch (responseError) {

            console.error(
              "Login response processing failed:",
              responseError
            );

            this.msg.error(
              responseError &&
              responseError.message
                ? responseError.message
                : "Unable to process login response."
            );

            this.finishLoginRequest();
          }
        },

        (httpError: any) => {

          console.error(
            "Login API HTTP error:",
            {
              status: httpError && httpError.status,
              message: httpError && httpError.message
            }
          );

          if (httpError && httpError.status === 401) {

            this.msg.error(
              "Login API returned 401 Unauthorized."
            );

          } else if (
            httpError &&
            httpError.status === 0
          ) {

            this.msg.error(
              "Unable to connect to the login server."
            );

          } else {

            this.msg.error(
              "Login failed. Please check the Network response."
            );
          }

          this.finishLoginRequest();
        }
      );

    } catch (error) {

      console.error(
        "SSO login processing error:",
        error
      );

      this.msg.error(
        error && error.message
          ? error.message
          : "Unable to process OUAC login information."
      );

      this.finishLoginRequest();
    }
  }

  // =====================================================
  // RESET LOGIN REQUEST STATE
  // =====================================================

  private finishLoginRequest(): void {

    this.loading = false;

    this.loginRequestInProgress = false;
  }

  // =====================================================
  // BUILD OUAC IFRAME URL
  // =====================================================

  private buildIframeUrl(sso: boolean): string {

    return (
      `${ouacServiceUrl}` +
      `&CallBack=/assets/login-callback.html?UserInfo` +
      `&CssLink=/assets/sso-login.css` +
      `&State=RxFdv2p7KsDq38CX` +
      `&SSO=${sso ? 1 : 0}`
    );
  }

  // =====================================================
  // RESET LOGIN IFRAME
  // =====================================================

  private resetLoginIframe(): void {

    const iframe = document.querySelector(
      ".login-iframe"
    );

    if (iframe) {

      iframe.setAttribute(
        "src",
        this.buildIframeUrl(false)
      );
    }
  }

  // =====================================================
  // OUAC POSTMESSAGE CALLBACK
  // =====================================================

  @HostListener("window:message", ["$event"])
  onMessage(event: MessageEvent): void {

    // IMPORTANT:
    // Add an exact OUAC origin allowlist here once the
    // actual callback origin is confirmed.
    //
    // Do not trust arbitrary cross-origin messages in a
    // production SSO implementation.

    const info: any = event.data;

    if (
      !info ||
      typeof info !== "object" ||
      info.msgType !== "login"
    ) {

      return;
    }

    console.log(
      "OUAC login callback received.",
      {
        hasUserInfo: typeof info.UserInfo === "string",
        userInfoLength:
          typeof info.UserInfo === "string"
            ? info.UserInfo.length
            : 0
      }
    );

    if (
      !info.UserInfo ||
      typeof info.UserInfo !== "string"
    ) {

      this.loading = false;

      this.msg.error(
        "OUAC login information was not received."
      );

      return;
    }

    this.submit(info.UserInfo);
  }

  // =====================================================
  // LOGIN SUCCESS
  // =====================================================

  async loginSuccess(
    user: any,
    token: any,
    ClientIP: any
  ): Promise<void> {

    try {

      // -----------------------------------------------
      // STORE USER INFORMATION
      // -----------------------------------------------

      this.settingService.setUser({

        userInfo:
          this.publicSvr.security.encryptForWeb(
            JSON.stringify(user)
          )

      });

      this.cookieService.set(
        "USER_NO",
        this.publicSvr.security.encryptForWeb(
          user.eeid
        )
      );

      this.cookieService.set(
        "USER_NAME",
        this.publicSvr.security.encryptForWeb(
          user.name
        )
      );

      this.cookieService.set("IS_FOREIGN", "");
      this.cookieService.set("SESSION_ID", "");
      this.cookieService.set("FIH_TOKEN", "");

      sessionStorage.setItem(
        "ClientIp",
        ClientIP || ""
      );

      // -----------------------------------------------
      // INSERT LOGIN DATA
      // -----------------------------------------------

      try {

        await this.publicSvr.insertLoginData(
          user.eeid,
          ClientIP
        );

      } catch (insertError) {

        console.error(
          "insertLoginData failed:",
          insertError
        );

        // Keep the existing behavior: login continues
        // even if login audit insertion fails.
      }

      // -----------------------------------------------
      // CLEAR REUSE TABS
      // -----------------------------------------------

      if (this.reuseTabService) {

        this.reuseTabService.clear();
      }

      // -----------------------------------------------
      // STORE AUTH TOKEN
      // -----------------------------------------------

      this.tokenService.set({

        token: token

      });

      console.log(
        "Login session stored. Loading startup data."
      );

      // -----------------------------------------------
      // LOAD STARTUP CONFIGURATION
      // -----------------------------------------------

      await this.startupSrv.load();

      this.finishLoginRequest();

      // -----------------------------------------------
      // LOAD SYSTEM MENU
      // -----------------------------------------------

      this.getSystemMenu();

    } catch (error) {

      console.error(
        "Login initialization failed:",
        error
      );

      this.finishLoginRequest();

      this.msg.error(
        "Login succeeded, but application initialization failed."
      );
    }
  }

  // =====================================================
  // INITIALIZATION
  // =====================================================

  ngOnInit(): void {

    const url = window.location.href;

    // -----------------------------------------------
    // FORCE HTTPS ON SPECIFIED DOMAINS
    // -----------------------------------------------

    if (
      window.location.protocol !== "https:" &&
      (
        window.location.hostname ===
          "lms.idpbgtn.efoxconn.com" ||
        window.location.hostname ===
          "lms.idpbgind.efoxconn.com"
      )
    ) {

      window.location.href =
        "https://" +
        window.location.host +
        window.location.pathname +
        window.location.search +
        window.location.hash;

      return;
    }

    // -----------------------------------------------
    // REDIRECT TN TO INDIA HOST
    // -----------------------------------------------

    if (
      window.location.hostname ===
      "lms.idpbgtn.efoxconn.com"
    ) {

      window.location.href = url.replace(
        "lms.idpbgtn.efoxconn.com",
        "lms.idpbgind.efoxconn.com"
      );

      return;
    }

    // -----------------------------------------------
    // BUILD IFRAME URL
    // -----------------------------------------------

    const currentUrl = this.location.href;

    const fromInfo =
      currentUrl.indexOf("?from=iportal") > -1;

    this.iframeSafeSrc =
      this.domSanitizer
        .bypassSecurityTrustResourceUrl(
          this.buildIframeUrl(fromInfo)
        );

    // -----------------------------------------------
    // READ TOPAGE PARAMETER
    // -----------------------------------------------

    const toPageIndex =
      currentUrl.indexOf("&toPage=");

    if (toPageIndex > -1) {

      this.toPage = currentUrl
        .slice(toPageIndex + 8)
        .replace(/%2F/g, "/");
    }

    // -----------------------------------------------
    // READ MENUID PARAMETER
    // -----------------------------------------------

    const menuIndex =
      currentUrl.indexOf("&menuId=");

    if (menuIndex > -1) {

      this.menuId = currentUrl.slice(
        menuIndex + 8
      );
    }
  }

  // =====================================================

  // =====================================================
  // GET SYSTEM MENU
  // =====================================================

  getSystemMenu(): void {

    const group: any[] = [

      {
        text: "主導航",
        group: true,
        children: []
      }

    ];

    const userInfo =
      this.publicSvr.getSettingUserINfo();

    if (
      !userInfo ||
      !userInfo.eeid
    ) {

      console.error(
        "User information not found."
      );

      this.msg.error(
        "User information not found."
      );

      return;
    }

    const params = {

      Func:
        "Permission management-GetUserPermitModuleBySystem",

      SystemId: 30373,

      EmpNo: userInfo.eeid

    };

    const param =
      this.publicSvr.dealPostParamsEncrypt(
        params
      );

    const config = {

      headers: {

        apiID:
          "ffff-1687574205823-1020819373-1797",

        userKey:
          this.publicSvr.USER_APPLIATION_WEB

      }

    };

    console.log(
      "Loading system menu for current user."
    );

    this.http.post(
      iwxServiceUrl,
      param,
      null,
      config
    ).subscribe(

      (response: any) => {

        try {

          if (
            !response ||
            !response.r
          ) {

            console.error(
              "Invalid menu API response."
            );

            this.msg.error(
              "Invalid menu response from server."
            );

            return;
          }

          // -------------------------------------------
          // DECRYPT MENU RESPONSE
          // -------------------------------------------

          const decrypted =
            this.publicSvr.security.decrypt(
              response.r,
              this.publicSvr.iwxKInfo
            );

          const result = JSON.parse(
            decrypted
          );

          // -------------------------------------------
          // CHECK PERMISSION
          // -------------------------------------------

          const hasAccess =
            result &&
            result.IsOK == "1" &&
            Array.isArray(result.List) &&
            result.List.length > 0;

          if (!hasAccess) {

            this.msg.error(
              "No menu access. Please contact the administrator.",
              {
                nzDuration: 3000
              }
            );

            this.publicSvr.logout();

            return;
          }

          // -------------------------------------------
          // BUILD MENU TREE
          // -------------------------------------------

          group[0].children =
            this.arrayToTree(
              result.List
            );

          this.menuService.add(
            group
          );

          // -------------------------------------------
          // NAVIGATION
          // -------------------------------------------

          if (this.toPage) {

            const userMenu =
              result.List
                .filter(
                  (item: any) => item.webUrl
                )
                .map(
                  (item: any) => ({
                    id: item.id,
                    webUrl: item.webUrl
                  })
                );

            this.goToPage(
              userMenu,
              this.toPage
            );

          } else if (this.menuId) {

            this.getToMenuData(
              group[0].children,
              this.menuId
            );

          } else {

            this.router.navigateByUrl(
              "/ism/dashboard"
            );
          }

        } catch (error) {

          console.error(
            "Menu decrypt/parse error:",
            error
          );

          this.msg.error(
            "Unable to process menu response."
          );
        }
      },

      (error: any) => {

        console.error(
          "Menu API HTTP error:",
          {
            status: error && error.status,
            message: error && error.message
          }
        );

        if (
          error &&
          error.status === 401
        ) {

          this.msg.error(
            "Menu API Unauthorized (401). Please check authentication and permissions."
          );

        } else {

          this.msg.error(
            "Unable to load menu permissions."
          );
        }
      }
    );
  }

  // =====================================================
  // NAVIGATION TO PAGE
  // =====================================================

  goToPage(
    userMenu: any[],
    toPageInfo: string
  ): void {

    const matches =
      userMenu.filter(
        (item) =>
          item.webUrl === toPageInfo
      );

    if (matches.length === 0) {

      console.error(
        "Requested page is not available in user permissions."
      );

      this.publicSvr.logout();

      return;
    }

    this.router.navigateByUrl(
      toPageInfo
    );
  }

  // =====================================================
  // FIND MENU BY ID
  // =====================================================

  getToMenuData(
    data: any[],
    menuId: any
  ): void {

    if (!Array.isArray(data)) {
      return;
    }

    const matches =
      data.filter(
        (item) => item.id == menuId
      );

    if (matches.length === 0) {

      data.forEach(
        (item) => {

          if (
            item.children &&
            item.children.length > 0
          ) {

            this.getToMenuData(
              item.children,
              menuId
            );
          }
        }
      );

    } else {

      this.getPage(
        matches[0]
      );
    }
  }

  // =====================================================
  // OPEN MENU PAGE
  // =====================================================

  getPage(data: any): void {

    if (
      data.children &&
      data.children.length > 0
    ) {

      this.getPage(
        data.children[0]
      );

    } else {

      this.router.navigateByUrl(
        data.link
      );

      localStorage.removeItem(
        "fromInfo"
      );

      localStorage.removeItem(
        "toPageInfo"
      );

      localStorage.removeItem(
        "userMenu"
      );

      localStorage.removeItem(
        "userMenuId"
      );
    }
  }

  // =====================================================
  // ARRAY TO TREE
  // =====================================================

  arrayToTree(
    list: any[],
    parentId: any = 0
  ): any[] {

    if (!Array.isArray(list)) {
      return [];
    }

    return list

      .filter(
        (item) =>
          item.parentId === parentId
      )

      .map(
        (item) => {

          let icon: MenuIcon = null;

          if (item.icon) {

            const parts =
              item.icon.split("|");

            icon = {

              type: "icon",

              value: parts[0],

              theme:
                parts[1] || "outline"

            } as MenuIcon;
          }

          return {

            text: item.name,

            icon: icon,

            link:
              item.menuType === "15"
                ? item.webUrl +
                  "/" +
                  item.permissionId
                : item.webUrl,

            parentId: item.parentId,

            children:
              this.arrayToTree(
                list,
                item.id
              ),

            menuType: item.menuType,

            id: item.id,

            permissionId:
              item.permissionId,

            show: true,

            urlKey: item.urlkey

          };
        }
      );
  }

    this.isForgetPassword = true;
  }

  closeForgetPasswordModal(): void {

    this.isForgetPassword = false;

    this.isCode = false;

    this.resetInfo = {

      account: "",

      code: "",

      newPassword: "",

      confirmPassword: "",

      email: ""

    };
  }

  // =====================================================
  // PASSWORD VALIDATION
  // =====================================================

  passwordChange(
    name: string
  ): boolean {

    const newPassword =
      this.resetInfo.newPassword;

    const confirmPassword =
      this.resetInfo.confirmPassword;

    const reg =
      /^(?=.*[A-Za-z])(?=.*\d)(?=.*[`~!@#$%^&*()_+<>?:"{},.\/\\;'[\]])[A-Za-z\d`~!@#$%^&*()_+<>?:"{},.\/\\;'[\]]{8,}$/;

    if (
      name === "newPassword" &&
      newPassword &&
      !reg.test(newPassword)
    ) {

      this.msg.error(
        "Password must contain at least 8 characters, including letters, numbers and special characters.",
        {
          nzDuration: 10000
        }
      );

      this.resetInfo.newPassword = "";

      return false;
    }

    if (
      name === "confirmPassword" &&
      confirmPassword &&
      !reg.test(confirmPassword)
    ) {

      this.msg.error(
        "Password must contain at least 8 characters, including letters, numbers and special characters.",
        {
          nzDuration: 10000
        }
      );

      this.resetInfo.confirmPassword = "";

      return false;
    }

    if (
      newPassword &&
      confirmPassword &&
      newPassword !== confirmPassword
    ) {

      this.msg.error(
        "The entered passwords do not match.",
        {
          nzDuration: 10000
        }
      );

      return false;
    }

    return true;
  }

  // =====================================================
  // SEND VERIFICATION CODE
  // =====================================================

  sendVerifivationCode(): void {

    if (!this.resetInfo.account) {

      this.msg.error(
        "Please enter your account."
      );

      return;
    }

    const params = {

      EmpNo: this.resetInfo.account,

      CaptchaCode: "",

      Func:
        "H5-SendVerificationCode-IWX"

    };

    this.publicSvr.permitRequest(

      params,

      {

        headers: {

          apiID:
            "ffff-1655357911834-16925411121-0153",

          userKey:
            this.publicSvr.USER_GQ

        }

      },

      true

    ).then(

      (result: any) => {

        if (
          result &&
          result.IsOK == "1"
        ) {

          this.resetInfo.email =
            result.Msg;

          this.msg.success(
            "Verification code sent."
          );

        } else {

          this.msg.error(
            (result && result.Msg) ||
            "Unable to send verification code."
          );
        }
      }

    ).catch(

      (error: any) => {

        console.error(
          "Send verification code error:",
          error
        );

        this.msg.error(
          "Unable to send verification code."
        );
      }
    );
  }

  // =====================================================
  // VERIFY CODE
  // =====================================================

  nextOption(): void {

    if (!this.resetInfo.code) {

      this.msg.error(
        "Please enter verification code."
      );

      return;
    }

    const params = {

      EmpNo: this.resetInfo.account,

      CaptchaCode: this.resetInfo.code,

      Func:
        "H5-CheckVerificationCode-IWX"

    };

    this.publicSvr.permitRequest(

      params,

      {

        headers: {

          apiID:
            "ffff-1655375927998-16925411121-0200",

          userKey:
            this.publicSvr.USER_GQ

        }

      },

      true

    ).then(

      (result: any) => {

        if (
          result &&
          result.IsOK == "1"
        ) {

          this.isCode = true;

          this.msg.success(
            "Verification code verified."
          );

        } else {

          this.msg.error(
            (result && result.Msg) ||
            "Verification failed."
          );
        }
      }

    ).catch(

      (error: any) => {

        console.error(
          "Verify code error:",
          error
        );

        this.msg.error(
          "Unable to verify code."
        );
      }
    );
  }

  // =====================================================
  // RESET PASSWORD
  // =====================================================

  resetPassword(): void {

    if (
      !this.resetInfo.account ||
      !this.resetInfo.code ||
      !this.resetInfo.newPassword ||
      !this.resetInfo.confirmPassword
    ) {

      this.msg.error(
        "All information is required."
      );

      return;
    }

    if (
      this.resetInfo.newPassword !==
      this.resetInfo.confirmPassword
    ) {

      this.msg.error(
        "The entered passwords do not match."
      );

      return;
    }

    if (
      !this.passwordChange(
        "newPassword"
      )
    ) {

      return;
    }

    const params = {

      EmpNo: this.resetInfo.account,

      Password:
        this.resetInfo.newPassword,

      CaptchaCode:
        this.resetInfo.code,

      Func:
        "H5-ChangeEmpPasswordV2-IWX"

    };

    this.publicSvr.permitRequest(

      params,

      {

        headers: {

          apiID:
            "ffff-1655376433685-16925411121-0217",

          userKey:
            this.publicSvr.USER_GQ

        }

      },

      true

    ).then(

      (result: any) => {

        if (
          result &&
          result.IsOK == "1"
        ) {

          this.msg.success(
            "Password changed successfully."
          );

          this.closeForgetPasswordModal();

        } else {

          this.msg.error(
            (result && result.Msg) ||
            "Password reset failed."
          );
        }
      }

    ).catch(

      (error: any) => {

        console.error(
          "Reset password error:",
          error
        );

        this.msg.error(
          "Unable to reset password."
        );
      }
    );
  }

  // =====================================================
  // MODAL CANCEL
  // =====================================================

  handleCancel(): void {

    this.isVisible = false;
  }
}
