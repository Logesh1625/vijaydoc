CREATE OR REPLACE PACKAGE CMS.F1_MEAL_MANAGEMENT
AS

    /* ============================================================
       INSERT F1 MENU
       ============================================================ */

    PROCEDURE INSERT_F1_MENU(
        V_GROUPS            IN VARCHAR2,
        V_MEAL_PREFERENCE   IN VARCHAR2,
        V_FX_BADGE          IN VARCHAR2,
        V_NAME              IN VARCHAR2,
        V_FIRST_DATE_FXBL   IN VARCHAR2,
        V_LAST_DATE_FXBL    IN VARCHAR2,
        V_ADD_BY            IN VARCHAR2,
        V_RES               OUT VARCHAR2,
        V_RESULT            OUT VARCHAR2
    );


    /* ============================================================
       BULK F1 MENU
       ============================================================ */

    PROCEDURE BULK_F1_MENU(
        V_GROUPS            IN VARCHAR2,
        V_MEAL_PREFERENCE   IN VARCHAR2,
        V_FX_BADGE          IN VARCHAR2,
        V_NAME              IN VARCHAR2,
        V_FIRST_DATE_FXBL   IN VARCHAR2,
        V_LAST_DATE_FXBL    IN VARCHAR2,
        V_ADD_BY            IN VARCHAR2,
        V_RES               OUT VARCHAR2,
        V_RESULT            OUT VARCHAR2
    );


    /* ============================================================
       GET F1 MENU MANAGEMENT
       ============================================================ */

    PROCEDURE GET_F1_MENU_MANAGE(
        V_EMP_ID        IN VARCHAR2,
        V_GROUPS        IN VARCHAR2,
        V_SITE          IN VARCHAR2,
        V_FX_BADGE      IN VARCHAR2,
        V_ADD_BY        IN VARCHAR2,
        L_START_PAGE    IN VARCHAR2,
        L_PAGE_SIZE     IN VARCHAR2,
        V_RES           OUT VARCHAR2,
        V_RESULT        OUT VARCHAR2,
        P_CURSOR        OUT SYS_REFCURSOR
    );


    /* ============================================================
       UPDATE F1 MENU
       ============================================================ */

    PROCEDURE UPDATE_F1_MENU_MANAGE(
        V_ID                IN NUMBER,
        V_GROUPS            IN VARCHAR2,
        V_MEAL_PREFERENCE   IN VARCHAR2,
        V_FX_BADGE          IN VARCHAR2,
        V_NAME              IN VARCHAR2,
        V_FIRST_DATE_FXBL   IN VARCHAR2,
        V_LAST_DATE_FXBL    IN VARCHAR2,
        V_EDIT_BY           IN VARCHAR2,
        V_RES               OUT VARCHAR2,
        V_RESULT            OUT VARCHAR2
    );


    /* ============================================================
       DELETE F1 MENU
       ============================================================ */

    PROCEDURE DELETE_F1_MENU_MANAGE(
        L_IDS       IN VARCHAR2,
        V_EDIT_BY   IN VARCHAR2,
        V_RES       OUT VARCHAR2,
        V_RESULT    OUT VARCHAR2
    );


    /* ============================================================
       GET F1 MENU BY FX BADGE
       ============================================================ */

    PROCEDURE GET_F1_MENU_BY_FXBADGE(
        V_FX_BADGE  IN VARCHAR2,
        V_RES       OUT VARCHAR2,
        V_RESULT    OUT VARCHAR2,
        P_CURSOR    OUT SYS_REFCURSOR
    );


    /* ============================================================
       INSERT MEAL SCAN LOG
       ============================================================ */

    PROCEDURE INSERT_MEAL_SCAN_LOG(
        V_FX_BADGE  IN VARCHAR2,
        V_RES       OUT VARCHAR2,
        V_RESULT    OUT VARCHAR2,
        P_CURSOR    OUT SYS_REFCURSOR
    );


    /* ============================================================
       GET MEAL SCAN LOG
       ============================================================ */

    PROCEDURE GET_MEAL_SCAN_LOG(
        V_SEARCH_TEXT       IN VARCHAR2,
        V_STATUS_FILTER     IN VARCHAR2,
        V_SITE              IN VARCHAR2,
        V_RES               OUT VARCHAR2,
        V_RESULT            OUT VARCHAR2,
        P_CURSOR            OUT SYS_REFCURSOR,
        P_SUMMARY_CURSOR    OUT SYS_REFCURSOR
    );


    /* ============================================================
       GET MEAL DASHBOARD

       IMPORTANT:
       V_EMP_ID is used to identify logged-in CMS user.

       V_SITE:
          NULL / '' / ALL = all sites belonging to user
          specific value  = only that site
       ============================================================ */

    PROCEDURE GET_MEAL_DASHBOARD(
        V_EMP_ID            IN VARCHAR2,
        V_DATE              IN VARCHAR2,
        V_SITE              IN VARCHAR2,
        V_RES               OUT VARCHAR2,
        V_RESULT            OUT VARCHAR2,
        P_CURSOR            OUT SYS_REFCURSOR,
        P_SUMMARY_CURSOR    OUT SYS_REFCURSOR
    );


END F1_MEAL_MANAGEMENT;
