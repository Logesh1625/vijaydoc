      t = a2, a2 = a1, a1 = t;

      inter = a1.filter(function (e) {
        if (a2.indexOf(e) !== -1) return true;
      });
      interStr = inter.join('');
      orgS1 = s1.toUpperCase();
      if (interStr.length > 0) {
        res = orgS1.indexOf(interStr);

        res = -1;
      }

      return res;
