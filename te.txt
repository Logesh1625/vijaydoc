const response = await axios.post(
  'https://microservice.abgind.foxconn.com/india/gateway',
  params,
  { headers }
);

const data = response?.data;

console.log('HTTP status:', response.status);
console.log('Gateway response:', data);

let res;

if (typeof data?.r === 'string' && data.r.trim()) {
  const decryptedResponse = decrypt(
    data.r,
    getdk(iwxKey)
  );

  console.log('Decrypted OTP response:', decryptedResponse);

  if (!decryptedResponse) {
    throw new Error('Unable to decrypt OTP API response.');
  }

  try {
    res = JSON.parse(decryptedResponse);
  } catch (parseError) {
    console.error('OTP response is not valid JSON.');
    throw new Error('Unable to parse OTP API response.');
  }
} else if (data && typeof data.message === 'string') {
  console.error('OTP API error:', data.message);
  setErrorMessage(data.message);
  return;
} else {
  console.error('Unexpected OTP API response:', data);
  setErrorMessage('Unexpected OTP API response.');
  return;
}
