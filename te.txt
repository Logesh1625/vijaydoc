import { Component, OnInit } from "@angular/core";
import { NzMessageService, NzDrawerRef } from "ng-zorro-antd";
import { I18NService } from "@core";
import { PublicNetService } from "src/app/routes/a_Public_net/Public_net.service";

@Component({
  selector: "app-F1-Details-add",
  templateUrl: "./F1-Details-add.html",
})
export class AddDetailsDrawComponent implements OnInit {

  constructor(
    private publicSvr: PublicNetService,
    private msgSvr: NzMessageService,
    private i18n: I18NService,
    private ref: NzDrawerRef
  ) {}

  groups: any = "";
  mealPreference: number | null = null;
  fxBadge: any = "";
  name: any = "";
  remark: string = "";

  firstDateFxbl: Date | null = null;
  lastDateFxbl: Date | null = null;

  option: any;
  editData: any;
  timer: any;

  mealPreferenceList: any[] = [];

  // ==========================================
  // DATE HANDLERS
  // ==========================================

  onFirstDateFxblChange(date: Date | null): void {
    this.firstDateFxbl = date;

    if (
      this.lastDateFxbl &&
      this.firstDateFxbl &&
      this.startOfDay(this.lastDateFxbl).getTime() <
        this.startOfDay(this.firstDateFxbl).getTime()
    ) {
      this.lastDateFxbl = null;
    }
  }

  disabledLastDateFxbl = (current: Date): boolean => {
    if (!this.firstDateFxbl || !current) {
      return false;
    }

    return (
      this.startOfDay(current).getTime() <
      this.startOfDay(this.firstDateFxbl).getTime()
    );
  };

  private startOfDay(date: Date): Date {
    const d = new Date(date);
    d.setHours(0, 0, 0, 0);
    return d;
  }

  // ==========================================
  // INIT
  // ==========================================

  ngOnInit(): void {

    if (this.editData) {
      this.groups = this.editData.Groups || "";
      this.fxBadge = this.editData.FxBadge || "";
      this.name = this.editData.Name || "";
      this.remark = this.editData.Remark || "";

      this.firstDateFxbl = this.parseDate(
        this.editData.FirstDateFxbl
      );

      this.lastDateFxbl = this.parseDate(
        this.editData.LastDateFxbl
      );

      console.log("EDIT DATA:", this.editData);
    }

    this.loadMealPreferences();
  }

  // ==========================================
  // LOAD MEAL PREFERENCE MASTER
  // ==========================================

  private loadMealPreferences(): void {

    this.publicSvr
      .getGuardPassCode("GuardPass-Meal Preference")
      .subscribe(
        (res: any) => {

          const list = Array.isArray(res)
            ? res
            : res && Array.isArray(res.Data)
              ? res.Data
              : [];

          this.mealPreferenceList = list.map((item: any) => ({
            label: String(item.label || item.NAME || item.name || "").trim(),
            value: Number(item.value !== undefined
              ? item.value
              : item.ID !== undefined
                ? item.ID
                : item.id)
          })).filter((item: any) =>
            item.label !== "" && Number.isFinite(item.value)
          );

          console.log(
            "MEAL PREFERENCE LIST:",
            this.mealPreferenceList
          );

          if (this.option === "Edit" && this.editData) {
            this.setEditMealPreference();
          }
        },
        (error: any) => {
          console.error("Meal Preference load error:", error);
          this.msgSvr.error("Failed to load Meal Preference list.");
        }
      );
  }

  // ==========================================
  // EDIT MODE: SELECT MEAL PREFERENCE
  // ==========================================

  private setEditMealPreference(): void {

    const rawValue =
      this.editData.MealPreference !== undefined &&
      this.editData.MealPreference !== null
        ? this.editData.MealPreference
        : this.editData.MEAL_PREFERENCE;

    if (rawValue === undefined || rawValue === null || rawValue === "") {
      this.mealPreference = null;
      return;
    }

    const input = String(rawValue).trim();

    // First, try matching by ID.
    const byId = this.mealPreferenceList.find(
      (item: any) => String(item.value) === input
    );

    if (byId) {
      this.mealPreference = byId.value;
      console.log("EDIT MEAL PREFERENCE MATCHED BY ID:", byId);
      return;
    }

    // If the API returns the display name, match by label.
    const byName = this.mealPreferenceList.find(
      (item: any) =>
        item.label.toUpperCase() === input.toUpperCase()
    );

    if (byName) {
      this.mealPreference = byName.value;
      console.log("EDIT MEAL PREFERENCE MATCHED BY NAME:", byName);
      return;
    }

    this.mealPreference = null;

    console.warn(
      "Meal Preference not found:",
      rawValue,
      this.mealPreferenceList
    );
  }

  // ==========================================
  // DATE PARSER
  // ==========================================

  parseDate(value: any): Date | null {

    if (!value) {
      return null;
    }

    if (value instanceof Date) {
      return value;
    }

    if (typeof value === "string") {

      const parts = value.trim().split(/[-\/]/);

      if (parts.length === 3) {

        let day: number;
        let month: number;
        let year: number;

        // Supports DD-MM-YYYY and YYYY-MM-DD.
        if (parts[0].length === 4) {
          year = Number(parts[0]);
          month = Number(parts[1]) - 1;
          day = Number(parts[2]);
        } else {
          day = Number(parts[0]);
          month = Number(parts[1]) - 1;
          year = Number(parts[2]);
        }

        const parsed = new Date(year, month, day);

        if (
          !isNaN(parsed.getTime()) &&
          parsed.getFullYear() === year &&
          parsed.getMonth() === month &&
          parsed.getDate() === day
        ) {
          return parsed;
        }
      }
    }

    return null;
  }

  // ==========================================
  // VALIDATION
  // ==========================================

  private validateForm(): boolean {

    if (!this.groups || !String(this.groups).trim()) {
      this.msgSvr.warning("Please enter Groups.");
      return false;
    }

    if (this.mealPreference === null || this.mealPreference === undefined) {
      this.msgSvr.warning("Please select Meal Preference.");
      return false;
    }

    if (!this.fxBadge || !String(this.fxBadge).trim()) {
      this.msgSvr.warning("Please enter FX Badge.");
      return false;
    }

    if (!this.name || !String(this.name).trim()) {
      this.msgSvr.warning("Please enter Name.");
      return false;
    }

    if (!this.firstDateFxbl || !this.lastDateFxbl) {
      this.msgSvr.warning("Please select First Date and Last Date.");
      return false;
    }

    if (
      this.startOfDay(this.firstDateFxbl).getTime() >
      this.startOfDay(this.lastDateFxbl).getTime()
    ) {
      this.msgSvr.warning(
        "First Date cannot be greater than Last Date."
      );
      return false;
    }

    return true;
  }

  // ==========================================
  // ADD
  // ==========================================

  add(): void {

    if (!this.validateForm()) {
      return;
    }

    const params = {
      Groups: this.groups,
      MealPreference: this.mealPreference,
      FxBadge: this.fxBadge,
      Name: this.name,
      Remark: this.remark,

      FirstDateFxbl: this.publicSvr.dateFormat(
        this.firstDateFxbl,
        "DD-MM-YYYY"
      ),

      LastDateFxbl: this.publicSvr.dateFormat(
        this.lastDateFxbl,
        "DD-MM-YYYY"
      ),

      AddBy: this.publicSvr.getSettingUserInfo().eeid,
      Func: "CMS-InsertF1Menu"
    };

    console.log("ADD PARAMS:", params);

    this.publicSvr
      .iwxPostF1(
        params,
        {
          headers: {
            apiID: "ffff-1789631992006-10208193125-1291",
            userKey: "59EC4918B4383A80B95F3612AE1E5FE1"
          }
        },
        true
      )
      .then((res: any) => {

        console.log("ADD RESPONSE:", res);

        if (res && res.IsOK == "1") {
          this.msgSvr.success("Success");
          this.ref.close(true);
        } else {
          this.msgSvr.error(
            res && res.Msg ? res.Msg : "Failed to add."
          );
        }
      })
      .catch((error: any) => {
        console.error("ADD ERROR:", error);
        this.msgSvr.error("Failed to add record.");
      });
  }

  // ==========================================
  // EDIT
  // ==========================================

  edit(): void {

    if (!this.validateForm()) {
      return;
    }

    const params = {
      Id: this.editData.Id,
      Groups: this.groups,
      MealPreference: this.mealPreference,
      FxBadge: this.fxBadge,
      Name: this.name,
      Remark: this.remark,

      FirstDateFxbl: this.publicSvr.dateFormat(
        this.firstDateFxbl,
        "DD-MM-YYYY"
      ),

      LastDateFxbl: this.publicSvr.dateFormat(
        this.lastDateFxbl,
        "DD-MM-YYYY"
      ),

      EditBy: this.publicSvr.getSettingUserInfo().eeid,
      Func: "CMS-UpdateF1MenuManage"
    };

    console.log("EDIT PARAMS:", params);

    this.publicSvr
      .iwxPostF1(
        params,
        {
          headers: {
            apiID: "ffff-1789634002932-10208193125-1317",
            userKey: "59EC4918B4383A80B95F3612AE1E5FE1"
          }
        },
        true
      )
      .then((res: any) => {

        console.log("EDIT RESPONSE:", res);

        if (res && res.IsOK == "1") {
          this.msgSvr.success("Success");
          this.ref.close(true);
        } else {
          this.msgSvr.error(
            res && res.Msg ? res.Msg : "Failed to update."
          );
        }
      })
      .catch((error: any) => {
        console.error("EDIT ERROR:", error);
        this.msgSvr.error("Failed to update record.");
      });
  }

  // ==========================================
  // CANCEL
  // ==========================================

  cancel(): void {
    this.ref.close();
  }

}


<nz-form-item nz-col [nzSpan]="24">

  <nz-form-label
    class="form-label"
    [nzSpan]="5"
    nzRequired
  >
    Meal Preference
  </nz-form-label>

  <nz-form-control [nzSpan]="14">

    <nz-select
      [(ngModel)]="mealPreference"
      name="mealPreference"
      required
      nzPlaceHolder="Please select"
      style="width: 100%;"
    >

      <nz-option
        *ngFor="let item of mealPreferenceList"
        [nzValue]="item.value"
        [nzLabel]="item.label"
      >
      </nz-option>

    </nz-select>

  </nz-form-control>

</nz-form-item>
