const Login = () => {

  const navigate = useNavigate();

  const classes = useStyles(theme);


  // ==========================================================
  // STATE
  // ==========================================================

  const [isLoading, setIsLoading] =
    useState(false);

  const [loginStatus, setLoginStatus] =
    useState('');

  const [account, setAccount] =
    useState('');

  const [password, setPassword] =
    useState('');

  const [errorMessage, setErrorMessage] =
    useState('');

  const [showPassword, setShowPassword] =
    useState(false);

  const [secretkey, setsecretkey] =

  const [otp, setOTP] =
    useState('');

  const [showQRCode, setShowQRCode] =
    useState(false);

    useState(false);


  // ==========================================================
  // SESSION VALUES
  // ==========================================================

  const issuccessfullylogin =
    sessionStorage.getItem(
      'issuccessfullylogin'
    );

  const isscuessqr =
    sessionStorage.getItem(
      'qrstatus'
    );


  // ==========================================================
  // ENCRYPTED VALUES
  // ==========================================================

  const originalBoolean = true;

  const encryptedBoolean =
    encrypt(
      originalBoolean.toString(),
      getdk(iwxKey)
    );


  const originalNumberBoolean = 0;

  const encryptedBooleannum =
    encrypt(
      originalNumberBoolean.toString(),
      getdk(iwxKey)
    );


  // ==========================================================
  // PASSWORD VISIBILITY
  // ==========================================================

  const handleTogglePassword = () => {

    setShowPassword(
      previous =>
        !previous
    );

  };


  // ==========================================================
  // ENTER KEY
  // ==========================================================

  const handleKeyPress = (event) => {

    if (event.key === 'Enter') {

      event.preventDefault();

      handleLogin(event);

    }

  };


  // ==========================================================
  // GET USER DETAILS
  // ==========================================================

  const fetchdata = async () => {

    try {

      const headers = {

        apiID:
          USER_DETAILS_API_ID,

        userKey:
          USER_KEY

      };


      const params = {

        EmpId:
          account.trim(),

        EmpName:
          '',

        Status:
          '',

        StartPage:
          1,

        PageSize:
          999999,

        Func:
          'UserManagement-Get_user_details_new'

      };


      console.log(
        '================================'
      );

      console.log(
        'USER DETAILS API'
      );

      console.log(
        'apiID:',
        headers.apiID
      );

      console.log(
        'EmpId:',
        params.EmpId
      );


      const res =
        await iwxPost(
          params,
          { headers },
          true
        );


      console.log(
        'USER DETAILS RESPONSE:',
        res
      );


      if (
        !res ||
        !res.List
      ) {

        console.error(
          'User details API returned no List'
        );

        return null;

      }


      // ======================================================
      // SAVE QR STATUS
      // ======================================================

      if (
        res.List.length > 0
      ) {

        const qrStatusValue =
          res.List[0].QrStatus;


        const qrencrypt =
          encrypt(
            String(qrStatusValue),
            getdk(iwxKey)
          );


        sessionStorage.setItem(
          'qrstatus',
          qrencrypt
        );

      }


      return res;


    } catch (error) {

      console.error(
        'USER DETAILS API ERROR:',
        error
      );


      if (error.response) {

        console.error(
          'HTTP STATUS:',
          error.response.status
        );

        console.error(
          'SERVER RESPONSE:',
          error.response.data
        );

      }


      return null;

    }

  };


  // ==========================================================
  // GET EMPLOYEE NAME
  // ==========================================================

  const fetchempName = async () => {

    try {

      const result =
        await fetchempbasicinfo(
          account.trim()
        );

      return result;

    } catch (error) {

      console.error(
        'EMPLOYEE NAME ERROR:',
        error
      );

      return null;

    }

  };


  // ==========================================================
  // GET ROLE DATA
  // ==========================================================

  const fetchRoleData = async () => {

    try {

      const headers = {

        apiID:
          ROLE_API_ID,

        userKey:
          USER_KEY

      };


      const params = {

        EmpId:
          account.trim(),

        Func:
          'UserManagement-Get_role_based_empid_new'

      };


      console.log(
        '================================'
      );

      console.log(
        'ROLE API'
      );

      console.log(
        'apiID:',
        headers.apiID
      );

      console.log(
        'EmpId:',
        params.EmpId
      );


      const res =
        await iwxPost(
          params,
          { headers },
          true
        );


      console.log(
        'ROLE API RESPONSE:',
        res
      );


      return res;


    } catch (error) {

      console.error(
        'ROLE API ERROR:',
        error
      );


      if (error.response) {

        console.error(
          'HTTP STATUS:',
          error.response.status
        );

        console.error(
          'SERVER RESPONSE:',
          error.response.data
        );

      }


      return null;

    }

  };


  // ==========================================================
  // STORE LOGIN LOG
  // ==========================================================

  const StoreLoginLog = async (
    msg,
    empname,
    roles
  ) => {

    try {

      const headers = {

        apiID:
          LOGIN_LOG_INSERT_API_ID,

        userKey:
          USER_KEY

      };


      const params = {

        EmpId:
          account.trim(),

        SystemIp:
          '',

        LoginStatus:
          msg,

        EmpName:
          empname &&
          empname[0]
            ? empname[0].EmployeeName
            : '',

        UserRole:
          roles &&
          roles.List &&
          roles.List[0]
            ? roles.List[0].RoleName
            : '',

        Func:
          'UserManagement-Insert_login_log_new'

      };


      console.log(
        '================================'
      );

      console.log(
        'LOGIN LOG API'
      );

      console.log(
        'REQUEST:',
        params
      );


      const res =
        await iwxPost(
          params,
          { headers },
          true
        );


      console.log(
        'LOGIN LOG RESPONSE:',
        res
      );


      if (
        !res ||
        !res.List
      ) {

        console.error(
          'Login log API returned no List'
        );

        return undefined;

      }


      return res.List[0]?.Id;


    } catch (error) {

      console.error(
        'LOGIN LOG ERROR:',
        error
      );

      return undefined;

    }

  };


  // ==========================================================
  // UPDATE QR STATUS / SECRET KEY
  // ==========================================================

  const updatateqrstatua = async () => {

    try {

      const acnt =
        sessionStorage.getItem(
          'account'
        );


      if (!acnt) {

        console.error(
          'Account not found in session'
        );

        return;

      }


      const decryptemp =
        decrypt(
          acnt,
          getdk(iwxKey)
        );


      const empnumber =
        decryptemp.replace(
          /"/g,
          ''
        );


      const headers = {

        apiID:
          'ffff-1743483629535-1020912129-2321',

        userKey:
          USER_KEY

      };


      const params = {

        EmpId:
          empnumber,

        SecretKey:
          secretkey,

        Func:
          'UserManagement-Update_secret_key_new'

      };


      console.log(
        'UPDATE SECRET KEY:',
        params
      );


      if (
        isscuessqr ===
        encryptedBooleannum
      ) {

        await iwxPost(
          params,
          { headers },
          true
        );

      }

    } catch (error) {

      console.error(
        'UPDATE QR STATUS ERROR:',
        error
      );

    }

  };


  // ==========================================================
  // GENERATE SECRET KEY
  // ==========================================================

  const handleGenerateSecret = async () => {

    try {

      const headers = {

        apiID:
          SECRET_GENERATE_API_ID,

        userKey:
          SECRET_GEN_USER_KEY

      };


      console.log(
        '================================'
      );

      console.log(
        'SECRET GENERATION API'
      );

      console.log(
        'URL:',
        SECRET_GEN_URL
      );


      const response =
        await axios.post(
          SECRET_GEN_URL,
          {},
          {
            headers
          }
        );


      console.log(
        'SECRET RESPONSE:',
        response.data
      );


      if (
        !response.data ||
        !response.data.r
      ) {

        console.error(
          'Secret generation rejected:',
          response.data
        );

        return;

      }


      const decrypted =
        decrypt(
          response.data.r,
          getdk(iwxKey)
        );


      const result =
        JSON.parse(
          decrypted
        );


      console.log(
        'SECRET RESULT:',
        result
      );


      if (result.secret) {

        setsecretkey(
          result.secret
        );

        setShowQRCode(
          true
        );

      }


    } catch (error) {

      console.error(
        'SECRET GENERATION ERROR:',
        error
      );


      if (error.response) {

        console.error(
          'HTTP STATUS:',
          error.response.status
        );

        console.error(
          'SERVER RESPONSE:',
          error.response.data
        );

      }

    }

  };


  // ==========================================================
  // GENERATE OTP URI
  // ==========================================================

  const generateOTPURI = () => {

    try {

      const acnt =
        sessionStorage.getItem(
          'account'
        );


      if (!acnt) {

        return '';

      }


      const decryptemp =
        decrypt(
          acnt,
          getdk(iwxKey)
        );


      const empnumber =
        decryptemp.replace(
          /"/g,
          ''
        );


      return (
        `otpauth://totp/DMS:${encodeURIComponent(empnumber)}` +
        `?secret=${secretkey}`
      );


    } catch (error) {

      console.error(
        'OTP URI ERROR:',
        error
      );

      return '';

    }

  };


  // ==========================================================
  // STORE OTP LOGIN LOG
  // ==========================================================

  const StoreOTPtologinlog = async (
    msg
  ) => {

    try {

      const getloginlogid =
        sessionStorage.getItem(
          'loginlogID'
        );


      if (!getloginlogid) {

        console.error(
          'loginlogID not found'
        );

        return;

      }


      const decryptloginlog =
        decrypt(
          getloginlogid,
          getdk(iwxKey)
        );


      const headers = {

        apiID:
          LOGIN_LOG_UPDATE_API_ID,

        userKey:
          USER_KEY

      };


      const params = {

        Update:
          'OTP',

        Id:
          decryptloginlog,

        Otp:
          otp,

        OtpStatus:
          msg,

        Func:
          'UserManagement-Update_login_log_new'

      };


      console.log(
        'OTP LOGIN LOG:',
        params
      );


      await iwxPost(
        params,
        { headers },
        true
      );


    } catch (error) {

      console.error(
        'OTP LOGIN LOG ERROR:',
        error
      );

    }

  };


  // ==========================================================
  // VERIFY OTP
  // ==========================================================

  const verifyOTP = async () => {

    if (!otp) {

      setErrorMessage(
        'Enter OTP.'
      );

      return;

    }


    setotploading(true);


    try {

      const acnt =
        sessionStorage.getItem(
          'account'
        );


      if (!acnt) {

        setErrorMessage(
          'Account session expired.'
        );

        return;

      }


      const decryptemp =
        decrypt(
          acnt,
          getdk(iwxKey)
        );


      const empnumber =
        decryptemp.replace(
          /"/g,
          ''
        );


      const headers = {

        apiID:
          USER_DETAILS_API_ID,

        userKey:
          USER_KEY

      };


      const params = {

        EmpId:
          empnumber,

        EmpName:
          '',

        Status:
          '',

        StartPage:
          1,

        PageSize:
          999999,

        Func:
          'UserManagement-Get_user_details_new'

      };


      console.log(
        'GET SECRET KEY FOR OTP'
      );


      const res =
        await iwxPost(
          params,
          { headers },
          true
        );


      console.log(
        'OTP USER DETAILS:',
        res
      );


      if (
        !res ||
        !res.List ||
        res.List.length === 0
      ) {

        setErrorMessage(
          'Unable to verify OTP. User details not found.'
        );

        return;

      }


      const secretkeyget =
        res.List[0]?.SecretKey;


      if (
        !secretkeyget &&
        isscuessqr !==
        encryptedBooleannum
      ) {

        setErrorMessage(
          'Secret key not found.'
        );

        return;

      }


      await verifyOTP22(
        secretkeyget
      );


    } catch (error) {

      console.error(
        'VERIFY OTP ERROR:',
        error
      );

      setErrorMessage(
        'Unable to verify OTP. Please try again.'
      );

    } finally {

      setotploading(false);

    }

  };


  // ==========================================================
  // VERIFY OTP AGAINST AUTHENTICATOR API
  // ==========================================================

  const verifyOTP22 = async (
    secretkeyget
  ) => {

    try {

      const secretToVerify =
        isscuessqr ===
        encryptedBooleannum
          ? secretkey
          : secretkeyget;


      if (!secretToVerify) {

        setErrorMessage(
          'Secret key is missing.'
        );

        return;

      }


      const headers = {

        apiID:
          OTP_VERIFY_API_ID,

        userKey:
          OTP_VERIFY_USER_KEY

      };


      // API documentation says otp = int
      const otpNumber =
        Number(otp);


      if (
        !Number.isInteger(
          otpNumber
        )
      ) {

        setErrorMessage(
          'Enter a valid OTP.'
        );

        return;

      }


      const params =
        dealPostParamsEncrypt({

          secret:
            secretToVerify,

          otp:
            otpNumber

        });


      console.log(
        '================================'
      );

      console.log(
        'OTP VERIFY API'
      );

      console.log(
        'apiID:',
        headers.apiID
      );


      const response =
        await axios.post(
          OTP_VERIFY_URL,
          params,
          {
            headers
          }
        );


      console.log(
        'OTP VERIFY RESPONSE:',
        response.data
      );


      if (
        !response.data ||
        !response.data.r
      ) {

        console.error(
          'OTP verification rejected:',
          response.data
        );

        setErrorMessage(
          'Unable to verify OTP. Please try again.'
        );

        return;

      }


      const decrypted =
        decrypt(
          response.data.r,
          getdk(iwxKey)
        );


      const result =
        JSON.parse(
          decrypted
        );


      console.log(
        'OTP RESULT:',
        result
      );


      // ======================================================
      // SUCCESS
      // ======================================================

      if (
        result.IsOK === 1 ||
        result.IsOK === '1'
      ) {

        await updatateqrstatua();


        sessionStorage.setItem(
          'isLoggedIn',
          encryptedBoolean
        );


        const loginresponse =
          sessionStorage.getItem(
            'loginresponse'
          );


        sessionStorage.setItem(
          'data',
          loginresponse || ''
        );


        await StoreOTPtologinlog(
          result.Msg
        );


        navigate(
          '/home/Dashboard',
          {
            state: {
              account
            }
          }
        );


      } else {

        setErrorMessage(
          result.Msg ||
          'Incorrect OTP'
        );


        await StoreOTPtologinlog(
          result.Msg ||
          'Incorrect OTP'
        );

      }


    } catch (error) {

      console.error(
        'OTP VERIFY ERROR:',
        error
      );


      if (error.response) {

        console.error(
          'OTP HTTP STATUS:',
          error.response.status
        );

        console.error(
          'OTP SERVER RESPONSE:',
          error.response.data
        );

      }


      setErrorMessage(
        'Unable to verify OTP. Please try again.'
      );

    }

  };


  // ==========================================================
  // MAIN LOGIN
  // ==========================================================

  const handleLogin = async (e) => {

    e.preventDefault();


    if (isLoading) {

      return;

    }


    setIsLoading(true);

    setErrorMessage('');


    try {

      // ======================================================
      // 1. VALIDATION
      // ======================================================

      const empId =
        account.trim();


      if (
        !empId ||
        !password
      ) {

        setErrorMessage(
          'Fill in all fields.'
        );

        return;

      }


      // ======================================================
      // 2. GET USER DETAILS
      // ======================================================

      const userData =
        await fetchdata();


      console.log(
        'FINAL USER DATA:',
        userData
      );


      if (
        !userData ||
        !userData.List
      ) {

        setErrorMessage(
          'Error checking account availability.'
        );

        return;

      }


      // ======================================================
      // 3. USER NOT FOUND
      // ======================================================

      if (
        userData.List.length === 0
      ) {

        setErrorMessage(
          "You don't have access to this system."
        );


        const empName =
          await fetchempName();


        const roles =
          await fetchRoleData();


        await StoreLoginLog(
          "You don't have access to this system",
          empName,
          roles
        );


        return;

      }


      // ======================================================
      // 4. FIND CURRENT EMPLOYEE
      // ======================================================

      const findempdata =
        userData.List.find(
          item =>
            item.EmpId === empId
        );


      if (!findempdata) {

        setErrorMessage(
          "You don't have access to this system."
        );


        return;

      }


      // ======================================================
      // 5. CHECK DELETE FLAG
      // ======================================================

      if (
        findempdata.DelFlag === '1' ||
        findempdata.DelFlag === 1
      ) {

        const empName =
          await fetchempName();


        const roles =
          await fetchRoleData();


        const msg =
          "You don't have access to this system";


        setErrorMessage(
          msg
        );


        await StoreLoginLog(
          msg,
          empName,
          roles
        );


        return;

      }


      // ======================================================
      // 6. SAVE ACCOUNT
      // ======================================================

      const accountt =
        dealPostParamsEncrypt(
          empId
        );


      sessionStorage.setItem(
        'account',
        accountt.s
      );


      // ======================================================
      // 7. GET EMPLOYEE NAME
      // ======================================================

      const fetchempname =
        await fetchempName();


      console.log(
        'EMPLOYEE NAME:',
        fetchempname
      );


      // ======================================================
      // 8. GET ROLE
      // ======================================================

      const fetchroles =
        await fetchRoleData();


      console.log(
        'ROLE DATA:',
        fetchroles
      );


      // ======================================================
      // 9. LOGIN API
      // ======================================================

      const encryptedParams =
        dealPostParamsEncrypt({

          Func:
            'APPFrame-AccountLoginPhoneSpv3-IWX',

          Account:
            empId,

          Password:
            password,

          OsType:
            '1',

          DeviceId:
            '1',

          OsVer:
            '1',

          AppVer:
            '1',

          DeviceType:
            '1',

          AppVersion:
            '1'

        });


      console.log(
        '================================'
      );

      console.log(
        'LOGIN CREDENTIAL API'
      );


      const response =
        await getlogincredentials(
          encryptedParams
        );


      console.log(
        'LOGIN RAW RESPONSE:',
        response
      );


      if (
        !response ||
        !response.r
      ) {

        setErrorMessage(
          'Login API returned no response.'
        );

        return;

      }


      // ======================================================
      // 10. DECRYPT LOGIN RESPONSE
      // ======================================================

      const decryptedResponse =
        decrypt(
          response.r,
          getdk(iwxKey)
        );


      const decryptedResponseObj =
        JSON.parse(
          decryptedResponse
        );


      console.log(
        'LOGIN RESPONSE:',
        decryptedResponseObj
      );


      // ======================================================
      // 11. LOGIN SUCCESS
      // ======================================================

      if (
        decryptedResponseObj.IsOK === '1' ||
        decryptedResponseObj.IsOK === 1
      ) {


        const loginlogresponse =
          await StoreLoginLog(
            decryptedResponseObj.Msg,
            fetchempname,
            fetchroles
          );


        if (
          loginlogresponse ===
            undefined ||
          loginlogresponse ===
            null
        ) {

          setErrorMessage(
            'Unable to record the login. Please try again.'
          );

          setLoginStatus(
            'error'
          );

          return;

        }


        // ==================================================
        // SAVE LOGIN LOG ID
        // ==================================================

        const encryptedloginlogid =
          encrypt(
            loginlogresponse.toString(),
            getdk(iwxKey)
          );


        sessionStorage.setItem(
          'loginlogID',
          encryptedloginlogid
        );


        // ==================================================
        // SAVE LOGIN RESPONSE
        // ==================================================

        sessionStorage.setItem(
          'loginresponse',
          response.r
        );


        // ==================================================
        // LOGIN SUCCESS STATUS
        // ==================================================

        sessionStorage.setItem(
          'issuccessfullylogin',
          encryptedBoolean
        );


        setLoginStatus(
          'success'
        );


      } else {


        // ==================================================
        // LOGIN FAILED
        // ==================================================

        const msg =
          decryptedResponseObj.Msg ||
          'Login failed';


        await StoreLoginLog(
          msg,
          fetchempname,
          fetchroles
        );


        setErrorMessage(
          msg
        );


        setLoginStatus(
          'error'
        );

      }


    } catch (error) {

      console.error(
        '================================'
      );

      console.error(
        'HANDLE LOGIN ERROR:',
        error
      );

      console.error(
        '================================'
      );


      if (error.response) {

        console.error(
          'HTTP STATUS:',
          error.response.status
        );

        console.error(
          'SERVER RESPONSE:',
          error.response.data
        );

      }


      setErrorMessage(
        'Login failed. Please try again.'
      );


      setLoginStatus(
        'error'
      );


    } finally {

      setIsLoading(
        false
      );

    }

  };


  // ==========================================================
  // BACK TO LOGIN
  // ==========================================================

  const handlebacklogin = () => {

    sessionStorage.removeItem(
      'issuccessfullylogin'
    );

    sessionStorage.removeItem(
      'qrstatus'
    );

    sessionStorage.removeItem(
      'loginresponse'
    );

    sessionStorage.removeItem(
      'loginlogID'
    );

    sessionStorage.removeItem(
      'account'
    );


    setAccount('');

    setPassword('');

    setOTP('');

    setsecretkey('');

    setShowQRCode(false);

    setErrorMessage('');

    setLoginStatus('');


    navigate(
      '/login'
    );

  };


  // ==========================================================
  // CLEAR ERROR WHEN INPUT CHANGES
  // ==========================================================

  useEffect(() => {

    setErrorMessage('');

  }, [
    account,
    password,
    otp
  ]);


  // ==========================================================
  // GENERATE SECRET / QR
  // ==========================================================

  useEffect(() => {

    const isLoggedIn =
      sessionStorage.getItem(
        'issuccessfullylogin'
      );


    if (
      isLoggedIn === null ||
      isLoggedIn === encryptedBoolean
    ) {

      handleGenerateSecret();

    }

  }, [
    issuccessfullylogin,
    loginStatus
  ]);


  // ==========================================================
  // UI
  // ==========================================================

  return (

    <Grid
      container
      component="main"
      sx={{

        minHeight: '100vh',

        backgroundImage:
          `url(${loginimage})`,

        backgroundRepeat:
          'no-repeat',

        backgroundSize:
          'cover',

        backgroundPosition:
          'center',

        position:
          'relative'

      }}
    >


      {/* ================================================
          LOGIN / MFA AREA
      ================================================= */}

      <Grid
        item
        xs={12}
        sx={{

          display:
            'flex',

          alignItems:
            'center',

          justifyContent:
            'center',

          minHeight:
            '90vh'

        }}
      >

        <Box
          className={
            classes.root
          }
          onKeyPress={
            handleKeyPress
          }
        >

          <Box
            className={
              classes.formm
            }
          >


            {/* ==========================================
                LOGO
            =========================================== */}

            <img
              src="/images/logo1.png"
              alt="MUI Image"
              style={{
                maxWidth:
                  '180px',
                marginBottom:
                  '10px'
              }}
            />


            {/* ==========================================
                MFA SCREEN
            =========================================== */}

            {issuccessfullylogin ===
            encryptedBoolean ? (

              <div>


                <Typography
                  sx={{
                    fontFamily:
                      '"Open Sans","Helvetica","Arial",sans-serif',

                    fontWeight:
                      600,

                    fontSize:
                      '1rem',

                    mb:
                      2
                  }}
                >

                  MFA Auth

                </Typography>


                {/* ======================================
                    QR CODE
                ======================================= */}

                {isscuessqr ===
                  encryptedBooleannum && (

                  <>

                    {showQRCode && (
                      <Box
                        sx={{
                          display:
                            'flex',

                          flexDirection:
                            'column',

                          alignItems:
                            'center',

                          mb:
                            2
                        }}
                      >

                        <QRCodeCanvas
                          value={
                            generateOTPURI()
                          }
                          size={128}
                        />


                        <Typography
                          sx={{
                            fontFamily:
                              '"Open Sans","Helvetica","Arial",sans-serif',

                            fontWeight:
                              600,

                            fontSize:
                              '10px',

                            mt:
                              1,

                            wordBreak:
                              'break-all'
                          }}
                        >

                          Secret Key:

                          <br />

                          {secretkey}

                        </Typography>

                      </Box>
                    )}

                  </>

                )}


                {/* ======================================
                    OTP INPUT
                ======================================= */}

                <TextField
                  fullWidth
                  type="text"
                  label="OTP"
                  placeholder="Enter OTP"
                  value={otp}
                  error={
                    Boolean(
                      errorMessage
                    )
                  }
                  helperText={
                    errorMessage
                  }
                  inputProps={{
                    inputMode:
                      'numeric',
                    maxLength:
                      6
                  }}
                  onChange={(e) => {

                    const value =
                      e.target.value.replace(
                        /\D/g,
                        ''
                      );

                    setOTP(
                      value
                    );

                  }}
                />


                {/* ======================================
                    VERIFY OTP
                ======================================= */}

                <Button
                  fullWidth
                  disabled={
                    !otp ||
                    otp.length < 6 ||
                    otploading
                  }
                  onClick={
                    verifyOTP
                  }
                  sx={{
                    mt:
                      3,
                    mb:
                      2
                  }}
                  variant="contained"
                >

                  {otploading ? (

                    <CircularProgress
                      size={24}
                      color="inherit"
                    />

                  ) : (

                    'Verify OTP'

                  )}

                </Button>


                {/* ======================================
                    BACK
                ======================================= */}

                <Button
                  onClick={
                    handlebacklogin
                  }
                  variant="contained"
                  color="secondary"
                >

                  Back to login

                </Button>


              </div>


            ) : (


              /* ========================================
                 LOGIN SCREEN
              ========================================= */

              <form
                onSubmit={
                  handleLogin
                }
              >


                <Typography
                  sx={{
                    fontFamily:
                      '"Open Sans","Helvetica","Arial",sans-serif',

                    fontWeight:
                      700,

                    fontSize:
                      '1rem',

                    mb:
                      1
                  }}
                >

                  Login

                </Typography>


                {/* ======================================
                    USERNAME
                ======================================= */}

                <TextField
                  margin="normal"
                  fullWidth
                  label="Username"
                  value={
                    account
                  }
                  autoComplete="username"
                  onChange={(e) =>
                    setAccount(
                      e.target.value
                    )
                  }
                />


                {/* ======================================
                    PASSWORD
                ======================================= */}

                <TextField
                  autoComplete="current-password"
                  margin="normal"
                  fullWidth
                  label="Password"
                  type={
                    showPassword
                      ? 'text'
                      : 'password'
                  }
                  value={
                    password
                  }
                  onChange={(e) =>
                    setPassword(
                      e.target.value
                    )
                  }

                  InputProps={{
                    endAdornment: (

                      <InputAdornment
                        position="end"
                      >

                        <IconButton
                          onClick={
                            handleTogglePassword
                          }
                          edge="end"
                        >

                          {showPassword ? (

                            <Visibility />

                          ) : (

                            <VisibilityOff />

                          )}

                        </IconButton>

                      </InputAdornment>

                    )
                  }}

                />


                {/* ======================================
                    ERROR
                ======================================= */}

                {errorMessage && (

                  <Typography
                    color="error"
                    sx={{
                      mt:
                        1,
                      fontSize:
                        '14px'
                    }}
                  >

                    {errorMessage}

                  </Typography>

                )}


                {/* ======================================
                    LOGIN BUTTON
                ======================================= */}

                <Button
                  type="submit"
                  fullWidth
                  variant="contained"
                  sx={{
                    mt:
                      3,
                    mb:
                      2
                  }}
                  disabled={
                    isLoading
                  }
                >

                  {isLoading ? (

                    <CircularProgress
                      size={24}
                      color="inherit"
                    />

                  ) : (

                    'Login'

                  )}

                </Button>


              </form>

            )}

          </Box>

        </Box>

      </Grid>


      {/* ==================================================
          FOOTER
      =================================================== */}

      <Grid
        item
        xs={12}
        sx={{
          display:
            'flex',

          alignItems:
            'center',

          justifyContent:
            'center',

          textAlign:
            'center',

          fontSize:
            '16px',

          pb:
            2
        }}
      >

        <div>

          <div
            style={{
              listStyleType:
                'none',
              color:
                '#000'
            }}
          >

            FXCN IT: Tel:533-26038 |
            Email:idpbg-it-tn-dev@mail.foxconn.com

          </div>


          <div>

            Copyright 2020-2025
            Foxconn A Business Group IT.
            All Rights Reserved

          </div>

        </div>

      </Grid>

    </Grid>

  );

};
