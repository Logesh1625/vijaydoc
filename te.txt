const handleGenerateSecret = async () => {
  const headers = {
    apiID: 'ffff-1694231787059-10208193125-0099',
    userKey: 'YOUR_EXISTING_USER_KEY',
  };

  try {
    setShowQRCode(false);
    setsecretkey('');
    setErrorMessage('');

    const response = await axios.post(
      'https://microservice.abgind.foxconn.com/india/gateway',
      {},
      { headers }
    );

    console.log(
      'Secret API HTTP status:',
      response.status
    );

    // API returns the secret directly
    const res = response.data;

    if (!res || typeof res.secret !== 'string' || !res.secret.trim()) {
      console.error(
        'Invalid secret API response structure'
      );

      throw new Error(
        'Secret field is missing from the API response'
      );
    }

    // Store the returned secret
    setsecretkey(res.secret);

    // Display QR code
    setShowQRCode(true);

    console.log('Secret generated successfully');

  } catch (error) {
    console.error(
      'Error generating secret:',
      error.response?.status,
      error.message
    );

    setErrorMessage(
      'Unable to generate MFA secret. Please try again or contact IT support.'
    );
  }
};
