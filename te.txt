const handleGenerateSecret = async () => {
  const headers = {
    apiID: 'ffff-1694231787059-10208193125-0099',
    userKey: 'YOUR_VALID_USER_KEY',
  };

  try {
    setErrorMessage('');

    const response = await axios.post(
      'https://microservice.abgind.foxconn.com/india/gateway',
      {},
      { headers }
    );

    console.log('HTTP Status:', response.status);
    console.log('API Response:', response.data);

    if (response.data?.r) {
      const decrypted = decrypt(
        response.data.r,
        getdk(iwxKey)
      );

      console.log('Decrypted API response:', decrypted);

      if (!decrypted) {
        throw new Error('Unable to decrypt API response');
      }

      const result = JSON.parse(decrypted);

      console.log(
        'Decrypted response fields:',
        Object.keys(result)
      );

      if (result.secret) {
        setsecretkey(result.secret);
        setShowQRCode(true);
      } else {
        throw new Error(
          'Secret field is missing. Check the API documentation.'
        );
      }
    } else {
      console.error(
        'Unexpected API response:',
        response.data
      );

      throw new Error(
        'Expected encrypted response field "r" was not returned.'
      );
    }
  } catch (error) {
    console.error(
      'Secret generation failed:',
      error.response?.data || error.message
    );

    setErrorMessage(
      'Unable to generate MFA secret. Please contact IT support.'
    );
  }
};
