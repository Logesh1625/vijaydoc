// ============================================================
// LOGIN COMPONENT
// ============================================================

const Login = () => {

  const navigate = useNavigate();

  const classes = useStyles(theme);


  // ==========================================================
  // STATE
  // ==========================================================

  const [isLoading, setIsLoading] =
    useState(false);

  const [loginStatus, setLoginStatus] =
    useState('');

  const [account, setAccount] =
    useState('');

  const [password, setPassword] =
    useState('');

  const [errorMessage, setErrorMessage] =
    useState('');

  const [showPassword, setShowPassword] =
    useState(false);

  const [secretkey, setSecretKey] =
    useState('');

  const [otp, setOTP] =
    useState('');

  const [showQRCode, setShowQRCode] =
    useState(false);

  const [otpLoading, setOtpLoading] =
    useState(false);

  const [secretLoading, setSecretLoading] =
    useState(false);


  // ==========================================================
  // ENCRYPTED FLAGS
  // ==========================================================

  const encryptedBoolean =
    encrypt(
      'true',
      getdk(iwxKey)
    );

  const encryptedBooleanNum =
    encrypt(
      '0',
      getdk(iwxKey)
    );


  // ==========================================================
  // SESSION VALUES
  // ==========================================================

  const issuccessfullylogin =
    sessionStorage.getItem(
      'issuccessfullylogin'
    );

  const isscuessqr =
    sessionStorage.getItem(
      'qrstatus'
    );


  // ==========================================================
  // PASSWORD VISIBILITY
  // ==========================================================

  const handleTogglePassword = () => {

    setShowPassword(
      previous => !previous
    );

  };


  // ==========================================================
  // ENTER KEY
  // ==========================================================

  const handleKeyPress = (event) => {

    if (event.key === 'Enter') {

      event.preventDefault();

      if (
        issuccessfullylogin ===
        encryptedBoolean
      ) {

        verifyOTP();

      } else {

        handleLogin(event);

      }

    }

  };


  // ==========================================================
  // FETCH USER DETAILS
  // ==========================================================

  const fetchdata = async () => {

    try {

      const headers = {

        apiID:
          USER_DETAILS_API_ID,

        userKey:
          USER_KEY

      };


      const params = {

        EmpId:
          account.trim(),

        EmpName:
          '',

        Status:
          '',

        StartPage:
          1,

        PageSize:
          999999,

        Func:
          'UserManagement-Get_user_details_new'

      };


      console.log(
        '======================================'
      );

      console.log(
        'USER DETAILS API'
      );

      console.log(
        'Employee:',
        account
      );

      console.log(
        'API ID:',
        USER_DETAILS_API_ID
      );

      console.log(
        'UserKey exists:',
        Boolean(USER_KEY)
      );

      console.log(
        'UserKey length:',
        USER_KEY.length
      );

      console.log(
        '======================================'
      );


      const response =
        await iwxPost(
          params,
          { headers },
          true
        );


      console.log(
        'USER DETAILS RESPONSE:',
        response
      );


      if (
        !response
      ) {

        console.error(
          'USER DETAILS: API returned null'
        );

        return null;

      }


      if (
        !Array.isArray(response.List)
      ) {

        console.error(
          'USER DETAILS: List missing',
          response
        );

        return null;

      }


      if (
        response.List.length === 0
      ) {

        console.log(
          'USER DETAILS: No employee found'
        );

        return response;

      }


      // ======================================================
      // QR STATUS
      // ======================================================

      const qrStatusValue =
        response.List[0]?.QrStatus;


      if (
        qrStatusValue !==
          undefined &&
        qrStatusValue !==
          null
      ) {

        const encryptedQRStatus =
          encrypt(
            String(qrStatusValue),
            getdk(iwxKey)
          );


        sessionStorage.setItem(
          'qrstatus',
          encryptedQRStatus
        );


        console.log(
          'QR STATUS:',
          qrStatusValue
        );

      }


      return response;

    } catch (error) {

      console.error(
        'fetchdata ERROR:',
        error
      );

      return null;

    }

  };


  // ==========================================================
  // EMPLOYEE BASIC INFORMATION
  // ==========================================================

  const fetchempName = async () => {

    try {

      const response =
        await fetchempbasicinfo(
          account.trim()
        );


      console.log(
        'EMPLOYEE BASIC INFO:',
        response
      );


      return response;

    } catch (error) {

      console.error(
        'EMPLOYEE BASIC INFO ERROR:',
        error
      );

      return null;

    }

  };


  // ==========================================================
  // FETCH ROLE
  // ==========================================================

  const fetchRoleData = async () => {

    try {

      const headers = {

        apiID:
          ROLE_API_ID,

        userKey:
          USER_KEY

      };


      const params = {

        EmpId:
          account.trim(),

        Func:
          'UserManagement-Get_role_based_empid_new'

      };


      console.log(
        'ROLE API CALL'
      );


      const response =
        await iwxPost(
          params,
          { headers },
          true
        );


      console.log(
        'ROLE RESPONSE:',
        response
      );


      return response;

    } catch (error) {

      console.error(
        'ROLE API ERROR:',
        error
      );

      return null;

    }

  };


  // ==========================================================
  // STORE LOGIN LOG
  // ==========================================================

  const StoreLoginLog = async (
    msg,
    empname,
    roles
  ) => {

    try {

      const headers = {

        apiID:
          LOGIN_LOG_INSERT_API_ID,

        userKey:
          USER_KEY

      };


      const employeeName =
        empname &&
        empname[0]
          ? empname[0].EmployeeName
          : '';


      const roleName =
        roles &&
        roles.List &&
        roles.List[0]
          ? roles.List[0].RoleName
          : '';


      const params = {

        EmpId:
          account.trim(),

        SystemIp:
          '',

        LoginStatus:
          msg,

        EmpName:
          employeeName,

        UserRole:
          roleName,

        Func:
          'UserManagement-Insert_login_log_new'

      };


      console.log(
        'LOGIN LOG:',
        params
      );


      const response =
        await iwxPost(
          params,
          { headers },
          true
        );


      console.log(
        'LOGIN LOG RESPONSE:',
        response
      );


      if (
        !response ||
        !Array.isArray(response.List)
      ) {

        return undefined;

      }


      return response.List[0]?.Id;

    } catch (error) {

      console.error(
        'StoreLoginLog ERROR:',
        error
      );

      return undefined;

    }

  };


  // ==========================================================
  // UPDATE SECRET KEY
  // ==========================================================

  const updateQRStatus = async () => {

    try {

      const encryptedAccount =
        sessionStorage.getItem(
          'account'
        );


      if (!encryptedAccount) {

        console.error(
          'Account not found'
        );

        return;

      }


      const decryptedAccount =
        decrypt(
          encryptedAccount,
          getdk(iwxKey)
        );


      if (!decryptedAccount) {

        return;

      }


      const empnumber =
        decryptedAccount.replace(
          /"/g,
          ''
        );


      const headers = {

        apiID:
          UPDATE_SECRET_API_ID,

        userKey:
          USER_KEY

      };


      const params = {

        EmpId:
          empnumber,

        SecretKey:
          secretkey,

        Func:
          'UserManagement-Update_secret_key_new'

      };


      if (
        isscuessqr ===
        encryptedBooleanNum
      ) {

        console.log(
          'Updating QR secret key'
        );


        const response =
          await iwxPost(
            params,
            { headers },
            true
          );


        console.log(
          'UPDATE SECRET RESPONSE:',
          response
        );

      }

    } catch (error) {

      console.error(
        'updateQRStatus ERROR:',
        error
      );

    }

  };


  // ==========================================================
  // GENERATE SECRET
  // ==========================================================

  const handleGenerateSecret = async () => {

    if (secretLoading) {

      return;

    }


    try {

      setSecretLoading(true);

      setErrorMessage('');


      const headers = {

        apiID:
          SECRET_GENERATION_API_ID,

        userKey:
          SECRET_GEN_USER_KEY

      };


      console.log(
        '======================================'
      );

      console.log(
        'SECRET GENERATION'
      );

      console.log(
        'URL:',
        SECRET_GEN_URL
      );

      console.log(
        'API ID:',
        SECRET_GENERATION_API_ID
      );

      console.log(
        '======================================'
      );


      const response =
        await axios.post(
          SECRET_GEN_URL,
          {},
          {
            headers,
            timeout: 30000
          }
        );


      console.log(
        'SECRET RAW RESPONSE:',
        response.data
      );


      if (
        !response.data
      ) {

        setErrorMessage(
          'Authentication server returned empty response.'
        );

        return;

      }


      let result = null;


      if (
        response.data.r
      ) {

        const decrypted =
          decrypt(
            response.data.r,
            getdk(iwxKey)
          );


        console.log(
          'SECRET DECRYPTED:',
          decrypted
        );


        if (!decrypted) {

          setErrorMessage(
            'Unable to decrypt authentication response.'
          );

          return;

        }


        result =
          JSON.parse(
            decrypted
          );

      } else if (
        typeof response.data ===
        'object'
      ) {

        result =
          response.data;

      }


      console.log(
        'SECRET RESULT:',
        result
      );


      if (
        result &&
        result.secret
      ) {

        setSecretKey(
          result.secret
        );

        setShowQRCode(
          true
        );

        setErrorMessage('');

      } else {

        console.error(
          'Secret not found:',
          result
        );

        setErrorMessage(
          result?.Msg ||
          'Unable to generate authentication secret.'
        );

      }

    } catch (error) {

      console.error(
        'SECRET GENERATION ERROR:',
        error
      );


      if (
        error.response
      ) {

        console.error(
          'SECRET STATUS:',
          error.response.status
        );

        console.error(
          'SECRET DATA:',
          error.response.data
        );

      }


      setErrorMessage(
        'Authentication server error.'
      );

    } finally {

      setSecretLoading(false);

    }

  };


  // ==========================================================
  // GENERATE OTP URI
  // ==========================================================

  const generateOTPURI = () => {

    const encryptedAccount =
      sessionStorage.getItem(
        'account'
      );


    if (!encryptedAccount) {

      return '';

    }


    const decryptedAccount =
      decrypt(
        encryptedAccount,
        getdk(iwxKey)
      );


    if (!decryptedAccount) {

      return '';

    }


    const empnumber =
      decryptedAccount.replace(
        /"/g,
        ''
      );


    return (
      `otpauth://totp/DMS:${encodeURIComponent(
        empnumber
      )}?secret=${secretkey}`
    );

  };


  // ==========================================================
  // STORE OTP LOGIN LOG
  // ==========================================================

  const StoreOTPtologinlog = async (
    msg
  ) => {

    try {

      const encryptedLoginLogID =
        sessionStorage.getItem(
          'loginlogID'
        );


      if (!encryptedLoginLogID) {

        console.error(
          'Login log ID not found'
        );

        return;

      }


      const decryptedLoginLogID =
        decrypt(
          encryptedLoginLogID,
          getdk(iwxKey)
        );


      if (!decryptedLoginLogID) {

        return;

      }


      const headers = {

        apiID:
          UPDATE_LOGIN_LOG_API_ID,

        userKey:
          USER_KEY

      };


      const params = {

        Update:
          'OTP',

        Id:
          decryptedLoginLogID,

        Otp:
          otp,

        OtpStatus:
          msg,

        Func:
          'UserManagement-Update_login_log_new'

      };


      await iwxPost(
        params,
        { headers },
        true
      );


    } catch (error) {

      console.error(
        'StoreOTPtologinlog ERROR:',
        error
      );

    }

  };


  // ==========================================================
  // VERIFY OTP
  // ==========================================================

  const verifyOTP = async () => {

    if (!otp) {

      setErrorMessage(
        'Please enter OTP.'
      );

      return;

    }


    if (
      otp.length !== 6
    ) {

      setErrorMessage(
        'Please enter a valid 6-digit OTP.'
      );

      return;

    }


    setOtpLoading(true);

    setErrorMessage('');


    try {

      const encryptedAccount =
        sessionStorage.getItem(
          'account'
        );


      if (!encryptedAccount) {

        setErrorMessage(
          'Account information not found.'
        );

        return;

      }


      const decryptedAccount =
        decrypt(
          encryptedAccount,
          getdk(iwxKey)
        );


      if (!decryptedAccount) {

        setErrorMessage(
          'Invalid account information.'
        );

        return;

      }


      const empnumber =
        decryptedAccount.replace(
          /"/g,
          ''
        );


      const headers = {

        apiID:
          USER_DETAILS_API_ID,

        userKey:
          USER_KEY

      };


      const params = {

        EmpId:
          empnumber,

        EmpName:
          '',

        Status:
          '',

        StartPage:
          1,

        PageSize:
          999999,

        Func:
          'UserManagement-Get_user_details_new'

      };


      const response =
        await iwxPost(
          params,
          { headers },
          true
        );


      console.log(
        'OTP USER DETAILS:',
        response
      );


      if (
        !response ||
        !Array.isArray(response.List) ||
        response.List.length === 0
      ) {

        setErrorMessage(
          'Unable to get OTP information.'
        );

        return;

      }


      const storedSecret =
        response.List[0]?.SecretKey;


      if (
        !storedSecret &&
        !secretkey
      ) {

        setErrorMessage(
          'Authentication secret not found.'
        );

        return;

      }


      await verifyOTP22(
        storedSecret
      );


    } catch (error) {

      console.error(
        'VERIFY OTP ERROR:',
        error
      );


      setErrorMessage(
        'Unable to verify OTP. Please try again.'
      );

    } finally {

      setOtpLoading(false);

    }

  };


  // ==========================================================
  // VERIFY OTP AUTHENTICATOR API
  // ==========================================================

  const verifyOTP22 = async (
    storedSecret
  ) => {

    try {

      const secretToVerify =
        isscuessqr ===
        encryptedBooleanNum
          ? secretkey
          : storedSecret;


      if (!secretToVerify) {

        setErrorMessage(
          'Secret key is missing.'
        );

        return;

      }


      const headers = {

        apiID:
          OTP_VERIFICATION_API_ID,

        userKey:
          OTP_VERIFY_USER_KEY

      };


      const params =
        dealPostParamsEncrypt({

          secret:
            secretToVerify,

          otp:
            otp

        });


      console.log(
        '======================================'
      );

      console.log(
        'OTP VERIFICATION'
      );

      console.log(
        'URL:',
        OTP_VERIFY_URL
      );

      console.log(
        'API ID:',
        OTP_VERIFICATION_API_ID
      );

      console.log(
        '======================================'
      );


      const response =
        await axios.post(
          OTP_VERIFY_URL,
          params,
          {
            headers,
            timeout: 30000
          }
        );


      console.log(
        'OTP RAW RESPONSE:',
        response.data
      );


      if (
        !response.data ||
        !response.data.r
      ) {

        setErrorMessage(
          'Unable to verify OTP. Please try again.'
        );

        return;

      }


      const decrypted =
        decrypt(
          response.data.r,
          getdk(iwxKey)
        );


      console.log(
        'OTP DECRYPTED:',
        decrypted
      );


      if (!decrypted) {

        setErrorMessage(
          'Unable to decrypt OTP response.'
        );

        return;

      }


      const result =
        JSON.parse(
          decrypted
        );


      console.log(
        'OTP RESULT:',
        result
      );


      // ======================================================
      // OTP SUCCESS
      // ======================================================

      if (
        result.IsOK === 1 ||
        result.IsOK === '1'
      ) {

        await updateQRStatus();


        sessionStorage.setItem(
          'isLoggedIn',
          encryptedBoolean
        );


        const loginResponse =
          sessionStorage.getItem(
            'loginresponse'
          );


        if (loginResponse) {

          sessionStorage.setItem(
            'data',
            loginResponse
          );

        }


        await StoreOTPtologinlog(
          result.Msg ||
          'OTP Success'
        );


        setOTP('');


        navigate(
          '/home/Dashboard',
          {
            state: {
              account:
                account
            }
          }
        );


      } else {

        const message =
          result.Msg ||
          'Incorrect OTP';


        setErrorMessage(
          message
        );


        await StoreOTPtologinlog(
          message
        );

      }


    } catch (error) {

      console.error(
        'OTP VERIFY ERROR:',
        error
      );


      if (
        error.response
      ) {

        console.error(
          'OTP STATUS:',
          error.response.status
        );

        console.error(
          'OTP RESPONSE:',
          error.response.data
        );

      }


      setErrorMessage(
        'OTP verification failed.'
      );

    }

  };


  // ==========================================================
  // LOGIN
  // ==========================================================

  const handleLogin = async (
    event
  ) => {

    event.preventDefault();


    if (isLoading) {

      return;

    }


    setIsLoading(true);

    setErrorMessage('');


    try {

      // ======================================================
      // VALIDATION
      // ======================================================

      const enteredAccount =
        account.trim();


      if (
        !enteredAccount ||
        !password
      ) {

        setErrorMessage(
          'Fill in all fields.'
        );

        return;

      }


      // ======================================================
      // GET USER DETAILS
      // ======================================================

      const userResponse =
        await fetchdata();


      if (
        !userResponse
      ) {

        setErrorMessage(
          'Error checking account availability.'
        );

        return;

      }


      if (
        !Array.isArray(
          userResponse.List
        )
      ) {

        setErrorMessage(
          'Invalid user details response.'
        );

        return;

      }


      const userList =
        userResponse.List;


      // ======================================================
      // FIND EMPLOYEE
      // ======================================================

      const findempdata =
        userList.filter(
          user =>
            String(
              user.EmpId
            )
              .trim()
              .toUpperCase() ===
            enteredAccount
              .toUpperCase()
        );


      console.log(
        'ENTERED ACCOUNT:',
        enteredAccount
      );

      console.log(
        'USER LIST:',
        userList
      );

      console.log(
        'MATCHED EMPLOYEE:',
        findempdata
      );


      // ======================================================
      // EMPLOYEE NOT FOUND
      // ======================================================

      if (
        findempdata.length === 0
      ) {

        const empName =
          await fetchempName();

        const roles =
          await fetchRoleData();


        const msg =
          "You don't have access to this system";


        setErrorMessage(
          msg
        );


        await StoreLoginLog(
          msg,
          empName,
          roles
        );


        return;

      }


      // ======================================================
      // DELETED USER
      // ======================================================

      if (
        String(
          findempdata[0]?.DelFlag
        ) === '1'
      ) {

        const empName =
          await fetchempName();

        const roles =
          await fetchRoleData();


        const msg =
          "You don't have access to this system";


        setErrorMessage(
          msg
        );


        await StoreLoginLog(
          msg,
          empName,
          roles
        );


        return;

      }


      // ======================================================
      // SAVE ACCOUNT
      // ======================================================

      const encryptedAccount =
        dealPostParamsEncrypt(
          enteredAccount
        );


      sessionStorage.setItem(
        'account',
        encryptedAccount.s
      );


      // ======================================================
      // EMPLOYEE NAME
      // ======================================================

      const empName =
        await fetchempName();


      // ======================================================
      // ROLE
      // ======================================================

      const roles =
        await fetchRoleData();


      // ======================================================
      // LOGIN API
      // ======================================================

      const encryptedParams =
        dealPostParamsEncrypt({

          Func:
            'APPFrame-AccountLoginPhoneSpv3-IWX',

          Account:
            enteredAccount,

          Password:
            password,

          OsType:
            '1',

          DeviceId:
            '1',

          OsVer:
            '1',

          AppVer:
            '1',

          DeviceType:
            '1',

          AppVersion:
            '1'

        });


      console.log(
        'LOGIN API START'
      );


      const loginResponse =
        await getlogincredentials(
          encryptedParams
        );


      console.log(
        'LOGIN RESPONSE:',
        loginResponse
      );


      if (
        !loginResponse ||
        !loginResponse.r
      ) {

        const msg =
          'Login service did not return a valid response.';


        setErrorMessage(
          msg
        );


        await StoreLoginLog(
          msg,
          empName,
          roles
        );


        return;

      }


      // ======================================================
      // DECRYPT LOGIN
      // ======================================================

      const decryptedLogin =
        decrypt(
          loginResponse.r,
          getdk(iwxKey)
        );


      console.log(
        'DECRYPTED LOGIN:',
        decryptedLogin
      );


      if (!decryptedLogin) {

        setErrorMessage(
          'Unable to decrypt login response.'
        );

        return;

      }


      const loginResult =
        JSON.parse(
          decryptedLogin
        );


      console.log(
        'LOGIN RESULT:',
        loginResult
      );


      // ======================================================
      // LOGIN SUCCESS
      // ======================================================

      if (
        loginResult.IsOK === '1' ||
        loginResult.IsOK === 1
      ) {

        const loginLogID =
          await StoreLoginLog(
            loginResult.Msg,
            empName,
            roles
          );


        if (
          loginLogID ===
            undefined ||
          loginLogID === null
        ) {

          setErrorMessage(
            'Unable to record the login. Please try again.'
          );

          setLoginStatus(
            'error'
          );

          return;

        }


        // ====================================================
        // SAVE LOGIN LOG ID
        // ====================================================

        const encryptedLoginLogID =
          encrypt(
            String(loginLogID),
            getdk(iwxKey)
          );


        sessionStorage.setItem(
          'loginlogID',
          encryptedLoginLogID
        );


        // ====================================================
        // SAVE LOGIN RESPONSE
        // ====================================================

        sessionStorage.setItem(
          'loginresponse',
          loginResponse.r
        );


        // ====================================================
        // LOGIN SUCCESS FLAG
        // ====================================================

        sessionStorage.setItem(
          'issuccessfullylogin',
          encryptedBoolean
        );


        setLoginStatus(
          'success'
        );


        setErrorMessage('');


      } else {

        // ====================================================
        // LOGIN FAILED
        // ====================================================

        const msg =
          loginResult.Msg ||
          'Invalid username or password.';


        await StoreLoginLog(
          msg,
          empName,
          roles
        );


        setErrorMessage(
          msg
        );


        setLoginStatus(
          'error'
        );

      }


    } catch (error) {

      console.error(
        'LOGIN ERROR:',
        error
      );


      if (
        error.response
      ) {

        console.error(
          'STATUS:',
          error.response.status
        );

        console.error(
          'RESPONSE:',
          error.response.data
        );

      }


      setErrorMessage(
        'Error checking account availability.'
      );


    } finally {

      setIsLoading(false);

    }

  };


  // ==========================================================
  // BACK TO LOGIN
  // ==========================================================

  const handlebacklogin = () => {

    sessionStorage.removeItem(
      'issuccessfullylogin'
    );

    sessionStorage.removeItem(
      'qrstatus'
    );

    sessionStorage.removeItem(
      'loginresponse'
    );

    sessionStorage.removeItem(
      'loginlogID'
    );

    sessionStorage.removeItem(
      'account'
    );


    setAccount('');

    setPassword('');

    setOTP('');

    setSecretKey('');

    setShowQRCode(false);

    setLoginStatus('');

    setErrorMessage('');


    navigate('/login');

  };


  // ==========================================================
  // CLEAR ERROR
  // ==========================================================

  useEffect(() => {

    setErrorMessage('');

  }, [
    account,
    password
  ]);


  // ==========================================================
  // GENERATE QR SECRET
  // ==========================================================

  useEffect(() => {

    const currentLogin =
      sessionStorage.getItem(
        'issuccessfullylogin'
      );


    if (
      currentLogin !==
      encryptedBoolean
    ) {

      return;

    }


    if (
      isscuessqr ===
      encryptedBooleanNum
    ) {

      handleGenerateSecret();

    }

  }, [
    loginStatus
  ]);


  // ==========================================================
  // UI
  // ==========================================================

  return (

    <Grid
      container
      component="main"
      sx={{

        minHeight:
          '100vh',

        backgroundImage:
          `url(${loginimage})`,

        backgroundRepeat:
          'no-repeat',

        backgroundSize:
          'cover',

        backgroundPosition:
          'center',

        position:
          'relative',

        '&::before': {

          content:
            '""',

          position:
            'absolute',

          inset:
            0,

          backgroundColor:
            'rgba(255,255,255,0.15)',

          zIndex:
            0

        }

      }}
    >


      {/* =====================================================
          LOGIN / MFA AREA
      ===================================================== */}

      <Grid
        item
        xs={12}
        sx={{

          display:
            'flex',

          alignItems:
            'center',

          justifyContent:
            'center',

          minHeight:
            'calc(100vh - 80px)',

          position:
            'relative',

          zIndex:
            1

        }}
      >

        <Box
          className={
            classes.root
          }

          onKeyDown={
            handleKeyPress
          }
        >

          <Box
            className={
              classes.formm
            }
          >


            {/* =================================================
                LOGO
            ================================================= */}

            <img
              src="/images/logo1.png"
              alt="DMS"
              style={{

                maxWidth:
                  '180px',

                maxHeight:
                  '70px',

                objectFit:
                  'contain',

                marginBottom:
                  '15px'

              }}
            />


            {/* =================================================
                MFA SCREEN
            ================================================= */}

            {issuccessfullylogin ===
            encryptedBoolean ? (

              <Box>

                <Typography
                  sx={{

                    fontFamily:
                      '"Open Sans","Helvetica","Arial",sans-serif',

                    fontWeight:
                      600,

                    fontSize:
                      '1rem',

                    mb:
                      2

                  }}
                >
                  MFA Auth
                </Typography>


                {/* ============================================
                    QR CODE
                ============================================ */}

                {isscuessqr ===
                  encryptedBooleanNum && (

                  <Box
                    sx={{

                      display:
                        'flex',

                      flexDirection:
                        'column',

                      alignItems:
                        'center',

                      mb:
                        2

                    }}
                  >

                    {secretLoading ? (

                      <CircularProgress
                        size={30}
                      />

                    ) : (

                      showQRCode &&
                      secretkey && (

                        <>

                          <QRCodeCanvas
                            value={
                              generateOTPURI()
                            }
                            size={128}
                          />


                          <Typography
                            sx={{

                              fontSize:
                                '10px',

                              fontWeight:
                                600,

                              mt:
                                1,

                              wordBreak:
                                'break-all'

                            }}
                          >

                            Secret Key:
                            <br />

                            {secretkey}

                          </Typography>

                        </>

                      )

                    )}

                  </Box>

                )}


                {/* ============================================
                    OTP
                ============================================ */}

                <TextField
                  fullWidth
                  type="text"
                  label="OTP"
                  placeholder="Enter OTP"
                  value={otp}
                  error={
                    Boolean(
                      errorMessage
                    )
                  }
                  helperText={
                    errorMessage
                  }
                  onChange={(
                    event
                  ) => {

                    const value =
                      event.target.value
                        .replace(
                          /\D/g,
                          ''
                        )
                        .slice(
                          0,
                          6
                        );


                    setOTP(
                      value
                    );

                  }}
                  inputProps={{

                    maxLength:
                      6,

                    inputMode:
                      'numeric'

                  }}
                />


                {/* ============================================
                    VERIFY BUTTON
                ============================================ */}

                <Button
                  fullWidth
                  disabled={
                    !otp ||
                    otp.length !== 6 ||
                    otpLoading
                  }
                  onClick={
                    verifyOTP
                  }
                  sx={{

                    mt:
                      3,

                    mb:
                      2

                  }}
                  variant="contained"
                >

                  {otpLoading ? (

                    <CircularProgress
                      size={24}
                      sx={{
                        color:
                          '#fff'
                      }}
                    />

                  ) : (

                    'Verify OTP'

                  )}

                </Button>


                {/* ============================================
                    BACK BUTTON
                ============================================ */}

                <Button
                  onClick={
                    handlebacklogin
                  }
                  variant="outlined"
                  fullWidth
                >
                  Back to login
                </Button>

              </Box>

            ) : (


              /* =================================================
                 LOGIN SCREEN
              ================================================= */

              <form
                onSubmit={
                  handleLogin
                }
              >

                <Typography
                  sx={{

                    fontFamily:
                      '"Open Sans","Helvetica","Arial",sans-serif',

                    fontWeight:
                      700,

                    fontSize:
                      '1rem',

                    mb:
                      1

                  }}
                >
                  Login
                </Typography>


                {/* ============================================
                    USERNAME
                ============================================ */}

                <TextField
                  margin="normal"
                  fullWidth
                  label="Username"
                  value={
                    account
                  }
                  autoComplete="username"
                  onChange={(
                    event
                  ) =>
                    setAccount(
                      event.target.value
                    )
                  }
                />


                {/* ============================================
                    PASSWORD
                ============================================ */}

                <TextField
                  autoComplete="current-password"
                  margin="normal"
                  fullWidth
                  label="Password"
                  type={
                    showPassword
                      ? 'text'
                      : 'password'
                  }
                  value={
                    password
                  }
                  onChange={(
                    event
                  ) =>
                    setPassword(
                      event.target.value
                    )
                  }
                  InputProps={{

                    endAdornment: (

                      <InputAdornment
                        position="end"
                      >

                        <IconButton
                          onClick={
                            handleTogglePassword
                          }
                          edge="end"
                        >

                          {showPassword ? (

                            <Visibility />

                          ) : (

                            <VisibilityOff />

                          )}

                        </IconButton>

                      </InputAdornment>

                    )

                  }}
                />


                {/* ============================================
                    ERROR
                ============================================ */}

                {errorMessage && (

                  <Typography
                    color="error"
                    sx={{

                      mt:
                        1,

                      fontSize:
                        '14px'

                    }}
                  >
                    {errorMessage}
                  </Typography>

                )}


                {/* ============================================
                    LOGIN BUTTON
                ============================================ */}

                <Button
                  type="submit"
                  fullWidth
                  variant="contained"
                  sx={{

                    mt:
                      3,

                    mb:
                      2

                  }}
                  disabled={
                    isLoading
                  }
                >

                  {isLoading ? (

                    <CircularProgress
                      size={24}
                      sx={{
                        color:
                          '#fff'
                      }}
                    />

                  ) : (

                    'LOGIN'

                  )}

                </Button>

              </form>

            )}

          </Box>

        </Box>

      </Grid>


      {/* =====================================================
          FOOTER
      ===================================================== */}

      <Grid
        item
        xs={12}
        sx={{

          display:
            'flex',

          alignItems:
            'center',

          justifyContent:
            'center',

          py:
            1,

          position:
            'relative',

          zIndex:
            1,

          backgroundColor:
            'rgba(255,255,255,0.75)'

        }}
      >

        <Box
          sx={{

            textAlign:
              'center',

            fontSize:
              '14px',

            color:
              '#000'

          }}
        >

          <div>
            FXCN IT: Tel:533-26038 |
            Email:
            idpbg-it-tn-dev@mail.foxconn.com
          </div>

          <div>
            Copyright 2020-2025
            Foxconn A Business Group IT.
            All Rights Reserved
          </div>

        </Box>

      </Grid>

    </Grid>

  );

};


export default Login;
