const handleGenerateSecret = async () => {
  const headers = {
    apiID: 'ffff-1694231787059-10208193125-0099',
    userKey: 'YOUR_EXISTING_USER_KEY',
  };

  try {
    setShowQRCode(false);
    setsecretkey('');
    setErrorMessage('');

    const response = await axios.post(
      'https://microservice.abgind.foxconn.com/india/gateway',
      {},
      { headers }
    );

    // Debug: inspect response structure without exposing values
    console.log('Secret API HTTP status:', response.status);

    console.log(
      'Secret API response keys:',
      Object.keys(response.data || {})
    );

    console.log(
      'Secret API response data type:',
      typeof response.data
    );

    // Check response
    if (!response?.data) {
      throw new Error('API returned an empty response');
    }

    // Expected encrypted response
    const encryptedResponse = response.data.r;

    if (
      typeof encryptedResponse !== 'string' ||
      !encryptedResponse.trim()
    ) {
      console.error(
        'Secret API response does not contain field r'
      );

      throw new Error(
        'Secret API response format is incorrect. Check the Network Response.'
      );
    }

    // Decrypt response
    const decryptedResponse = decrypt(
      encryptedResponse,
      getdk(iwxKey)
    );

    if (!decryptedResponse) {
      throw new Error(
        'Unable to decrypt the secret response'
      );
    }

    // Parse JSON
    const result = JSON.parse(decryptedResponse);

    if (!result?.secret) {
      throw new Error(
        'The decrypted response does not contain a secret'
      );
    }

    setsecretkey(result.secret);
    setShowQRCode(true);

  } catch (error) {
    console.error(
      'Secret generation failed:',
      error.message
    );

    setErrorMessage(
      'Unable to generate MFA secret. Please contact IT support.'
    );
  }
};
