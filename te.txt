import { Component, OnInit } from "@angular/core";
import { NzMessageService, NzDrawerRef } from "ng-zorro-antd";
import { I18NService } from "@core";
import { PublicNetService } from "src/app/routes/a_Public_net/Public_net.service";

@Component({
  selector: "app-F1-Details-add",
  templateUrl: "./F1-Details-add.html",
})
export class AddDetailsDrawComponent implements OnInit {

  constructor(
    private publicSvr: PublicNetService,
    private msgSvr: NzMessageService,
    private i18n: I18NService,
    private ref: NzDrawerRef
  ) { }

  // ==========================================
  // FORM FIELDS
  // ==========================================

  groups: any = "";

  mealPreference: any = null;
  fxBadge: any = "";
  name: any = "";
  remark: string = "";

  firstDateFxbl: Date | null = null;
  lastDateFxbl: Date | null = null;

  option: any;
  editData: any;
  timer: any;

  mealPreferenceList: any[] = [];

  // ==========================================
  // DATE HANDLERS
  // ==========================================

  onFirstDateFxblChange(date: Date): void {
    this.firstDateFxbl = date;

    // If Last Date is earlier than First Date, clear it
    if (this.lastDateFxbl && this.firstDateFxbl && this.lastDateFxbl < this.firstDateFxbl) {
      this.lastDateFxbl = null;
    }
  }

  disabledLastDateFxbl = (current: Date): boolean => {
    if (!this.firstDateFxbl) {
      return false;
    }
    return current < this.startOfDay(this.firstDateFxbl);
  };

  private startOfDay(date: Date): Date {
    const d = new Date(date);
    d.setHours(0, 0, 0, 0);
    return d;
  }

  // ==========================================
  // INIT
  // ==========================================

  ngOnInit(): void {

    // LOAD EDIT DATA
    if (this.editData) {

      this.groups = this.editData.Groups;

      this.fxBadge = this.editData.FxBadge;
      this.name = this.editData.Name;
      this.remark = this.editData.Remark || "";

      this.firstDateFxbl = this.parseDate(this.editData.FirstDateFxbl);
      this.lastDateFxbl = this.parseDate(this.editData.LastDateFxbl);

      console.log("EDIT DATA:", JSON.stringify(this.editData));
    }

    // LOAD MEAL PREFERENCE MASTER
    this.publicSvr
      .getGuardPassCode("GuardPass-Meal Preference")
      .subscribe((res: any) => {

        this.mealPreferenceList = res || [];

        console.log("MEAL PREFERENCE LIST:", JSON.stringify(this.mealPreferenceList));

        // EDIT MODE: convert NAME -> ID
        if (this.option === "Edit" && this.editData) {

          const selectedMeal = this.mealPreferenceList.find(
            (item: any) =>
              String(item.label).trim().toUpperCase() ===
              String(this.editData.MealPreference).trim().toUpperCase()
          );

          if (selectedMeal) {
            this.mealPreference = Number(selectedMeal.value);
          } else {
            this.mealPreference = null;
            console.warn("Meal Preference not found:", this.editData.MealPreference);
          }

          console.log("FINAL MEAL PREFERENCE ID:", this.mealPreference);
        }
      });
  }

  // ==========================================
  // DATE PARSER
  // ==========================================

  parseDate(value: any): Date | null {

    if (!value) {
      return null;
    }

    if (value instanceof Date) {
      return value;
    }

    if (typeof value === "string") {

      const parts = value.split(/[-\/]/);

      if (parts.length === 3) {

        const day = parseInt(parts[0], 10);
        const month = parseInt(parts[1], 10) - 1;
        const year = parseInt(parts[2], 10);

        const parsed = new Date(year, month, day);

        if (!isNaN(parsed.getTime())) {
          return parsed;
        }
      }
    }

    return null;
  }

  // ==========================================
  // ADD
  // ==========================================

  add(): void {

    const params = {

      Groups: this.groups,

      // ID is sent
      MealPreference: this.mealPreference,

      FxBadge: this.fxBadge,

      Name: this.name,

      Remark: this.remark,

      FirstDateFxbl: this.firstDateFxbl
        ? this.publicSvr.dateFormat(this.firstDateFxbl, "DD-MM-YYYY")
        : "",

      LastDateFxbl: this.lastDateFxbl
        ? this.publicSvr.dateFormat(this.lastDateFxbl, "DD-MM-YYYY")
        : "",

      AddBy: this.publicSvr.getSettingUserInfo().eeid,

      Func: "CMS-InsertF1Menu"
    };

    console.log("DEBUG - Add params:", JSON.stringify(params));

    this.publicSvr
      .iwxPostF1(
        params,
        {
          headers: {
            apiID: "ffff-1789631992006-10208193125-1291",
            userKey: "59EC4918B4383A80B95F3612AE1E5FE1"
          }
        },
        true
      )
      .then((res: any) => {

        console.log("DEBUG - Add server response:", res);

        if (res && res.IsOK == "1") {
          this.msgSvr.success("Success");
          this.ref.close(true);
        } else {
          this.msgSvr.error(res.Msg || "Failed to add.");
        }
      });
  }

  // ==========================================
  // EDIT
  // ==========================================

  edit(): void {

    const params = {

      Id: this.editData.Id,

      Groups: this.groups,

      // ID is sent
      MealPreference: this.mealPreference,

      FxBadge: this.fxBadge,

      Name: this.name,

      Remark: this.remark,

      FirstDateFxbl: this.firstDateFxbl
        ? this.publicSvr.dateFormat(this.firstDateFxbl, "DD-MM-YYYY")
        : "",

      LastDateFxbl: this.lastDateFxbl
        ? this.publicSvr.dateFormat(this.lastDateFxbl, "DD-MM-YYYY")
        : "",

      EditBy: this.publicSvr.getSettingUserInfo().eeid,

      Func: "CMS-UpdateF1MenuManage"
    };

    console.log("DEBUG - Edit params being sent:", JSON.stringify(params));

    this.publicSvr
      .iwxPostF1(
        params,
        {
          headers: {
            apiID: "ffff-1789634002932-10208193125-1317",
            userKey: "59EC4918B4383A80B95F3612AE1E5FE1"
          }
        },
        true
      )
      .then((res: any) => {

        console.log("DEBUG - Edit server response:", res);

        if (res && res.IsOK == "1") {
          this.msgSvr.success("Success");
          this.ref.close(true);
        } else {
          this.msgSvr.error(res.Msg || "Failed to update.");
        }
      });
  }

  // ==========================================
  // CANCEL
  // ==========================================

  cancel(): void {
    this.ref.close();
  }

}
