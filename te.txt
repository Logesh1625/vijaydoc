PROCEDURE BULK_F1_MENU(
    V_DATA              IN CLOB,
    V_ADD_BY            IN VARCHAR2,
    V_RES               OUT VARCHAR2,
    V_RESULT            OUT VARCHAR2
)
IS

    L_OBJECT        VARCHAR2(32767);

    L_GROUPS        VARCHAR2(100);
    L_MEAL_PREF     VARCHAR2(100);
    L_FX_BADGE      VARCHAR2(100);
    L_NAME          VARCHAR2(200);
    L_FIRST_DATE    VARCHAR2(30);
    L_LAST_DATE     VARCHAR2(30);

    L_COUNT         NUMBER := 0;
    L_INSERT_COUNT  NUMBER := 0;
    L_SKIP_COUNT    NUMBER := 0;

    /* =====================================================
       GET SIMPLE JSON STRING VALUE

       Example JSON:
       {
          "GROUPS":"A",
          "MEAL_PREFERENCE":"Indian Veg",
          "FX_BADGE":"12345"
       }

       This parser is intended for the simple JSON generated
       by the Angular Excel upload.
       ===================================================== */
    FUNCTION GET_JSON_VALUE(
        P_OBJECT IN VARCHAR2,
        P_KEY    IN VARCHAR2
    )
    RETURN VARCHAR2
    IS
        L_VALUE VARCHAR2(4000);
    BEGIN

        L_VALUE :=
            REGEXP_SUBSTR(
                P_OBJECT,
                '"' ||
                P_KEY ||
                '"[[:space:]]*:[[:space:]]*"([^"]*)"',
                1,
                1,
                NULL,
                1
            );

        RETURN L_VALUE;

    EXCEPTION
        WHEN OTHERS THEN
            RETURN NULL;
    END GET_JSON_VALUE;


BEGIN

    /* =====================================================
       VALIDATE INPUT
       ===================================================== */

    IF V_DATA IS NULL
       OR DBMS_LOB.GETLENGTH(V_DATA) = 0
    THEN

        V_RES := '0';
        V_RESULT := 'Bulk data is empty';

        RETURN;

    END IF;


    /* =====================================================
       LOOP THROUGH JSON OBJECTS

       Expected format:

       [
         {
           "GROUPS":"...",
           "MEAL_PREFERENCE":"...",
           "FX_BADGE":"...",
           "NAME":"...",
           "FIRST_DATE_FXBL":"...",
           "LAST_DATE_FXBL":"..."
         },
         {
           ...
         }
       ]
       ===================================================== */

    L_COUNT := 1;


    LOOP

        L_OBJECT :=
            REGEXP_SUBSTR(
                V_DATA,
                '\{[^}]*\}',
                1,
                L_COUNT
            );


        /* -------------------------------------------------
           NO MORE OBJECTS
           ------------------------------------------------- */

        EXIT WHEN L_OBJECT IS NULL;


        /* =================================================
           READ JSON VALUES
           ================================================= */

        L_GROUPS :=
            GET_JSON_VALUE(
                L_OBJECT,
                'GROUPS'
            );


        L_MEAL_PREF :=
            GET_JSON_VALUE(
                L_OBJECT,
                'MEAL_PREFERENCE'
            );


        L_FX_BADGE :=
            GET_JSON_VALUE(
                L_OBJECT,
                'FX_BADGE'
            );


        L_NAME :=
            GET_JSON_VALUE(
                L_OBJECT,
                'NAME'
            );


        L_FIRST_DATE :=
            GET_JSON_VALUE(
                L_OBJECT,
                'FIRST_DATE_FXBL'
            );


        L_LAST_DATE :=
            GET_JSON_VALUE(
                L_OBJECT,
                'LAST_DATE_FXBL'
            );


        /* =================================================
           VALIDATE BADGE
           ================================================= */

        IF L_FX_BADGE IS NOT NULL
           AND TRIM(L_FX_BADGE) <> ''
        THEN

            /* =============================================
               CHECK DUPLICATE
               ============================================= */

            SELECT COUNT(1)
            INTO L_COUNT
            FROM CMS.C_F1_MEAL_INFO M
            WHERE UPPER(TRIM(M.FX_BADGE)) =
                  UPPER(TRIM(L_FX_BADGE))
              AND NVL(M.DEL_FLAG,0) = 0;


            IF L_COUNT = 0 THEN

                /* =========================================
                   INSERT
                   ========================================= */

                INSERT INTO CMS.C_F1_MEAL_INFO
                (
                    ID,
                    GROUPS,
                    MEAL_PREFERENCE,
                    FX_BADGE,
                    NAME,
                    FIRST_DATE_FXBL,
                    LAST_DATE_FXBL,
                    ADD_BY,
                    ADD_DATE,
                    DEL_FLAG
                )
                VALUES
                (
                    CMS.SEQ_C_F1_MEAL_INFO.NEXTVAL,

                    L_GROUPS,

                    L_MEAL_PREF,

                    L_FX_BADGE,

                    L_NAME,

                    CASE
                        WHEN L_FIRST_DATE IS NOT NULL
                             AND TRIM(L_FIRST_DATE) <> ''
                        THEN
                            TO_DATE(
                                L_FIRST_DATE,
                                'DD-MM-YYYY'
                            )
                        ELSE
                            NULL
                    END,

                    CASE
                        WHEN L_LAST_DATE IS NOT NULL
                             AND TRIM(L_LAST_DATE) <> ''
                        THEN
                            TO_DATE(
                                L_LAST_DATE,
                                'DD-MM-YYYY'
                            )
                        ELSE
                            NULL
                    END,

                    V_ADD_BY,

                    SYSDATE,

                    0
                );


                L_INSERT_COUNT :=
                    L_INSERT_COUNT + 1;


            ELSE

                L_SKIP_COUNT :=
                    L_SKIP_COUNT + 1;

            END IF;

        END IF;


        /* =================================================
           NEXT JSON OBJECT
           ================================================= */

        L_COUNT :=
            L_COUNT + 1;

    END LOOP;


    /* =====================================================
       COMMIT
       ===================================================== */

    COMMIT;


    /* =====================================================
       RESULT
       ===================================================== */

    IF L_INSERT_COUNT = 0 THEN

        V_RES := '0';

        V_RESULT :=
            'No records inserted. Duplicate or invalid data.';

    ELSE

        V_RES := '1';

        V_RESULT :=
            'Bulk upload completed. Inserted: ' ||
            L_INSERT_COUNT ||
            ', Skipped: ' ||
            L_SKIP_COUNT;

    END IF;


EXCEPTION

    WHEN OTHERS THEN

        ROLLBACK;

        V_RES := '0';

        V_RESULT :=
            'Bulk upload failed: ' ||
            SQLERRM;

END BULK_F1_MENU;
