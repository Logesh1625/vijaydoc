private isActiveUser(item: any): boolean {

  if (!item) {
    return false;
  }

  var delFlag = 0;

  if (
    item.DelFlag !== undefined &&
    item.DelFlag !== null &&
    item.DelFlag !== ''
  ) {
    delFlag = item.DelFlag;

  } else if (
    item.DEL_FLAG !== undefined &&
    item.DEL_FLAG !== null &&
    item.DEL_FLAG !== ''
  ) {
    delFlag = item.DEL_FLAG;

  } else if (
    item.delFlag !== undefined &&
    item.delFlag !== null &&
    item.delFlag !== ''
  ) {
    delFlag = item.delFlag;

  } else if (
    item.del_flag !== undefined &&
    item.del_flag !== null &&
    item.del_flag !== ''
  ) {
    delFlag = item.del_flag;
  }

  return Number(delFlag) === 0;
}
