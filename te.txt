CREATE OR REPLACE PROCEDURE DMS.RUN_SUMMARY_REPORT_PK
IS
    v_seq_number NUMBER;
BEGIN

    INSERT INTO DMS.C_SUMMARY_REPORT_T
    (
        ID,
        WORK_DATE,
        DORM_NAME,
        DORM_LOCATION,
        Overall_Dorm_Capacity,
        Overall_Floor_Capacity,
        Occupied_Floor_Capacity,
        Vacant_Floor_Capacity,
        SUM_ROOM_CAPACITY,
        Occupied_Room_Capacity,
        Vacant_Room_Capacity,
        Sum_of_Bed_Capacity,
        Occupied_Bed_Capacity,
        Vacant_Bed_Capacity,
        occupied_bed_ratio,
        vacant_bed_ratio,
        DORM_ID
    )

    SELECT
        DMS.C_SUMMARY_REPORT_SEQ.NEXTVAL,
        WORK_DATE,
        DORM_NAME,
        DORM_LOCATION,
        DORM_CAPACITY,
        FLOOR_CAPACITY,
        Occupied_Floor_Capacity,
        Vacant_Floor_Capacity,
        SUM_ROOM_CAPACITY,
        Occupied_Room_Capacity,
        Vacant_Room_Capacity,
        Sum_of_Bed_Capacity,
        Occupied_Bed_Capacity,
        Vacant_Bed_Capacity,
        occupied_bed_ratio,
        vacant_bed_ratio,
        DORM_ID

    FROM
    (
        SELECT
            TRUNC(SYSDATE) AS WORK_DATE,
            D.DORM_NAME,
            D.DORM_LOCATION,
            D.DORM_CAPACITY,
            D.FLOOR_CAPACITY,

            COUNT(DISTINCT CASE
                WHEN B.CREATE_BED_FORM_STATUS = 1
                THEN B.FLOOR_ID
            END) AS Occupied_Floor_Capacity,

            COUNT(DISTINCT CASE
                WHEN F.CREATE_FLOOR_FORM_STATUS != 3
                THEN F.ID
            END) AS Vacant_Floor_Capacity,

            -- Active Floor Room Capacity
            (
                SELECT NVL(SUM(F2.ROOM_CAPACITY), 0)
                FROM DMS.C_FLOOR_CREATION_T F2
                WHERE F2.DORM_ID = D.ID
                  AND F2.DEL_FLAG = '0'
            ) AS SUM_ROOM_CAPACITY,

            -- Occupied Room Capacity
            (
                SELECT NVL(SUM(F2.ROOM_CAPACITY), 0)
                FROM DMS.C_FLOOR_CREATION_T F2
                WHERE F2.DORM_ID = D.ID
                  AND F2.DEL_FLAG = '0'
            )
            -
            COUNT(DISTINCT CASE
                WHEN B.CREATE_BED_FORM_STATUS NOT IN (1,3)
                THEN R.ID
            END) AS Occupied_Room_Capacity,

            -- Active Room Bed Capacity
            NVL(
                (
                    SELECT SUM(R2.BED_CAPACITY)
                    FROM DMS.C_ROOM_CREATION_T R2
                    WHERE R2.DORM_ID = D.ID
                      AND R2.DEL_FLAG = '0'
                ), 0
            ) AS Sum_of_Bed_Capacity,

            COUNT(CASE
                WHEN B.CREATE_BED_FORM_STATUS = 1
                THEN 1
            END) AS Occupied_Bed_Capacity,

            COUNT(CASE
                WHEN B.CREATE_BED_FORM_STATUS = 0
                THEN 1
            END) AS Vacant_Bed_Capacity,

            CASE
                WHEN MAX(D.DORM_CAPACITY) > 0
                THEN
                    (
                        COUNT(DISTINCT CASE
                            WHEN B.CREATE_BED_FORM_STATUS = 1
                            THEN B.ID
                        END)
                        / MAX(D.DORM_CAPACITY)
                    ) * 100
                ELSE 0
            END AS occupied_bed_ratio,

            CASE
                WHEN MAX(D.DORM_CAPACITY) > 0
                THEN
                    (
                        (
                            MAX(D.DORM_CAPACITY)
                            -
                            COUNT(DISTINCT CASE
                                WHEN B.CREATE_BED_FORM_STATUS = 1
                                THEN B.ID
                            END)
                        )
                        / MAX(D.DORM_CAPACITY)
                    ) * 100
                ELSE 0
            END AS vacant_bed_ratio,

            D.ID AS DORM_ID

        FROM DMS.C_DORM_CREATION_T D

        LEFT JOIN DMS.C_FLOOR_CREATION_T F
            ON D.ID = F.DORM_ID
            AND F.DEL_FLAG = '0'

        LEFT JOIN DMS.C_ROOM_CREATION_T R
            ON F.ID = R.FLOOR_ID
            AND R.DEL_FLAG = '0'

        LEFT JOIN DMS.C_BED_CREATION_T B
            ON R.ID = B.ROOM_ID
            AND B.DEL_FLAG = '0'

        WHERE D.DEL_FLAG = '0'

        GROUP BY
            D.DORM_NAME,
            D.DORM_LOCATION,
            D.DORM_CAPACITY,
            D.FLOOR_CAPACITY,
            D.ID
    );

END RUN_SUMMARY_REPORT_PK;
/
