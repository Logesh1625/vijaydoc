getGuardPassCode(
  params?: any,
  site?: any,
  isShowLabel?: any
): Observable<any> {

  const code = params.slice(10);

  const _param = {
    TypeName: code,
    Func: "CMS-GetBaseTypeInfo",
    Site: site,
  };

  const config = {
    headers: {
      apiID: "ffff-1710991146185-10208193125-0170",
      userKey: this.USER_TN,
    },
  };

  const param = this.dealPostParamsEncrypt(_param);

  return this.http
    .post(iwxServiceUrl, param, null, config)
    .pipe(
      map((resp: any) => {
        if (!resp || !resp.r) {
          return [];
        }

        const res = JSON.parse(
          this.security.decrypt(resp.r, this.iwxKInfo)
        );

        console.log("Meal Preference API response:", res);

        if (
          String(res.IsOK) !== "1" ||
          !Array.isArray(res.List)
        ) {
          return [];
        }

        return res.List
          .filter((item: any) => item)
          .map((item: any) => ({
            label: item.Name,
            value: isShowLabel
              ? item.Name
              : Number(item.BaseCodeId),
          }));
      })
    );
}
