batchAdd(data: any[]) {

  console.log("Before filter:", data.length);

  const isSampleRow = (row: any): boolean => {
    const first = String(
      row["First Day at Factory"] || ""
    ).trim().toUpperCase();

    const last = String(
      row["Last Day at Factory"] || ""
    ).trim().toUpperCase();

    return first === "DD-MM-YYYY" &&
           last === "DD-MM-YYYY";
  };

  this.excelData = (data || []).filter(
    (row: any) => !isSampleRow(row)
  );

  console.log("After filter:", this.excelData.length);

  console.log(
    "Last imported row:",
    this.excelData[this.excelData.length - 1]
  );

  this.sussessUpLoad.total = this.excelData.length;

  // Keep the remaining existing validation logic unchanged.
}
